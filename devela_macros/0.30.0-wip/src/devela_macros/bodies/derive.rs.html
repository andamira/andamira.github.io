5use proc_macro::{Delimiter, Group, Punct, Spacing, TokenStream as TS, TokenTree as TT};

7pub(crate) fn body_macro_apply(args: TS, input: TS) -> TS {
8    macro_call(args.into_iter().collect(), input)
9}

11pub(crate) fn body_macro_derive(args: TS, input: TS) -> TS {
12    let args = match split_top_commas(args.into_iter().collect()) {
13        Ok(args) => args,
14        Err(err) => return err,
15    };
16    let mut parsed = Vec::with_capacity(args.len());
17    for arg in &args {
18        let Ok(head) = parse_macro_head(arg.clone()) else {
19            return error(
20                "expected `path::to::Macro`, `path::to::Macro!`, or `path::to::Macro!(...)`",
21            );
22        };
23        parsed.push(head);
24    }
25    // Pure classic derive path.
26    if parsed.iter().all(|head| !head.had_bang) {
27        if parsed.iter().any(|head| head.args.is_some()) {
28            return error("macro derives with arguments must end in `!`");
29        }
30        let mut out = real_derive(join_commas(args));
31        out.extend(input);
32        return out;
33    }
34    let mut out = TS::new();
35    let mut real_derives = Vec::new();
36    for (arg, head) in args.into_iter().zip(parsed) {
37        if head.had_bang {
38            out.extend(macro_call_parsed(head, input.clone()));
39        } else if head.args.is_some() {
40            return error("macro derives with arguments must end in `!`");
41        } else {
42            real_derives.push(arg);
43        }
44    }
45    let mut real_derives = join_commas(real_derives);
46    if !real_derives.is_empty() {
47        real_derives.extend([comma()]);
48    }
49    real_derives.extend("::devela::__macro_derive_helpers,".parse::<TS>().unwrap());
50    out.extend(real_derive(real_derives));
51    out.extend(input);
52    out
53}
54pub(crate) fn body_macro_derive_with(args: TS, input: TS) -> TS {
55    let args = match split_top_commas(args.into_iter().collect()) {
56        Ok(args) => args,
57        Err(err) => return err,
58    };
59    let mut out = TS::new();
60    for arg in args {
61        out.extend(macro_call(arg, input.clone()));
62    }
63    // permits inert helper attrs such as `#[__custom(...)]` on the original item
64    out.extend(macro_attrs_derive());
65    out.extend(input);
66    out
67}

71struct MacroHead {
72    path: Vec<TT>,
73    args: Option<Group>,
74    had_bang: bool,
75}
76// This accepts: m m! m("x") m!("x") path::to::m("x", "y") path::to::m!("x", "y")
77fn parse_macro_head(mut tts: Vec<TT>) -> Result<MacroHead, ()> {
78    strip_trailing_comma(&mut tts);
79    let args = if matches!(
80        tts.last(),
81        Some(TT::Group(g)) if g.delimiter() == Delimiter::Parenthesis
82    ) {
83        match tts.pop().unwrap() {
84            TT::Group(g) => Some(g),
85            _ => unreachable!(),
86        }
87    } else {
88        None
89    };
90    let had_bang = path_bang(&tts)?;
91    if had_bang {
92        strip_trailing_bang(&mut tts)?;
93    }
94    Ok(MacroHead { path: tts, args, had_bang })
95}
96fn macro_call(head: Vec<TT>, input: TS) -> TS {
97    let Ok(head) = parse_macro_head(head) else {
98        return error("expected `path::to::macro`, optionally followed by `(...)`");
99    };
100    macro_call_parsed(head, input)
101}
102fn macro_call_parsed(mut head: MacroHead, input: TS) -> TS {
103    head.path.push(bang());
104    let mut body = TS::new();
105    if let Some(args) = head.args {
106        body.extend([TT::Group(args)]);
107    }
108    body.extend(input);
109    head.path.push(TT::Group(Group::new(Delimiter::Brace, body)));
110    head.path.into_iter().collect()
111}
112fn real_derive(derives: TS) -> TS {
113    let mut inner: TS = "::core::prelude::v1::derive".parse().unwrap();
114    inner.extend([TT::Group(Group::new(Delimiter::Parenthesis, derives))]);
115
116    TS::from_iter([
117        TT::Punct(Punct::new('#', Spacing::Alone)),
118        TT::Group(Group::new(Delimiter::Bracket, inner)),
119    ])
120}
121fn path_bang(tts: &[TT]) -> Result<bool, ()> {
122    let mut i = 0;
123    eat_colons(tts, &mut i);
124    loop {
125        match tts.get(i) {
126            Some(TT::Ident(_)) => i += 1,
127            _ => return Err(()),
128        }
129        if i == tts.len() {
130            return Ok(false);
131        }
132        if matches!(tts.get(i), Some(tt) if is_bang(tt)) {
133            i += 1;
134            return if i == tts.len() { Ok(true) } else { Err(()) };
135        }
136        if !eat_colons(tts, &mut i) {
137            return Err(());
138        }
139    }
140}
141fn split_top_commas(tts: Vec<TT>) -> Result<Vec<Vec<TT>>, TS> {
142    if tts.is_empty() {
143        return Err(error("expected at least one path"));
144    }
145    let mut out = vec![Vec::new()];
146    for tt in tts {
147        if is_comma(&tt) {
148            if out.last().unwrap().is_empty() {
149                return Err(error("empty argument"));
150            }
151            out.push(Vec::new());
152        } else {
153            out.last_mut().unwrap().push(tt);
154        }
155    }
156    // accept one trailing comma
157    if out.last().is_some_and(Vec::is_empty) {
158        out.pop();
159    }
160    Ok(out)
161}

163fn join_commas(parts: Vec<Vec<TT>>) -> TS {
164    let mut out = TS::new();
165    for (i, part) in parts.into_iter().enumerate() {
166        if i != 0 {
167            out.extend([comma()]);
168        }
169        out.extend(part);
170    }
171
172    out
173}
174fn comma() -> TT {
175    TT::Punct(Punct::new(',', Spacing::Alone))
176}
177fn bang() -> TT {
178    TT::Punct(Punct::new('!', Spacing::Alone))
179}
180fn is_comma(tt: &TT) -> bool {
181    matches!(tt, TT::Punct(p) if p.as_char() == ',')
182}
183fn is_bang(tt: &TT) -> bool {
184    matches!(tt, TT::Punct(p) if p.as_char() == '!')
185}
186fn macro_attrs_derive() -> TS {
187    real_derive("::devela::__macro_derive_helpers,".parse().unwrap())
188}
189fn strip_trailing_bang(tts: &mut Vec<TT>) -> Result<(), ()> {
190    match tts.pop() {
191        Some(TT::Punct(p)) if p.as_char() == '!' => Ok(()),
192        _ => Err(()),
193    }
194}
195fn strip_trailing_comma(tts: &mut Vec<TT>) {
196    if tts.last().is_some_and(is_comma) {
197        tts.pop();
198    }
199}
200fn error(msg: &str) -> TS {
201    format!("::core::compile_error!({msg:?});").parse().unwrap()
202}
203fn eat_colons(tts: &[TT], i: &mut usize) -> bool {
204    let ok = matches!(tts.get(*i),
205            Some(TT::Punct(p)) if p.as_char() == ':' && p.spacing() == Spacing::Joint)
206        && matches!(tts.get(*i + 1),
207            Some(TT::Punct(p)) if p.as_char() == ':' && p.spacing() == Spacing::Alone);
208    if ok {
209        *i += 2;
210    }
211    ok
