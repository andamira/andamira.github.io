65impl SpectrumKey {
66    /// Returns this key's keyboard half-row index (`0..=7`).
67    #[must_use]
68    pub const fn half_row(self) -> u8 {
69        (self as u8) >> 3
70    }
71    /// Returns this key's bit index within its half-row (`0..=4`).
72    #[must_use]
73    pub const fn bit(self) -> u8 {
74        (self as u8) & 0x07
75    }
76    /// Returns this key's bit mask within its half-row.
77    #[must_use]
78    pub const fn mask(self) -> u8 {
79        1 << self.bit()
80    }
81    /// Returns the complete 16-bit Z80 I/O address selecting this key's half-row.
82    // FEFE  Caps Z X C V
83    // FDFE  A S D F G
84    // FBFE  Q W E R T
85    // F7FE  1 2 3 4 5
86    // EFFE  0 9 8 7 6
87    // DFFE  P O I U Y
88    // BFFE  Enter L K J H
89    // 7FFE  Space Sym M N B
90    #[must_use]
91    pub const fn port(self) -> u16 {
92        let high = !(1_u8 << self.half_row());
93        ((high as u16) << 8) | 0xFE
94    }
95}

98/// A snapshot of the ZX Spectrum keyboard matrix.
99#[doc = crate::_doc_meta!{
100    location("computer/zx", struct SpectrumKeys),
101    test_size_of(SpectrumKeys = 8|64; niche !Option),
102}]
103#[repr(transparent)]
104#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
105pub struct SpectrumKeys([u8; 8]);
106
107impl SpectrumKeys {
108    /// No keys pressed.
109    pub const NONE: Self = Self([0; 8]);
110
111    pub(crate) const fn new(rows: [u8; 8]) -> Self {
112        Self(rows)
113    }
114
115    /// Returns whether `key` is pressed.
116    #[must_use]
117    #[inline(always)]
118    pub const fn is_pressed(&self, key: SpectrumKey) -> bool {
119        self.0[key.half_row() as usize] & key.mask() != 0
120    }
121    /// Returns whether `key` became pressed since `previous`.
122    #[must_use]
123    #[inline(always)]
124    pub const fn just_pressed(&self, previous: &Self, key: SpectrumKey) -> bool {
125        let (row, mask) = (key.half_row() as usize, key.mask());
126        self.0[row] & !previous.0[row] & mask != 0
127    }
128    /// Returns whether `key` became released since `previous`.
129    #[must_use]
130    #[inline(always)]
131    pub const fn just_released(&self, previous: &Self, key: SpectrumKey) -> bool {
132        let (row, mask) = (key.half_row() as usize, key.mask());
133        previous.0[row] & !self.0[row] & mask != 0
134    }
135}
