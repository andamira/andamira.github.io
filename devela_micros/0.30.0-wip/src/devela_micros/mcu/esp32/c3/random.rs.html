10/// Access to the ESP32-C3 hardware random-number generator.
11#[doc = crate::_doc_meta!{
12    location("mcu/esp32", struct Esp32C3Rng),
13    test_size_of(Esp32C3Rng = 0),
14}]
15/// The hardware state may receive entropy from physical sources, but true-random
16/// output is only guaranteed while a supported main entropy source is active.
17///
18/// Without RF or the internal SAR-ADC entropy source, output should be treated
19/// as pseudo-random and is represented conservatively as weak entropy.
20#[derive(Debug)]
21pub struct Esp32C3Rng(());
22
23impl Esp32C3Rng {
24    /// RNG data register.
25    ///
26    /// `SYSCON_RND_DATA_REG = DR_REG_SYSCON_BASE + 0xB0`.
27    pub const DATA: EspReg32 = EspReg32::new(0x6002_60B0);
28
29    #[cfg(feature = "unsafe_mmio")]
30    pub(crate) const unsafe fn new_unchecked() -> Self {
31        Self(())
32    }
33
34    // /// Qualifies this RNG handle for use as a physical entropy source.
35    // ///
36    // /// # Safety
37    // /// A supported physical entropy source must already be active
38    // /// and must remain active while the returned handle is used.
39    // #[must_use]
40    // #[cfg(feature = "unsafe_mmio")]
41    // pub const unsafe fn assume_entropy(self) -> Esp32C3Entropy {
42    //     Esp32C3Entropy(self)
43    // }
44
45    /// Reads one 32-bit value from the hardware RNG state.
46    #[must_use]
47    #[cfg(feature = "unsafe_mmio")]
48    pub fn next_u32(&mut self) -> u32 {
49        // SAFETY: construction establishes that this is the active ESP32-C3 RNG.
50        unsafe { Self::DATA.read() }
51    }
52}
53
54#[cfg(feature = "unsafe_mmio")]
55impl RandTry for Esp32C3Rng {
56    type Error = Infallible;
57
58    const RAND_OUTPUT_BITS: u32 = 32;
59    const RAND_STATE_BITS: u32 = 0;
60    const RAND_QUALITIES: RandQualities = RandQualities::EXTERNAL.with_entropy_weak();
61
62    fn rand_try_next_u64(&mut self) -> Result<u64, Self::Error> {
63        let lo = self.next_u32() as u64;
64        let hi = self.next_u32() as u64;
65        Ok(lo | hi << 32)
66    }
67
68    fn rand_try_next_u32(&mut self) -> Result<u32, Self::Error> {
69        Ok(self.next_u32())
70    }
71}
