25/// The ESP32-C3 is a single-core 32-bit RISC-V microcontroller with a
26/// maximum CPU frequency of 160 MHz, 384 KiB of ROM, and 400 KiB of SRAM.
27///
28/// It integrates 2.4 GHz Wi-Fi and Bluetooth LE, alongside peripherals
29/// including two UARTs, I²C, SPI, timers, watchdogs, and USB Serial/JTAG.
30/// The main ESP32-C3 variants provide GPIO0 through GPIO21.
31///
32/// devela currently provides direct-boot startup support together with
33/// low-level access to GPIO, I²C0, UART0, and USB Serial/JTAG.
34///
35/// # GPIO
36///
37/// Simple digital output has two independent pieces of state:
38///
39/// - [`GPIO_OUT`] holds the output levels.
40/// - [`GPIO_ENABLE`] selects which pins actively drive those levels.
41///
42/// Their `W1TS` and `W1TC` variants set or clear selected bits without
43/// requiring a read-modify-write.
44///
45/// IO MUX and the GPIO matrix provide the wider pin-routing system.
46/// Peripheral use therefore may require more than the basic GPIO registers.
47///
48/// See also:
49///
50/// - [ESP32-C3 datasheet]
51/// - [ESP32-C3 Technical Reference Manual]
52/// - [ESP32-C3 hardware reference]
53///
54/// [`GPIO_OUT`]: #associatedconstant.GPIO_OUT
55/// [`GPIO_ENABLE`]: #associatedconstant.GPIO_ENABLE
56/// [ESP32-C3 datasheet]: https://documentation.espressif.com/esp32-c3_datasheet_en.pdf
57/// [ESP32-C3 Technical Reference Manual]: https://documentation.espressif.com/esp32-c3_technical_reference_manual_en.pdf
58/// [ESP32-C3 hardware reference]: https://docs.espressif.com/projects/esp-idf/en/stable/esp32c3/hw-reference/index.html
59#[derive(Debug)]
60pub struct McuEsp32C3;

124/// # Clock and timing
125impl McuEsp32C3 {
126    /// External main-crystal (`XTAL_CLK`) frequency, in hertz.
127    pub const XTAL_HZ: u32 = 40_000_000;
128
129    /// System Timer (`SYSTIMER`) peripheral base address.
130    pub const SYSTIMER_BASE: u32 = 0x6002_3000;
131
132    /// System Timer (`SYSTIMER`) peripheral.
133    pub const SYSTIMER: Esp32C3SystemTimer = Esp32C3SystemTimer::new(Self::SYSTIMER_BASE);
134
135    /// Enables System Timer counter `UNIT0`.
136    ///
137    /// Enables the SYSTIMER APB clock, releases the peripheral from reset,
138    /// and enables `UNIT0` counting.
139    ///
140    /// This preserves the existing SYSTIMER configuration
141    /// and does not explicitly reload or clear the counter.
142    ///
143    /// # Safety
144    /// This must execute on the active ESP32-C3 device.
145    ///
146    /// SYSTIMER and its system clock/reset state must not be concurrently configured.
147    #[must_use]
148    #[cfg(feature = "unsafe_mmio")]
149    pub unsafe fn prepare_systimer_unit0() -> Esp32C3SystemTimer {
150        const SYSTEM_PERIP_CLK_EN0: EspReg32 = EspReg32::new(0x600c_0010);
151        const SYSTEM_PERIP_RST_EN0: EspReg32 = EspReg32::new(0x600c_0018);
152        const SYSTIMER_CLOCK_ENABLE: u32 = 1 << 29;
153        const SYSTIMER_RESET: u32 = 1 << 29;
154        const UNIT0_WORK_ENABLE: u32 = 1 << 30;
155        unsafe {
156            let clock = SYSTEM_PERIP_CLK_EN0;
157            clock.write(clock.read() | SYSTIMER_CLOCK_ENABLE);
158
159            let reset = SYSTEM_PERIP_RST_EN0;
160            reset.write(reset.read() & !SYSTIMER_RESET);
161
162            let timer = Self::SYSTIMER;
163            let config = timer.config_reg();
164            config.write(config.read() | UNIT0_WORK_ENABLE);
165
166            timer
167        }
168    }
169}

171/// # I²C
172#[cfg(feature = "unsafe_mmio")]
173impl McuEsp32C3 {
174    /// Enables I²C0, routes it through `sda` and `scl`,
175    /// and configures it as a master at `bus_hz` from the 40 MHz XTAL.
176    ///
177    /// The routed pins are configured for open-drain operation with input
178    /// enabled and their weak internal pull-ups enabled.
179    ///
180    /// # Safety
181    /// I²C0 and both GPIOs must not be concurrently configured or accessed.
182    ///
183    /// While the returned controller is alive, I²C0 and its routed pins
184    /// must not be accessed through another raw hardware handle.
185    pub unsafe fn prepare_i2c0(
186        sda: Esp32C3Pin,
187        scl: Esp32C3Pin,
188        bus_hz: u32,
189    ) -> I2cController<EspI2c> {
190        const SYSTEM_PERIP_CLK_EN0: EspReg32 = EspReg32::new(0x600C_0010);
191        const SYSTEM_PERIP_RST_EN0: EspReg32 = EspReg32::new(0x600C_0018);
192        const I2C0_CLOCK: u32 = 1 << 7;
193        const I2C0_RESET: u32 = 1 << 7;
194
195        const SCL_SIGNAL: u8 = 53;
196        const SDA_SIGNAL: u8 = 54;
197
198        unsafe {
199            let clock = SYSTEM_PERIP_CLK_EN0;
200            clock.write(clock.read() | I2C0_CLOCK);
201
202            let reset = SYSTEM_PERIP_RST_EN0;
203            reset.write(reset.read() | I2C0_RESET);
204            reset.write(reset.read() & !I2C0_RESET);
205
206            Self::prepare_i2c0_pin(sda, SDA_SIGNAL);
207            Self::prepare_i2c0_pin(scl, SCL_SIGNAL);
208
209            Self::I2C0.configure_master_xtal(Self::XTAL_HZ, bus_hz);
210            I2cController::new_unchecked(Self::I2C0)
211        }
212    }
213    unsafe fn prepare_i2c0_pin(pin: Esp32C3Pin, signal: u8) {
214        unsafe {
215            pin.configure_open_drain_pullup();
216            pin.matrix_route_output(signal);
217            pin.matrix_route_input(signal);
218        }
219    }
220}

222/// # UART
223#[cfg(feature = "unsafe_mmio")]
224impl McuEsp32C3 {
225    /// Enables UART0 and selects its direct IO-MUX pins:
226    /// GPIO20 for RX and GPIO21 for TX.
227    ///
228    /// This does not configure baud rate or framing.
229    ///
230    /// # Safety
231    /// UART0 and GPIO20/21 must not be concurrently configured.
232    pub unsafe fn prepare_uart0_default() {
233        const SYSTEM_PERIP_CLK_EN0: EspReg32 = EspReg32::new(0x600C_0010);
234        const SYSTEM_PERIP_RST_EN0: EspReg32 = EspReg32::new(0x600C_0018);
235
236        const UART0_CLOCK: u32 = 1 << 2;
237        const UART0_RESET: u32 = 1 << 2;
238        const UART_RST_CORE: u32 = 1 << 23;
239        const UART_MEMORY_CLOCK: u32 = 1 << 24;
240
241        const IO_MUX_RX: EspReg32 = EspReg32::new(0x6000_9054);
242        const IO_MUX_TX: EspReg32 = EspReg32::new(0x6000_9058);
243
244        const INPUT_ENABLE: u32 = 1 << 9;
245        const FUNCTION_MASK: u32 = 0b111 << 12;
246
247        unsafe {
248            // Keep UART0 and its FIFO memory clocked.
249            let clock = SYSTEM_PERIP_CLK_EN0;
250            clock.write(clock.read() | UART0_CLOCK | UART_MEMORY_CLOCK);
251
252            // Reset the UART peripheral and core together.
253            //
254            // The ESP32-C3 requires the core reset around the peripheral-reset
255            // pulse to avoid transient UART output during initialization.
256            let reset = SYSTEM_PERIP_RST_EN0;
257            let uart_clock = Self::UART0.clock_config_reg();
258
259            reset.write(reset.read() & !UART0_RESET);
260            uart_clock.write(uart_clock.read() | UART_RST_CORE);
261            reset.write(reset.read() | UART0_RESET);
262            reset.write(reset.read() & !UART0_RESET);
263            uart_clock.write(uart_clock.read() & !UART_RST_CORE);
264
265            // Function 0 directly connects GPIO20/21 to U0RXD/U0TXD.
266            let rx = IO_MUX_RX;
267            rx.write((rx.read() & !FUNCTION_MASK) | INPUT_ENABLE);
268
269            let tx = IO_MUX_TX;
270            tx.write(tx.read() & !FUNCTION_MASK);
271        }
272    }
273}

275/// # Boot watchdogs
276#[cfg(feature = "unsafe_mmio")]
277impl McuEsp32C3 {
278    /// Completes the watchdog handoff from ROM flash boot to direct-boot code.
279    ///
280    /// Direct boot bypasses the usual SDK startup that takes ownership of
281    /// watchdog state established during boot. Leaving that state active can
282    /// reset a long-running application shortly after a cold power-on.
283    ///
284    /// This disables the Timer Group 0 MWDT and RTC WDT, including their
285    /// flash-boot modes, and enables automatic feeding of the Super WDT.
286    ///
287    /// # Safety
288    /// No other code may concurrently configure these watchdogs.
289    pub unsafe fn handoff_boot_watchdogs() {
290        const WDT_WRITE_KEY: u32 = 0x50D8_3AA1;
291        const SWD_WRITE_KEY: u32 = 0x8F1D_312A;
292
293        const WDT_ENABLE: u32 = 1 << 31;
294
295        const MWDT_FLASHBOOT_ENABLE: u32 = 1 << 14;
296        const MWDT_CONFIG_UPDATE: u32 = 1 << 22;
297
298        const RWDT_FLASHBOOT_ENABLE: u32 = 1 << 12;
299
300        const SWD_AUTO_FEED_ENABLE: u32 = 1 << 31;
301
302        unsafe {
303            // Timer Group 0 MWDT.
304            Self::TIMG0_WDT_WPROTECT.write(WDT_WRITE_KEY);
305
306            let config = Self::TIMG0_WDT_CONFIG0.read();
307            Self::TIMG0_WDT_CONFIG0
308                .write((config & !(WDT_ENABLE | MWDT_FLASHBOOT_ENABLE)) | MWDT_CONFIG_UPDATE);
309
310            Self::TIMG0_WDT_WPROTECT.write(0);
311
312            // RTC WDT.
313            Self::RTC_WDT_WPROTECT.write(WDT_WRITE_KEY);
314
315            let config = Self::RTC_WDT_CONFIG0.read();
316            Self::RTC_WDT_CONFIG0.write(config & !(WDT_ENABLE | RWDT_FLASHBOOT_ENABLE));
317
318            Self::RTC_WDT_WPROTECT.write(0);
319
320            // Super WDT.
321            //
322            // Unlike the digital watchdogs above, keep the SWD operational but
323            // neutralize its boot-time reset behavior through automatic feeding.
324            Self::RTC_SWD_WPROTECT.write(SWD_WRITE_KEY);
325
326            let config = Self::RTC_SWD_CONF.read();
327            Self::RTC_SWD_CONF.write(config | SWD_AUTO_FEED_ENABLE);
328
329            Self::RTC_SWD_WPROTECT.write(0);
330        }
331    }
332}
