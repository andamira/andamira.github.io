15/// Provides low-level master configuration and blocking transfers
16/// through the controller's command and FIFO engine.
17///
18/// Chip-specific clock enabling and GPIO-matrix routing are handled separately.
19#[repr(transparent)]
20#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
21pub struct EspI2c(u32);

85#[allow(dead_code)]
86impl EspI2c {
87    const CTRL_CONF_UPDATE: u32 = 1 << 11;
88    const CTRL_FSM_RESET: u32 = 1 << 10;
89    const CTRL_ARBITRATION: u32 = 1 << 9;
90    const CTRL_RX_LSB_FIRST: u32 = 1 << 7;
91    const CTRL_TX_LSB_FIRST: u32 = 1 << 6;
92    const CTRL_TRANS_START: u32 = 1 << 5;
93    const CTRL_MASTER: u32 = 1 << 4;
94    const CTRL_SCL_OPEN_DRAIN: u32 = 1 << 1;
95    const CTRL_SDA_OPEN_DRAIN: u32 = 1 << 0;
96
97    const STATUS_BUS_BUSY: u32 = 1 << 4;
98
99    const FIFO_TX_RESET: u32 = 1 << 13;
100    const FIFO_RX_RESET: u32 = 1 << 12;
101    const FIFO_NONFIFO: u32 = 1 << 10;
102
103    const INT_NACK: u32 = 1 << 10;
104    const INT_TIMEOUT: u32 = 1 << 8;
105    const INT_ARBITRATION: u32 = 1 << 5;
106    const INT_ALL: u32 = 0x3ffff;
107
108    const CMD_RESTART: u32 = 6;
109    const CMD_WRITE: u32 = 1;
110    const CMD_STOP: u32 = 2;
111    const CMD_END: u32 = 4;
112    const CMD_DONE: u32 = 1 << 31;
113
114    const POLL_LIMIT: u32 = 1_000_000;
115
116    #[must_use]
117    const fn command(op: u32, bytes: u8, check_ack: bool) -> u32 {
118        bytes as u32
119            | ((check_ack as u32) << 8)
120            // expected ACK = 0
121            | (op << 11)
122    }
123    fn next_slice_byte(
124        slices: &[&[u8]],
125        slice_index: &mut usize,
126        byte_index: &mut usize,
127    ) -> Option<u8> {
128        while *slice_index < slices.len() {
129            let slice = slices[*slice_index];
130            if *byte_index < slice.len() {
131                let byte = slice[*byte_index];
132                *byte_index += 1;
133                return Some(byte);
134            }
135            *slice_index += 1;
136            *byte_index = 0;
137        }
138        None
139    }
140}

142#[cfg(feature = "unsafe_mmio")]
143impl EspI2c {
144    unsafe fn modify(reg: EspReg32, mask: u32, value: u32) {
145        unsafe {
146            reg.write((reg.read() & !mask) | (value & mask));
147        }
148    }
149    unsafe fn reset_fifos(self) {
150        let reg = self.fifo_conf_reg();
151        unsafe {
152            let value = reg.read() & !Self::FIFO_NONFIFO;
153            reg.write(value | Self::FIFO_TX_RESET | Self::FIFO_RX_RESET);
154            reg.write(value);
155        }
156    }
157    unsafe fn update(self) {
158        let reg = self.control_reg();
159        unsafe { reg.write(reg.read() | Self::CTRL_CONF_UPDATE) };
160    }
161    unsafe fn reset_fsm(self) {
162        let reg = self.control_reg();
163        unsafe { reg.write(reg.read() | Self::CTRL_FSM_RESET) };
164    }
165    unsafe fn run_commands(self, last_command: u8) -> Result<(), I2cError> {
166        unsafe {
167            self.int_clr_reg().write(Self::INT_ALL);
168            self.update();
169            let control = self.control_reg();
170            control.write(control.read() | Self::CTRL_TRANS_START);
171            for _ in 0..Self::POLL_LIMIT {
172                let status = self.int_raw_reg().read();
173                let error = if status & Self::INT_NACK != 0 {
174                    Some(I2cError::Nack)
175                } else if status & Self::INT_TIMEOUT != 0 {
176                    Some(I2cError::Timeout)
177                } else if status & Self::INT_ARBITRATION != 0 {
178                    Some(I2cError::ArbitrationLost)
179                } else {
180                    None
181                };
182                if let Some(error) = error {
183                    self.int_clr_reg().write(Self::INT_ALL);
184                    self.reset_fsm();
185                    return Err(error);
186                }
187                if self.command_reg(last_command).read() & Self::CMD_DONE != 0 {
188                    self.int_clr_reg().write(Self::INT_ALL);
189                    return Ok(());
190                }
191            }
192            self.reset_fsm();
193            Err(I2cError::Timeout)
194        }
195    }
196}

198#[cfg(feature = "unsafe_mmio")]
199impl EspI2c {
200    /// Configures master mode using an XTAL source clock.
201    ///
202    /// The timing calculation follows Espressif's ESP32-C3 low-level HAL.
203    ///
204    /// # Panics
205    /// Panics if either frequency is zero
206    /// or the requested timing cannot be represented by this controller.
207    ///
208    /// # Safety
209    /// The controller must be clocked and exclusively configured by this caller.
210    pub unsafe fn configure_master_xtal(self, source_hz: u32, bus_hz: u32) {
211        assert!(source_hz != 0);
212        assert!(bus_hz != 0);
213
214        let clkm_div = (source_hz as u64 / (bus_hz as u64 * 1024) + 1) as u32;
215        let sclk_hz = source_hz / clkm_div;
216        let half_cycle = sclk_hz / bus_hz / 2;
217        assert!((4..=512).contains(&half_cycle), "I²C bus timing is not representable");
218        let wait_high = if bus_hz >= 80_000 { half_cycle / 2 - 2 } else { half_cycle / 4 };
219        let scl_high = half_cycle - wait_high;
220
221        let sda_hold = half_cycle / 4;
222        let sda_sample = half_cycle / 2;
223        let setup = half_cycle;
224        let hold = half_cycle;
225
226        let timeout = 32 - (5 * half_cycle).leading_zeros() + 2;
227
228        assert!(clkm_div <= 256);
229        assert!(half_cycle <= 512);
230        assert!(wait_high <= 0x7f);
231        assert!(timeout <= 0x1f);
232
233        unsafe {
234            // Master, MSB first, open-drain peripheral outputs, no arbitration.
235            let ctrl = self.control_reg();
236            ctrl.write(
237                (ctrl.read()
238                    & !(Self::CTRL_ARBITRATION
239                        | Self::CTRL_RX_LSB_FIRST
240                        | Self::CTRL_TX_LSB_FIRST))
241                    | Self::CTRL_MASTER
242                    | Self::CTRL_SCL_OPEN_DRAIN
243                    | Self::CTRL_SDA_OPEN_DRAIN,
244            );
245
246            self.reset_fifos();
247            self.int_ena_reg().write(0);
248
249            // XTAL source: sclk_sel = 0, fractional divider = 0.
250            Self::modify(self.clock_reg(), 0x003f_ffff, (clkm_div - 1) | (1 << 21));
251
252            Self::modify(self.scl_low_reg(), 0x1ff, half_cycle - 1);
253            Self::modify(self.scl_high_reg(), 0xffff, scl_high | (wait_high << 9));
254
255            Self::modify(self.sda_hold_reg(), 0x1ff, sda_hold - 1);
256            Self::modify(self.sda_sample_reg(), 0x1ff, sda_sample - 1);
257
258            Self::modify(self.rstart_setup_reg(), 0x1ff, setup - 1);
259            Self::modify(self.stop_setup_reg(), 0x1ff, setup - 1);
260            Self::modify(self.start_hold_reg(), 0x1ff, hold - 1);
261            Self::modify(self.stop_hold_reg(), 0x1ff, hold - 1);
262
263            Self::modify(self.timeout_reg(), 0x3f, timeout | (1 << 5));
264
265            self.update();
266        }
267    }

278    /// Performs one blocking write transaction to a 7-bit I²C target.
279    ///
280    /// Transfers larger than the hardware FIFO are continued transparently
281    /// without releasing the bus.
282    ///
283    /// # Safety
284    /// The controller and its routed pins must be configured for this bus
285    /// and must not be concurrently accessed.
286    pub unsafe fn write_blocking(self, address: I2cAddr7, bytes: &[u8]) -> Result<(), I2cError> {
287        unsafe { self.write_slices_blocking(address, &[bytes]) }
288    }
289
290    /// Performs one blocking write transaction from a sequence of byte slices.
291    ///
292    /// The slices are concatenated into a single I²C transfer without inserting
293    /// STOP or repeated-START conditions between them. Empty slices are ignored.
294    ///
295    /// Transfers larger than the hardware FIFO are continued transparently
296    /// without releasing the bus.
297    ///
298    /// Passing no data slices performs an address-only transaction, equivalent
299    /// to [`probe_blocking`](Self::probe_blocking).
300    ///
301    /// # Safety
302    /// The controller and its routed pins must be configured for this bus
303    /// and must not be concurrently accessed.
304    pub unsafe fn write_slices_blocking(
305        self,
306        address: I2cAddr7,
307        slices: &[&[u8]],
308    ) -> Result<(), I2cError> {
309        unsafe {
310            is! { self.status_reg().read() & Self::STATUS_BUS_BUSY != 0, self.reset_fsm() }
311
312            self.reset_fifos();
313            self.data_reg().write(address.write_address_byte() as u32);
314            self.command_reg(0).write(Self::command(Self::CMD_RESTART, 0, false));
315            self.command_reg(1).write(Self::command(Self::CMD_WRITE, 1, true));
316
317            let (mut slice_index, mut byte_index) = (0, 0);
318            let mut next = Self::next_slice_byte(slices, &mut slice_index, &mut byte_index);
319
320            // An empty payload is an address-only transaction.
321            if next.is_none() {
322                self.command_reg(2).write(Self::command(Self::CMD_STOP, 0, false));
323                return self.run_commands(2);
324            }
325
326            let mut first = true;
327            while next.is_some() {
328                // The first FIFO load already contains the address byte.
329                let capacity = if first { Self::FIFO_LEN - 1 } else { Self::FIFO_LEN };
330
331                let mut count = 0usize;
332                while count < capacity {
333                    let Some(byte) = next else { break };
334                    self.data_reg().write(byte as u32);
335                    count += 1;
336                    next = Self::next_slice_byte(slices, &mut slice_index, &mut byte_index);
337                }
338                let write_command = if first { 2 } else { 0 };
339
340                self.command_reg(write_command).write(Self::command(
341                    Self::CMD_WRITE,
342                    count as u8,
343                    true,
344                ));
345                let last_command = write_command + 1;
346
347                if next.is_none() {
348                    self.command_reg(last_command).write(Self::command(Self::CMD_STOP, 0, false));
349                } else {
350                    // Pause the command engine without generating STOP.
351                    // Refill the FIFO and continue the same bus transaction.
352                    self.command_reg(last_command).write(Self::command(Self::CMD_END, 0, false));
353                }
354                self.run_commands(last_command)?;
355                first = false;
356            }
357            Ok(())
358        }
359    }
