8/// An AVR USART described by its control, baud-rate, and data registers.
9#[doc = crate::_doc_meta!{
10    location("mcu/avr", struct AvrUsart),
11    test_size_of(AvrUsart = 12|96; niche !Option),
12}]
13/// The operational API provides asynchronous 8N1 configuration
14/// and byte-oriented polling for reception and transmission.
15///
16/// Values can safely be copied and inspected. Operations that access the
17/// described registers are unsafe because the addresses must correspond to
18/// the active device and access must respect the peripheral's hardware state.
19///
20/// # Methods
21///
22/// The operational API is grouped by USART function:
23///
24/// - [Configuration](#configuration) — selects framing, baud rate,
25///   and which sides of the USART are enabled.
26/// - [Receiver](#receiver) — controls reception, reports readiness,
27///   and reads bytes using blocking or non-blocking operations.
28/// - [Transmitter](#transmitter) — controls transmission, reports readiness,
29///   and queues bytes using blocking or non-blocking operations.
30///
31/// Semantic and datasheet register accessors are also provided for lower-level use.
32#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
33pub struct AvrUsart {
34    ucsra: AvrReg8,
35    ucsrb: AvrReg8,
36    ucsrc: AvrReg8,
37    ubrrl: AvrReg8,
38    ubrrh: AvrReg8,
39    udr: AvrReg8,
40}

79#[allow(dead_code)]
80impl AvrUsart {
81    // UCSRnA
82    const RXC: u8 = 1 << 7; // Receive complete.
83    const UDRE: u8 = 1 << 5; // Data-register empty.
84    const U2X: u8 = 1 << 1; // Double asynchronous transmission speed.
85
86    // UCSRnB
87    const RXEN: u8 = 1 << 4; // Receiver enable.
88    const TXEN: u8 = 1 << 3; // Transmitter enable.
89    const UCSZ2: u8 = 1 << 2; // Character-size bit 2.
90
91    // UCSRnC
92    const UCSZ1: u8 = 1 << 2; // Character-size bit 1.
93    const UCSZ0: u8 = 1 << 1; // Character-size bit 0.
94
95    /// Selects the closest representable asynchronous baud setting.
96    ///
97    /// Returns `(UBRRn, double_speed)`.
98    fn baud_setting(clock_hz: u32, baud: u32) -> Option<(u16, bool)> {
99        // Returns: (UBRRn, absolute-error numerator, effective divisor).
100        fn candidate(clock_hz: u32, baud: u32, factor: u32) -> Option<(u16, u64, u64)> {
101            is! { clock_hz == 0 || baud == 0, return None }
102            let (clock, baud, factor) = (clock_hz as u64, baud as u64, factor as u64);
103
104            // baud = clock / (factor × (UBRR + 1))
105            // Round UBRR + 1 to the nearest integer.
106            let denominator = factor * baud;
107            let divider = (clock + denominator / 2) / denominator;
108
109            // UBRR is 12-bit: 0..=4095, therefore divider is 1..=4096.
110            is! { divider == 0 || divider > 4096, return None }
111            let effective_divisor = factor * divider;
112
113            // The actual baud error is:
114            //
115            // |clock / effective_divisor - baud|
116            // =
117            // |clock - baud × effective_divisor| / effective_divisor
118            let error = clock.abs_diff(baud * effective_divisor);
119            Some(((divider - 1) as u16, error, effective_divisor))
120        }
121        let normal = candidate(clock_hz, baud, 16);
122        let double = candidate(clock_hz, baud, 8);
123        match (normal, double) {
124            (None, None) => None,
125            (Some((ubrr, _, _)), None) => Some((ubrr, false)),
126            (None, Some((ubrr, _, _))) => Some((ubrr, true)),
127            (
128                Some((normal_ubrr, normal_error, normal_divisor)),
129                Some((double_ubrr, double_error, double_divisor)),
130            ) => {
131                // Compare the two rational errors without floating point.
132                // Ties prefer normal-speed mode.
133                if normal_error * double_divisor <= double_error * normal_divisor {
134                    Some((normal_ubrr, false))
135                } else {
136                    Some((double_ubrr, true))
137                }
138            }
139        }
140    }
141}

143#[cfg(feature = "unsafe_mmio")]
144impl AvrUsart {
145    unsafe fn configure_8n1(self, clock_hz: u32, baud: u32, enable: u8) {
146        let Some((ubrr, double_speed)) = Self::baud_setting(clock_hz, baud) else {
147            panic!("AVR USART baud rate is not representable");
148        };
149        unsafe {
150            // Take ownership of this USART configuration.
151            // Disables RX, TX and USART interrupts while it is reconfigured.
152            self.control_reg().write(0);
153
154            // U2X selects normal (÷16) or double-speed (÷8) asynchronous mode.
155            self.status_reg().write(if double_speed { Self::U2X } else { 0 });
156
157            // Updating UBRRnL activates the new divider, so write high first.
158            self.baud_high_reg().write((ubrr >> 8) as u8);
159            self.baud_low_reg().write(ubrr as u8);
160
161            self.set_frame_8n1();
162            self.control_reg().write(enable);
163        }
164    }
165}

167/// # Configuration
168#[cfg(feature = "unsafe_mmio")]
169impl AvrUsart {
170    /// Selects asynchronous 8N1 framing: 8 data bits, no parity, and 1 stop bit.
171    ///
172    /// This does not change the baud rate or enable or disable the receiver or transmitter.
173    ///
174    /// # Safety
175    /// This must describe a USART on the active device and not be concurrently used or configured.
176    pub unsafe fn set_frame_8n1(self) {
177        let control = self.control_reg();
178        unsafe {
179            // UCSZn2 = 0; together with UCSZn1:0 = 0b11 this selects 8-bit data.
180            control.write(control.read() & !Self::UCSZ2);
181            // Asynchronous, no parity, one stop bit, UCSZn1:0 = 0b11.
182            self.frame_reg().write(Self::UCSZ1 | Self::UCSZ0);
183        }
184    }
185
186    /// Configures a transmit-only asynchronous 8N1 connection.
187    ///
188    /// Selects the closest representable baud rate for `clock_hz` and `baud`,
189    /// configures 8 data bits, no parity and 1 stop bit, and enables the transmitter.
190    ///
191    /// The receiver and USART interrupts are left disabled.
192    ///
193    /// # Panics
194    /// Panics if `clock_hz` or `baud` is zero,
195    /// or if the requested rate cannot be represented by the AVR baud generator.
196    ///
197    /// # Safety
198    /// This must describe a USART on the active device
199    /// and not be concurrently used or configured.
200    pub unsafe fn configure_tx_8n1(self, clock_hz: u32, baud: u32) {
201        unsafe { self.configure_8n1(clock_hz, baud, Self::TXEN) };
202    }
203    /// Configures a full-duplex asynchronous 8N1 connection.
204    ///
205    /// Selects the closest representable baud rate for `clock_hz` and `baud`,
206    /// configures 8 data bits, no parity and 1 stop bit, and enables both
207    /// the receiver and transmitter.
208    ///
209    /// USART interrupts are left disabled.
210    ///
211    /// # Panics
212    /// Panics if `clock_hz` or `baud` is zero,
213    /// or if the requested rate cannot be represented by the AVR baud generator.
214    ///
215    /// # Safety
216    /// This must describe a USART on the active device
217    /// and not be concurrently used or configured.
218    pub unsafe fn configure_rx_tx_8n1(self, clock_hz: u32, baud: u32) {
219        unsafe { self.configure_8n1(clock_hz, baud, Self::RXEN | Self::TXEN) };
220    }
221}

223/// # Receiver
224#[cfg(feature = "unsafe_mmio")]
225impl AvrUsart {
226    /// Returns whether a received byte is ready to be read.
227    ///
228    /// # Safety
229    /// The USART must belong to the active device.
230    #[must_use]
231    pub unsafe fn rx_ready(self) -> bool {
232        unsafe { self.status_reg().read() & Self::RXC != 0 }
233    }
234    /// Enables the receiver, preserving the other control bits.
235    ///
236    /// # Safety
237    /// The USART must belong to the active device and not be concurrently used or configured.
238    pub unsafe fn enable_rx(self) {
239        let reg = self.control_reg();
240        unsafe { reg.write(reg.read() | Self::RXEN) };
241    }
242    /// Disables the receiver, preserving the other control bits.
243    ///
244    /// # Safety
245    /// The USART must belong to the active device and not be concurrently used or configured.
246    pub unsafe fn disable_rx(self) {
247        let reg = self.control_reg();
248        unsafe { reg.write(reg.read() & !Self::RXEN) };
249    }
250
251    /// Attempts to read one received byte without waiting.
252    ///
253    /// Returns `None` if no received byte is available.
254    ///
255    /// The receiver must already be enabled.
256    ///
257    /// # Safety
258    /// The USART must belong to the active device and not be concurrently read.
259    #[must_use]
260    pub unsafe fn try_read_byte(self) -> Option<u8> {
261        if unsafe { self.rx_ready() } {
262            Some(unsafe { self.data_reg().read() })
263        } else {
264            None
265        }
266    }
267    /// Reads one received byte, waiting until one is available.
268    ///
269    /// The receiver must already be enabled.
270    ///
271    /// # Safety
272    /// The USART must belong to the active device and not be concurrently read.
273    #[must_use]
274    pub unsafe fn read_byte_blocking(self) -> u8 {
275        while !unsafe { self.rx_ready() } {}
276        unsafe { self.data_reg().read() }
277    }
278}

280/// # Transmitter
281#[cfg(feature = "unsafe_mmio")]
282impl AvrUsart {
283    /// Returns whether the transmit data register can accept another byte.
284    ///
285    /// # Safety
286    /// The USART must belong to the active device.
287    #[must_use]
288    pub unsafe fn tx_ready(self) -> bool {
289        unsafe { self.status_reg().read() & Self::UDRE != 0 }
290    }
291    /// Enables the transmitter, preserving the other control bits.
292    ///
293    /// # Safety
294    /// The USART must belong to the active device and not be concurrently used or configured.
295    pub unsafe fn enable_tx(self) {
296        let reg = self.control_reg();
297        unsafe { reg.write(reg.read() | Self::TXEN) };
298    }
299    /// Disables the transmitter, preserving the other control bits.
300    ///
301    /// # Safety
302    /// The USART must belong to the active device and not be concurrently used or configured.
303    pub unsafe fn disable_tx(self) {
304        let reg = self.control_reg();
305        unsafe { reg.write(reg.read() & !Self::TXEN) };
306    }
307
308    /// Attempts to queue one byte for transmission without waiting.
309    ///
310    /// Returns `true` if the byte was accepted,
311    /// or `false` if the transmit data register was still busy.
312    ///
313    /// The transmitter must already be enabled.
314    ///
315    /// # Safety
316    /// The USART must belong to the active device and not be concurrently written.
317    #[must_use]
318    pub unsafe fn try_write_byte(self, byte: u8) -> bool {
319        if unsafe { self.tx_ready() } {
320            unsafe { self.data_reg().write(byte) };
321            true
322        } else {
323            false
324        }
325    }
326    /// Queues one byte for transmission, waiting until the USART can accept it.
327    ///
328    /// Waits for space in the transmit data register;
329    /// it does not wait for the byte to finish shifting onto the wire.
330    ///
331    /// The transmitter must already be enabled.
332    ///
333    /// # Safety
334    /// The USART must belong to the active device and not be concurrently written.
335    pub unsafe fn write_byte_blocking(self, byte: u8) {
336        while !unsafe { self.tx_ready() } {}
337        unsafe { self.data_reg().write(byte) };
338    }
339    /// Queues all bytes for transmission, waiting for space as needed.
340    ///
341    /// Returns after the final byte has been accepted by the USART,
342    /// which may be before that byte has finished transmitting on the wire.
343    ///
344    /// The transmitter must already be enabled.
345    ///
346    /// # Safety
347    /// The USART must belong to the active device and not be concurrently written.
348    pub unsafe fn write_bytes_blocking(self, bytes: &[u8]) {
349        for &byte in bytes {
350            unsafe { self.write_byte_blocking(byte) };
351        }
352    }
353}
