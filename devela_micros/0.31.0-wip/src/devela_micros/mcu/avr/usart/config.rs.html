3#[cfg(feature = "unsafe_mmio")]
4impl AvrUsart {
5    unsafe fn configure_8n1(self, clock_hz: u32, baud: u32, enable: u8) {
6        let Some((ubrr, double_speed)) = Self::baud_setting(clock_hz, baud) else {
7            panic!("AVR USART baud rate is not representable");
8        };
9        unsafe {
10            // Take ownership of this USART configuration.
11            // Disables RX, TX and USART interrupts while it is reconfigured.
12            self.control_reg().write(0);
13
14            // U2X selects normal (÷16) or double-speed (÷8) asynchronous mode.
15            self.status_reg().write(if double_speed { Self::U2X } else { 0 });
16
17            // Updating UBRRnL activates the new divider, so write high first.
18            self.baud_high_reg().write((ubrr >> 8) as u8);
19            self.baud_low_reg().write(ubrr as u8);
20
21            self.set_frame_8n1();
22            self.control_reg().write(enable);
23        }
24    }
25}

28#[cfg(feature = "unsafe_mmio")]
29impl AvrUsart {
30    /// Selects asynchronous 8N1 framing: 8 data bits, no parity, and 1 stop bit.
31    ///
32    /// This does not change the baud rate or enable or disable the receiver or transmitter.
33    ///
34    /// # Safety
35    /// This must describe a USART on the active device and not be concurrently used or configured.
36    pub unsafe fn set_frame_8n1(self) {
37        let control = self.control_reg();
38        unsafe {
39            // UCSZn2 = 0; together with UCSZn1:0 = 0b11 this selects 8-bit data.
40            control.write(control.read() & !Self::UCSZ2);
41            // Asynchronous, no parity, one stop bit, UCSZn1:0 = 0b11.
42            self.frame_reg().write(Self::UCSZ1 | Self::UCSZ0);
43        }
44    }
45
46    /// Configures a transmit-only asynchronous 8N1 connection.
47    ///
48    /// Selects the closest representable baud rate for `clock_hz` and `baud`,
49    /// configures 8 data bits, no parity and 1 stop bit, and enables the transmitter.
50    ///
51    /// The receiver and USART interrupts are left disabled.
52    ///
53    /// # Panics
54    /// Panics if `clock_hz` or `baud` is zero,
55    /// or if the requested rate cannot be represented by the AVR baud generator.
56    ///
57    /// # Safety
58    /// This must describe a USART on the active device
59    /// and not be concurrently used or configured.
60    pub unsafe fn configure_tx_8n1(self, clock_hz: u32, baud: u32) {
61        unsafe { self.configure_8n1(clock_hz, baud, Self::TXEN) };
62    }
63    /// Configures a full-duplex asynchronous 8N1 connection.
64    ///
65    /// Selects the closest representable baud rate for `clock_hz` and `baud`,
66    /// configures 8 data bits, no parity and 1 stop bit, and enables both
67    /// the receiver and transmitter.
68    ///
69    /// USART interrupts are left disabled.
70    ///
71    /// # Panics
72    /// Panics if `clock_hz` or `baud` is zero,
73    /// or if the requested rate cannot be represented by the AVR baud generator.
74    ///
75    /// # Safety
76    /// This must describe a USART on the active device
77    /// and not be concurrently used or configured.
78    pub unsafe fn configure_rx_tx_8n1(self, clock_hz: u32, baud: u32) {
79        unsafe { self.configure_8n1(clock_hz, baud, Self::RXEN | Self::TXEN) };
80    }
81}

85#[allow(dead_code, reason = "safe helpers used by unsafe-gated code")]
86impl AvrUsart {
87    // UCSRnA
88    pub(super) const RXC: u8 = 1 << 7; // Receive complete.
89    pub(super) const UDRE: u8 = 1 << 5; // Data-register empty.
90    pub(super) const U2X: u8 = 1 << 1; // Double asynchronous transmission speed.
91
92    // UCSRnB
93    pub(super) const RXEN: u8 = 1 << 4; // Receiver enable.
94    pub(super) const TXEN: u8 = 1 << 3; // Transmitter enable.
95    pub(super) const UCSZ2: u8 = 1 << 2; // Character-size bit 2.
96
97    // UCSRnC
98    pub(super) const UCSZ1: u8 = 1 << 2; // Character-size bit 1.
99    pub(super) const UCSZ0: u8 = 1 << 1; // Character-size bit 0.
100
101    /// Selects the closest representable asynchronous baud setting.
102    ///
103    /// Returns `(UBRRn, double_speed)`.
104    fn baud_setting(clock_hz: u32, baud: u32) -> Option<(u16, bool)> {
105        // Returns: (UBRRn, absolute-error numerator, effective divisor).
106        fn candidate(clock_hz: u32, baud: u32, factor: u32) -> Option<(u16, u64, u64)> {
107            is! { clock_hz == 0 || baud == 0, return None }
108            let (clock, baud, factor) = (clock_hz as u64, baud as u64, factor as u64);
109
110            // baud = clock / (factor × (UBRR + 1))
111            // Round UBRR + 1 to the nearest integer.
112            let denominator = factor * baud;
113            let divider = (clock + denominator / 2) / denominator;
114
115            // UBRR is 12-bit: 0..=4095, therefore divider is 1..=4096.
116            is! { divider == 0 || divider > 4096, return None }
117            let effective_divisor = factor * divider;
118
119            // The actual baud error is:
120            //
121            // |clock / effective_divisor - baud|
122            // =
123            // |clock - baud × effective_divisor| / effective_divisor
124            let error = clock.abs_diff(baud * effective_divisor);
125            Some(((divider - 1) as u16, error, effective_divisor))
126        }
127        let normal = candidate(clock_hz, baud, 16);
128        let double = candidate(clock_hz, baud, 8);
129        match (normal, double) {
130            (None, None) => None,
131            (Some((ubrr, _, _)), None) => Some((ubrr, false)),
132            (None, Some((ubrr, _, _))) => Some((ubrr, true)),
133            (
134                Some((normal_ubrr, normal_error, normal_divisor)),
135                Some((double_ubrr, double_error, double_divisor)),
136            ) => {
137                // Compare the two rational errors without floating point.
138                // Ties prefer normal-speed mode.
139                if normal_error * double_divisor <= double_error * normal_divisor {
140                    Some((normal_ubrr, false))
141                } else {
142                    Some((double_ubrr, true))
143                }
144            }
145        }
146    }
147}
