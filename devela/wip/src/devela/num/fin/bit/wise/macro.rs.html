16/// A mnemonic vocabulary of bitwise operations, with selectable range backends.
17#[doc = crate::_doc_meta!{
18    location("num/fin", macro bitwise),
19}]
20/// This macro requires no capability feature or `Bitwise` implementation.
21/// It operates on unsigned integer primitives; all bit indices are zero-based.
22/// Operations return values without mutating the inputs and evaluate each argument once.
23///
24/// Range operations use an inline backend by default, or delegate to `Bitwise`
25/// methods when the `bitwise` feature is enabled. The implementation can also be
26/// selected per invocation, without changing the operation's semantics:
27/// - `bitwise![@inline mask u8; 2..=5]` forces primitive expressions.
28/// - `bitwise![@methods mask u8; 2..=5]` requires the `bitwise` feature.
29///
30/// The same prefixes work with `extract` and `insert`. Selection and validation
31/// take place within the macro; neither backend is required to validate ranges.
32///
33/// The extended `Bitwise` namespace and `BitOps` trait require the `bitwise` feature.
34///
35/// # Single bits
36/// - `test`, `set`, `clear`, `toggle` interpret the last argument as a bit *index*.
37///
38/// # Masks
39/// - `any`, `all` interpret the last argument as a bit *mask*.
40/// - An empty mask makes `any` false and `all` true.
41///
42/// # Hacker's bit tricks
43/// - `clear_lowest`: remove the lowest set bit (zero stays zero).
44/// - `isolate_lowest`: isolate the lowest set bit (zero stays zero).
45/// - `isolate_trailing`: isolate the lowest *unset* bit (all ones yields zero).
46///
47/// # Inclusive ranges
48/// - `mask T; START..=END`: construct a `T` mask covering the range.
49/// - `extract BITS; START..=END`: shift the selected bits down to bit zero.
50/// - `insert BITS, VALUE; START..=END`: replace the region, truncating `VALUE`.
51///
52/// For range forms, both bounds must be within the integer width and
53/// `START <= END`. Invalid ranges panic in all build modes. Full-width
54/// ranges are supported without an overflowing shift.
55///
56/// Single-bit forms follow Rust's ordinary shift behavior for invalid indices
57/// (debug overflow checks; do not pass an out-of-range index).
58///
59/// Range bounds must be a single token tree (a literal, identifier, or a
60/// parenthesized expression), matching the syntax of `set!` and `bitfield!`.
61///
62/// # Corresponding `Bitwise` methods
63///
64/// These are **semantic counterparts**. With the `bitwise` feature, the three range
65/// operations delegate to these methods by default, but `@inline` can override that.
66/// The links use the `u8` implementation; all six unsigned primitives are supported.
67///
68/// | `bitwise!` arm | `Bitwise` method |
69/// |---|---|
70/// | `test BITS; bit N` | [`is_set`](Bitwise::<u8>::is_set) |
71/// | `set BITS; bit N` | [`set`](Bitwise::<u8>::set) |
72/// | `clear BITS; bit N` | [`unset`](Bitwise::<u8>::unset) |
73/// | `toggle BITS; bit N` | [`flip`](Bitwise::<u8>::flip) |
74/// | `any BITS; mask M` | [`is_set_mask`](Bitwise::<u8>::is_set_mask) |
75/// | `mask T; START..=END` | [`mask_range`](Bitwise::<u8>::mask_range) |
76/// | `extract BITS; START..=END` | [`get_value_range`](Bitwise::<u8>::get_value_range) |
77/// | `insert BITS, VALUE; START..=END` | [`set_value_range`](Bitwise::<u8>::set_value_range) |
78///
79/// `all` checks an arbitrary mask and has no direct `Bitwise` equivalent;
80/// [`is_set_range`](Bitwise::<u8>::is_set_range) instead checks a contiguous range.
81/// The three hacker idioms likewise have no direct counterparts.
82/// Unlike the unchecked `Bitwise` range methods, `bitwise!` range arms
83/// panic for invalid ranges in **all** build modes.
84///
85/// # Examples
86/// ```
87/// use devela::bitwise;
88///
89/// let bits = 0b1010_0000_u8;
90/// assert!(bitwise![test bits; bit 7]);
91/// assert_eq!(bitwise![set bits; bit 0], 0b1010_0001);
92/// assert_eq!(bitwise![clear bits; bit 7], 0b0010_0000);
93/// assert_eq!(bitwise![toggle bits; bit 5], 0b1000_0000);
94///
95/// assert!(bitwise![any bits; mask 0b1000_0001]);
96/// assert!(!bitwise![all bits; mask 0b1000_0001]);
97///
98/// assert_eq!(bitwise![mask u8; 3..=6], 0b0111_1000);
99/// assert_eq!(bitwise![extract 0b1011_0110_u8; 1..=4], 0b1011);
100/// assert_eq!(bitwise![insert 0b1000_0001_u8, 0b101_u8; 1..=3], 0b1000_1011);
101/// ```
102#[macro_export]
103#[cfg_attr(cargo_primary_package, doc(hidden))]
104macro_rules! bitwise· {
105    (
106
107    /* Single-bit operations */
108
109     test $bits:expr; bit $index:expr) => {{
110        let (bits, index) = ($bits, $index);
111        (bits & (1 << index)) != 0
112    }};
113    (set $bits:expr; bit $index:expr) => {{
114        let (bits, index) = ($bits, $index);
115        bits | (1 << index)
116    }};
117    (clear $bits:expr; bit $index:expr) => {{
118        let (bits, index) = ($bits, $index);
119        bits & !(1 << index)
120    }};
121    (toggle $bits:expr; bit $index:expr) => {{
122        let (bits, index) = ($bits, $index);
123        bits ^ (1 << index)
124    }};
125    (
126
127    /* Mask predicates */
128
129     any $bits:expr; mask $mask:expr) => {{
130        let (bits, mask) = ($bits, $mask);
131        (bits & mask) != 0
132    }};
133    (all $bits:expr; mask $mask:expr) => {{
134        let (bits, mask) = ($bits, $mask);
135        (bits & mask) == mask
136    }};
137    (
138
139    /* Hacker's bit tricks */
140
141     clear_lowest $bits:expr) => {{
142        let bits = $bits;
143        bits & bits.wrapping_sub(1)
144    }};
145    (isolate_lowest $bits:expr) => {{
146        let bits = $bits;
147        bits & bits.wrapping_neg()
148    }};
149    (isolate_trailing $bits:expr) => {{
150        let bits = $bits;
151        !bits & bits.wrapping_add(1)
152    }};
153    (
154
155    /* Inclusive bit ranges: feature-selected default. */
156
157     mask $T:ty; $start:tt ..= $end:tt) => {
158        $crate::__bitwise!(% dispatch __bitwise_backend; mask $T; $start ..= $end)
159    };
160    (extract $bits:expr; $start:tt ..= $end:tt) => {
161        $crate::__bitwise!(% dispatch __bitwise_backend; extract $bits; $start ..= $end)
162    };
163    (insert $bits:expr, $value:expr; $start:tt ..= $end:tt) => {
164        $crate::__bitwise!(% dispatch __bitwise_backend; insert $bits, $value; $start ..= $end)
165    };
166    (
167
168    /* Range backend selection: available per invocation. */
169
170     @inline $($range:tt)+) => {
171            $crate::__bitwise!(% dispatch __bitwise_inline; $($range)+)
172    };
173    (@methods $($range:tt)+) => {
174        $crate::__bitwise!(% dispatch __bitwise_methods; $($range)+)
175    };
176}

178pub use bitwise· as bitwise;

181#[doc(hidden)]
182#[macro_export]
183macro_rules! __bitwise· {
184    /* Shared validation and u32 conversion, independent of the backend. */
185    (% dispatch $backend:ident; mask $T:ty; $start:tt ..= $end:tt) => {{
186        let (start, end) = (($start) as u128, ($end) as u128);
187        assert!(start <= end && end < <$T>::BITS as u128, "bitwise!: invalid bit range");
188        $crate::$backend!(mask $T; start as u32, end as u32)
189    }};
190    (% dispatch $backend:ident; extract $bits:expr; $start:tt ..= $end:tt) => {{
191        let (bits, start, end) = ($bits, ($start) as u128, ($end) as u128);
192        let bit_width = (::core::mem::size_of_val(&bits) * 8) as u128;
193        assert!(start <= end && end < bit_width, "bitwise!: invalid bit range");
194        $crate::$backend!(extract bits; start as u32, end as u32)
195    }};
196    (% dispatch $backend:ident; insert $bits:expr, $value:expr; $start:tt ..= $end:tt) => {{
197        let (bits, value, start, end) =
198            ($bits, $value, ($start) as u128, ($end) as u128);
199        let bit_width = (::core::mem::size_of_val(&bits) * 8) as u128;
200        assert!(start <= end && end < bit_width, "bitwise!: invalid bit range");
201        $crate::$backend!(insert bits, value; start as u32, end as u32)
202    }};
203}

214#[doc(hidden)]
215#[macro_export]
216macro_rules! __bitwise_inline· {
217    (mask $T:ty; $start:expr, $end:expr) => {{
218        let (start, end): (u32, u32) = ($start, $end);
219        let high = if end == <$T>::BITS - 1 {
220            !0 as $T
221        } else {
222            (1 as $T).wrapping_shl(end.wrapping_add(1)).wrapping_sub(1)
223        };
224        let low = if start == 0 { 0 as $T } else { (1 as $T).wrapping_shl(start).wrapping_sub(1) };
225        high & !low
226    }};
227    (extract $bits:expr; $start:expr, $end:expr) => {{
228        let (bits, start, end) = ($bits, $start, $end);
229        let bit_width = (::core::mem::size_of_val(&bits) * 8) as u32;
230        let width = end.wrapping_sub(start).wrapping_add(1);
231        let all = bits | !bits;
232        let mask = if width == bit_width { all } else { !all.wrapping_shl(width) };
233        bits.wrapping_shr(start) & mask
234    }};
235    (insert $bits:expr, $value:expr; $start:expr, $end:expr) => {{
236        let (bits, value, start, end) = ($bits, $value, $start, $end);
237        let bit_width = (::core::mem::size_of_val(&bits) * 8) as u32;
238        let width = end.wrapping_sub(start).wrapping_add(1);
239        let all = bits | !bits;
240        let low_mask = if width == bit_width { all } else { !all.wrapping_shl(width) };
241        let mask = low_mask.wrapping_shl(start);
242        (bits & !mask) | ((value & low_mask).wrapping_shl(start))
243    }};
244}

248#[doc(hidden)]
249#[cfg(feature = "bitwise")]
250#[macro_export]
251macro_rules! __bitwise_methods· {
252    (mask $T:ty; $start:expr, $end:expr) => {
253        $crate::Bitwise::<$T>::mask_range($start, $end).0
254    };
255    (extract $bits:expr; $start:expr, $end:expr) => {
256        $crate::Bitwise($bits).get_value_range($start, $end).0
257    };
258    (insert $bits:expr, $value:expr; $start:expr, $end:expr) => {
259        $crate::Bitwise($bits).set_value_range($value, $start, $end).0
260    };
261}

277    #[test]
278    fn basic_bits_and_masks() {
279        let bits = 0b1010_0000_u8;
280        assert!(bitwise![test bits; bit 7]);
281        assert!(!bitwise![test bits; bit 0]);
282        assert_eq!(bitwise![set bits; bit 0], 0b1010_0001);
283        assert_eq!(bitwise![clear bits; bit 7], 0b0010_0000);
284        assert_eq!(bitwise![toggle bits; bit 5], 0b1000_0000);
285        assert!(bitwise![any bits; mask 0b1000_0001]);
286        assert!(!bitwise![all bits; mask 0b1000_0001]);
287        assert!(!bitwise![any bits; mask 0]);
288        assert!(bitwise![all bits; mask 0]);
289    }
290    #[test]
291    fn hacker_idioms() {
292        let bits = 0b1011_0000_u8;
293        assert_eq!(bitwise![clear_lowest bits], 0b1010_0000);
294        assert_eq!(bitwise![isolate_lowest bits], 0b0001_0000);
295        assert_eq!(bitwise![isolate_trailing bits], 0b0000_0001);
296        assert_eq!(bitwise![clear_lowest 0_u8], 0);
297        assert_eq!(bitwise![isolate_lowest 0_u8], 0);
298        assert_eq!(bitwise![isolate_trailing u8::MAX], 0);
299    }
300    #[test]
301    #[cfg(feature = "bitwise")]
302    fn range_backends_are_equivalent() {
303        macro_rules! compare {
304            ($T:ty) => {{
305                let end = <$T>::BITS - 1;
306                let bits = !0 as $T;
307                let value = !0 as $T;
308                let pattern = 0b1010_0110 as $T;
309                let replacement = 0b1101 as $T;
310                assert_eq!(
311                    __bitwise_inline!(mask $T; 0_u32, end),
312                    __bitwise_methods!(mask $T; 0_u32, end),
313                );
314                assert_eq!(
315                    __bitwise_inline!(extract bits; 0_u32, end),
316                    __bitwise_methods!(extract bits; 0_u32, end),
317                );
318                assert_eq!(
319                    __bitwise_inline!(insert 0 as $T, value; 0_u32, end),
320                    __bitwise_methods!(insert 0 as $T, value; 0_u32, end),
321                );
322                assert_eq!(
323                    __bitwise_inline!(mask $T; 1_u32, 3_u32),
324                    __bitwise_methods!(mask $T; 1_u32, 3_u32),
325                );
326                assert_eq!(
327                    __bitwise_inline!(extract bits; 1_u32, 3_u32),
328                    __bitwise_methods!(extract bits; 1_u32, 3_u32),
329                );
330                assert_eq!(
331                    __bitwise_inline!(insert 0 as $T, value; 1_u32, 3_u32),
332                    __bitwise_methods!(insert 0 as $T, value; 1_u32, 3_u32),
333                );
334                assert_eq!(
335                    __bitwise_inline!(extract pattern; 1_u32, 3_u32),
336                    __bitwise_methods!(extract pattern; 1_u32, 3_u32),
337                );
338                assert_eq!(
339                    __bitwise_inline!(insert pattern, replacement; 1_u32, 3_u32),
340                    __bitwise_methods!(insert pattern, replacement; 1_u32, 3_u32),
341                );
342            }};
343        }
344        compare!(u8);
345        compare!(u16);
346        compare!(u32);
347        compare!(u64);
348        compare!(u128);
349        compare!(usize);
350    }

365    fn inclusive_ranges() {
366        assert_eq!(bitwise![mask u8; 3..=6], 0b0111_1000);
367        assert_eq!(bitwise![mask u8; 0..=7], u8::MAX);
368        assert_eq!(bitwise![mask u16; 0..=15], u16::MAX);
369        assert_eq!(bitwise![mask u32; 0..=31], u32::MAX);
370        assert_eq!(bitwise![mask u64; 0..=63], u64::MAX);
371        assert_eq!(bitwise![mask u128; 0..=127], u128::MAX);
372        assert_eq!(bitwise![mask usize; 0..=(usize::BITS - 1)], usize::MAX);
373        assert_eq!(bitwise![extract 0b1011_0110_u8; 1..=4], 0b1011);
374        assert_eq!(bitwise![extract u8::MAX; 0..=7], u8::MAX);

377        assert_eq!(bitwise![insert 0_u128, u128::MAX; 0..=127], u128::MAX);
