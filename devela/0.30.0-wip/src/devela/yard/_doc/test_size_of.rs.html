3//! Defines [`_doc_test_size_of!`].
4//
5
6#[doc = crate::_tags!(internal assert mem)]
7/// Emits compact rustdoc metadata for a type's checked stack size.
8#[doc = crate::_doc_meta!{
9    location("yard", macro _doc_test_size_of),
10}]
11/// The visible fragment shows the expected `size_of::<T>()` value.
12/// A hidden doctest calls [`test_size_of!`][crate::test_size_of]
13/// to verify the layout during doctests.
14///
15/// It can also explicitly record whether wrapping `T` in `Option` preserves
16/// its stack size. Omitting the niche suffix leaves that relation unspecified.
17///
18/// # Forms
19/// - `Type = N`: checks `devela::Type`.
20/// - `Type = N|B`: also documents and checks the expected bit count.
21/// - `Type = N; niche Option`: also checks `Option<Type>` has the same size.
22/// - `Type = N; niche !Option`: also checks `Option<Type>` has a different size.
23/// - `Label: Type = N`: shows `Type` but uses `Label` as the metadata label.
24/// - `abs Label: path::Type = N`: uses the type path exactly as written.
25///
26/// The niche suffixes check size preservation only:
27/// - `niche Option`: `size_of::<Option<T>>() == size_of::<T>()`.
28/// - `niche !Option`: `size_of::<Option<T>>() != size_of::<T>()`.
29///
30/// All forms accept outer attributes before the type entry. These attributes
31/// are shown in the visible metadata and re-emitted before the hidden doctest.
32///
33/// # Examples
34/// ```ignore
35/// #[doc = crate::_doc_test_size_of!(RasterFormat = 4)]
36/// #[doc = crate::_doc_test_size_of!(RasterFormat = 4|32)]
37///
38/// #[doc = crate::_doc_test_size_of!(u8 = 1|8; niche !Option)]
39/// #[doc = crate::_doc_test_size_of!(NonZeroU8 = 1|8; niche Option)]
40///
41/// #[doc = crate::_doc_test_size_of!(PcmRawBuf_Slice: PcmRawBuf<&[u8]> = 24)]
42/// #[doc = crate::_doc_test_size_of!(EventWindowCompact: EventWindow = 16|128; niche Option)]
43/// #[doc = crate::_doc_test_size_of!(abs Local: crate::Local = 8)]
44/// ```
45///
46/// Conditional size metadata:
47/**
48```ignore
49#[doc = crate::_doc_test_size_of!(
50    #[cfg(target_pointer_width = "64")]
51    Handle = 8|64; niche Option
52)]
53#[doc = crate::_doc_test_size_of!(
54    #[cfg(target_pointer_width = "32")]
55    Handle = 4|32; niche Option
56)]
57```
58**/
59#[cfg_attr(cargo_primary_package, doc(hidden))]
60#[cfg_attr(not(feature = "__docs_internal"), doc(hidden))]
61#[cfg_attr(nightly_doc, doc(cfg(feature = "__docs_internal")))]
62#[macro_export]
63macro_rules! _doc_test_size_of· {
64    // Unnamed public-root type:
65    //
66    // #[doc = _doc_test_size_of!(RasterFormat = 4)]
67    // #[doc = _doc_test_size_of!(EventWindow = 16|128; niche Option)]
68    // #[doc = _doc_test_size_of!(EventWindow = 16|128; niche !Option)]
69    ($(#[$meta:meta])* $ty:ty = $bytes:literal
70     $(| $bits:literal)? ; niche ! $wrap:ident $(,)?) => {
71        $crate::_doc_test_size_of!(@doc [$(#[$meta])*] [! $wrap]
72            stringify!($ty), concat!("devela::", stringify!($ty)),
73            stringify!($ty), stringify!($bytes) $(, stringify!($bits))?
74        )
75    };
76    ($(#[$meta:meta])* $ty:ty = $bytes:literal
77     $(| $bits:literal)? $(; niche $wrap:ident)? $(,)?) => {
78        $crate::_doc_test_size_of!(@doc [$(#[$meta])*] [$($wrap)?]
79            stringify!($ty), concat!("devela::", stringify!($ty)),
80            stringify!($ty), stringify!($bytes) $(, stringify!($bits))?
81        )
82    };
83    // Named public-root type:
84    //
85    // #[doc = _doc_test_size_of!(PcmRawBuf_Slice: PcmRawBuf<&[u8]> = 24)]
86    // #[doc = _doc_test_size_of!(PcmRawBuf_Slice: PcmRawBuf<&[u8]> = 24|192)]
87    // #[doc = _doc_test_size_of!(EventWindowCompact: EventWindow = 16|128; niche Option)]
88    // #[doc = _doc_test_size_of!(EventWindowCompact: EventWindow = 16|128; !niche Option)]
89    ($(#[$meta:meta])* $name:ident : $ty:ty = $bytes:literal
90     $(| $bits:literal)? ; niche ! $wrap:ident $(,)?) => {
91        $crate::_doc_test_size_of!(@doc [$(#[$meta])*] [! $wrap]
92            stringify!($name), concat!("devela::", stringify!($ty)),
93            stringify!($ty), stringify!($bytes) $(, stringify!($bits))?
94        )
95    };
96    ($(#[$meta:meta])* $name:ident : $ty:ty = $bytes:literal
97     $(| $bits:literal)? $(; niche $wrap:ident)? $(,)?) => {
98        $crate::_doc_test_size_of!(@doc [$(#[$meta])*] [$($wrap)?]
99            stringify!($name), concat!("devela::", stringify!($ty)),
100            stringify!($ty), stringify!($bytes) $(, stringify!($bits))?
101        )
102    };
103    // Explicit path escape hatch:
104    //
105    // #[doc = _doc_test_size_of!(abs PcmRawBuf_Slice: crate::PcmRawBuf<&[u8]> = 24)]
106    // #[doc = _doc_test_size_of!(abs PcmRawBuf_Slice: ::devela::PcmRawBuf<&[u8]> = 24)]
107    // #[doc = _doc_test_size_of!(abs EventWindow: crate::EventWindow = 16|128; niche Option)]
108    // #[doc = _doc_test_size_of!(abs EventWindow: crate::EventWindow = 16|128; !niche Option)]
109    ($(#[$meta:meta])* abs $name:ident : $ty:ty = $bytes:literal
110     $(| $bits:literal)? ; niche ! $wrap:ident $(,)?) => {
111        $crate::_doc_test_size_of!(@doc [$(#[$meta])*] [! $wrap]
112            stringify!($name), stringify!($ty), stringify!($ty),
113            stringify!($bytes) $(, stringify!($bits))?
114        )
115    };
116    ($(#[$meta:meta])* abs $name:ident : $ty:ty = $bytes:literal
117     $(| $bits:literal)? $(; niche $wrap:ident)? $(,)?) => {
118        $crate::_doc_test_size_of!(@doc [$(#[$meta])*] [$($wrap)?]
119            stringify!($name), stringify!($ty), stringify!($ty),
120            stringify!($bytes) $(, stringify!($bits))?
121        )
122    };
123    // Core doc emitter.
124    (@doc [$(#[$meta:meta])*] [$($niche:tt)*]
125        $label:expr, $test_ty:expr, $shown_ty:expr, $bytes:expr $(, $bits:expr)?
126    ) => {
127        concat!(
128            "\n\n",
129            "<sup class='_doc_test_size_of' title='stack size, checked by hidden doctest'>",
130            "📦 `size_of::<", $shown_ty, ">() == ", $bytes, "` bytes",
131            $(" / ", $bits, " bits",)?
132            $crate::_doc_test_size_of!(@meta $(#[$meta])*),
133            "</sup>",
134            $crate::_doc_test_size_of!(@niche_doc [$($niche)*]),
135            "\n\n",
136            "<div hidden class='devela-hide-next'></div>\n\n",
137            "```rust\n",
138            $( "# ", stringify!(#[$meta]), "\n", )*
139            "# devela::test_size_of!(assert ", $test_ty, " = ", $bytes,
140            $("|", $bits,)?
141            $crate::_doc_test_size_of!(@niche_test [$($niche)*]),
142            ");\n",
143            "```\n",
144        )
145    };
146    // Niche-size documentation badge.
147    (@niche_doc []) => { "" };
148    (@niche_doc [Option]) => {
149        concat!("\n", "<sup class='_doc_niche' ",
150            "title='checked by hidden doctest: Option<T> has the same stack size as T'>",
151            "⚗️`Option<T>` &#x1F7F0; `T`", "</sup>" // = emoji
152        )
153    };
154    (@niche_doc [! Option]) => {
155        concat!("\n", "<sup class='_doc_niche' ",
156            "title='checked by hidden doctest: Option<T> does not have the same stack size as T'>",
157            "⚗️`Option<T>` &#x2757;&#x1F7F0; `T`", "</sup>" // != emoji
158        )
159    };
160    // Niche-size doctest suffix.
161    (@niche_test []) => { "" };
162    (@niche_test [Option]) => { "; niche Option" };
163    (@niche_test [! Option]) => { "; niche !Option" };
164    // Attribute printer helpers.
165    (@meta) => { "" };
166    (@meta $(#[$meta:meta])+) => {
167        concat!("  &#35;&#xfe0f;&#x20e3; ", $(stringify!(#[$meta]), " ",)+) // [#] emoji
168    };
169}
170#[doc(inline)]
171pub use _doc_test_size_of· as _doc_test_size_of;
