95    /// All possible integer operation errors.
96    pub enum IntError {
97        +tag: _tags!(no),
98        DOC_NOT_IMPLEMENTED: +const NotImplemented => NotImplemented,
99        +tag: _tags!(no),
100        DOC_NOT_SUPPORTED: +const NotSupported => NotSupported,
101
102        // used by ops: core, combinatoric, modulo, prime, root:
103        DOC_OVERFLOW: +const Overflow(s|0: Option<Sign>) => Overflow(*s),
104        // used by ops: modulo:
105        DOC_NO_INVERSE: +const NoInverse => NoInverse,
106        // used by ops: combinatoric, root:
107        DOC_NON_NEGATIVE_REQUIRED: +const NonNegativeRequired => NonNegativeRequired,
108        // used by ops: division, modulo, root:
109        DOC_NON_ZERO_REQUIRED: +const NonZeroRequired => NonZeroRequired,
110        // used by ops: combinatoric, factor:
111        DOC_MISMATCHED_SIZES: +const MismatchedSizes => MismatchedSizes,
112
113        // DOC_POSITIVE_REQUIRED: +const PositiveRequired => PositiveRequired,
114        // DOC_INCOMPATIBLE_BOUNDS: +const IncompatibleBounds => IncompatibleBounds,
115    }
