16#[doc = crate::_tags!(code value)]
17/// Hooks operations into a value without breaking its flow.
18#[doc = crate::_doc_meta!{
19    location("code/ops", trait Hook),
20}]
21/// `Hook` provides a way to intercept a value,
22/// allowing mutation (`hook`) or observation (`tap`) while preserving identity.
23///
24/// Use this trait when working with APIs that mutate in place or return `()`,
25/// but where fluent sequencing is still desired.
26///
27/// This is the ergonomic, runtime counterpart of the [`hook!`] macro.
28///
29/// # Examples
30/// ```
31/// # extern crate devela as devela;
32/// use devela::Hook;
33///
34/// # #[cfg(feature = "alloc")] {
35/// let v = vec![3, 2, 1, 5]
36///     .hook(|v| v.sort())
37///     .tap(|v| assert_eq![v, &[1, 2, 3, 5]])
38///     .hook(|v| v.push(7));
39/// assert_eq![v, vec![1, 2, 3, 5, 7]];
40/// # }
41/// ```
42/// # Design
43/// `Hook` preserves value identity.
44/// For transformations that replace the value, see [`Morph`].
45#[rustfmt::skip]
46#[doc = crate::_doc_vendor!("apply")]
47pub trait Hook: Sized {
48    /// Hooks a mutation step into the value and returns it.
49    ///
50    /// The provided function receives a mutable reference to the value,
51    /// allowing in-place modification while preserving flow.
52    ///
53    /// This is the method form of the [`hook!`] macro.
54    #[must_use]
55    fn hook<F>(mut self, f: F) -> Self where F: FnOnce(&mut Self) { f(&mut self); self }
56
57    /// Taps into the value for observation and returns it unchanged.
58    ///
59    /// The provided function receives a shared reference to the value.
60    /// This is useful for inspection, logging, or assertions.
61    #[must_use]
62    fn tap<F>(self, f: F) -> Self where F: FnOnce(&Self) { f(&self); self }
63}
64impl<T: Sized> Hook for T {}

66#[doc = crate::_tags!(code value)]
67/// Hooks one or more mutation steps into a value and returns it.
68#[doc = crate::_doc_meta!{
69    location("code/ops", macro hook),
70}]
71/// `hook!` binds a value to a local mutable name, applies a sequence of
72/// side-effecting operations to it, and then returns the value.
73///
74/// This macro preserves the value's identity: steps may mutate it, but do not replace it.
75///
76/// This is the trait-free, const-capable mechanical form of [`Hook`].
77///
78/// Steps can be written as:
79/// - a comma- or semicolon-separated list of expressions, or
80/// - a sequence of adjacent `{ ... }` blocks.
81///
82/// Block syntax is useful when a step requires multiple statements
83/// or local control flow.
84///
85/// All steps are applied in order to the same value.
86///
87/// # Examples
88/// ```
89/// use devela::{const_assert, is, hook};
90///
91/// /* expression syntax */
92/// const ARRAY: [u8; 4] = [3, 1, 4, 1];
93/// const DATA: [u8; 4] = hook!(ARRAY; |v| v[0] = 9; v[3] = 7);
94/// const_assert![eq_buf &DATA, &[9, 1, 4, 7]];
95///
96/// /* block syntax */
97/// const INPUT: [u8; 4] = [3, 0, 9, 1];
98/// const OUTPUT: [u8; 4] = hook!(INPUT, |v| { v[0] = 1; is!(v[2] >7 , v[2] -= 7) } { v[3] += 5 });
99/// const_assert![eq_buf &OUTPUT, &[1, 0, 2, 6]];
100/// ```
101/// For transformations that replace the value, see [`morph!`].
102#[macro_export]
103#[cfg_attr(cargo_primary_package, doc(hidden))]
104macro_rules! hook· {
105    // (comma separated)
106    // expr-sequence form
107    ($t:expr, |$v:ident| $($step:expr),+ $(,)?) => {{
108        let mut $v = $t; $( $step; )+ $v
109    }};
110    // block-sequence form
111    ($t:expr, |$v:ident| $( { $($body:tt)* } $(,)? )+) => {{
112        let mut $v = $t; $( { $($body)* }; )+ $v
113    }};
114
115    // (semicolon separated)
116    // expr-sequence form
117    ($t:expr; |$v:ident| $($step:expr);+ $(;)?) => {{
118        let mut $v = $t; $( $step; )+ $v
119    }};
120    // block-sequence form
121    ($t:expr; |$v:ident| $( { $($body:tt)* } $(;)? )+) => {{
122        let mut $v = $t; $( { $($body)* }; )+ $v
123    }};
124}
125#[doc(inline)]
126pub use hook· as hook;

128#[doc = crate::_tags!(code value)]
129/// Morphs a value by threading it through a function.
130#[doc = crate::_doc_meta!{
131    location("code/ops", trait Morph),
132}]
133/// `Morph` provides a fluent way to pass a value (by value, shared reference,
134/// or exclusive reference) into a transformation and return the result.
135///
136/// Unlike [`Hook`], `Morph` does not preserve value identity:
137/// each operation yields a new result.
138///
139/// This is the ergonomic, runtime counterpart of the [`morph!`] macro.
140///
141/// # Examples
142/// ```
143/// # extern crate devela as devela;
144/// use devela::Morph;
145///
146/// let x = 3.morph(|v| v * 2).morph(|v| v + 1);
147/// assert_eq![x, 7];
148/// ```
149///
150/// ```compile_fail
151/// # extern crate devela as devela;
152/// use devela::Morph;
153///
154/// // We can mutate it, but we do not receive the value back.
155/// let v: Vec<i32> = vec![3, 2, 1, 5].morph_mut(|it| it.sort());
156/// ```
157#[rustfmt::skip]
158#[doc = crate::_doc_vendor!("apply")]
159pub trait Morph<R> {
160    /// Morphs the value into a new one and returns it.
161    ///
162    /// The provided function consumes the value and produces a replacement.
163    #[must_use]
164    fn morph<F>(self, f: F) -> R where F: FnOnce(Self) -> R, Self: Sized { f(self) }
165
166    /// Morphs the value by shared reference and returns the result.
167    ///
168    /// The provided function receives `&Self` and may compute any result from it.
169    #[must_use]
170    fn morph_ref<F>(&self, f: F) -> R where F: FnOnce(&Self) -> R { f(self) }
171
172    /// Morphs the value by exclusive reference and returns the result.
173    ///
174    /// The provided function receives `&mut Self` and may mutate it before producing the result.
175    #[must_use]
176    fn morph_mut<F>(&mut self, f: F) -> R where F: FnOnce(&mut Self) -> R { f(self) }
177}
178impl<T: ?Sized, R> Morph<R> for T {}

180#[doc = crate::_tags!(code value)]
181/// Morphs a value through one or more transformation steps and returns the result.
182#[doc = crate::_doc_meta!{
183    location("code/ops", macro morph),
184}]
185/// `morph!` expresses a left-to-right transformation pipeline
186/// without explicit intermediate bindings in user code.
187///
188/// Conceptually, `morph!(x, a, b, c)` corresponds to `c(b(a(x)))`.
189///
190/// A single binding introduces the pipeline variable,
191/// which is reused implicitly by subsequent steps.
192/// Each step may change the value's type.
193///
194/// This macro is const-capable and does not create closures.
195/// It is the trait-free mechanical form of [`Morph`].
196///
197/// Steps can be written as:
198/// - a comma-separated list of expressions, or
199/// - a semicolon-separated list of expressions.
200///
201/// # Examples
202/// ```
203/// use devela::{const_assert, is, morph};
204///
205/// const S: &str = morph!(3u8, |v| v as usize, v * 2, v + 1, if v == 7 { "7" } else { "not7" });
206/// const_assert![eq_str S, "7"];
207/// ```
208/// For pipelines that preserve value identity and mutate in place, see [`hook!`].
209#[macro_export]
210#[cfg_attr(cargo_primary_package, doc(hidden))]
211macro_rules! morph· {
212    // (comma separated)
213    // base case: single-step
214    ($t:expr, |$v:ident| $e:expr $(,)?) => {{ let $v = $t; $e }};
215    // recursive case: multiple steps
216    ($t:expr, |$v:ident| $e:expr, $($rest:expr),+ $(,)?) => {{
217        morph!( { let $v = $t; $e }, |$v| $($rest),+ )
218    }};
219
220    // (semicolon separated)
221    // base case: single-step
222    ($t:expr; |$v:ident| $e:expr $(;)?) => {{ let $v = $t; $e }};
223    // recursive case: multiple steps
224    ($t:expr; |$v:ident| $e:expr; $($rest:expr);+ $(;)?) => {{
225        morph!( { let $v = $t; $e }; |$v| $($rest);+ )
226    }};
227}
228#[doc(inline)]
229pub use morph· as morph;
