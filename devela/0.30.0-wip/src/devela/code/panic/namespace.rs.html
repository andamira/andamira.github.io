3//! Defines the [`Panic`] namespace.
4//
5// - https://doc.rust-lang.org/stable/std/panic/index.html
6
7#[cfg(feature = "std")]
8use crate::{Any, Box, PanicHookInfo, PanicUnwindSafe, ThreadResult};
9use ::std::panic::{catch_unwind, panic_any, resume_unwind, set_hook, take_hook};
10
11#[doc = crate::_tags!(namespace)]
12/// Panic-related operations.
13#[doc = crate::_doc_meta!{
14    location("code/panic", struct Panic),
15}]
16#[cfg_attr(nightly_doc, doc(cfg(feature = "std")))]
17#[derive(Debug)]
18pub struct Panic;
19
20#[cfg(feature = "std")]
21#[cfg_attr(nightly_doc, doc(cfg(feature = "std")))]
22impl Panic {
23    /// Invokes a closure, capturing the cause of an unwinding panic if one occurs.
24    ///
25    /// See `std::panic::`[`catch_unwind`].
26    pub fn catch<F: FnOnce() -> R + PanicUnwindSafe, R>(f: F) -> ThreadResult<R> {
27        catch_unwind(f)
28    }
29
30    /// Panic the current thread with the given message as the panic payload.
31    ///
32    /// See `std::panic::`[`panic_any`].
33    pub fn any<M: 'static + Any + Send>(msg: M) -> ! {
34        panic_any(msg)
35    }
36
37    /// Triggers a panic without invoking the panic hook.
38    ///
39    /// See `std::panic::`[`resume_unwind`].
40    pub fn resume(payload: Box<dyn Any + Send>) -> ! {
41        resume_unwind(payload)
42    }
43
44    /// Registers a custom panic hook, replacing the previously registered hook.
45    pub fn set_hook(hook: Box<dyn Fn(&PanicHookInfo<'_>) + Sync + Send + 'static>) {
46        set_hook(hook);
47    }
48
49    /// Unregisters the current panic hook, returns it and registers.
50    pub fn take_hook() -> Box<dyn Fn(&PanicHookInfo<'_>) + Sync + Send + 'static> {
51        take_hook()
52    }
53}
