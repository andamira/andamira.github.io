23/// # Build environment
24impl Build {
25    /// Returns the current crate's name.
26    pub fn crate_name() -> String {
27        env::var("CARGO_CRATE_NAME").expect("CARGO_CRATE_NAME not set")
28    }
29
30    /// Retuns the path of `OUT_DIR`.
31    pub fn out_dir() -> PathBuf {
32        PathBuf::from(env::var("OUT_DIR").expect("OUT_DIR not set"))
33    }
34    /// Retuns the path of `CARGO_MANIFEST_DIR`.
35    pub fn manifest_dir() -> PathBuf {
36        PathBuf::from(env::var("CARGO_MANIFEST_DIR").expect("CARGO_MANIFEST_DIR not set"))
37    }
38    /// Retuns the path of `CARGO_MANIFEST_PATH`.
39    pub fn manifest_path() -> PathBuf {
40        PathBuf::from(env::var("CARGO_MANIFEST_PATH").expect("CARGO_MANIFEST_PATH not set"))
41    }
42    /// Marks an environment variable as affecting build-script reruns.
43    // https://doc.rust-lang.org/cargo/reference/build-scripts.html#rerun-if-env-changed
44    pub fn rerun_if_env_changed(var: &str) {
45        println!("cargo:rerun-if-env-changed={var}");
46    }
47}

49/// # Print
50#[rustfmt::skip]
51impl Build {
52    /// Prints a message to *stdout* from the build script.
53    pub fn println(msg: impl Display) { println!("cargo:warning={}", msg); }
54
55    /// Prints a heading message to *stdout*, underlined.
56    pub fn println_heading(msg: &str) {
57        Self::println("");
58        Self::println(msg);
59        Self::println("-".repeat(msg.len()));
60    }
61    /// Prints the value of an environment variable.
62    pub fn println_var(var: &str) {
63        if let Ok(v) = env::var(var) {
64            Self::println(fmt::from_fn(|f| write!(f, "· {var}: {v}")));
65        } else {
66            Self::println(fmt::from_fn(|f| write!(f, "x {var}:")));
67        }
68    }
69    /// Prints the value of an encoded environment variable,
70    /// replacing unit separator characters `'0x1f'` with spaces.
71    ///
72    /// It accepts a new name to show for the decoded variable.
73    pub fn println_var_encoded(var: &str, new_var_name: &str) {
74        if let Ok(ev) = env::var(var) {
75            let v = ev.replace('\x1f', " ");
76            Self::println(format!["· {new_var_name}(*): {v}"]);
77        } else {
78            Self::println(format!["x {new_var_name}:"]);
79        }
80    }
81    /// Prints the build script `start` or end message to *stdout*.
82    pub fn println_start_end(name: &str, start: bool) {
83        let msg = if start { format!("~ Start of {name} ~") }
84            else { Self::println(""); format!("~ End of {name} ~") };
85        let line = "~".repeat(msg.len());
86        Self::println(&line); Self::println(&msg); Self::println(&line);
87    }
88}

90/// # Emit `cfg` flags
91#[rustfmt::skip]
92impl Build {
93    /// Emits a custom `cfg` flag for the current crate.
94    pub fn emit_flag(flag: &str) { println!("cargo:rustc-cfg={flag}"); }
95    /// Registers a custom `cfg` flag with rustc's checked-cfg lint.
96    pub fn emit_check_cfg(flag: &str) { println!("cargo:rustc-check-cfg=cfg({flag})"); }
97    /// Emits a custom `cfg` flag and registers it for checked cfg validation.
98    pub fn emit_checked_flag(flag: &str) { Self::emit_check_cfg(flag); Self::emit_flag(flag); }
99    /// Emits a custom `cfg` flag if `condition` is true. Returns whether the flag was emitted.
100    pub fn emit_flag_if(flag: &str, condition: bool) -> bool {
101        Self::emit_check_cfg(flag);
102        if condition { Self::emit_flag(flag); }
103        condition
104    }
105    /// Emits a compile-time environment variable for the current crate.
106    pub fn emit_env(key: &str, value: impl Display) { println!("cargo:rustc-env={key}={value}"); }
107    /// Emits a compile-time environment marker with an empty value.
108    pub fn emit_env_marker(key: &str) { println!("cargo:rustc-env={key}="); }
109}

111/// # Native libraries
112#[rustfmt::skip]
113impl Build {
114    /// Returns whether a native library is discoverable.
115    ///
116    /// Uses `pkg-config`/`pkgconf` first, then falls back to direct file search
117    /// in common platform library paths.
118    #[must_use]
119    pub fn has_lib(lib: &str) -> bool { Self::lib_found_source(lib).is_some() }
120
121    /// Emits `flag` if a native library is discoverable.
122    ///
123    /// The flag is emitted as-is. It may be a normal cfg flag or a devela
124    /// reflective flag such as `ffi_xcb_shm··`.
125    ///
126    /// Returns whether the flag was emitted.
127    ///
128    /// # Features
129    /// If the feature `__disable_native_libs` is enabled then this always returns `false`.
130    #[must_use]
131    pub fn emit_flag_if_lib(flag: &str, lib: &str) -> bool {
132        Self::emit_check_cfg(flag);
133        Self::rerun_if_lib_env_changed();
134        let found = Self::lib_found_source(lib);
135        if found.is_some() { Self::emit_flag(flag); }
136        #[cfg(feature = "__dbg")]
137        match &found { Some(source) =>
138            Self::println(format!("· lib {lib}: found via {source}; emitted {flag}")),
139            None => Self::println(format!("x lib {lib}: not found; skipped {flag}")),
140        }
141        found.is_some()
142    }
143
144    /* private helpers*/
145
146    fn rerun_if_lib_env_changed() {
147        for var in [ "PKG_CONFIG", "PKG_CONFIG_PATH", "PKG_CONFIG_LIBDIR",
148            "PKG_CONFIG_SYSROOT_DIR", "LIBRARY_PATH", "LD_LIBRARY_PATH",
149            "DYLD_LIBRARY_PATH", "LIB", "PATH", "CC", ] {
150            Self::rerun_if_env_changed(var);
151        }
152    }
153    // NOTE: If the feature `__disable_native_libs` is enabled then this always returns `false`.
154    fn lib_found_source(lib: &str) -> Option<String> {
155        if cfg!(feature = "__disable_native_libs") { return None; }
156        if Self::lib_found_pkg_config(lib) { return Some("pkg-config".to_string()); }
157        Self::find_lib_file(lib).map(|path| path.display().to_string())
158    }
159    fn lib_found_pkg_config(lib: &str) -> bool {
160        if let Some(cmd) = env::var_os("PKG_CONFIG") { return Self::pkg_config_exists(cmd, lib); }
161        Self::pkg_config_exists("pkg-config", lib) || Self::pkg_config_exists("pkgconf", lib)
162    }
163    fn pkg_config_exists(cmd: impl AsRef<std::ffi::OsStr>, lib: &str) -> bool {
164        Command::new(cmd).arg("--exists").arg(lib)
165            .stdin(Stdio::null()).stdout(Stdio::null()).stderr(Stdio::null())
166            .status().is_ok_and(|status| status.success())
167    }
168    fn find_lib_file(lib: &str) -> Option<PathBuf> {
169        let filenames = Self::lib_filenames(lib);
170        Self::lib_search_dirs().into_iter()
171            .find_map(|dir| filenames.iter().map(|file| dir.join(file)).find(|path| path.is_file()))
172    }
173    fn lib_filenames(lib: &str) -> Vec<String> {
174        let lib = lib.strip_prefix("lib").unwrap_or(lib);
175        let target_os = env::var("CARGO_CFG_TARGET_OS").unwrap_or_default();
176        match target_os.as_str() {
177            "windows" => vec![format!("{lib}.lib"), format!("{lib}.dll"), format!("lib{lib}.a")],
178            "macos" | "ios" => vec![format!("lib{lib}.dylib"), format!("lib{lib}.a")],
179            _ => vec![format!("lib{lib}.so"), format!("lib{lib}.a")],
180        }
181    }
182    fn lib_search_dirs() -> Vec<PathBuf> {
183        let mut dirs = Vec::new();
184        for var in ["LIBRARY_PATH", "LD_LIBRARY_PATH", "DYLD_LIBRARY_PATH", "LIB"] {
185            if let Some(value) = env::var_os(var) { dirs.extend(env::split_paths(&value)); }
186        }
187        Self::push_cc_library_dirs(&mut dirs);
188        Self::push_common_library_dirs(&mut dirs);
189        dirs.sort(); dirs.dedup();
190        dirs
191    }
192    fn push_cc_library_dirs(dirs: &mut Vec<PathBuf>) {
193        let cc = env::var_os("CC").unwrap_or_else(|| "cc".into());
194        let Ok(output) = Command::new(cc).arg("-print-search-dirs").output() else { return; };
195        if !output.status.success() { return; }
196        let stdout = String::from_utf8_lossy(&output.stdout);
197        for line in stdout.lines() {
198            if let Some(paths) = line.strip_prefix("libraries: =") {
199                dirs.extend(env::split_paths(paths));
200            }
201        }
202    }
203    fn push_common_library_dirs(dirs: &mut Vec<PathBuf>) {
204        let target_os = env::var("CARGO_CFG_TARGET_OS").unwrap_or_default();
205        match target_os.as_str() {
206            "linux" | "freebsd" | "netbsd" | "openbsd" => {
207                if let Some(multiarch) = Self::linux_multiarch_dir() {
208                    dirs.push(format!("/usr/lib/{multiarch}").into());
209                    dirs.push(format!("/lib/{multiarch}").into());
210                }
211                dirs.extend([
212                    "/usr/local/lib".into(), "/usr/lib".into(), "/lib".into(),
213                    "/usr/lib64".into(), "/lib64".into(), ]);
214            }
215            "macos" | "ios" => {
216                dirs.extend([
217                    "/usr/local/lib".into(), "/opt/homebrew/lib".into(), "/usr/lib".into(), ]);
218            }
219            "windows" => {
220                if let Some(value) = env::var_os("PATH") { dirs.extend(env::split_paths(&value)); }
221            }
222            _ => {}
223        }
224    }
225    fn linux_multiarch_dir() -> Option<&'static str> {
226        let arch = env::var("CARGO_CFG_TARGET_ARCH").ok()?;
227        match arch.as_str() {
228            "x86_64" => Some("x86_64-linux-gnu"),
229            "x86" => Some("i386-linux-gnu"),
230            "aarch64" => Some("aarch64-linux-gnu"),
231            "arm" => Some("arm-linux-gnueabihf"),
232            "riscv64" => Some("riscv64-linux-gnu"),
233            "powerpc64" => Some("powerpc64-linux-gnu"),
234            "s390x" => Some("s390x-linux-gnu"),
235            _ => None,
236        }
237    }
238}

257    }
