6crate::enumset! {
7    #[doc = crate::_tags!(runtime event)]
8    /// External control notice directed at an application.
9    #[doc = crate::_doc_meta!{
10        location("run/app", enum AppControl),
11        test_size_of(AppControl = 1|8; niche Option),
12
13    }]
14    #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
15    pub enum AppControl(
16        #[doc = crate::_tags!(runtime signal set)]
17        /// A set of application control notices.
18        #[doc = crate::_doc_meta!{
19            location("run/app", struct AppControlSet),
20            test_size_of(AppControlSet = 2|16; niche !Option),
21        }]
22        pub AppControlSet: u16
23    ) {
24        /// User requested interruption, usually `Ctrl+C`.
25        Interrupt,
26        /// User requested a stronger quit, usually `Ctrl+\` on Unix terminals.
27        Quit,
28        /// Polite termination request from the environment.
29        Terminate,
30        /// Session, terminal, or controlling process disappeared.
31        Hangup,
32        /// Application/session was asked to suspend.
33        Suspend,
34        /// Application/session resumed after suspension.
35        Resume,
36        /// Application/session is being closed.
37        Close,
38        /// User session is logging out.
39        Logout,
40        /// System is shutting down.
41        Shutdown,
42    }
43}

45#[cfg(feature = "_linux_abi")]
46mod impl_linux {
47    use crate::{AppControl, AppControlSet, LinuxSignal, LinuxSignalSet, whilst};
48
49    impl AppControl {
50        /// Returns the Linux signal normally used to deliver this control notice.
51        #[must_use]
52        pub const fn to_linux_signal(self) -> Option<LinuxSignal> {
53            match self {
54                Self::Interrupt => Some(LinuxSignal::SIGINT),
55                Self::Quit => Some(LinuxSignal::SIGQUIT),
56                Self::Terminate => Some(LinuxSignal::SIGTERM),
57                Self::Hangup => Some(LinuxSignal::SIGHUP),
58                Self::Suspend => Some(LinuxSignal::SIGTSTP),
59                Self::Resume => Some(LinuxSignal::SIGCONT),
60                /* unmapped on linux */
61                // Self::Close => None,
62                // Self::Logout => None,
63                // Self::Shutdown => None,
64                _ => None,
65            }
66        }
67    }
68    impl AppControlSet {
69        /// Returns the Linux signals needed to receive these app controls.
70        #[must_use]
71        pub const fn to_linux_signals(self) -> LinuxSignalSet {
72            let mut out = LinuxSignalSet::new();
73            whilst! { i in 0..AppControl::VARIANTS; {
74                let control = AppControl::ALL[i];
75                if control.is_in(self) {
76                    if let Some(signal) = control.to_linux_signal() {
77                        out = out.with(signal.to_set());
78                    }
79                }
80            }}
81            out
82        }
83    }
84    impl LinuxSignal {
85        /// Returns the portable application control notice represented by this signal.
86        #[must_use]
87        pub const fn to_app_control(self) -> Option<AppControl> {
88            match self {
89                Self::SIGINT => Some(AppControl::Interrupt),
90                Self::SIGQUIT => Some(AppControl::Quit),
91                Self::SIGTERM => Some(AppControl::Terminate),
92                Self::SIGHUP => Some(AppControl::Hangup),
93                Self::SIGTSTP => Some(AppControl::Suspend),
94                Self::SIGCONT => Some(AppControl::Resume),
95                /* mapped to EventWindow */
96                // Self::SIGWINCH => None,
97                _ => None,
98            }
99        }
100    }
