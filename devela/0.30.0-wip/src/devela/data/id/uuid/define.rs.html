13/// A standardized portable 128-bit identifier.
14#[doc = crate::_doc_meta!{
15    location("data/id/uuid", struct Uuid),
16    test_size_of(Uuid = 16|128; niche !Option),
17}]
18#[must_use]
19#[repr(transparent)]
20#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
21pub struct Uuid([u8; 16]);
22
23word! { impl Uuid([u8; 16]); }

25impl Uuid {
26    /* constants */
27
28    /// The nil UUID, with all bits set to zero.
29    pub const NIL: Self = Self([0; 16]);
30
31    /// The maximum UUID, with all bits set to one.
32    pub const MAX: Self = Self([0xFF; 16]);
33
34    /// The length of the standard textual UUID representation.
35    pub const STR_LEN: usize = 36;
36
37    /// The maximum Unix millisecond timestamp representable by UUID version 7.
38    pub const V7_UNIX_TS_MS_MAX: u64 = 0xFFFF_FFFF_FFFF;
39
40    /* construction */
41
42    /// Constructs an IETF UUID by setting its version and variant fields.
43    pub(super) const fn from_ietf_bytes(mut bytes: [u8; 16], version: UuidVersion) -> Self {
44        bytes[6] = (bytes[6] & 0x0F) | (version.number() << 4);
45        bytes[8] = (bytes[8] & 0x3F) | 0x80;
46        Self::from_bytes(bytes)
47    }
48
49    /// Creates a version 4 UUID from 128 bits of random or pseudo-random input.
50    ///
51    /// The version and variant bits replace six bits of the input,
52    /// leaving 122 random bits in the resulting UUID.
53    pub const fn from_random_v4(random: [u8; 16]) -> Self {
54        Self::from_ietf_bytes(random, UuidVersion::V4)
55    }
56    /// Generates a deterministic version 4 UUID using Pcg32.
57    ///
58    /// This is suitable when reproducibility is desired;
59    /// it does not provide cryptographic unpredictability.
60    pub const fn from_pcg32_v4(rng: &mut Pcg32) -> Self {
61        let mut random = [0u8; 16];
62        rng.fill_bytes(&mut random);
63        Self::from_random_v4(random)
64    }
65
66    /// Creates a version 7 UUID from a Unix millisecond timestamp
67    /// and 80 bits of random input.
68    ///
69    /// The timestamp occupies the standardized 48-bit `unix_ts_ms` field.
70    /// The version and variant fields replace six bits of `random`,
71    /// leaving 74 random bits in the resulting UUID.
72    ///
73    /// Returns `None` if `unix_ts_ms` does not fit in 48 bits.
74    #[rustfmt::skip]
75    pub const fn from_random_v7(unix_ts_ms: u64, random: [u8; 10]) -> Option<Self> {
76        is! { unix_ts_ms > 0xFFFF_FFFF_FFFF, return None }
77        let ts = unix_ts_ms.to_be_bytes();
78        let bytes = [
79            ts[2], ts[3], ts[4], ts[5], ts[6], ts[7],
80            random[0], random[1],
81            random[2], random[3], random[4], random[5],
82            random[6], random[7], random[8], random[9],
83        ];
84        Some(Self::from_ietf_bytes(bytes, UuidVersion::V7))
85    }
86    /// Generates a version 7 UUID using a Unix millisecond timestamp
87    /// and deterministic randomness from `Pcg32`.
88    ///
89    /// This is suitable when reproducibility is desired;
90    /// it does not provide cryptographic unpredictability.
91    pub const fn from_pcg32_v7(unix_ts_ms: u64, rng: &mut Pcg32) -> Option<Self> {
92        is! { unix_ts_ms > 0xFFFF_FFFF_FFFF, return None } // avoids advancing rng
93        let mut random = [0u8; 10];
94        rng.fill_bytes(&mut random);
95        Self::from_random_v7(unix_ts_ms, random)
96    }
97    /// Generates a version 7 UUID using the current time from an absolute `TimeSource`
98    /// and deterministic randomness from `Pcg32`.
99    ///
100    /// Returns `None` if the source is not absolute or if its current
101    /// Unix millisecond timestamp exceeds the UUID version 7 range.
102    #[cfg(feature = "time")]
103    pub fn from_pcg32_v7_now<T, P>(rng: &mut Pcg32) -> Option<Self>
104    where
105        T: TimeSource<P>,
106        P: TimePoint,
107    {
108        is! { !T::time_is_absolute(), return None }
109        Self::from_pcg32_v7(T::time_now_millis(), rng)
110    }
111    /// Generates a version 7 UUID using the current time from an absolute `TimeSourceCfg`
112    /// and deterministic randomness from `Pcg32`.
113    #[cfg(feature = "time")]
114    pub fn from_pcg32_v7_now_cfg<T, P>(cfg: T::Config, rng: &mut Pcg32) -> Option<Self>
115    where
116        T: TimeSourceCfg<P>,
117        P: TimePoint,
118    {
119        is! { !T::time_is_absolute(cfg), return None }
120        Self::from_pcg32_v7(T::time_now_millis(cfg), rng)
121    }
122
123    /// Creates a UUID from its 16-byte representation.
124    pub const fn from_bytes(bytes: [u8; 16]) -> Self {
125        Self(bytes)
126    }
127    /// Creates a UUID from a `u128` interpreted in big-endian byte order.
128    pub const fn from_u128(value: u128) -> Self {
129        Self(value.to_be_bytes())
130    }
131    /// Creates a UUID from a non-NIL niche-optimized representation.
132    pub const fn from_non_nil(non_nil: UuidNonNil) -> Self {
133        non_nil.into_uuid()
134    }

136    /* representation */
137
138    /// Returns a reference to the 16-byte representation.
139    pub const fn as_bytes(&self) -> &[u8; 16] {
140        &self.0
141    }
142    /// Returns this UUID as a big-endian `u128`.
143    pub const fn as_u128(self) -> u128 {
144        u128::from_be_bytes(self.0)
145    }
146    /// Returns this UUID in its standard hyphenated textual representation,
147    /// using lowercase hexadecimal digits.
148    ///
149    /// It requires a byte `buffer` of at least [`STR_LEN`][Self::STR_LEN] bytes.
150    pub const fn as_str_into(self, buffer: &mut [u8]) -> Option<&str> {
151        is! { buffer.len() < Self::STR_LEN, return None }
152        let bytes = self.0;
153        let mut dst = 0;
154        whilst! { src in 0..16; {
155            if src == 4 || src == 6 || src == 8 || src == 10 {
156                buffer[dst] = b'-';
157                dst += 1;
158            }
159            let byte = bytes[src];
160            buffer[dst] = Ascii::digit_lower(byte >> 4);
161            buffer[dst + 1] = Ascii::digit_lower(byte & 0x0F);
162            dst += 2;
163        }}
164        unwrap![ok_some Str::from_utf8(slice![buffer, ..Self::STR_LEN])]
165    }
166    /// Returns the 16-byte representation.
167    pub const fn into_bytes(self) -> [u8; 16] {
168        self.0
169    }
170    /// Converts this UUID into a non-NIL niche-optimized representation.
171    pub const fn into_non_nil(self) -> Option<UuidNonNil> {
172        UuidNonNil::from_uuid(self)
173    }
174    /// Returns the Unix millisecond timestamp encoded by a version 7 UUID.
175    ///
176    /// Returns `None` if this is not a recognized IETF version 7 UUID.
177    #[must_use]
178    pub const fn unix_ts_ms_v7(self) -> Option<u64> {
179        is! { !matches!(self.version(), Some(UuidVersion::V7)), return None }
180        Some(u64::from_be_bytes([
181            0, 0, self.0[0], self.0[1], self.0[2], self.0[3], self.0[4], self.0[5],
182        ]))
183    }

185    /* classification */
186
187    /// Returns whether this is the nil UUID.
188    pub const fn is_nil(self) -> bool {
189        self.as_u128() == 0
190    }
191    /// Returns whether this is the maximum UUID.
192    pub const fn is_max(self) -> bool {
193        self.as_u128() == u128::MAX
194    }
195    /// Returns the UUID variant.
196    pub const fn variant(self) -> UuidVariant {
197        let byte = self.0[8];
198        if byte & 0x80 == 0 {
199            UuidVariant::Ncs
200        } else if byte & 0xC0 == 0x80 {
201            UuidVariant::Ietf
202        } else if byte & 0xE0 == 0xC0 {
203            UuidVariant::Microsoft
204        } else {
205            UuidVariant::Future
206        }
207    }
208    /// Returns the version number for an IETF UUID.
209    ///
210    /// Returns `None` for UUIDs belonging to another variant.
211    pub const fn version_number(self) -> Option<u8> {
212        match self.variant() {
213            UuidVariant::Ietf => Some(self.0[6] >> 4),
214            _ => None,
215        }
216    }
217    /// Returns the recognized UUID version.
218    ///
219    /// Returns `None` for another variant or for an unused or
220    /// reserved version number.
221    pub const fn version(self) -> Option<UuidVersion> {
222        match self.version_number() {
223            Some(number) => UuidVersion::from_number(number),
224            None => None,
225        }
226    }
227    /// Parses the standard hyphenated textual UUID representation.
228    ///
229    /// Hexadecimal digits may use either case.
230    pub const fn parse_str(string: &str) -> Result<Self, TextParseError> {
231        let input = string.as_bytes();
232        if input.len() < Self::STR_LEN {
233            return Err(TextParseError::unexpected_eof(TextCursor::new_prim(input.len() as _)));
234        }
235        if input.len() > Self::STR_LEN {
236            return Err(TextParseError::trailing_input(TextCursor::new_prim(Self::STR_LEN as _)));
237        }
238        let mut bytes = [0u8; 16];
239        let mut src = 0usize;
240        whilst! { dst in 0..16; {
241            if src == 8 || src == 13 || src == 18 || src == 23 {
242                if input[src] != b'-' {
243                    let cursor = TextCursor::new_prim(src as _);
244                    return Err(TextParseError::unexpected_byte(cursor, b'-', Some(input[src])));
245                }
246                src += 1;
247            }
248           let hi = unwrap! { some_ok_or? Ascii::hex_digit_value(input[src]),
249                TextParseError::invalid_digit(TextCursor::new_prim(src as _)) };
250           let lo = unwrap! { some_ok_or? Ascii::hex_digit_value(input[src + 1]),
251                TextParseError::invalid_digit(TextCursor::new_prim((src + 1) as _)) };
252            bytes[dst] = hi << 4 | lo;
253            src += 2;
254        }}
255        Ok(Self::from_bytes(bytes))
256    }

298impl FromRandTry for Uuid {
299    /// Constructs a version 4 UUID from a fallible random source.
300    ///
301    /// `FromRandTry` maps to UUID version 4 because version 4 is defined
302    /// entirely by random or pseudo-random input.
303    fn from_rand_try<R: RandTry + ?Sized>(rng: &mut R) -> Result<Self, R::Error> {
304        let mut random = [0u8; 16];
305        rng.rand_try_fill_bytes(&mut random)?;
306        Ok(Self::from_random_v4(random))
307    }
