11#[doc = crate::_tags!(uid state)]
12/// A stateful monotonic UUID version 7 generator.
13#[doc = crate::_doc_meta!{
14    location("data/id/uuid", struct UuidV7Generator),
15    test_size_of(UuidV7Generator = 16|128; niche !Option),
16}]
17/// The generator retains its preceding UUID as state. Each successfully
18/// generated UUID is strictly greater than the preceding one.
19///
20/// When the supplied timestamp advances, a new UUID is initialized from that
21/// timestamp and fresh random material. When the timestamp is unchanged or
22/// moves backwards, the previous timestamp is retained and its `rand_b` field
23/// is advanced by a positive increment derived from the supplied random material.
24///
25/// This preserves UUID ordering across repeated millisecond timestamps
26/// and clock rollback within one generator state.
27///
28/// Generation returns `None` if the timestamp exceeds the UUID version 7 range
29/// or if `rand_b` cannot be advanced without overflow. The generator state
30/// is unchanged when generation fails.
31///
32/// The ordering guarantee applies to a single state lineage.
33/// Independently restored generators are not coordinated.
34#[must_use]
35#[derive(Debug, PartialEq, Eq)]
36pub struct UuidV7Generator {
37    last: Option<UuidNonNil>,
38}
39
40impl UuidV7Generator {
41    /* constructors */
42
43    /// Creates an empty UUIDv7 generator.
44    pub const fn new() -> Self {
45        Self { last: None }
46    }
47    /// Restores a generator using a UUID version 7 value as its preceding state.
48    ///
49    /// Returns `None` if `last` is not a recognized IETF version 7 UUID.
50    pub const fn from_last(last: UuidNonNil) -> Option<Self> {
51        unwrap![some_or last.version(), UuidVersion::V7 => Some(Self { last: Some(last) }), None]
52    }
53
54    /* state */
55
56    /// Returns the UUID currently retained as generator state.
57    pub const fn last(&self) -> Option<UuidNonNil> {
58        self.last
59    }
60    /// Returns the UUID currently retained as generator state,
61    /// consuming the generator.
62    pub const fn into_last(self) -> Option<UuidNonNil> {
63        self.last
64    }
65
66    /* generation */
67
68    /// Generates the next monotonic UUID version 7
69    /// from a Unix millisecond timestamp and random material.
70    ///
71    /// If `unix_ts_ms` is greater than the timestamp retained by the generator,
72    /// it begins a new timestamp using `random`. Otherwise the retained timestamp
73    /// is preserved and `random` is used to derive a positive `rand_b` increment.
74    ///
75    /// Returns `None` if `unix_ts_ms` exceeds the UUID version 7 timestamp range,
76    /// or if the monotonic `rand_b` field would overflow.
77    ///
78    /// The generator state is unchanged on failure.
79    pub const fn next_random(&mut self, unix_ts_ms: u64, random: [u8; 10]) -> Option<UuidNonNil> {
80        is! { unix_ts_ms > Uuid::V7_UNIX_TS_MS_MAX, return None }
81        let last = unwrap![some_or self.last, return self._start_timestamp(unix_ts_ms, random)];
82        let last_ts = unwrap![some? last.into_uuid().unix_ts_ms_v7()];
83        is! { unix_ts_ms > last_ts,
84            self._start_timestamp(unix_ts_ms, random), self._advance(last, random)
85        }
86    }
87    /// Generates the next monotonic UUID version 7 using a Unix millisecond timestamp
88    /// and deterministic randomness from [`Pcg32`].
89    ///
90    /// This is suitable when reproducibility is desired;
91    /// it does not provide cryptographic unpredictability.
92    ///
93    /// Returns `None` if `unix_ts_ms` exceeds the UUID version 7 timestamp range,
94    /// or if the monotonic `rand_b` field would overflow.
95    ///
96    /// An out-of-range timestamp is rejected without advancing `rng`.
97    pub const fn next_pcg32(&mut self, unix_ts_ms: u64, rng: &mut Pcg32) -> Option<UuidNonNil> {
98        is! { unix_ts_ms > Uuid::V7_UNIX_TS_MS_MAX, return None }
99        let mut random = [0; 10];
100        rng.fill_bytes(&mut random);
101        self.next_random(unix_ts_ms, random)
102    }
103    /// Generates the next monotonic UUID version 7 using the current time from
104    /// an absolute [`TimeSource`] and deterministic randomness from [`Pcg32`].
105    ///
106    /// Returns `None` if the source is not absolute, if its current Unix
107    /// millisecond timestamp exceeds the UUID version 7 range,
108    /// or if the monotonic `rand_b` field would overflow.
109    #[cfg(feature = "time")]
110    pub fn next_pcg32_now<T, P>(&mut self, rng: &mut Pcg32) -> Option<UuidNonNil>
111    where
112        T: TimeSource<P>,
113        P: TimePoint,
114    {
115        is! { !T::time_is_absolute(), return None }
116        self.next_pcg32(T::time_now_millis(), rng)
117    }
118    /// Generates the next monotonic UUID version 7 using the current time from
119    /// an absolute configured [`TimeSourceCfg`] and deterministic randomness
120    /// from [`Pcg32`].
121    ///
122    /// Returns `None` if the source is not absolute, if its current Unix
123    /// millisecond timestamp exceeds the UUID version 7 range,
124    /// or if the monotonic `rand_b` field would overflow.
125    #[cfg(feature = "time")]
126    pub fn next_pcg32_now_cfg<T, P>(
127        &mut self,
128        cfg: T::Config,
129        rng: &mut Pcg32,
130    ) -> Option<UuidNonNil>
131    where
132        T: TimeSourceCfg<P>,
133        P: TimePoint,
134    {
135        is! { !T::time_is_absolute(cfg), return None }
136        self.next_pcg32(T::time_now_millis(cfg), rng)
137    }
138    /// Generates the next monotonic UUID version 7 using a Unix millisecond timestamp
139    /// and a fallible random source.
140    ///
141    /// Returns:
142    /// - `Ok(Some(uuid))` on successful generation,
143    /// - `Ok(None)` if the timestamp is outside the UUID version 7 range
144    ///   or the monotonic `rand_b` field would overflow,
145    /// - `Err(error)` if the random source fails.
146    ///
147    /// The generator state is unchanged on `Ok(None)` or `Err`.
148    pub fn next_rand_try<R: RandTry + ?Sized>(
149        &mut self,
150        unix_ts_ms: u64,
151        rng: &mut R,
152    ) -> Result<Option<UuidNonNil>, R::Error> {
153        is! { unix_ts_ms > Uuid::V7_UNIX_TS_MS_MAX, return Ok(None) }
154        let mut random = [0; 10];
155        rng.rand_try_fill_bytes(&mut random)?;
156        Ok(self.next_random(unix_ts_ms, random))
157    }
158
159    /* helpers */
160
161    /// Maximum value of the 62-bit UUIDv7 `rand_b` field.
162    const RAND_B_MAX: u64 = 0x3FFF_FFFF_FFFF_FFFF;
163
164    const fn _start_timestamp(
165        &mut self,
166        unix_ts_ms: u64,
167        mut random: [u8; 10],
168    ) -> Option<UuidNonNil> {
169        // rand_b begins at random[2].
170        // Reserve its most-significant bit as a rollover guard.
171        random[2] &= 0x1F;
172        let uuid = unwrap![some? Uuid::from_random_v7(unix_ts_ms, random)];
173        let uuid = unwrap![some_guaranteed_or_ub UuidNonNil::from_uuid(uuid)]; // safe for UUIDv7 
174        self.last = Some(uuid);
175        Some(uuid)
176    }
177    const fn _advance(&mut self, last: UuidNonNil, random: [u8; 10]) -> Option<UuidNonNil> {
178        let mut bytes = last.into_bytes();
179        let mut rand_b = read_at!(bytes, 8, @8);
180        rand_b[0] &= 0x3F;
181        let rand_b = u64::from_be_bytes(rand_b);
182        let increment = (u16::from_be_bytes(read_at!(random, 0, @2)) & 0x0FFF) as u64 + 1;
183        let next = unwrap![some_if? rand_b.checked_add(increment), |next| next <= Self::RAND_B_MAX];
184        let mut next = next.to_be_bytes();
185        next[0] = 0x80 | (next[0] & 0x3F);
186        write_at!(bytes, 8, @8 next);
187        let uuid = unwrap![some? UuidNonNil::from_bytes(bytes)];
188        self.last = Some(uuid);
189        Some(uuid)
190    }
191}
