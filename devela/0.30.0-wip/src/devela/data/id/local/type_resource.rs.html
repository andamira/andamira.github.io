13/// Defines zero-cost, zero-sized, type-safe *resource* IDs.
14#[doc = crate::_doc_meta!{
15    location("data/id/local", macro type_resource),
16}]
17/// This macro generates zero-sized types associated with an inner ID type.
18/// These types enable strong type safety at compile time
19/// while adding no runtime overhead.
20///
21/// By associating resources with unique types, the system enforces correct
22/// usage of identifiers and reduces errors from mixing unrelated IDs.
23///
24/// Each generated type provides a `new` constructor method returning a
25/// [`TypeResource`] instance of the associated inner type.
26///
27/// Unlike [`type_marker!`][crate::type_marker], which generates purely marker
28/// types with no data, `type_resource!` ties each type to an inner ID type
29/// for handling type-safe resources.
30///
31/// # Examples
32/// ```
33/// use devela::type_resource;
34///
35/// type_resource![Id0:u8]; // single definition and resource
36/// type_resource![Id1,Id2:u16]; // multiple definitions, same resource
37/// type_resource![Id3,Id4:u32; Id5:u64; Id6,Id7:i8]; // diferent resources
38/// ```
39/// See also: [`TypeResource`] and [`TypeResourced`].
40#[macro_export]
41#[cfg_attr(cargo_primary_package, doc(hidden))]
42macro_rules! type_resource {
43    // One or multiple resources share the same ID data type
44    ($($name:ident),+ : $inner:ty) => {
45        $(
46            /// Type-safe resource ID with both phantom-based and newtype-based constructors.
47            pub struct $name;
48
49            impl $name {
50                /// Creates a new `TypeResource`.
51                #[allow(unused)]
52                pub const fn new(data: $inner) -> $crate::TypeResource<$name> {
53                    $crate::TypeResource::new(data)
54                }
55            }
56
57            /// Associates the resource with its inner ID type for the newtype-based system.
58            impl $crate::TypeResourced for $name {
59                type TypeData = $inner;
60            }
61        )+
62    };
63
64    // Multiple groups of resources have different ID data types
65    ($($($name:ident),+ : $inner:ty);+ $(;)?) => {
66        $( type_resource!($($name),+ : $inner); )+
67    };
68}
69#[doc(inline)]
70pub use type_resource;

73/// Represents an association between a resource and its inner data type.
74#[doc = crate::_doc_meta!{
75    location("data/id/local", trait TypeResourced),
76}]
77///
78/// See also: [`TypeResource`] and [`type_resource!`].
79pub trait TypeResourced {
80    /// The inner type that serves as the unique identifier for the resource.
81    type TypeData;
82}
83
84#[doc = crate::_tags!(code uid)]
85/// A newtype-based ID that associates a resource with its inner ID.
86#[doc = crate::_doc_meta!{
87    location("data/id/local", struct TypeResource),
88}]
89///
90/// # Examples
91/// ```
92/// use devela::type_resource;
93///
94/// type_resource![Id1, Id2: u32];
95/// let (id1, id2) = (Id1::new(42), Id2::new(42));
96///
97/// assert_eq!(42, *id1.get());
98/// assert_eq!(*id1.get(), *id2.get());
99/// ```
100/// ```compile_fail
101/// use devela::type_resource;
102///
103/// # type_resource![Id1, Id2: u32];
104/// # let (id1, id2) = (Id1::new(42), Id2::new(42));
105/// assert_eq!(id1, id2); // type mismatch compile error
106/// ```
107///
108/// See also: [`type_resource!`], [`TypeResourced`].
109#[repr(transparent)]
110#[must_use]
111pub struct TypeResource<T: TypeResourced> {
112    data: T::TypeData,
113}
114
115impl<T: TypeResourced> TypeResource<T> {
116    /// Creates a new `TypeResource` instance with the given inner ID.
117    pub const fn new(data: T::TypeData) -> Self {
118        TypeResource { data }
119    }
120    /// Gets a reference to the ID data.
121    pub const fn get(&self) -> &T::TypeData {
122        &self.data
123    }
124    /// Takes ownership of the ID data.
125    pub fn take(self) -> T::TypeData {
126        self.data
127    }
128}
