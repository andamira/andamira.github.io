11/// Maps backend/native identifiers into stable, compact internal IDs.
12#[doc = crate::_doc_meta!{
13    location("data/id/local", struct IdRegistry),
14    test_size_of(IdRegistry<i32, 16> = 132|1056; niche !Option),

17/// Used throughout the library anywhere a subsystem needs to allocate
18/// a small set of stable identifiers from arbitrary external handles.
19#[derive(Clone, Debug)]
20pub struct IdRegistry<Id, const MAX: usize> {
21    next: u32,
22    entries: [(u32, Id); MAX],
23}
24impl<Id: Copy, const MAX: usize> IdRegistry<Id, MAX> {
25    /// Creates an empty registry.
26    pub const fn new(default: Id) -> Self {
27        Self { next: 0, entries: [(0, default); MAX] }
28    }
29    /// Returns an existing id for `native`, or inserts a new one.
30    pub fn intern(&mut self, native: u32, mk: impl Fn(u32) -> Id) -> Id {
31        // lookup
32        for (raw, id) in self.entries.iter() {
33            if *raw == native {
34                return *id;
35            }
36        }
37        // allocate new
38        let id = mk(self.next);
39        self.next += 1;
40        for slot in self.entries.iter_mut() {
41            if slot.0 == 0 {
42                *slot = (native, id);
43                return id;
44            }
45        }
46        panic!("IdRegistry FULL"); // TEMP
47    }
48}
