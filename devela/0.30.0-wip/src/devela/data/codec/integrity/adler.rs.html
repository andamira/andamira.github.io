14/// Adler-32 checksum.
15#[doc = crate::_doc_meta!{
16    location("data/codec", struct Adler32),
17    test_size_of(Adler32 = 4|32; niche !Option),
18}]
19/// A lightweight checksum consisting of two 16-bit accumulators (`a` and `b`)
20/// updated modulo 65521, the largest prime below 2^16.
21/// The final checksum packs `b` in the high 16 bits and `a` in the low 16 bits.
22///
23/// Adler-32 is **byte-order sensitive**: the checksum reflects the exact
24/// sequence of input bytes. When used together with Rust's `Hash` trait,
25/// results may differ between little-endian and big-endian targets unless
26/// the input is serialized in a fixed byte order.
27///
28/// This type supports incremental updates via [`write_bytes`][Self::write_bytes],
29/// making it suitable for checksumming streaming data.
30///
31/// For details, see <https://en.wikipedia.org/wiki/Adler-32>.
32#[must_use]
33#[derive(Debug, Copy, Clone)]
34pub struct Adler32 {
35    /// Running sum of bytes.
36    a: u16,
37    /// Running sum of all previous a values.
38    b: u16,
39}
40
41#[rustfmt::skip]
42impl ConstInit for Adler32 { const INIT: Self = Self::new(); }
43#[rustfmt::skip]
44impl Default for Adler32 { fn default() -> Self { Self::new() } }
45#[rustfmt::skip]
46impl Hasher for Adler32 {
47    fn finish(&self) -> u64 { u64::from(self.checksum()) }
48    fn write(&mut self, bytes: &[u8]) { self.write_bytes(bytes); }
49}
50
51/// Compile-time friendly API.
52#[rustfmt::skip]
53impl Adler32 {
54    /* constructors */
55
56    /// Creates a new Adler-32 state with the initial value `1`.
57    ///
58    /// This corresponds to the standard Adler-32 initial state `(a=1, b=0)`.
59    pub const fn new() -> Self { Self { a: 1, b: 0 } }
60
61    /// Builds a checksum state from a previously computed Adler-32 value.
62    pub const fn from_checksum(sum: u32) -> Self {
63        Adler32 { a: sum as u16, b: (sum >> 16) as u16 }
64    }
65
66    /* operations */
67
68    /// Returns the current Adler-32 checksum as a packed `u32`.
69    #[must_use]
70    pub const fn checksum(&self) -> u32 {
71        ((self.b as u32) << 16) | self.a as u32
72    }
73    /// Computes the Adler-32 checksum of a byte slice.
74    ///
75    /// This is a convenience wrapper that does not require
76    /// constructing an [`Adler32`] value explicitly.
77    #[must_use]
78    pub const fn checksum_bytes(bytes: &[u8]) -> u32 {
79        let mut new = Adler32::new();
80        new.write_bytes(bytes);
81        new.checksum()
82    }
83    /// Updates the checksum with the given bytes.
84    ///
85    /// This method may be called multiple times to checksum data streams.
86    pub const fn write_bytes(&mut self, bytes: &[u8]) {
87        self.update_chunked(bytes);
88    }
89    /// Internal Adler-32 update routine.
90    ///
91    /// Implements the standard Adler-32 update step:
92    /// ```text
93    /// a = (a + data[i]) mod 65521
94    /// b = (b + a)        mod 65521
95    /// ```
96    /// using chunked reduction to avoid accumulator overflow.
97    const fn update_chunked(&mut self, bytes: &[u8]) {
98        const MOD: u32 = 65521;
99        const OUTER_CHUNK: usize = 5552 * INNER_CHUNK;
100        const INNER_CHUNK: usize = 4;
101        let (mut a, mut b) = (self.a as u32, self.b as u32);
102        let mut a_vec = Lane4::<u32>([0; INNER_CHUNK]);
103        let mut b_vec = a_vec;
104        let (bytes, final_remainder) = bytes.split_at(bytes.len() - bytes.len() % INNER_CHUNK);
105        // vectorized outer processing (full OUTER_CHUNK blocks)
106        let (outer_count, remainder_outer_chunk) = Slice::chunks_exact::<OUTER_CHUNK>(bytes);
107        whilst! { i in 0..outer_count; {
108            let outer_chunk = unwrap![some Slice::chunk::<OUTER_CHUNK>(bytes, i)];
109
110            // vectorized inner accumulation over 4-byte groups
111            let (inner_count, _) = Slice::chunks_exact::<INNER_CHUNK>(outer_chunk);
112            whilst! { j in 0..inner_count; {
113                let byte_vec = Slice::chunk::<INNER_CHUNK>(outer_chunk, j).unwrap();
114                let val = Lane4::<u32>::from_byte_values(byte_vec);
115                a_vec.add_assign_plain(val);
116                b_vec.add_assign_plain(a_vec);
117            }}
118            b += OUTER_CHUNK as u32 * a;
119            a_vec.rem_scalar_assign_plain(MOD);
120            b_vec.rem_scalar_assign_plain(MOD);
121            b %= MOD;
122        }}
123        // vectorized remainder after the outer loop
124        let (inner_count, _) = Slice::chunks_exact::<INNER_CHUNK>(remainder_outer_chunk);
125        whilst! { i in 0..inner_count; {
126            let byte_vec = unwrap![some Slice::chunk::<4>(remainder_outer_chunk, i)];
127            let val = Lane4::<u32>::from_byte_values(byte_vec);
128            a_vec.add_assign_plain(val);
129            b_vec.add_assign_plain(a_vec);
130        }}
131        // combine vector lanes into scalar a and b
132        b += remainder_outer_chunk.len() as u32 * a;
133        a_vec.rem_scalar_assign_plain(MOD);
134        b_vec.rem_scalar_assign_plain(MOD);
135        b %= MOD;
136        b_vec.mul_scalar_assign_plain(4);
137        b_vec.0[1] += MOD - a_vec.0[1];
138        b_vec.0[2] += (MOD - a_vec.0[2]) * 2;
139        b_vec.0[3] += (MOD - a_vec.0[3]) * 3;
140        whilst! { i in 0..a_vec.0.len(); { a += a_vec.0[i]; }}
141        whilst! { i in 0..b_vec.0.len(); { b += b_vec.0[i]; }}
142        // final scalar remainder (bytes < 4)
143        whilst! { i in 0..final_remainder.len(); {
144            a += final_remainder[i] as u32;
145            b += a;
146        }}
147        // reduce to 16-bit result
148        self.a = (a % MOD) as u16;
149        self.b = (b % MOD) as u16;
150    }
151}

153#[cfg(test)]
154#[rustfmt::skip]
155mod _test {
156    use crate::{Adler32, Hash};
157
158    #[test]
159    fn checksum_bytes() {
160        assert_eq![0x00000001, Adler32::checksum_bytes(b"")];
161        assert_eq![0x00620062, Adler32::checksum_bytes(b"a")];
162        assert_eq![0x024D0127, Adler32::checksum_bytes(b"abc")];
163        assert_eq![0x29750586, Adler32::checksum_bytes(b"message digest")];
164        assert_eq![0x64A607E0, Adler32::checksum_bytes(b"ABCDEFGHIJKLMNOPQRSTUVWXYZ")];
165        assert_eq![0x11E60398, Adler32::checksum_bytes(b"Wikipedia")];
166    }
167    #[test]
168    fn write_bytes() {
169        let full = Adler32::checksum_bytes(b"HelloWorld");
170        let mut st = Adler32::new();
171        st.write_bytes(b"Hello");
172        st.write_bytes(b"World");
173        assert_eq!(full, st.checksum());
174    }
175    #[test]
176    fn many_small_updates() {
177        let input = b"abcdefghij1234567890";
178        let mut a = Adler32::new();
179        a.write_bytes(input);
180        let full = a.checksum();
181        let mut b = Adler32::new();
182        for byte in input {
183            b.write_bytes(&[*byte]);
184        }
185        assert_eq!(full, b.checksum());
186    }
187    #[test]
188    fn hash_struct() {
189        #[derive(Hash)]
190        struct Packet { id: u16, flag: bool, value: u32, }
191        let pkt = Packet { id: 0x1234, flag: true, value: 0xCAFEBABE };
192        let mut ad = Adler32::new();
193        pkt.hash(&mut ad);
194        if cfg!(target_endian = "little") { assert_eq!(ad.checksum(), 0x09D00388); }
195        if cfg!(target_endian = "big") { assert_eq!(ad.checksum(), 0x0A160388); }
196    }
