5/// Normal left/right guard pattern.
6pub(super) const GUARD: u8 = 0b101;

8/// Centre guard pattern.

11/// Number set A, from leftmost to rightmost module.
12#[rustfmt::skip]
13pub(super) const SET_A: [u8; 10] = [
14    0b000_1101, // 0
15    0b001_1001, // 1
16    0b001_0011, // 2
17    0b011_1101, // 3
18    0b010_0011, // 4
19    0b011_0001, // 5
20    0b010_1111, // 6
21    0b011_1011, // 7
22    0b011_0111, // 8
23    0b000_1011, // 9
24];
25
26pub(super) const fn set_b(digit: u8) -> u8 {
27    reverse7(SET_A[digit as usize] ^ 0x7F)
28}
29pub(super) const fn set_c(digit: u8) -> u8 {
30    SET_A[digit as usize] ^ 0x7F
31}
32pub(super) const fn reverse7(pattern: u8) -> u8 {
33    pattern.reverse_bits() >> 1
34}
35pub(super) const fn decode_a(pattern: u8) -> Option<u8> {
36    whilst! { digit in 0..SET_A.len(); {
37        is! { SET_A[digit] == pattern, return Some(digit as u8) }
38    }}
39    None
40}
41pub(super) const fn decode_b(pattern: u8) -> Option<u8> {
42    decode_a(reverse7(pattern) ^ 0x7F)
43}
44pub(super) const fn decode_c(pattern: u8) -> Option<u8> {
45    decode_a(pattern ^ 0x7F)
46}
47pub(super) const fn ean_check_digit(data: &[u8]) -> Option<u8> {
48    lets! { mut sum = 0_usize, mut i = data.len(), mut weight = 3_usize }
49    while i != 0 {
50        i -= 1;
51        let digit = data[i];
52        is! { digit > 9, return None }
53        sum += digit as usize * weight;
54        weight = 4 - weight; // 3 ↔ 1
55    }
56    Some(((10 - sum % 10) % 10) as u8)
57}
