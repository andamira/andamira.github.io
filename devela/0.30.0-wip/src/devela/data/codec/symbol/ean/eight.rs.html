3//! Implements EAN-8 for [`Ean<8>`][crate::Ean].
4//
5
6use super::_helper;
7use crate::{Ean, is, read_at, unwrap, whilst, write_at};
8
9impl Ean<8> {
10    /// Number of digits preceding the check digit.
11    pub const DATA_DIGITS: usize = 7;
12
13    /// Number of logical modules, excluding quiet zones.
14    pub const MODULES: usize = 67;
15
16    /* digits */
17
18    /// Calculates the mandatory check digit for seven GTIN-8 data digits.
19    ///
20    /// Returns `None` if any input value is greater than 9.
21    #[must_use]
22    pub const fn check_digit(data: [u8; 7]) -> Option<u8> {
23        _helper::ean_check_digit(&data)
24    }
25    /// Appends the calculated check digit to seven GTIN-8 data digits.
26    ///
27    /// Returns `None` if any input value is greater than 9.
28    #[must_use]
29    pub const fn with_check_digit(data: [u8; 7]) -> Option<[u8; 8]> {
30        let check = unwrap![some? Self::check_digit(data)];
31        let mut digits = [0; 8];
32        write_at![digits, 0, @7 data, check];
33        Some(digits)
34    }
35    /// Returns whether all digits are decimal and the check digit is correct.
36    #[must_use]
37    pub const fn is_valid(digits: [u8; 8]) -> bool {
38        let data = read_at![digits, 0, @7];
39        unwrap![some_or Self::check_digit(data), check => check == digits[7], false]
40    }
41
42    /* encoding */
43
44    /// Encodes a valid EAN-8 digit sequence into its 67 logical modules.
45    ///
46    /// Returns `None` if a digit is not decimal or the check digit is invalid.
47    #[must_use]
48    pub const fn encode(digits: [u8; 8]) -> Option<u128> {
49        is! { !Self::is_valid(digits), return None }
50        let mut modules = _helper::GUARD as u128;
51        whilst! { i in 0..4; {
52            modules = (modules << 7) | _helper::SET_A[digits[i] as usize] as u128;
53        }}
54        modules = (modules << 5) | _helper::CENTER as u128;
55        whilst! { i in ..8; {
56            modules = (modules << 7) | _helper::set_c(digits[i]) as u128;
57        }}
58        Some((modules << 3) | _helper::GUARD as u128)
59    }
60
61    /* decoding */
62
63    /// Decodes an exact 67-module EAN-8 pattern.
64    ///
65    /// Returns `None` for malformed guards, invalid digit patterns,
66    /// excess high bits, or an invalid check digit.
67    #[must_use]
68    pub const fn decode(modules: u128) -> Option<[u8; 8]> {
69        is! { modules >> Self::MODULES != 0, return None } // The representation must be canonical
70        // Guards.
71        if ((modules >> 64) & 0b111) != _helper::GUARD as u128
72            || ((modules >> 31) & 0b1_1111) != _helper::CENTER as u128
73            || (modules & 0b111) != _helper::GUARD as u128
74        {
75            return None;
76        }
77        let mut digits = [0_u8; 8];
78        // Left half: number set A.
79        whilst! { i in 0..4; {
80            let shift = 57 - i * 7;
81            let pattern = ((modules >> shift) & 0x7F) as u8;
82            digits[i] = unwrap![some? _helper::decode_a(pattern)];
83        }}
84        // Right half: number set C, the bitwise complement of A.
85        whilst! { i in ..8; {
86            let shift = 24 - (i - 4) * 7;
87            let pattern = ((modules >> shift) & 0x7F) as u8;
88            digits[i] = unwrap![some? _helper::decode_c(pattern)];
89        }}
90        is! { Self::is_valid(digits), Some(digits), None }
91    }
92}

94#[cfg(test)]
95mod _test {
96    use super::*;
97    type Ean8 = Ean<8>;
98
99    #[test]
100    fn check_digits() {
101        assert_eq!(Ean8::check_digit([9, 6, 3, 8, 5, 0, 7]), Some(4));
102        assert_eq!(Ean8::check_digit([0; 7]), Some(0));
103        assert_eq!(Ean8::with_check_digit([9, 6, 3, 8, 5, 0, 7]), Some([9, 6, 3, 8, 5, 0, 7, 4]),);
104        assert_eq!(Ean8::check_digit([0, 1, 2, 3, 4, 5, 10]), None);
105    }
106    #[test]
107    fn validity() {
108        assert!(Ean8::is_valid([9, 6, 3, 8, 5, 0, 7, 4]));
109        assert!(Ean8::is_valid([0; 8]));
110        assert!(!Ean8::is_valid([9, 6, 3, 8, 5, 0, 7, 3]));
111        assert!(!Ean8::is_valid([9, 6, 3, 8, 5, 0, 10, 4]));
112    }
113    #[test]
114    fn canonical_symbol() {
115        let digits = [9, 6, 3, 8, 5, 0, 7, 4];
116        let expected = 0x5_16BD_EB75_4EE5_12E5;
117        assert_eq!(Ean8::encode(digits), Some(expected));
118        assert_eq!(Ean8::decode(expected), Some(digits));
119    }
120    #[test]
121    fn rejects_invalid_encoding_input() {
122        assert_eq!(Ean8::encode([9, 6, 3, 8, 5, 0, 7, 3]), None);
123        assert_eq!(Ean8::encode([9, 6, 3, 8, 5, 0, 10, 4]), None);
124    }
125    #[test]
126    fn rejects_invalid_module_patterns() {
127        let valid = Ean8::encode([9, 6, 3, 8, 5, 0, 7, 4]).unwrap();
128        // Bits outside the canonical 67-bit representation.
129        assert_eq!(Ean8::decode(valid | (1_u128 << 67)), None);
130        // Corrupted left, centre and right guards.
131        assert_eq!(Ean8::decode(valid ^ (1_u128 << 66)), None);
132        assert_eq!(Ean8::decode(valid ^ (1_u128 << 33)), None);
133        assert_eq!(Ean8::decode(valid ^ 1), None);
134    }
135    #[test]
136    fn roundtrip_examples() {
137        for data in [
138            [0, 0, 0, 0, 0, 0, 0],
139            [9, 6, 3, 8, 5, 0, 7],
140            [1, 2, 3, 4, 5, 6, 7],
141            [9, 9, 9, 9, 9, 9, 9],
142        ] {
143            let digits = Ean8::with_check_digit(data).unwrap();
144            let modules = Ean8::encode(digits).unwrap();
145            assert_eq!(Ean8::decode(modules), Some(digits));
146        }
147    }
