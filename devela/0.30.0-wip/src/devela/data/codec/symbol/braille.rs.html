3//! Defines [`BrailleByte`].

5
6use crate::{Str, charu, is, slice, unwrap, whilst, write_at};
7
8#[doc = crate::_tags!(codec namespace)]
9/// Reversibly maps bytes to Unicode Braille-pattern characters.
10#[doc = crate::_doc_meta!{
11    location("data/codec", struct BrailleByte),
12    test_size_of(BrailleByte = 0),
13}]
14///
15/// Each Braille cell uses 3 UTF-8 bytes.
16/// ```text
17/// b7 b3
18/// b6 b2
19/// b5 b1
20/// b4 b0
21/// ```
22/// The mapping is reversible for all 256 byte values.
23///
24/// # Example
25/// ```
26/// use devela::BrailleByte;
27///
28/// let input = [0x00, 0x55, 0xAA, 0xFF];
29/// let mut encoded = [0; 12];
30///
31/// let braille = BrailleByte::encode_to_slice(&input, &mut encoded).unwrap();
32/// assert_eq!(braille, "⠀⣒⠭⣿");
33///
34/// let mut decoded = [0; 4];
35/// assert_eq!(BrailleByte::decode_from_slice(braille, &mut decoded).unwrap(), input);
36/// ```
37#[derive(Debug)]
38pub struct BrailleByte;
39
40impl BrailleByte {
41    const BASE: u32 = 0x2800;
42    const END: u32 = 0x28FF;
43
44    /// UTF-8 length of one encoded Braille cell.
45    pub const UTF8_LEN: usize = 3;
46
47    /* encoding */
48
49    /// Returns the UTF-8 encoded length for `input_len` bytes.
50    #[must_use]
51    pub const fn encoded_len(input_len: usize) -> usize {
52        unwrap![some_expect input_len.checked_mul(Self::UTF8_LEN), "encoded length overflow"]
53    }
54    /// Encodes `byte` as one Unicode Braille-pattern character.
55    #[must_use]
56    pub const fn encode(byte: u8) -> char {
57        unwrap![some char::from_u32(
58            Self::BASE + Self::byte_to_pattern(byte) as u32
59        )]
60    }
61    /// Encodes `byte` as a UTF-8-backed Unicode scalar.
62    pub const fn encode_charu(byte: u8) -> charu {
63        charu::from_char(Self::encode(byte))
64    }
65    /// Encodes `input` as Braille-pattern UTF-8 into `output`.
66    ///
67    /// Returns the written prefix as a string slice,
68    /// or `None` if `output` is too small.
69    pub const fn encode_to_slice<'a>(input: &[u8], output: &'a mut [u8]) -> Option<&'a str> {
70        let len = unwrap![some? input.len().checked_mul(Self::UTF8_LEN)];
71        is! { output.len() < len, return None }
72        let mut offset = 0;
73        whilst! { i in 0..input.len(); {
74            write_at![output, +=offset, #Self::encode(input[i])];
75        }}
76        unwrap![ok_some Str::from_utf8(slice![output, ..offset])]
77    }
78
79    /* decoding */
80
81    /// Returns the decoded byte length for a structurally valid encoded length.
82    #[must_use]
83    pub const fn decoded_len(input_len: usize) -> Option<usize> {
84        is! { input_len.is_multiple_of(Self::UTF8_LEN), Some(input_len / Self::UTF8_LEN), None }
85    }
86    /// Decodes a Unicode Braille-pattern character into its represented byte.
87    ///
88    /// Returns `None` if `ch` is outside `U+2800..=U+28FF`.
89    #[must_use]
90    pub const fn decode(ch: char) -> Option<u8> {
91        let scalar = ch as u32;
92        if scalar >= Self::BASE && scalar <= Self::END {
93            Some(Self::pattern_to_byte((scalar - Self::BASE) as u8))
94        } else {
95            None
96        }
97    }
98    /// Decodes Braille-pattern `input` into bytes.
99    ///
100    /// Returns the written output prefix, or `None` if `input` contains a
101    /// non-Braille character or `output` is too small.
102    pub const fn decode_from_slice<'a>(input: &str, output: &'a mut [u8]) -> Option<&'a [u8]> {
103        is! { !input.len().is_multiple_of(Self::UTF8_LEN), return None }
104        let len = input.len() / Self::UTF8_LEN;
105        is! { output.len() < len, return None }
106        let mut chars = Str::chars(input);
107        let mut offset = 0;
108        while let Some(ch) = chars.next_char() {
109            output[offset] = unwrap![some? Self::decode(ch)];
110            offset += 1;
111        }
112        Some(slice![output, ..offset])
113    }
114
115    /* private*/
116
117    const fn byte_to_pattern(b: u8) -> u8 {
118        ((b & 0x80) >> 7)
119            | ((b & 0x40) >> 5)
120            | ((b & 0x20) >> 3)
121            | ((b & 0x10) << 2)
122            | (b & 0x08)
123            | ((b & 0x04) << 2)
124            | ((b & 0x02) << 4)
125            | ((b & 0x01) << 7)
126    }
127    const fn pattern_to_byte(p: u8) -> u8 {
128        ((p & 0x01) << 7)
129            | ((p & 0x02) << 5)
130            | ((p & 0x04) << 3)
131            | ((p & 0x40) >> 2)
132            | (p & 0x08)
133            | ((p & 0x10) >> 2)
134            | ((p & 0x20) >> 4)
135            | ((p & 0x80) >> 7)
136    }
137}

139#[cfg(test)]
140mod _test {
141    use super::*;
142
143    #[test]
144    fn canonical_patterns() {
145        assert_eq!(BrailleByte::encode(0x00), '⠀');
146        assert_eq!(BrailleByte::encode(0x01), '⢀');
147        assert_eq!(BrailleByte::encode(0x02), '⠠');
148        assert_eq!(BrailleByte::encode(0x04), '⠐');
149        assert_eq!(BrailleByte::encode(0x08), '⠈');
150        assert_eq!(BrailleByte::encode(0x10), '⡀');
151        assert_eq!(BrailleByte::encode(0x20), '⠄');
152        assert_eq!(BrailleByte::encode(0x40), '⠂');
153        assert_eq!(BrailleByte::encode(0x80), '⠁');
154        assert_eq!(BrailleByte::encode(0xFF), '⣿');
155    }
156    #[test]
157    fn lengths() {
158        assert_eq!(BrailleByte::encoded_len(0), 0);
159        assert_eq!(BrailleByte::encoded_len(4), 12);
160        assert_eq!(BrailleByte::decoded_len(12), Some(4));
161        assert_eq!(BrailleByte::decoded_len(11), None);
162    }
163    #[test]
164    fn scalar_roundtrip_all_bytes() {
165        for byte in 0..=u8::MAX {
166            assert_eq!(BrailleByte::decode(BrailleByte::encode(byte)), Some(byte));
167        }
168    }
169    #[test]
170    fn rejects_non_braille() {
171        assert_eq!(BrailleByte::decode('\0'), None);
172        assert_eq!(BrailleByte::decode('A'), None);
173        assert_eq!(BrailleByte::decode('\u{27FF}'), None);
174        assert_eq!(BrailleByte::decode('\u{2900}'), None);
175    }
176    #[test]
177    fn slice_roundtrip() {
178        let input = b"hello";
179        let mut encoded = [0; 15];
180        let braille = BrailleByte::encode_to_slice(input, &mut encoded).unwrap();
181        assert_eq!(braille, "⠎⢖⠞⠞⢾");
182        assert_eq!(braille.len(), BrailleByte::encoded_len(input.len()));
183        let mut decoded = [0; 5];
184        let bytes = BrailleByte::decode_from_slice(braille, &mut decoded).unwrap();
185        assert_eq!(bytes, input);
186    }
187    #[test]
188    fn slice_rejects_invalid_input_and_small_buffers() {
189        let mut small = [0; 2];
190        assert_eq!(BrailleByte::encode_to_slice(b"x", &mut small), None);
191        let mut output = [0; 4];
192        assert_eq!(BrailleByte::decode_from_slice("abc", &mut output), None);
193        assert_eq!(BrailleByte::decode_from_slice("⠁A", &mut output), None);
194        let mut none = [];
195        assert_eq!(BrailleByte::decode_from_slice("⠁", &mut none), None);
196    }
