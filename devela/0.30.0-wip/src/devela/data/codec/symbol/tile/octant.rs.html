6use crate::{Str, charu, is, slice, unwrap, whilst, write_at};
7
8#[doc = crate::_tags!(codec namespace)]
9/// Reversibly maps bytes to 2×4 Unicode octant tiles.
10#[doc = crate::_doc_meta!{
11    location("data/codec/symbol", struct Octant),
12    test_size_of(Octant = 0),
13}]
14/// Bits map to filled octants as follows:
15/// ```text
16/// b7 b3
17/// b6 b2
18/// b5 b1
19/// b4 b0
20/// ```
21/// A cleared bit is empty and a set bit is filled.
22///
23/// The empty pattern is encoded as a space.
24/// The mapping is reversible for all 256 byte values.
25///
26/// # Example
27/// ```
28/// use devela::Octant;
29///
30/// assert_eq!(Octant::encode(0x00), ' ');
31/// assert_eq!(Octant::encode(0xF0), '▌');
32/// assert_eq!(Octant::encode(0x0F), '▐');
33/// assert_eq!(Octant::encode(0xFF), '█');
34/// assert_eq!(Octant::decode('█'), Some(0xFF));
35/// ```
36#[derive(Debug)]
37pub struct Octant;
38
39impl Octant {
40    const BASE: u32 = 0x1CD00;
41    const END: u32 = 0x1CDE5;
42
43    /// Number of representable octant patterns.
44    pub const PATTERN_COUNT: usize = 256;
45
46    /// Maximum UTF-8 length of one encoded octant character.
47    pub const MAX_UTF8_LEN: usize = 4;
48
49    /* encode */
50
51    /// Returns the UTF-8 encoded length of `input`.
52    #[must_use]
53    pub const fn encoded_len(input: &[u8]) -> usize {
54        unwrap![some_expect Self::encoded_len_checked(input), "encoded length overflow"]
55    }
56    /// Encodes a 2×4 octant pattern.
57    #[must_use]
58    pub const fn encode(pattern: u8) -> char {
59        let mask = Self::pattern_to_unicode_mask(pattern);
60        is! { let Some(ch) = Self::reused_char(mask), return ch }
61        unwrap![some char::from_u32( Self::BASE + Self::dedicated_index(mask) as u32)]
62    }
63    /// Encodes a 2×4 octant pattern as a UTF-8-backed Unicode scalar.
64    pub const fn encode_charu(pattern: u8) -> charu {
65        charu::from_char(Self::encode(pattern))
66    }
67    /// Encodes octant patterns as UTF-8 characters into `output`.
68    ///
69    /// Returns the written prefix, or `None` if `output` is too small.
70    pub const fn encode_to_slice<'a>(input: &[u8], output: &'a mut [u8]) -> Option<&'a str> {
71        let len = unwrap![some? Self::encoded_len_checked(input)];
72        is! { output.len() < len, return None }
73        let mut offset = 0;
74        whilst! { i in 0..input.len(); {
75            write_at![output, +=offset, #Self::encode(input[i])];
76        }}
77        unwrap![ok_some Str::from_utf8(slice![output, ..offset])]
78    }
79
80    /* decode */
81
82    /// Returns the decoded pattern count of `input`.
83    ///
84    /// Returns `None` if `input` contains a character outside this mapping.
85    #[must_use]
86    pub const fn decoded_len(input: &str) -> Option<usize> {
87        let mut chars = Str::chars(input);
88        let mut len = 0;
89        while let Some(ch) = chars.next_char() {
90            unwrap![some? Self::decode(ch)];
91            len += 1;
92        }
93        Some(len)
94    }
95    /// Decodes a Unicode octant tile into its 2×4 pattern.
96    ///
97    /// A space decodes to the empty pattern.
98    #[must_use]
99    pub const fn decode(ch: char) -> Option<u8> {
100        if let Some(mask) = Self::reused_mask(ch) {
101            return Some(Self::unicode_mask_to_pattern(mask));
102        }
103        let scalar = ch as u32;
104        if scalar >= Self::BASE && scalar <= Self::END {
105            let index = (scalar - Self::BASE) as u8;
106            let mask = Self::mask_from_dedicated_index(index);
107            Some(Self::unicode_mask_to_pattern(mask))
108        } else {
109            None
110        }
111    }
112    /// Decodes Unicode octant tiles into patterns.
113    ///
114    /// Returns the written output prefix, or `None` if `input` contains
115    /// another character or `output` is too small.
116    pub const fn decode_from_slice<'a>(input: &str, output: &'a mut [u8]) -> Option<&'a [u8]> {
117        let len = unwrap![some? Self::decoded_len(input)];
118        is! { output.len() < len, return None }
119        let mut chars = Str::chars(input);
120        let mut offset = 0;
121        while let Some(ch) = chars.next_char() {
122            output[offset] = unwrap![some? Self::decode(ch)];
123            offset += 1;
124        }
125        Some(slice![output, ..offset])
126    }
127
128    /* private */
129
130    const fn encoded_len_checked(input: &[u8]) -> Option<usize> {
131        let mut len = 0_usize;
132        whilst! { i in 0..input.len(); {
133            len = unwrap![some? len.checked_add(Self::encode(input[i]).len_utf8())];
134        }}
135        Some(len)
136    }
137    /// Converts our column-major spatial pattern to Unicode octant numbering.
138    const fn pattern_to_unicode_mask(p: u8) -> u8 {
139        ((p & 0b1000_0000) >> 7)
140            | ((p & 0b0000_1000) >> 2)
141            | ((p & 0b0100_0000) >> 4)
142            | ((p & 0b0000_0100) << 1)
143            | ((p & 0b0010_0000) >> 1)
144            | ((p & 0b0000_0010) << 4)
145            | ((p & 0b0001_0000) << 2)
146            | ((p & 0b0000_0001) << 7)
147    }
148    /// Converts Unicode octant numbering to our column-major pattern.
149    const fn unicode_mask_to_pattern(m: u8) -> u8 {
150        ((m & 0b0000_0001) << 7)
151            | ((m & 0b0000_0010) << 2)
152            | ((m & 0b0000_0100) << 4)
153            | ((m & 0b0000_1000) >> 1)
154            | ((m & 0b0001_0000) << 1)
155            | ((m & 0b0010_0000) >> 4)
156            | ((m & 0b0100_0000) >> 2)
157            | ((m & 0b1000_0000) >> 7)
158    }
159    /// Returns the character reused for a Unicode octant mask.
160    const fn reused_char(mask: u8) -> Option<char> {
161        whilst! { i in 0..Self::REUSED.len(); {
162            let (candidate, ch) = Self::REUSED[i];
163            is! { candidate == mask, return Some(ch) }
164            is! { candidate > mask, break }
165        }}
166        None
167    }
168    /// Returns the Unicode octant mask represented by a reused character.
169    const fn reused_mask(ch: char) -> Option<u8> {
170        whilst! { i in 0..Self::REUSED.len(); {
171            let (mask, candidate) = Self::REUSED[i];
172            is! { candidate == ch, return Some(mask) }
173        }}
174        None
175    }
176    /// Returns the offset of a non-reused mask in the dedicated Unicode range.
177    const fn dedicated_index(mask: u8) -> u8 {
178        let mut skipped = 0_u8;
179        whilst! { i in 0..Self::REUSED.len(); {
180            let reused = Self::REUSED[i].0;
181            is! { reused >= mask, break }
182            skipped += 1;
183        }}
184        mask - skipped
185    }
186    /// Restores a Unicode octant mask from its dedicated-range offset.
187    const fn mask_from_dedicated_index(index: u8) -> u8 {
188        let mut mask = index;
189        whilst! { i in 0..Self::REUSED.len(); {
190            let reused = Self::REUSED[i].0;
191            is! { reused <= mask, mask += 1, break }
192        }}
193        mask
194    }
195    /// Octant masks represented by characters outside the dedicated range.
196    ///
197    /// Masks use Unicode's row-major octant numbering:
198    /// ```text
199    /// 1 2
200    /// 3 4
201    /// 5 6
202    /// 7 8
203    /// ```
204    #[rustfmt::skip]
205    const REUSED: [(u8, char); 26] = [
206        (0x00, ' '),
207        (0x01, '\u{1CEA8}'),
208        (0x02, '\u{1CEAB}'),
209        (0x03, '\u{1FB82}'),
210        (0x05, '▘'),
211        (0x0A, '▝'),
212        (0x0F, '▀'),
213        (0x14, '\u{1FBE6}'),
214        (0x28, '\u{1FBE7}'),
215        (0x3F, '\u{1FB85}'),
216        (0x40, '\u{1CEA3}'),
217        (0x50, '▖'),
218        (0x55, '▌'),
219        (0x5A, '▞'),
220        (0x5F, '▛'),
221        (0x80, '\u{1CEA0}'),
222        (0xA0, '▗'),
223        (0xA5, '▚'),
224        (0xAA, '▐'),
225        (0xAF, '▜'),
226        (0xC0, '▂'),
227        (0xF0, '▄'),
228        (0xF5, '▙'),
229        (0xFA, '▟'),
230        (0xFC, '▆'),
231        (0xFF, '█'),
232    ];
233}

235#[cfg(test)]
236mod _test {
237    use super::*;
238
239    #[test]
240    fn canonical_patterns() {
241        assert_eq!(Octant::encode(0x00), ' ');
242        // Individual subcells, top-to-bottom by columns.
243        assert_eq!(Octant::encode(0b1000_0000), '\u{1CEA8}'); // 1: TL
244        assert_eq!(Octant::encode(0b0000_1000), '\u{1CEAB}'); // 2: TR
245        assert_eq!(Octant::encode(0b0100_0000), '\u{1CD00}'); // 3
246        assert_eq!(Octant::encode(0b0000_0100), '\u{1CD03}'); // 4
247        assert_eq!(Octant::encode(0b0010_0000), '\u{1CD09}'); // 5
248        assert_eq!(Octant::encode(0b0000_0010), '\u{1CD18}'); // 6
249        assert_eq!(Octant::encode(0b0001_0000), '\u{1CEA3}'); // 7: BL
250        assert_eq!(Octant::encode(0b0000_0001), '\u{1CEA0}'); // 8: BR
251        // Existing block characters reused by the octant repertoire.
252        assert_eq!(Octant::encode(0xF0), '▌');
253        assert_eq!(Octant::encode(0x0F), '▐');
254        assert_eq!(Octant::encode(0xCC), '▀');
255        assert_eq!(Octant::encode(0x33), '▄');
256        assert_eq!(Octant::encode(0xFF), '█');
257        // Bounds of the dedicated range.
258        assert_eq!(Octant::encode(0x40), '\u{1CD00}');
259        assert_eq!(Octant::encode(0x7F), '\u{1CDE5}');
260    }
261    #[test]
262    fn scalar_roundtrip_all_patterns() {
263        for pattern in u8::MIN..=u8::MAX {
264            let ch = Octant::encode(pattern);
265            assert_eq!(Octant::decode(ch), Some(pattern));
266        }
267    }
268    #[test]
269    fn rejects_invalid() {
270        assert_eq!(Octant::decode('A'), None);
271        assert_eq!(Octant::decode('●'), None);
272        assert_eq!(Octant::decode('\u{1CCFF}'), None);
273        assert_eq!(Octant::decode('\u{1CDE6}'), None);
274    }
275    #[test]
276    fn lengths() {
277        assert_eq!(Octant::encoded_len(&[]), 0);
278        assert_eq!(Octant::encoded_len(&[0x00]), 1); // space
279        assert_eq!(Octant::encoded_len(&[0xF0]), 3); // ▌
280        assert_eq!(Octant::encoded_len(&[0x80]), 4); // U+1CEA8
281        assert_eq!(Octant::encoded_len(&[0x40]), 4); // U+1CD00
282        assert_eq!(Octant::encoded_len(&[0x00, 0xF0, 0x80]), 8);
283    }
284    #[test]
285    fn slice_roundtrip_all_patterns() {
286        let mut input = [0_u8; 256];
287        let mut i = 0;
288        while i < input.len() {
289            input[i] = i as u8;
290            i += 1;
291        }
292        let mut encoded = [0; 256 * Octant::MAX_UTF8_LEN];
293        let text = Octant::encode_to_slice(&input, &mut encoded).unwrap();
294        assert_eq!(Octant::encoded_len(&input), text.len());
295        let mut decoded = [0; 256];
296        assert_eq!(Octant::decode_from_slice(text, &mut decoded), Some(input.as_slice()));
297    }
298    #[test]
299    fn slice_rejects_invalid_input_and_small_buffers() {
300        let mut encoded = [0; 3];
301        assert_eq!(Octant::encode_to_slice(&[0x80], &mut encoded), None);
302        let mut decoded = [0; 1];
303        assert_eq!(Octant::decode_from_slice("▌▐", &mut decoded), None);
304        let mut decoded = [0; 4];
305        assert_eq!(Octant::decode_from_slice("▌A▐", &mut decoded), None);
306    }
