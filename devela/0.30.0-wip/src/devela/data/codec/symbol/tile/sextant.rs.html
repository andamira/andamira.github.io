6use crate::{Str, charu, is, slice, unwrap, whilst, write_at};

8#[doc = crate::_tags!(codec namespace)]
9/// Reversibly maps 6-bit patterns to 2×3 Unicode sextant mosaics.
10#[doc = crate::_doc_meta!{
11    location("data/codec/symbol", struct Sextant),
12    test_size_of(Sextant = 0),
13}]
14/// Bits map to filled sextants as follows:
15/// ```text
16/// b5 b2
17/// b4 b1
18/// b3 b0
19/// ```
20/// A cleared bit is empty and a set bit is filled.
21///
22/// The empty pattern is encoded as a space.
23///
24/// # Example
25/// ```
26/// use devela::Sextant;
27///
28/// assert_eq!(Sextant::encode(0b000000), Some(' '));
29/// assert_eq!(Sextant::encode(0b100000), Some('🬀'));
30/// assert_eq!(Sextant::encode(0b111000), Some('▌'));
31/// assert_eq!(Sextant::encode(0b111111), Some('█'));
32/// assert_eq!(Sextant::decode('▐'), Some(0b000111));
33/// ```
34#[derive(Debug)]
35pub struct Sextant;
36
37impl Sextant {
38    const BASE: u32 = 0x1FB00;
39    const END: u32 = 0x1FB3B;
40
41    /// Number of representable sextant patterns.
42    pub const PATTERN_COUNT: usize = 64;
43
44    /// Maximum UTF-8 length of one encoded sextant character.
45    pub const MAX_UTF8_LEN: usize = 4;
46
47    /* encode */
48
49    /// Returns the UTF-8 encoded length of `input`.
50    ///
51    /// Returns `None` if any pattern is greater than `0b11_1111`.
52    #[must_use]
53    pub const fn encoded_len(input: &[u8]) -> Option<usize> {
54        let mut len = 0_usize;
55        whilst! { i in 0..input.len(); {
56            let ch = unwrap![some? Self::encode(input[i])];
57            len = unwrap![some? len.checked_add(ch.len_utf8())];
58        }}
59        Some(len)
60    }
61    /// Encodes a 2×3 sextant pattern.
62    ///
63    /// Returns `None` if `pattern > 0b11_1111`.
64    #[must_use]
65    pub const fn encode(pattern: u8) -> Option<char> {
66        is! { pattern >= Self::PATTERN_COUNT as u8, return None }
67        let mask = Self::pattern_to_unicode_mask(pattern);
68        let scalar = match mask {
69            0 => return Some(' '),
70            21 => return Some('▌'), // sextants 1,3,5
71            42 => return Some('▐'), // sextants 2,4,6
72            63 => return Some('█'),
73            1..=20 => Self::BASE + mask as u32 - 1,
74            22..=41 => Self::BASE + mask as u32 - 2,
75            43..=62 => Self::BASE + mask as u32 - 3,
76            _ => return None,
77        };
78        char::from_u32(scalar)
79    }
80    /// Encodes a 2×3 sextant pattern as a UTF-8-backed Unicode scalar.
81    #[must_use]
82    pub const fn encode_charu(pattern: u8) -> Option<charu> {
83        Some(charu::from_char(unwrap![some? Self::encode(pattern)]))
84    }
85    /// Encodes sextant patterns as UTF-8 characters into `output`.
86    ///
87    /// Returns the written prefix, or `None` if a pattern is invalid
88    /// or `output` is too small.
89    pub const fn encode_to_slice<'a>(input: &[u8], output: &'a mut [u8]) -> Option<&'a str> {
90        let len = unwrap![some? Self::encoded_len(input)];
91        is! { output.len() < len, return None }
92        let mut offset = 0;
93        whilst! { i in 0..input.len(); {
94            let ch = unwrap![some? Self::encode(input[i])];
95            write_at![output, +=offset, #ch];
96        }}
97        unwrap![ok_some Str::from_utf8(slice![output, ..offset])]
98    }
99
100    /* decode */
101
102    /// Returns the decoded pattern count of `input`.
103    ///
104    /// Returns `None` if `input` contains a character outside this mapping.
105    #[must_use]
106    pub const fn decoded_len(input: &str) -> Option<usize> {
107        let mut chars = Str::chars(input);
108        let mut len = 0;
109        while let Some(ch) = chars.next_char() {
110            unwrap![some? Self::decode(ch)];
111            len += 1;
112        }
113        Some(len)
114    }
115    /// Decodes a Unicode sextant mosaic into its 2×3 pattern.
116    ///
117    /// A space decodes to the empty pattern.
118    #[must_use]
119    pub const fn decode(ch: char) -> Option<u8> {
120        let scalar = ch as u32;
121        let mask = match ch {
122            ' ' => 0,
123            '▌' => 21,
124            '▐' => 42,
125            '█' => 63,
126            _ if scalar >= Self::BASE && scalar <= Self::END => {
127                let index = (scalar - Self::BASE) as u8;
128                is! { index < 20, index + 1, is! { index < 40, index + 2, index + 3 } }
129            }
130            _ => return None,
131        };
132        Some(Self::unicode_mask_to_pattern(mask))
133    }
134    /// Decodes Unicode sextant mosaics into patterns.
135    ///
136    /// Returns the written output prefix, or `None` if `input` contains
137    /// another character or `output` is too small.
138    pub const fn decode_from_slice<'a>(input: &str, output: &'a mut [u8]) -> Option<&'a [u8]> {
139        let len = unwrap![some? Self::decoded_len(input)];
140        is! { output.len() < len, return None }
141        let mut chars = Str::chars(input);
142        let mut offset = 0;
143        while let Some(ch) = chars.next_char() {
144            output[offset] = unwrap![some? Self::decode(ch)];
145            offset += 1;
146        }
147        Some(slice![output, ..offset])
148    }
149
150    /* private */
151
152    /// Converts our column-major spatial pattern to Unicode sextant numbering.
153    const fn pattern_to_unicode_mask(p: u8) -> u8 {
154        ((p & 0b10_0000) >> 5)
155            | ((p & 0b00_0100) >> 1)
156            | ((p & 0b01_0000) >> 2)
157            | ((p & 0b00_0010) << 2)
158            | ((p & 0b00_1000) << 1)
159            | ((p & 0b00_0001) << 5)
160    }
161    /// Converts Unicode sextant numbering to our column-major pattern.
162    const fn unicode_mask_to_pattern(m: u8) -> u8 {
163        ((m & 0b00_0001) << 5)
164            | ((m & 0b00_0010) << 1)
165            | ((m & 0b00_0100) << 2)
166            | ((m & 0b00_1000) >> 2)
167            | ((m & 0b01_0000) >> 1)
168            | ((m & 0b10_0000) >> 5)
169    }
170}

172#[cfg(test)]
173mod _test {
174    use super::*;
175
176    #[test]
177    fn canonical_patterns() {
178        assert_eq!(Sextant::encode(0b000000), Some(' '));
179        // Individual subcells, top-to-bottom by columns.
180        assert_eq!(Sextant::encode(0b100000), Some('🬀')); // 1: TL
181        assert_eq!(Sextant::encode(0b000100), Some('🬁')); // 2: TR
182        assert_eq!(Sextant::encode(0b010000), Some('🬃')); // 3: ML
183        assert_eq!(Sextant::encode(0b000010), Some('🬇')); // 4: MR
184        assert_eq!(Sextant::encode(0b001000), Some('🬏')); // 5: BL
185        assert_eq!(Sextant::encode(0b000001), Some('🬞')); // 6: BR
186        // Existing Block Elements reused by the sextant repertoire.
187        assert_eq!(Sextant::encode(0b111000), Some('▌'));
188        assert_eq!(Sextant::encode(0b000111), Some('▐'));
189        assert_eq!(Sextant::encode(0b111111), Some('█'));
190        // Horizontal thirds.
191        assert_eq!(Sextant::encode(0b100100), Some('🬂'));
192        assert_eq!(Sextant::encode(0b010010), Some('🬋'));
193        assert_eq!(Sextant::encode(0b001001), Some('🬭'));
194    }
195    #[test]
196    fn scalar_roundtrip_all_patterns() {
197        for pattern in 0..64 {
198            let ch = Sextant::encode(pattern).unwrap();
199            assert_eq!(Sextant::decode(ch), Some(pattern));
200        }
201    }
202    #[test]
203    fn rejects_invalid() {
204        assert_eq!(Sextant::encode(64), None);
205        assert_eq!(Sextant::encode(u8::MAX), None);
206        assert_eq!(Sextant::decode('A'), None);
207        assert_eq!(Sextant::decode('▖'), None);
208    }
209    #[test]
210    fn lengths() {
211        assert_eq!(Sextant::encoded_len(&[]), Some(0));
212        assert_eq!(Sextant::encoded_len(&[0]), Some(1));
213        assert_eq!(Sextant::encoded_len(&[0b111000]), Some(3));
214        assert_eq!(Sextant::encoded_len(&[0b100000]), Some(4));
215        assert_eq!(Sextant::encoded_len(&[0, 0b111000, 0b100000]), Some(8));
216    }
217    #[test]
218    fn slice_roundtrip_all_patterns() {
219        let mut input = [0_u8; 64];
220        let mut i = 0;
221        while i < input.len() {
222            input[i] = i as u8;
223            i += 1;
224        }
225        let mut encoded = [0; 64 * Sextant::MAX_UTF8_LEN];
226        let text = Sextant::encode_to_slice(&input, &mut encoded).unwrap();
227        assert_eq!(Sextant::encoded_len(&input), Some(text.len()));
228        let mut decoded = [0; 64];
229        assert_eq!(Sextant::decode_from_slice(text, &mut decoded), Some(input.as_slice()));
230    }
231}
