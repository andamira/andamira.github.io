7/// Defines [`Oneof`] and implements all the methods.
8// #[macro_export]
9// #[doc(hidden)]
10// #[cfg_attr(cargo_primary_package, doc(hidden))]
11// #[allow(clippy::crate_in_macro_def, reason = "impl_const_init arm")]
12macro_rules! _data_value_impl_oneof {
13    /* points of entry, using hardcoded argument list */
14
15    // main point of entry
16    // (defines Oneof & implements everything)
17    () => { _data_value_impl_oneof!(%canonical); };
18    //
19    // var_name : var_idx(0-based) + var_nth(1-based) : nth_suffix
20    ($($T:ident : $idx:literal + $nth:literal : $suf:literal),* $(,)?) => {
21        _data_value_impl_oneof!(define_enum: $($T:$nth:$suf),+);
22        _data_value_impl_oneof!(impl_prefixes: $($T:$idx+$nth:$suf),+);
23        _data_value_impl_oneof!(methods_general: $($T:$idx+$nth:$suf),+);
24        _data_value_impl_oneof!(methods_individ: $($T:$idx+$nth:$suf),+);
25    };
26
27    // Implements prefix-dependent functionality for every non-empty arity.
28    (impl_prefixes: $($T:ident : $idx:literal + $nth:literal : $suf:literal),+) => {
29        _data_value_impl_oneof!(%impl_prefixes: (); $($T:$idx+$nth:$suf),+);
30    };
31    // Adds the next variant to the active prefix and implements that arity.
32    (%impl_prefixes:
33        ($($active:ident : $active_idx:literal),*);
34        $next:ident : $idx:literal + $nth:literal : $suf:literal
35        $(, $rest:ident : $rest_idx:literal + $rest_nth:literal : $rest_suf:literal)*
36    ) => {
37        _data_value_impl_oneof!(%impl_prefix: $nth;
38            ($($active:$active_idx,)* $next:$idx);
39            ($($rest),*)
40        );
41        _data_value_impl_oneof!(%impl_prefixes:
42            ($($active:$active_idx,)* $next:$idx);
43            $($rest:$rest_idx+$rest_nth:$rest_suf),*
44        );
45    };
46    // Stops after the complete prefix.
47    (%impl_prefixes: ($($active:ident : $active_idx:literal),*);) => {};
48
49    // Implements functionality that depends on the active prefix length.
50    (%impl_prefix: $len:literal;
51        ($first:ident : $first_idx:literal $(, $T:ident : $idx:literal)*);
52        ($($inactive:ident),*)
53    ) => {
54        impl<$first: Default $(, $T)*> Default for Oneof<$len, $first $(, $T)*> {
55            fn default() -> Self { Self::_0($first::default()) }
56        }
57        impl<$first: crate::ConstInit $(, $T)*> crate::ConstInit for Oneof<$len, $first $(, $T)*> {
58            const INIT: Self = Self::_0($first::INIT);
59        }
60
61        impl<$first: Clone + Default $(, $T: Clone + Default)*> Oneof<$len, $first $(, $T)*> {
62            /// Returns a tuple with the active variant's inner value in its
63            /// corresponding position, `Default::default()` for the other
64            /// active positions, and `()` for inactive trailing positions.
65            pub fn into_tuple_defaults(self) -> (
66                $first,
67                $($T,)*
68                $(_data_value_impl_oneof!(%unit $inactive),)*
69            ) { $crate::paste! {
70                let index = self.variant_index();
71
72                (
73                    if $first_idx == index {
74                        self.clone().[<into $first>]().unwrap()
75                    } else {
76                        Default::default()
77                    },
78                    $(
79                        if $idx == index {
80                            self.clone().[<into $T>]().unwrap()
81                        } else {
82                            Default::default()
83                        },
84                    )*
85                    $(_data_value_impl_oneof!(%unit $inactive),)*
86                )
87            }}
88        }
89    };
90    // Expands to unit in either type or expression position.
91    (%unit $_T:ident) => { () };
92
93    /* helpers */
94
95    // hardcoded canonical list (12 variants)
96    (%canonical $($prefix:tt)*) => { _data_value_impl_oneof! { $($prefix)*
97        _0:0+1:"st", _1:1+2:"nd", _2:2+3:"rd", _3:3+4:"th", _4:4+5:"th", _5:5+6:"th",
98        _6:6+7:"th", _7:7+8:"th", _8:8+9:"th", _9:9+10:"th", _10:10+11:"th", _11:11+12:"th",
99    }};
100    // maps the canonical list to one of idents and forwards them to the prefixed arm.
101    (%map_ident $prefix:ident:
102     $($T:ident : $idx:literal + $nth:literal : $suf:literal),* $(,)?) => {
103        _data_value_impl_oneof!($prefix: $($T),*);
104    };
105    // maps the canonical list to one of idents:nth:suf and forwards them to the prefixed arm.
106    (%map_ident_nth_suf $prefix:ident:
107     $($T:ident : $idx:literal + $nth:literal : $suf:literal),* $(,)?) => {
108        _data_value_impl_oneof!($prefix: $($T:$nth:$suf),+);
109    };
110
111    /* main arms */
112
113    (define_enum: $($variant:ident : $nth:literal : $suf:literal),+) => { $crate::paste! {
114        #[doc = crate::_tags!(code)]
115        /// A generic, parameterized *enum* for expressing structured alternatives.
116        #[doc = crate::_doc_meta!{
117            location("data/value", enum Oneof),
118        }]
119        /// Omitted trailing variant types default to [`Infallible`][crate::Infallible],
120        /// making those variants uninhabited.
121        ///
122        /// The first `LEN` variants are expected to be active,
123        /// with all remaining variants set to `Infallible`.
124        #[non_exhaustive]
125        #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
126        pub enum Oneof<const LEN: usize, $($variant = $crate::Infallible),+> {
127            $(
128                #[doc = "The " $nth $suf " variant."]
129                $variant($variant)
130            ),+
131        }
132    }};
133    (
134    // Implements:
135    // - LEN
136    // - MAX_ARITY
137    // - variant_index
138    // - is_variant_index
139    //
140    // - validate
141    //
142    // - into_tuple_options
143    // - as_tuple_ref_options
144    methods_general: $($T:ident : $idx:literal + $nth:literal : $suf:literal),+) => {
145        /// # Structural methods.
146        impl<const LEN:usize,  $($T),+ > Oneof<LEN, $($T),+> {
147            /// The number of active leading variants.
148            pub const LEN: usize = {
149                assert![LEN <= Self::MAX_ARITY, "LEN must be <= MAX_ARITY"];
150                LEN
151            };
152            /// The maximum number of generic type parameters in this enum.
153            pub const MAX_ARITY: usize = $crate::ident_total!($($T),+);
154            // pub const MAX_ARITY: usize = $crate::capture_last![literal $($nth),+]; // BENCH
155
156            /// Returns the current variant index (0-based).
157            pub const fn variant_index(&self) -> usize {
158                match self { $( Oneof::$T(_) => $idx ),+ }
159            }
160            /// Checks whether the current variant is at `index` (0-based).
161            pub const fn is_variant_index(&self, index: usize) -> bool {
162                self.variant_index() == index
163            }
164        }
165        impl<const LEN: usize, $($T: 'static),+> Oneof<LEN, $($T),+> {
166            /// Validates that the first `LEN` variants are active
167            /// and that all remaining variants are `Infallible`.
168            #[allow(unused_assignments, reason = "wont be read in all cases")]
169            // WAIT const PartialEq for TypeId
170            pub fn validate() -> bool {
171                let mut active_count = 0;
172                let mut inactive_found = false;
173                $(
174                    if $crate::TypeId::of::<$T>() == $crate::TypeId::of::<$crate::Infallible>() {
175                        inactive_found = true;
176                    } else {
177                        if inactive_found { return false; }
178                        active_count += 1;
179                    }
180                )+
181                LEN == active_count
182            }
183        }
184        /// # Conversion methods.
185        // IMPROVE: add a const fn for T: Copy (into_tuple_copy_options / into_tuple_const_init)
186        impl<const LEN: usize, $($T: Clone),+ > Oneof<LEN, $($T),+> {
187            /// Returns a tuple with `Some(value)` for the active variant and `None` elsewhere.
188            pub fn into_tuple_options(self) -> ($(Option<$T>),+) { $crate::paste! {
189                let index = self.variant_index();
190                ( $(
191                    if $idx == index {
192                        self.clone().[<into $T>]()
193                    } else {
194                        None::<$T>
195                    }
196                ),+ )
197            }}
198        }
199        impl<const LEN: usize, $($T),+ > Oneof<LEN, $($T),+> {
200            /// Returns a tuple with `Some(&value)` for the active variant and `None` elsewhere.
201            // FUTURE: make the `()` types not wrapped in option.
202            pub const fn as_tuple_ref_options(&self) -> ($(Option<&$T>),+) { $crate::paste! {
203                ( $(
204                    if $idx == self.variant_index() {
205                        self.[<as_ref $T>]()
206                    } else {
207                        None::<&$T>
208                    }
209                ),+ )
210            }}
211        }
212    };
213    (
214    // Implements methods acting over individual fields.
215    methods_individ: $($T:ident : $idx:literal + $nth:literal : $suf:literal),+) => {
216        /// # Variant-specific methods.
217        impl<const LEN: usize, $($T),+> Oneof<LEN, $($T),+> {
218            // _data_value_impl_oneof!(methods_field_access: $($T),+);
219            _data_value_impl_oneof!(methods_field_access: $($T:$idx+$nth:$suf),+);
220            _data_value_impl_oneof!(methods_map: $($T),+);
221            // _data_value_impl_oneof!(methods_map: $($T:$idx+$nth),+);
222        }
223    };
224    (
225    // implements:
226    // - is_*
227    // - into_*
228    // - as_ref_*
229    // - as_mut_*
230    methods_field_access: $($T:ident : $idx:literal + $nth:literal : $suf:literal),+) => {
231        $( _data_value_impl_oneof! { %methods_field_access: $T : $idx + $nth : $suf} )+
232    };
233    (%methods_field_access: $T:ident : $idx:literal + $nth:literal : $suf:literal
234    ) => { $crate::paste! {
235        #[doc = "Returns `true` if there's a value in variant [`"
236            $T "`](#variant." $T ") (The " $nth $suf ")."]
237        pub const fn [<is $T>](&self) -> bool { matches!(self, Oneof::$T(_)) }
238
239        #[doc = "Returns a shared reference to the inner value in variant `" $T "`, if present."]
240        pub const fn [<as_ref $T>](&self) -> Option<&$T> {
241            $crate::is![let Self::$T($T) = self, Some($T), None]
242        }
243        #[doc = "Returns an exclusive reference to the inner value in variant`" $T "`, if present."]
244        pub const fn [<as_mut $T>](&mut self) -> Option<&mut $T> {
245            $crate::is![let Self::$T($T) = self, Some($T), None]
246        }
247        #[doc = "Returns a copy of the value in variant `" $T "`, if present."]
248        pub const fn [<copy $T >](self) -> Option<$T> where Self: Copy {
249            $crate::is![let Self::$T($T) = self, Some($T), None]
250        }
251        #[doc = "Returns the owned value in variant `" $T "`, if present."]
252        #[doc = "<hr/>"] // separator after the last method
253        pub fn [<into $T>](self) -> Option<$T> {
254            $crate::is![let Self::$T($T) = self, Some($T), None]
255        }
256    }};
257    (
258    // implements:
259    // - map_*
260    methods_map: $first:ident $(, $rest:ident)*) => {
261        // For the first variant, the `$before` list is empty.
262        _data_value_impl_oneof!(%methods_map: $first, (), ($($rest),*));
263        // Then, delegate to the helper macro with the first element as the accumulator.
264        _data_value_impl_oneof!(%methods_map_helper: ($first), ($($rest),*));
265
266        // NOTE: generates something like the following (e.g. for 6 variants):
267        //
268        // impl_map_method!(_0, (/*$before*/), (_1, _2, _3, _4, _5));
269        // impl_map_method!(_1, (_0), (_2, _3, _4, _5));
270        // impl_map_method!(_2, (_0, _1), (_3, _4, _5));
271        // impl_map_method!(_3, (_0, _1, _2), (_4, _5));
272        // impl_map_method!(_4, (_0, _1, _2, _3), (_5));
273        // impl_map_method!(_5, (_0, _1, _2, _3, _4), (/*$after*/));
274    };
275    (
276    %methods_map: $T:ident, ( $($before:ident),* ), ( $($after:ident),* )) => { $crate::paste! {
277        #[doc = "Transforms the inner value in variant`" $T
278        "` using `f`, leaving other variants unchanged."]
279        pub fn [<map $T>]<NEW>(self, f: impl FnOnce($T) -> NEW)
280            -> Oneof<LEN, $($before,)* NEW, $($after,)* > {
281            match self {
282                $( Self::$before(val) => Oneof::$before(val), )*
283                Self::$T(val) => Oneof::$T(f(val)),
284                $( Self::$after(val) => Oneof::$after(val), )*
285            }
286        }
287        // NOTE: Generates methods like the following (e.g. for variant _2, of 6 total):
288        //
289        // pub fn map_c<NEW>(self, f: impl FnOnce(_2) -> NEW) -> Oneof<_0, _1, NEW, _3, _4, _5> {
290        //     match self {
291        //         Self::_0(t) => Oneof::_0(t),    // $before
292        //         Self::_1(t) => Oneof::_1(t),    // …
293        //         Self::_2(t) => Oneof::_2(f(t)), // $T
294        //         Self::_3(t) => Oneof::_3(t),    // $after
295        //         Self::_4(t) => Oneof::_4(t),    // …
296        //         Self::_5(t) => Oneof::_5(t),    // …
297        //     }
298        // }
299    }};
300    // Stop when there are no types left in the `$after` list.
301    (%methods_map_helper: ($($before:ident),*), ()) => {};
302    // Recursively take the next type as the current one.
303    (%methods_map_helper: ($($before:ident),*), ($first:ident $(, $rest:ident)*)) => {
304        _data_value_impl_oneof!(%methods_map: $first, ($($before),*), ($($rest),*));
305        // Append the current type to the "before" list and continue.
306        _data_value_impl_oneof!(%methods_map_helper: ($($before,)* $first), ($($rest),*));
307    };
308}
