2//! Defines [`graph_adj!`].
3//

5#[doc = crate::_tags!(construction data_structure topol)]
6/// Defines a directed adjacency graph with static or allocating storage.
7#[doc = crate::_doc_meta!{
8    location("data/topol/graph", macro graph_adj),
9}]
10///
11/// The graph stores connectivity only. Vertex and edge application
12/// data can be associated externally through the generated handles.
13///
14/// Edges are directed. Self-loops, parallel edges, and cycles are allowed.
15///
16/// New edges are prepended to each source vertex's adjacency chain, so outgoing
17/// edges are traversed in reverse insertion order. Edge handles themselves are
18/// assigned in global insertion order.
19///
20/// The vertex domain is fixed for the graph's lifetime: statically
21/// through the vertex count, or at construction for the allocating backend.
22///
23/// Edge handles remain stable until `clear()`. Clearing invalidates them
24/// contextually, later insertion may reuse the same handle values.
25///
26/// The initial representation is append-only:
27/// individual edges cannot be removed, while `clear()` removes all edges.
28///
29/// `static` uses fixed arrays and is the default. `alloc` uses dynamic storage.
30///
31/// # Examples
32/// ```
33/// use devela::{NonMaxU8, NonMaxU16, graph_adj};
34///
35/// graph_adj! {
36///     [
37///         vertex: u8 + NonMaxU8;
38///         edge: u16 + NonMaxU16;
39///     ]
40///     pub MyGraph;
41///     pub MyVertex;
42///     pub MyEdge;
43/// }
44///
45/// let mut graph = MyGraph::<4, 8>::new();
46/// let a = graph.vertex(0).unwrap();
47/// let b = graph.vertex(1).unwrap();
48/// let edge = graph.add_edge(a, b).unwrap();
49///
50/// assert_eq!(graph.edge_target(edge), Some(b));
51/// assert!(graph.has_edge(a, b));
52/// ```
53/// See also:
54/// - [`GraphAdjExample`], [`GraphAdjVertexExample`], [`GraphAdjEdgeExample`],
55/// - [`GraphAdjAllocExample`], [`GraphAdjAllocVertexExample`], [`GraphAdjAllocEdgeExample`].
56///
57/// [`GraphAdjExample`]: crate::GraphAdjExample
58/// [`GraphAdjVertexExample`]: crate::GraphAdjVertexExample
59/// [`GraphAdjEdgeExample`]: crate::GraphAdjEdgeExample
60/// [`GraphAdjAllocExample`]: crate::GraphAdjAllocExample
61/// [`GraphAdjAllocVertexExample`]: crate::GraphAdjAllocVertexExample
62/// [`GraphAdjAllocEdgeExample`]: crate::GraphAdjAllocEdgeExample
63#[macro_export]
64#[cfg_attr(cargo_primary_package, doc(hidden))]
65macro_rules! graph_adj· {
66    (
67        [
68            vertex: $vprim:ident $(+ $VertexIndex:ty)?;
69            edge: $eprim:ident $(+ $EdgeIndex:ty)?;
70        ]
71
72        $(#[$graph_attr:meta])*
73        $vis:vis $Graph:ident $( : $kind:ident)?;
74
75        $(#[$vertex_attr:meta])*
76        $vvis:vis $Vertex:ident;
77
78        $(#[$edge_attr:meta])*
79        $evis:vis $Edge:ident $(;)?
80    ) => {
81        $crate::graph_adj! { %normalize_vertex
82            [kind: $($kind)?]
83            [vertex: $vprim $(+ $VertexIndex)?]
84            [edge: $eprim $(+ $EdgeIndex)?]
85            [graph: $(#[$graph_attr])* $vis $Graph]
86            [vertex_handle: $(#[$vertex_attr])* $vvis $Vertex]
87            [edge_handle: $(#[$edge_attr])* $evis $Edge]
88        }
89    };
90
91    /* normalize representations */
92
93    (%normalize_vertex
94        [kind: $($kind:ident)?]
95        [vertex: $vprim:ident]
96        $($rest:tt)*
97    ) => {
98        $crate::graph_adj! {
99            %normalize_edge
100            [kind: $($kind)?]
101            [vertex: $vprim + $vprim]
102            $($rest)*
103        }
104    };
105    (%normalize_vertex
106        [kind: $($kind:ident)?]
107        [vertex: $vprim:ident + $VertexIndex:ty]
108        $($rest:tt)*
109    ) => {
110        $crate::graph_adj! {
111            %normalize_edge
112            [kind: $($kind)?]
113            [vertex: $vprim + $VertexIndex]
114            $($rest)*
115        }
116    };
117
118    (%normalize_edge
119        [kind: $($kind:ident)?]
120        [vertex: $vprim:ident + $VertexIndex:ty]
121        [edge: $eprim:ident]
122        $($rest:tt)*
123    ) => {
124        $crate::graph_adj! {
125            %generate
126            [kind: $($kind)?]
127            [vertex: $vprim + $VertexIndex]
128            [edge: $eprim + $eprim]
129            $($rest)*
130        }
131    };
132    (%normalize_edge
133        [kind: $($kind:ident)?]
134        [vertex: $vprim:ident + $VertexIndex:ty]
135        [edge: $eprim:ident + $EdgeIndex:ty]
136        $($rest:tt)*
137    ) => {
138        $crate::graph_adj! {
139            %generate
140            [kind: $($kind)?]
141            [vertex: $vprim + $VertexIndex]
142            [edge: $eprim + $EdgeIndex]
143            $($rest)*
144        }
145    };
146
147    /* generate shared handle family */
148
149    (%generate
150        [kind: $($kind:ident)?]
151        [vertex: $vprim:ident + $VertexIndex:ty]
152        [edge: $eprim:ident + $EdgeIndex:ty]
153        [graph: $(#[$graph_attr:meta])* $vis:vis $Graph:ident]
154        [vertex_handle: $(#[$vertex_attr:meta])* $vvis:vis $Vertex:ident]
155        [edge_handle: $(#[$edge_attr:meta])* $evis:vis $Edge:ident]
156    ) => {
157        $crate::handle! {
158            [index: $vprim + $VertexIndex;]
159            $(#[$vertex_attr])*
160            $vvis $Vertex;
161        }
162        $crate::handle! {
163            [index: $eprim + $EdgeIndex;]
164            $(#[$edge_attr])*
165            $evis $Edge;
166        }
167
168        $crate::graph_adj! { %backend
169            [kind: $($kind)?]
170            [vertex: $vprim + $VertexIndex]
171            [edge: $eprim + $EdgeIndex]
172            [graph: $(#[$graph_attr])* $vis $Graph]
173            [vertex_handle: $vvis $Vertex]
174            [edge_handle: $evis $Edge]
175        }
176    };
177
178    /* backend dispatch */
179
180    (%backend
181        [kind:]
182        $($rest:tt)*
183    ) => {
184        $crate::graph_adj! { %backend [kind: static] $($rest)* }
185    };
186
187    (%backend
188        [kind: static]
189        [vertex: $vprim:ident + $VertexIndex:ty]
190        [edge: $eprim:ident + $EdgeIndex:ty]
191        [graph: $(#[$graph_attr:meta])* $vis:vis $Graph:ident]
192        [vertex_handle: $vvis:vis $Vertex:ident]
193        [edge_handle: $evis:vis $Edge:ident]
194    ) => {
195        $crate::__graph_adj_impl_array! {
196            [vertex: $vprim + $VertexIndex;]
197            [edge: $eprim + $EdgeIndex;]
198            $(#[$graph_attr])* $vis $Graph;
199            $vvis $Vertex;
200            $evis $Edge;
201        }
202    };
203
204    (%backend
205        [kind: alloc]
206        [vertex: $vprim:ident + $VertexIndex:ty]
207        [edge: $eprim:ident + $EdgeIndex:ty]
208        [graph: $(#[$graph_attr:meta])* $vis:vis $Graph:ident]
209        [vertex_handle: $vvis:vis $Vertex:ident]
210        [edge_handle: $evis:vis $Edge:ident]
211    ) => {
212        $crate::__graph_adj_impl_vec! {
213            [vertex: $vprim + $VertexIndex;]
214            [edge: $eprim + $EdgeIndex;]
215            $(#[$graph_attr])* $vis $Graph;
216            $vvis $Vertex;
217            $evis $Edge;
218        }
219    };
220}
221#[doc(inline)]
222pub use graph_adj· as graph_adj;
