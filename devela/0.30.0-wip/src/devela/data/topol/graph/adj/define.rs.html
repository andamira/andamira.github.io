3//! Defines [`graph_adj!`].
4//
5
6#[doc = crate::_tags!(construction data_structure topol)]
7/// Defines a directed adjacency graph with static or allocating storage.
8#[doc = crate::_doc_meta!{
9    location("data/topol/graph", macro graph_adj),
10}]
11///
12/// The graph stores connectivity only. Vertex and edge application
13/// data can be associated externally through the generated handles.
14///
15/// Edges are directed. Self-loops, parallel edges, and cycles are allowed.
16///
17/// New edges are prepended to each source vertex's adjacency chain, so outgoing
18/// edges are traversed in reverse insertion order. Edge handles themselves are
19/// assigned in global insertion order.
20///
21/// The vertex domain is fixed for the graph's lifetime: statically
22/// through the vertex count, or at construction for the allocating backend.
23///
24/// Edge handles remain stable until `clear()`. Clearing invalidates them
25/// contextually, later insertion may reuse the same handle values.
26///
27/// The initial representation is append-only:
28/// individual edges cannot be removed, while `clear()` removes all edges.
29///
30/// `static` uses fixed arrays and is the default. `alloc` uses dynamic storage.
31///
32/// # Examples
33/// ```
34/// use devela::{NonMaxU8, NonMaxU16, graph_adj};
35///
36/// graph_adj! {
37///     [
38///         vertex: u8 + NonMaxU8;
39///         edge: u16 + NonMaxU16;
40///     ]
41///     pub MyGraph;
42///     pub MyVertex;
43///     pub MyEdge;
44/// }
45///
46/// let mut graph = MyGraph::<4, 8>::new();
47/// let a = graph.vertex(0).unwrap();
48/// let b = graph.vertex(1).unwrap();
49/// let edge = graph.add_edge(a, b).unwrap();
50///
51/// assert_eq!(graph.edge_target(edge), Some(b));
52/// assert!(graph.has_edge(a, b));
53/// ```
54/// See also:
55/// - [`GraphAdjExample`], [`GraphAdjVertexExample`], [`GraphAdjEdgeExample`],
56/// - [`GraphAdjAllocExample`], [`GraphAdjAllocVertexExample`], [`GraphAdjAllocEdgeExample`].
57///
58/// [`GraphAdjExample`]: crate::GraphAdjExample
59/// [`GraphAdjVertexExample`]: crate::GraphAdjVertexExample
60/// [`GraphAdjEdgeExample`]: crate::GraphAdjEdgeExample
61/// [`GraphAdjAllocExample`]: crate::GraphAdjAllocExample
62/// [`GraphAdjAllocVertexExample`]: crate::GraphAdjAllocVertexExample
63/// [`GraphAdjAllocEdgeExample`]: crate::GraphAdjAllocEdgeExample
64#[macro_export]
65#[cfg_attr(cargo_primary_package, doc(hidden))]
66macro_rules! graph_adj· {
67    (
68        [
69            vertex: $vprim:ident $(+ $VertexIndex:ty)?;
70            edge: $eprim:ident $(+ $EdgeIndex:ty)?;
71        ]
72
73        $(#[$graph_attr:meta])*
74        $vis:vis $Graph:ident $( : $kind:ident)?;
75
76        $(#[$vertex_attr:meta])*
77        $vvis:vis $Vertex:ident;
78
79        $(#[$edge_attr:meta])*
80        $evis:vis $Edge:ident $(;)?
81    ) => {
82        $crate::graph_adj! { %normalize_vertex
83            [kind: $($kind)?]
84            [vertex: $vprim $(+ $VertexIndex)?]
85            [edge: $eprim $(+ $EdgeIndex)?]
86            [graph: $(#[$graph_attr])* $vis $Graph]
87            [vertex_handle: $(#[$vertex_attr])* $vvis $Vertex]
88            [edge_handle: $(#[$edge_attr])* $evis $Edge]
89        }
90    };
91
92    /* normalize representations */
93
94    (%normalize_vertex
95        [kind: $($kind:ident)?]
96        [vertex: $vprim:ident]
97        $($rest:tt)*
98    ) => {
99        $crate::graph_adj! {
100            %normalize_edge
101            [kind: $($kind)?]
102            [vertex: $vprim + $vprim]
103            $($rest)*
104        }
105    };
106    (%normalize_vertex
107        [kind: $($kind:ident)?]
108        [vertex: $vprim:ident + $VertexIndex:ty]
109        $($rest:tt)*
110    ) => {
111        $crate::graph_adj! {
112            %normalize_edge
113            [kind: $($kind)?]
114            [vertex: $vprim + $VertexIndex]
115            $($rest)*
116        }
117    };
118
119    (%normalize_edge
120        [kind: $($kind:ident)?]
121        [vertex: $vprim:ident + $VertexIndex:ty]
122        [edge: $eprim:ident]
123        $($rest:tt)*
124    ) => {
125        $crate::graph_adj! {
126            %generate
127            [kind: $($kind)?]
128            [vertex: $vprim + $VertexIndex]
129            [edge: $eprim + $eprim]
130            $($rest)*
131        }
132    };
133    (%normalize_edge
134        [kind: $($kind:ident)?]
135        [vertex: $vprim:ident + $VertexIndex:ty]
136        [edge: $eprim:ident + $EdgeIndex:ty]
137        $($rest:tt)*
138    ) => {
139        $crate::graph_adj! {
140            %generate
141            [kind: $($kind)?]
142            [vertex: $vprim + $VertexIndex]
143            [edge: $eprim + $EdgeIndex]
144            $($rest)*
145        }
146    };
147
148    /* generate shared handle family */
149
150    (%generate
151        [kind: $($kind:ident)?]
152        [vertex: $vprim:ident + $VertexIndex:ty]
153        [edge: $eprim:ident + $EdgeIndex:ty]
154        [graph: $(#[$graph_attr:meta])* $vis:vis $Graph:ident]
155        [vertex_handle: $(#[$vertex_attr:meta])* $vvis:vis $Vertex:ident]
156        [edge_handle: $(#[$edge_attr:meta])* $evis:vis $Edge:ident]
157    ) => {
158        $crate::handle! {
159            [index: $vprim + $VertexIndex;]
160            $(#[$vertex_attr])*
161            $vvis $Vertex;
162        }
163        $crate::handle! {
164            [index: $eprim + $EdgeIndex;]
165            $(#[$edge_attr])*
166            $evis $Edge;
167        }
168
169        $crate::graph_adj! { %backend
170            [kind: $($kind)?]
171            [vertex: $vprim + $VertexIndex]
172            [edge: $eprim + $EdgeIndex]
173            [graph: $(#[$graph_attr])* $vis $Graph]
174            [vertex_handle: $vvis $Vertex]
175            [edge_handle: $evis $Edge]
176        }
177    };
178
179    /* backend dispatch */
180
181    (%backend
182        [kind:]
183        $($rest:tt)*
184    ) => {
185        $crate::graph_adj! { %backend [kind: static] $($rest)* }
186    };
187
188    (%backend
189        [kind: static]
190        [vertex: $vprim:ident + $VertexIndex:ty]
191        [edge: $eprim:ident + $EdgeIndex:ty]
192        [graph: $(#[$graph_attr:meta])* $vis:vis $Graph:ident]
193        [vertex_handle: $vvis:vis $Vertex:ident]
194        [edge_handle: $evis:vis $Edge:ident]
195    ) => {
196        $crate::__graph_adj_impl_array! {
197            [vertex: $vprim + $VertexIndex;]
198            [edge: $eprim + $EdgeIndex;]
199            $(#[$graph_attr])* $vis $Graph;
200            $vvis $Vertex;
201            $evis $Edge;
202        }
203    };
204
205    (%backend
206        [kind: alloc]
207        [vertex: $vprim:ident + $VertexIndex:ty]
208        [edge: $eprim:ident + $EdgeIndex:ty]
209        [graph: $(#[$graph_attr:meta])* $vis:vis $Graph:ident]
210        [vertex_handle: $vvis:vis $Vertex:ident]
211        [edge_handle: $evis:vis $Edge:ident]
212    ) => {
213        $crate::__graph_adj_impl_vec! {
214            [vertex: $vprim + $VertexIndex;]
215            [edge: $eprim + $EdgeIndex;]
216            $(#[$graph_attr])* $vis $Graph;
217            $vvis $Vertex;
218            $evis $Edge;
219        }
220    };
221}
222#[doc(inline)]
223pub use graph_adj· as graph_adj;
