3//! Array implementations over fixed native arrays.

6use super::validate_storage_len;
7use crate::{Array, ArrayLayout, MismatchedCapacity, Slice};
8
9/// # Methods over a fixed native array.
10#[rustfmt::skip]
11impl<T, const LEN: usize, const RANK: usize> Array<[T; LEN], RANK> {
12    /* constructors */
13
14    /// Creates an owning array over the fixed native `storage`.
15    ///
16    /// Extra backing elements are permitted.
17    ///
18    /// # Errors
19    /// Returns [`MismatchedCapacity`] if `storage` does not cover
20    /// every physical position addressed by `layout`.
21    pub fn try_from_array(
22        storage: [T; LEN],
23        layout: ArrayLayout<RANK>,
24    ) -> Result<Self, MismatchedCapacity> {
25        match validate_storage_len(LEN, layout) {
26            Ok(()) => Ok(Self { data: storage, layout }),
27            Err(error) => Err(error),
28        }
29    }
30    /// Creates an owning array over a fixed native backing array in const contexts.
31    ///
32    /// This is the `T: Copy` counterpart of [`try_from_array`][Self::try_from_array].
33    /// Extra backing elements are permitted.
34    ///
35    /// # Errors
36    /// Returns [`MismatchedCapacity`] if `storage` does not cover every
37    /// physical position addressed by `layout`.
38    pub const fn try_from_array_copy(
39        storage: [T; LEN],
40        layout: ArrayLayout<RANK>,
41    ) -> Result<Self, MismatchedCapacity> where T: Copy {
42        match validate_storage_len(LEN, layout) {
43            Ok(()) => Ok(Self { data: storage, layout }),
44            Err(error) => Err(error),
45        }
46    }
47
48    /* physical storage */
49
50    /// Returns the complete physical backing slice.
51    ///
52    /// This is not necessarily the array's logical ravel order.
53    pub const fn storage(&self) -> &[T] {
54        &self.data
55    }
56    /// Returns the complete exclusive physical backing slice.
57    pub const fn storage_mut(&mut self) -> &mut [T] {
58        &mut self.data
59    }
60    /// Returns exclusive access to the fixed native backing array.
61    ///
62    /// Unlike dynamically sized backing storage,
63    /// exposing this value cannot change the physical storage length.
64    pub const fn data_mut(&mut self) -> &mut [T; LEN] {
65        &mut self.data
66    }
67    /// Returns the backing-storage length.
68    pub const fn storage_len(&self) -> usize {
69        LEN
70    }
71
72    /* logical access */
73
74    /// Returns a shared reference to the element at `coord`.
75    ///
76    /// Returns `None` if the coordinate is outside the logical shape.
77    pub const fn get(&self, coord: [usize; RANK]) -> Option<&T> {
78        match self.layout.storage_index(coord) {
79            Some(index) => Slice::get(&self.data, index),
80            None => None,
81        }
82    }
83    /// Returns an exclusive reference to the element at `coord`.
84    ///
85    /// Returns `None` if the coordinate is outside the logical shape.
86    pub const fn get_mut(&mut self, coord: [usize; RANK]) -> Option<&mut T> {
87        match self.layout.storage_index(coord) {
88            Some(index) => Slice::get_mut(&mut self.data, index),
89            None => None,
90        }
91    }
92
93    /* reborrowing */
94
95    /// Returns a shared slice-backed array reborrowed from this array.
96    pub const fn reborrow(&self) -> Array<&[T], RANK> {
97        Array {
98            data: &self.data,
99            layout: self.layout,
100        }
101    }
102    /// Returns an exclusive slice-backed array reborrowed from this array.
103    pub const fn reborrow_mut(&mut self) -> Array<&mut [T], RANK> {
104        Array {
105            data: &mut self.data,
106            layout: self.layout,
107        }
108    }
109}

111#[cfg(test)]
112mod _test {
113    use super::*;
114    use crate::{ArrayShape, StringU8, const_assert};
115
116    const SHAPE: ArrayShape<2> = ArrayShape::new([2, 3]);
117    const LAYOUT: ArrayLayout<2> = match ArrayLayout::dense_last(SHAPE) {
118        Ok(layout) => layout,
119        Err(_) => panic!("unexpected layout overflow"),
120    };
121    const FIXED: Array<[u8; 6], 2> = match Array::try_from_array_copy([0, 1, 2, 3, 4, 5], LAYOUT) {
122        Ok(array) => array,
123        Err(_) => panic!("insufficient storage"),
124    };
125    const FIXED_VALUE: Option<u8> = FIXED.get([1, 2]).copied();
126    const MUTATED_DATA: [u8; 6] = const_mutated_data();
127    const fn const_mutated_data() -> [u8; 6] {
128        let mut array = match Array::try_from_array_copy([0, 1, 2, 3, 4, 5], LAYOUT) {
129            Ok(array) => array,
130            Err(_) => panic!("insufficient storage"),
131        };
132        match array.get_mut([1, 2]) {
133            Some(value) => *value = 9,
134            None => panic!("valid coordinate rejected"),
135        }
136        *array.data()
137    }
138
139    #[test]
140    const fn const_shared_access() {
141        const_assert!(eq FIXED_VALUE.unwrap(), 5);
142    }
143    #[test]
144    const fn const_exclusive_access() {
145        const_assert!(eq_buf & MUTATED_DATA, &[0u8, 1, 2, 3, 4, 9]);
146    }
147    #[test]
148    fn constructor_infers_fixed_backing() {
149        let array = Array::try_from_array([0, 1, 2, 3, 4, 5], LAYOUT).unwrap();
150        let _: Array<[u8; 6], 2> = array;
151    }
152    #[test]
153    fn fixed_array_queries() {
154        let array = Array::try_from_array([0, 1, 2, 3, 4, 5, 6, 7], LAYOUT).unwrap();
155        assert_eq!(array.rank(), 2);
156        assert_eq!(array.shape(), SHAPE);
157        assert_eq!(array.layout(), LAYOUT);
158        assert_eq!(array.element_count(), 6);
159        assert!(!array.is_empty());
160        assert_eq!(array.storage_len(), 8);
161        assert_eq!(array.storage(), &[0, 1, 2, 3, 4, 5, 6, 7]);
162        assert_eq!(array.data(), &[0, 1, 2, 3, 4, 5, 6, 7]);
163        assert_eq!(array.get([0, 0]).copied(), Some(0));
164        assert_eq!(array.get([0, 2]).copied(), Some(2));
165        assert_eq!(array.get([1, 0]).copied(), Some(3));
166        assert_eq!(array.get([1, 2]).copied(), Some(5));
167        assert_eq!(array.get([2, 0]), None);
168        assert_eq!(array.get([0, 3]), None);
169    }
170    #[test]
171    fn rejects_short_fixed_storage() {
172        let error = Array::try_from_array([0, 1, 2, 3, 4], LAYOUT).unwrap_err();
173        assert_eq!(error, MismatchedCapacity::too_small(5, 6));
174        assert_eq!(error.missing(), Some(1));
175    }
176    #[test]
177    fn exposes_safe_mutable_fixed_backing() {
178        let mut array = Array::try_from_array([0, 1, 2, 3, 4, 5], LAYOUT).unwrap();
179        array.data_mut()[0] = 8;
180        array.storage_mut()[1] = 7;
181        *array.get_mut([1, 2]).unwrap() = 9;
182        assert_eq!(array.into_data(), [8, 7, 2, 3, 4, 9]);
183    }
184    #[test]
185    fn reborrows_fixed_storage() {
186        let mut array = Array::try_from_array([0, 1, 2, 3, 4, 5], LAYOUT).unwrap();
187        {
188            let shared = array.reborrow();
189            let _: Array<&[u8], 2> = shared;
190            assert_eq!(shared.get([1, 2]).copied(), Some(5));
191        }
192        {
193            let mut exclusive = array.reborrow_mut();
194            let _: &mut Array<&mut [u8], 2> = &mut exclusive;
195            *exclusive.get_mut([0, 0]).unwrap() = 9;
196        }
197        assert_eq!(array.get([0, 0]).copied(), Some(9));
198    }
199    #[test]
200    fn shared_access_does_not_require_copy() {
201        #[derive(Debug, PartialEq, Eq)]
202        struct Token(u8);
203        let layout = ArrayLayout::dense_last(ArrayShape::new([2])).unwrap();
204        let array = Array::try_from_array([Token(7), Token(8)], layout).unwrap();
205        assert_eq!(array.get([0]), Some(&Token(7)));
206        assert_eq!(array.get([1]), Some(&Token(8)));
207    }
208    #[test]
209    fn ownership_recovery_preserves_non_copy_values() {
210        let layout = ArrayLayout::dense_last(ArrayShape::new([2])).unwrap();
211        let array = Array::try_from_array(
212            [StringU8::<5>::from_str("left").unwrap(), StringU8::from_str("right").unwrap()],
213            layout,
214        )
215        .unwrap();
216        let (storage, recovered_layout) = array.into_parts();
217        assert_eq!(storage[0], "left");
218        assert_eq!(storage[1], "right");
219        assert_eq!(recovered_layout, layout);
220    }
221    #[test]
222    fn scalar_fixed_array() {
223        let layout = ArrayLayout::dense_last(ArrayShape::<0>::new([])).unwrap();
224        let array = Array::try_from_array([42], layout).unwrap();
225        assert_eq!(array.element_count(), 1);
226        assert_eq!(array.get([]).copied(), Some(42));
227    }
228    #[test]
229    fn empty_fixed_array() {
230        let layout = ArrayLayout::dense_last(ArrayShape::new([2, 0, 3])).unwrap();
231        let array = Array::<[u8; 0], 3>::try_from_array([], layout).unwrap();
232        assert!(array.is_empty());
233        assert_eq!(array.storage_len(), 0);
234        assert_eq!(array.get([0, 0, 0]), None);
235    }
236}
