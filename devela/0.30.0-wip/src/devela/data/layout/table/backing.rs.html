20const fn row_iter_parts(layout: TableLayout, row: usize) -> Option<(usize, usize, usize)> {
21    is! { row >= layout.rows(), return None }
22    let count = layout.columns();
23    // A row still exists in a rows×0 table; it simply contains no cells.
24    is! { count == 0, return Some((0, 0, 1)) }
25    let start = unwrap![some_expect layout.storage_index(TableCoord::new(row, 0)),
26        "valid table row has no storage position"];
27    let stride = layout.column_stride();
28    assert!(stride > 0, "non-empty table row has non-positive stride");
29    Some((start, count, stride as usize))
30}
31const fn column_iter_parts(layout: TableLayout, column: usize) -> Option<(usize, usize, usize)> {
32    is! { column >= layout.columns(), return None }
33    let count = layout.rows();
34    // A column still exists in a 0×columns table.
35    is! { count == 0, return Some((0, 0, 1)) }
36    let start = unwrap![some_expect layout.storage_index(TableCoord::new(0, column)),
37        "valid table column has no storage position"];
38    let stride = layout.row_stride();
39    assert!(stride > 0, "non-empty table column has non-positive stride");
40    Some((start, count, stride as usize))
41}
42const fn row_iter_ref<'a, T>(
43    storage: &'a [T],
44    layout: TableLayout,
45    row: usize,
46) -> Option<StridedIter<'a, T>> {
47    unwrap![=some_map row_iter_parts(layout, row),
48        |(start, count, stride)| StridedIter::from_count(storage, start, count, stride)]
49}
50const fn column_iter_ref<'a, T>(
51    storage: &'a [T],
52    layout: TableLayout,
53    column: usize,
54) -> Option<StridedIter<'a, T>> {
55    unwrap![=some_map column_iter_parts(layout, column),
56        |(start, count, stride)| StridedIter::from_count(storage, start, count, stride)]
57}
58const fn row_iter_mut<'a, T>(
59    storage: &'a mut [T],
60    layout: TableLayout,
61    row: usize,
62) -> Option<StridedIterMut<'a, T>> {
63    unwrap![=some_map row_iter_parts(layout, row),
64        |(start, count, stride)| StridedIterMut::from_count(storage, start, count, stride)]
65}
66const fn column_iter_mut<'a, T>(
67    storage: &'a mut [T],

71    unwrap![=some_map column_iter_parts(layout, column),
72        |(start, count, stride)| StridedIterMut::from_count(storage, start, count, stride)]
73}

77#[rustfmt::skip]
78impl<T, const LEN: usize> Table<[T; LEN]> {
79    /* construction */
80
81    /// Creates an owning table over fixed-array `storage`.
82    ///
83    /// Extra backing elements are permitted.
84    ///
85    /// # Errors
86    /// Returns [`MismatchedCapacity`] if `storage` does not cover every
87    /// physical position addressed by `layout`.
88    pub fn try_from_array(storage: [T; LEN], layout: TableLayout)
89        -> Result<Self, MismatchedCapacity> {
90        match Array::try_from_array(storage, layout.as_array()) {
91            Ok(array) => Ok(Self::from_array(array)),
92            Err(error) => Err(error),
93        }
94    }
95
96    /// Creates an owning table over a fixed native backing array in const contexts.
97    ///
98    /// This is the `T: Copy` counterpart of [`try_from_array`][Self::try_from_array].
99    ///
100    /// # Errors
101    /// Returns [`MismatchedCapacity`] if `storage` does not cover every
102    /// physical position addressed by `layout`.
103    pub const fn try_from_array_copy(storage: [T; LEN], layout: TableLayout)
104        -> Result<Self, MismatchedCapacity> where T: Copy {
105        match Array::try_from_array_copy(storage, layout.as_array()) {
106            Ok(array) => Ok(Self::from_array(array)),
107            Err(error) => Err(error),
108        }
109    }
110
111    /* physical storage */
112
113    /// Returns the complete physical backing slice.
114    ///
115    /// Physical storage order may differ from logical table order.
116    pub const fn storage(&self) -> &[T] { self.array.storage() }
117
118    /// Returns the complete exclusive physical backing slice.
119    pub const fn storage_mut(&mut self) -> &mut [T] { self.array.storage_mut() }
120
121    /// Returns exclusive access to the fixed native backing array.
122    pub const fn data_mut(&mut self) -> &mut [T; LEN] { self.array.data_mut() }
123
124    /// Returns the backing-storage length.
125    pub const fn storage_len(&self) -> usize { self.array.storage_len() }
126
127    /* logical access */
128
129    /// Returns a shared reference to the cell at `coord`.
130    pub const fn get(&self, coord: TableCoord) -> Option<&T> {
131        self.array.get(coord.as_array())
132    }
133    /// Returns an exclusive reference to the cell at `coord`.
134    pub const fn get_mut(&mut self, coord: TableCoord) -> Option<&mut T> {
135        self.array.get_mut(coord.as_array())
136    }
137
138    /* reborrowing */
139
140    /// Returns a shared slice-backed table reborrowed from this table.
141    pub const fn reborrow(&self) -> Table<&[T]> {
142        Table::from_array(self.array.reborrow())
143    }
144    /// Returns an exclusive slice-backed table reborrowed from this table.
145    pub const fn reborrow_mut(&mut self) -> Table<&mut [T]> {
146        Table::from_array(self.array.reborrow_mut())
147    }
148
149    /* logical traversal */
150
151    /// Returns an iterator over the cells of `row`.
152    ///
153    /// Returns `None` if `row` is out of bounds.
154    pub const fn row_iter(&self, row: usize) -> Option<StridedIter<'_, T>> {
155        row_iter_ref(self.storage(), self.layout(), row)
156    }
157    /// Returns a mutable lending iterator over the cells of `row`.
158    ///
159    /// Returns `None` if `row` is out of bounds.
160    pub const fn row_iter_mut(&mut self, row: usize) -> Option<StridedIterMut<'_, T>> {
161        let layout = self.layout();
162        row_iter_mut(self.storage_mut(), layout, row)
163    }
164    /// Returns an iterator over the cells of `column`.
165    ///
166    /// Returns `None` if `column` is out of bounds.
167    pub const fn column_iter(&self, column: usize) -> Option<StridedIter<'_, T>> {
168        column_iter_ref(self.storage(), self.layout(), column)
169    }
170    /// Returns a mutable lending iterator over the cells of `column`.
171    ///
172    /// Returns `None` if `column` is out of bounds.
173    pub const fn column_iter_mut(&mut self, column: usize) -> Option<StridedIterMut<'_, T>> {
174        let layout = self.layout();
175        column_iter_mut(self.storage_mut(), layout, column)
176    }
177}
