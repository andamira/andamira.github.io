4#[macro_export]
5macro_rules! __intern_string_impl_vec· {
6    (
7        [index: $iprim:ident + $Index:ty;]
8        [cursor: $cprim:ident + $Cursor:ty;]
9        [arena: $StringArena:ident;]
10
11        $(#[$interner_attr:meta])*
12        $vis:vis $Interner:ident;
13
14        $svis:vis $Symbol:ident;
15    ) => {
16        /* packed canonical strings */
17
18        $crate::__arena_string_impl_vec! {
19            [index: $iprim + $Index;]
20            [cursor: $cprim + $Cursor;]
21
22            $StringArena;
23            $Symbol;
24            [mark:]
25        }
26
27        /* interner */
28
29        $(#[$interner_attr])*
30        #[derive(Clone, Debug)]
31        $vis struct $Interner {
32            strings: $StringArena,
33
34            /// Active open-addressed lookup slots.
35            ///
36            /// `EMPTY` represents vacancy; otherwise the value is a
37            /// canonical-string index.
38            slots: $crate::Vec<$crate::MaybeNiche<$Index>>,
39        }
40
41        impl Default for $Interner {
42            fn default() -> Self { Self::new() }
43        }
44        impl $crate::ConstInit for $Interner {
45            const INIT: Self = Self::new();
46        }
47
48        #[allow(dead_code)]
49        impl $Interner {
50            /// Slot value representing an unoccupied probe position.
51            const EMPTY: $crate::MaybeNiche<$Index> = $crate::MaybeNiche::<$Index>::MAX;
52
53            /// Maximum number of canonical strings supported by the
54            /// symbol representation.
55            ///
56            /// One representable index value is reserved for `EMPTY`.
57            $vis const MAX_CAPACITY: usize = {
58                $crate::unwrap![ok_or $crate::MaybeNiche::<$Index>::MAX.try_to_usize(), usize::MAX]
59            };
60            /// Maximum packed-byte length supported by the cursor representation.
61            $vis const MAX_BYTE_CAPACITY: usize = $StringArena::MAX_BYTE_CAPACITY;
62
63            /* construction */
64
65            /// Creates an empty allocating string interner.
66            #[must_use]
67            $vis const fn new() -> Self {
68                Self { strings: $StringArena::new(), slots: $crate::Vec::new() }
69            }
70
71            /// Creates an empty interner with initial string, byte, and
72            /// lookup-slot capacities.
73            ///
74            /// The capacities describe initially reserved storage and do not
75            /// impose fixed limits. Lookup slots grow and rehash as necessary.
76            ///
77            /// # Panics
78            /// Panics if the requested string or byte capacity exceeds its
79            /// configured representation, or if allocation fails.
80            #[must_use]
81            $vis fn with_capacity(
82                string_capacity: usize,
83                byte_capacity: usize,
84                slot_capacity: usize,
85            ) -> Self {
86                assert!(string_capacity <= Self::MAX_CAPACITY,
87                    "intern_string! string capacity exceeds its symbol representation");
88                let strings = $StringArena::with_capacity(string_capacity, byte_capacity);
89                let mut slots = $crate::Vec::with_capacity(slot_capacity);
90                slots.resize(slot_capacity, Self::EMPTY);
91                Self { strings, slots }
92            }
93
94            /* canonical-string capacity */
95
96            /// Returns the canonical-string capacity available without reallocating.
97            #[must_use]
98            $vis fn capacity(&self) -> usize {
99                ::core::cmp::min(self.strings.capacity(), Self::MAX_CAPACITY)
100            }
101            /// Returns the number of canonical strings.
102            #[must_use]
103            $vis const fn len(&self) -> usize { self.strings.len() }
104            /// Returns whether no strings have been interned.
105            #[must_use]
106            $vis const fn is_empty(&self) -> bool { self.strings.is_empty() }
107
108            /* lookup capacity */
109
110            /// Returns the current number of active lookup slots.
111            #[must_use]
112            $vis const fn slot_capacity(&self) -> usize { self.slots.len() }
113
114            /* byte capacity */
115
116            /// Returns the packed-byte capacity available without reallocating.
117            #[must_use]
118            $vis fn byte_capacity(&self) -> usize {
119                self.strings.byte_capacity()
120            }
121            /// Returns the packed UTF-8 byte length.
122            #[must_use]
123            $vis const fn byte_len(&self) -> usize {
124                self.strings.byte_len()
125            }
126            /// Returns how many packed bytes fit without reallocating.
127            #[must_use]
128            $vis fn byte_remaining(&self) -> usize {
129                self.strings.byte_remaining()
130            }
131            /// Returns whether `string` can be represented by this interner.
132            ///
133            /// An already canonical string is always accepted.
134            ///
135            /// For a new string this checks the symbol and byte coordinate
136            /// representations. Lookup storage may grow and rehash.
137            #[must_use]
138            $vis fn can_intern(&self, string: &str) -> bool {
139                if self.find(string).is_some() { true } else { self._can_insert_new(string) }
140            }
141
142            /* access */
143
144            /// Returns whether `symbol` currently resolves to a canonical string.
145            #[must_use]
146            $svis const fn contains(&self, symbol: $Symbol) -> bool {
147                self.strings.contains(symbol)
148            }
149            /// Returns the canonical string identified by `symbol`.
150            #[must_use]
151            $svis fn get(&self, symbol: $Symbol) -> Option<&str> {
152                self.strings.get(symbol)
153            }
154            /// Returns the canonical symbol for `string`, if already interned.
155            #[must_use]
156            $svis fn find(&self, string: &str) -> Option<$Symbol> {
157                self._probe(string).0
158            }
159            /// Returns all packed canonical UTF-8 bytes.
160            #[must_use]
161            $vis fn as_bytes(&self) -> &[u8] {
162                self.strings.as_bytes()
163            }
164
165            /* mutation */
166
167            /// Returns the canonical symbol for `string`, inserting it if absent.
168            ///
169            /// Equal strings always resolve to the same symbol.
170            ///
171            /// Lookup storage grows and rehashes when every active slot is
172            /// occupied. Existing symbols remain unchanged by rehashing.
173            ///
174            /// Returns `None` when a new canonical string cannot be represented.
175            $svis fn intern(&mut self, string: &str) -> Option<$Symbol> {
176                let (found, vacant) = self._probe(string);
177                if let Some(symbol) = found { return Some(symbol); }
178                // Check semantic representability before growing lookup storage.
179                if !self._can_insert_new(string) { return None; }
180                let slot = match vacant {
181                    Some(slot) => slot,
182                    None => {
183                        // No active slot exists, or the lookup table is full.
184                        if !self._grow_slots() { return None; }
185                        let (_, vacant) = self._probe(string);
186                        $crate::unwrap![some? vacant]
187                    }
188                };
189                // Establish a lookup destination before mutating packed storage.
190                let symbol = self.strings.insert(string)?;
191                self.slots[slot] = $crate::MaybeNiche::<$Index>::new(symbol.get_index());
192                Some(symbol)
193            }
194            /// Removes every canonical string and resets the lookup table.
195            ///
196            /// Allocated storage and the current number of lookup slots are retained.
197            ///
198            /// Previously issued symbols may resolve again after later interning.
199            $vis fn clear(&mut self) {
200                self.strings.clear();
201                self.slots.fill(Self::EMPTY);
202            }
203
204            /* iteration */
205
206            /// Iterates over canonical strings in insertion order.
207            $vis fn iter(&self) -> impl Iterator<Item = &str> + '_ {
208                self.strings.iter()
209            }
210            /// Iterates over canonical symbols in insertion order.
211            $svis fn symbols(&self) -> impl Iterator<Item = $Symbol> + '_ {
212                self.strings.ids()
213            }
214            /// Iterates over canonical symbols and strings in insertion order.
215            $svis fn entries(&self) -> impl Iterator<Item = ($Symbol, &str)> + '_ {
216                self.strings.entries()
217            }
218
219            /* private */
220
221            /// Returns whether one new canonical string can be represented.
222            fn _can_insert_new(&self, string: &str) -> bool {
223                self.len() < Self::MAX_CAPACITY && self.strings.can_insert(string)
224            }
225            /// Returns the matching symbol or first vacant lookup slot.
226            fn _probe(&self, string: &str) -> (Option<$Symbol>, Option<usize>) {
227                let nn = (None, None);
228                let slots = self.slots.len();
229                if slots == 0 { return nn; }
230                let mut slot = $crate::HasherFx::<usize>::hash_bytes(string.as_bytes()) % slots;
231                $crate::whilst! { probed in 0..slots; {
232                    let stored = self.slots[slot];
233                    if stored.get_prim() == Self::EMPTY.get_prim() { return (None, Some(slot)); }
234                    let index = $crate::unwrap![ok_or stored.try_to_usize(), return nn];
235                    let symbol = $crate::unwrap![ok_or <$Symbol>::try_from_usize(index), return nn];
236                    let candidate = $crate::unwrap![some_or self.strings.get(symbol), return nn];
237                    if $crate::Slice::<u8>::eq(candidate.as_bytes(), string.as_bytes()) {
238                        return (Some(symbol), None);
239                    }
240                    slot += 1;
241                    if slot == slots { slot = 0; }
242                }}
243                nn
244            }
245            /// Expands the active lookup table and rehashes all canonical strings.
246            fn _grow_slots(&mut self) -> bool {
247                let old = self.slots.len();
248                let new = $crate::is![old == 0, 1,
249                    $crate::unwrap![some_or old.checked_mul(2), return false]];
250                self._rehash(new);
251                true
252            }
253            /// Rebuilds the lookup table with `slot_count` active slots.
254            ///
255            /// Canonical strings and their symbols remain unchanged.
256            fn _rehash(&mut self, slot_count: usize) {
257                debug_assert!(slot_count >= self.len());
258                self.slots.resize(slot_count, Self::EMPTY);
259                self.slots.fill(Self::EMPTY);
260                for (symbol, string) in self.strings.entries() {
261                    let mut slot = $crate::HasherFx::<usize>::hash_bytes(string.as_bytes())
262                        % slot_count;
263                    loop {
264                        if self.slots[slot].get_prim() == Self::EMPTY.get_prim() {
265                            self.slots[slot] =
266                                $crate::MaybeNiche::<$Index>::new(symbol.get_index());
267                            break;
268                        }
269                        slot += 1;
270                        if slot == slot_count { slot = 0; }
271                    }
272                }
273            }
274        }
275    };
276}
277pub use __intern_string_impl_vec· as __intern_string_impl_vec;
