4#[macro_export]
5macro_rules! __arena_bytes_impl_array· {
6    (
7        [cursor: $cprim:ident]
8        [arena: $(#[$arena_attr:meta])* $vis:vis $Arena:ident]
9        [handle: $hvis:vis $Handle:ident]
10        [mark: $($mvis:vis $Mark:ident)?]
11        [internal: $_:ty]
12        [module: $module:ident]
13        ($_d:tt)
14    ) => {
15        $(#[$arena_attr])*
16        #[derive(Clone, Debug)]
17        $vis struct $Arena<const CAP: usize> {
18            data: [$crate::MaybeByte; CAP],
19
20            /// End of the initialized byte prefix.
21            len: $cprim,
22        }
23
24        impl<const CAP: usize> $crate::ConstInit for $Arena<CAP> {
25            const INIT: Self = Self::new();
26        }
27        impl<const CAP: usize> Default for $Arena<CAP> {
28            fn default() -> Self { Self::new() }
29        }
30        impl<const CAP: usize> Eq for $Arena<CAP> {}
31        impl<const CAP: usize> PartialEq for $Arena<CAP> {
32            fn eq(&self, other: &Self) -> bool { $Arena::eq(self, other) }
33        }
34
35        #[allow(dead_code, private_interfaces)]
36        impl<const CAP: usize> $Arena<CAP> {
37            /* private helpers */
38
39            /// Ensures every arena position can also be represented by its handle.
40            ///
41            /// `CAP` itself must be representable because a full arena has
42            /// `len == CAP`, and a span may cover the entire arena.
43            const _GUARD_CAPACITY: () = assert!(
44                <$Handle>::try_from_usize(CAP, CAP).is_ok(),
45                "arena_bytes! capacity exceeds its cursor or handle representation",
46            );
47
48            /// Returns the written byte length in the machine indexing domain.
49            const fn _len_usize(&self) -> usize { self.len as usize }
50
51            /// Resolves a handle into a validated half-open byte range.
52            const fn _span_usize(&self, h: $Handle) -> Option<(usize, usize)> {
53                let start = $crate::unwrap![ok_some? h.get_offset_usize()];
54                let len = $crate::unwrap![ok_some? h.get_len_usize()];
55                let end = $crate::unwrap![some? start.checked_add(len)];
56                if end > self._len_usize() { return None; }
57                Some((start, end))
58            }
59            /// Resolves a repeated handle span into a validated byte range.
60            const fn _span_repeat_usize(&self, h: $Handle, count: $cprim)
61                -> Option<(usize, usize)> {
62                let (start, first_end) = $crate::unwrap![some? self._span_usize(h)];
63                let len = first_end - start;
64                let total = $crate::unwrap![some? len.checked_mul(count as usize)];
65                let end = $crate::unwrap![some? start.checked_add(total)];
66                if end > self._len_usize() { return None; }
67                Some((start, end))
68            }
69
70            /* construction */
71
72            /// Returns a new empty arena.
73            $vis const fn new() -> Self {
74                let () = Self::_GUARD_CAPACITY;
75                Self { data: <$_>::new_array(), len: 0 }
76            }
77
78            /* capacity */
79
80            /// Returns the total byte capacity.
81            $vis const fn capacity(&self) -> $cprim { CAP as $cprim }
82
83            /// Returns the occupied byte length.
84            #[must_use]
85            $vis const fn len(&self) -> $cprim { self.len }
86
87            /// Returns whether the arena contains no bytes.
88            #[must_use]
89            $vis const fn is_empty(&self) -> bool { self.len == 0 }
90
91            /// Returns whether no additional byte can be written.
92            #[must_use]
93            $vis const fn is_full(&self) -> bool { self.len == CAP as $cprim }
94
95            /// Returns the remaining byte capacity.
96            #[must_use]
97            $vis const fn remaining(&self) -> $cprim { CAP as $cprim - self.len }
98
99            /// Returns whether `n` additional bytes fit in the arena.
100            #[must_use]
101            $vis const fn can_write(&self, n: $cprim) -> bool {
102                let end = $crate::unwrap![some_or? self.len.checked_add(n), false];
103                end <= CAP as $cprim
104            }
105
106            /* snapshot and rollback */
107
108            $(
109                /// Returns a mark at the current byte frontier.
110                $mvis const fn mark(&self) -> $Mark { <$Mark>::new(self.len) }
111
112                /// Retracts the byte frontier to `mark`, reclaiming the written suffix.
113                ///
114                /// Returns `false` without changing the arena if `mark` lies ahead
115                /// of the current frontier.
116                ///
117                /// Handles whose spans extend beyond the new frontier no longer resolve.
118                /// This invalidation is not permanent: later writes may reuse their coordinates.
119                $mvis const fn rollback(&mut self, mark: $Mark) -> bool {
120                    if mark.0 <= self.len { self.len = mark.0; true } else { false }
121                }
122            )?
123
124            /* whole storage */
125
126            /// Returns a byte slice over all the written data.
127            $vis const fn as_bytes(&self) -> &[u8] {
128                <$_>::slice_bytes(&self.data, 0, self._len_usize())
129            }
130            /// Returns an exclusive byte slice over all the written data.
131            $vis const fn as_bytes_mut(&mut self) -> &mut [u8] {
132                let len = self._len_usize();
133                <$_>::slice_bytes_mut(&mut self.data, 0, len)
134            }
135            /// Compares two arenas for equality.
136            #[must_use]
137            $vis const fn eq(&self, other: &Self) -> bool {
138                $crate::Slice::<u8>::eq(self.as_bytes(), other.as_bytes())
139            }
140
141            /* byte spans */
142
143            /// Appends `len` bytes initialized to `byte`.
144            ///
145            /// # Errors
146            /// Returns `None` if the complete span
147            /// does not fit in the remaining arena capacity.
148            $hvis const fn push_filled(&mut self, len: usize, byte: u8) -> Option<$Handle> {
149                let start = self._len_usize();
150                let end = $crate::unwrap![some? start.checked_add(len)];
151                if end > CAP { return None; }
152                let handle = $crate::unwrap![ok_some? <$Handle>::try_from_usize(start, len)];
153                $crate::whilst! { i in 0..len; {
154                    <$_>::write_byte(&mut self.data, start + i, byte);
155                }}
156                self.len = end as $cprim;
157                Some(handle)
158            }
159            /// Appends `len` zero-initialized bytes.
160            ///
161            /// # Errors
162            /// Returns `None` if the complete span
163            /// does not fit in the remaining arena capacity.
164            $hvis const fn push_zeroed(&mut self, len: usize) -> Option<$Handle> {
165                self.push_filled(len, 0)
166            }
167            /// Writes a byte slice into the arena.
168            ///
169            /// # Errors
170            /// Returns `None` if the complete byte slice
171            /// does not fit in the remaining arena capacity.
172            $hvis const fn push_bytes(&mut self, bytes: &[u8]) -> Option<$Handle> {
173                let start = self._len_usize();
174                let end = $crate::unwrap![some? start.checked_add(bytes.len())];
175                if end > CAP { return None; }
176                let handle =
177                    $crate::unwrap![ok_some? <$Handle>::try_from_usize(start, bytes.len())];
178                $crate::whilst! { i in 0..bytes.len(); {
179                    <$_>::write_byte(&mut self.data, start + i, bytes[i]);
180                }}
181                // `_GUARD_CAPACITY` proves every value through CAP fits `$cprim`.
182                self.len = end as $cprim;
183                Some(handle)
184            }
185            /// Returns the bytes described by `handle`.
186            $hvis const fn read_bytes(&self, handle: $Handle) -> Option<&[u8]> {
187                let (start, end) = $crate::unwrap![some? self._span_usize(handle)];
188                Some(<$_>::slice_bytes(&self.data, start, end))
189            }
190            /// Returns the bytes described by `handle` exclusively.
191            $hvis const fn read_bytes_mut(&mut self, handle: $Handle) -> Option<&mut [u8]> {
192                let (start, end) = $crate::unwrap![some? self._span_usize(handle)];
193                Some(<$_>::slice_bytes_mut(&mut self.data, start, end))
194            }
195            /// Replaces the bytes described by `handle`.
196            $hvis const fn replace_bytes(&mut self, handle: $Handle, new: &[u8]) -> bool {
197                let dst = $crate::unwrap![some_or? self.read_bytes_mut(handle), false];
198                if dst.len() != new.len() { return false; }
199                dst.copy_from_slice(new);
200                true
201            }
202
203            /* string spans */
204
205            /// Writes a UTF-8 string into the arena.
206            ///
207            /// The returned handle describes exactly the string's UTF-8 bytes.
208            /// No length prefix or terminator is stored.
209            ///
210            /// # Errors
211            /// Returns `None` if the complete UTF-8 byte span
212            /// does not fit in the remaining arena capacity.
213            $hvis const fn push_str(&mut self, val: &str) -> Option<$Handle> {
214                self.push_bytes(val.as_bytes())
215            }
216            /// Returns the UTF-8 string described by `handle`.
217            ///
218            /// # Errors
219            /// Returns `None` if the handle does not describe a span within the
220            /// written prefix or if its bytes are not valid UTF-8.
221            $hvis const fn read_str(&self, handle: $Handle) -> Option<&str> {
222                let bytes = $crate::unwrap![some? self.read_bytes(handle)];
223                $crate::unwrap![ok_some $crate::Str::from_utf8(bytes)]
224            }
225            /// Returns the UTF-8 string described by `handle` exclusively.
226            ///
227            /// # Errors
228            /// Returns `None` if the handle does not describe a span within the
229            /// written prefix or if its bytes are not valid UTF-8.
230            $hvis const fn read_str_mut(&mut self, handle: $Handle) -> Option<&mut str> {
231                let bytes = $crate::unwrap![some? self.read_bytes_mut(handle)];
232                $crate::unwrap![ok_some $crate::Str::from_utf8_mut(bytes)]
233            }
234            /// Replaces the UTF-8 string described by `handle`.
235            ///
236            /// Returns `false` if the handle is invalid or `val` has a different
237            /// byte length from the described span.
238            $hvis const fn replace_str(&mut self, handle: $Handle, val: &str) -> bool {
239                self.replace_bytes(handle, val.as_bytes())
240            }
241
242            /* single bytes */
243
244            /// Writes a single byte into the arena.
245            ///
246            /// # Errors
247            /// Returns `None` if the byte does not fit in the remaining arena capacity.
248            $hvis const fn push_byte(&mut self, byte: u8) -> Option<$Handle> {
249                self.push_bytes(&[byte])
250            }
251            /// Reads the single byte described by `h`.
252            $hvis const fn read_byte(&self, h: $Handle) -> Option<u8> {
253                let (start, end) = $crate::unwrap![some? self._span_usize(h)];
254                if end - start != 1 { return None; }
255                Some(<$_>::read_byte(&self.data, start))
256            }
257            /// Returns the single byte described by `h` exclusively.
258            $hvis const fn read_byte_mut(&mut self, h: $Handle) -> Option<&mut u8> {
259                let (start, end) = $crate::unwrap![some? self._span_usize(h)];
260                if end - start != 1 { return None; }
261                Some(<$_>::read_byte_mut(&mut self.data, start))
262            }
263            /// Replaces the single byte described by `h`.
264            $hvis const fn replace_byte(&mut self, h: $Handle, new: u8) -> bool {
265                let dst = $crate::unwrap![some_or? self.read_byte_mut(h), false];
266                *dst = new;
267                true
268            }
269
270            /* views over multiple values */
271
272            /// Returns a shared slice starting at `handle`,
273            /// and spanning `count` items of its length.
274            ///
275            /// Returns `None` if the repeated span overflows
276            /// or extends beyond the written prefix.
277            $hvis const fn view_bytes(&self, h: $Handle, count: $cprim) -> Option<&[u8]> {
278                let (start, end) = $crate::unwrap![some? self._span_repeat_usize(h, count)];
279                Some(<$_>::slice_bytes(&self.data, start, end))
280            }
281
282            /// Returns an exclusive slice starting at `handle`,
283            /// and spanning `count` items of its length.
284            ///
285            /// Returns `None` if the repeated span overflows
286            /// or extends beyond the written prefix.
287            $hvis const fn view_bytes_mut(&mut self, h: $Handle, count: $cprim)
288                -> Option<&mut [u8]> {
289                let (start, end) = $crate::unwrap![some? self._span_repeat_usize(h, count)];
290                Some(<$_>::slice_bytes_mut(&mut self.data, start, end))
291            }
292
293            /* shrinking the arena */
294
295            /// Truncates the arena if `handle` describes its final region.
296            $hvis const fn truncate_last(&mut self, h: $Handle) -> bool {
297                let (_, end) = $crate::unwrap![some_or? self._span_usize(h), false];
298                if end != self._len_usize() { return false; }
299                self.len = h.get_offset_prim();
300                true
301            }
302            /// Copies the final stored span into `dst` and removes it.
303            $hvis const fn pop_into(&mut self, h: $Handle, dst: &mut [u8]) -> bool {
304                let src = $crate::unwrap![some_or? self.read_bytes(h), false];
305                if src.len() != dst.len() { return false; }
306                dst.copy_from_slice(src);
307                self.truncate_last(h)
308            }
309            /// Removes all written bytes.
310            $vis const fn clear(&mut self) { self.len = 0; }
311        }
312
313        /* primitives */
314
315        #[allow(dead_code, private_interfaces)]
316        /// Implements push, read and replace for primitives.
317        impl<const CAP: usize> $Arena<CAP> {
318            $module::_impl_arena_methods_for_prims!();
319
320            /* bool */
321
322            /// Pushes a `bool`. Returns its handle on success.
323            ///
324            /// # Errors
325            /// Returns `None` if the value does not fit in the remaining arena capacity.
326            $hvis const fn push_bool(&mut self, val: bool) -> Option<$Handle> {
327                self.push_byte(val as u8)
328            }
329            /// Reads a `bool` from the given `handle`.
330            ///
331            /// # Errors
332            /// Returns `None` if the handle is invalid or incomplete.
333            $hvis const fn read_bool(&self, handle: $Handle) -> Option<bool> {
334                match self.read_byte(handle) {
335                    Some(0) => Some(false),
336                    Some(1) => Some(true),
337                    _ => None,
338                }
339            }
340            /// Replaces a `bool` from the given `handle`. Returns `true` on success.
341            $hvis const fn replace_bool(&mut self, handle: $Handle, val: bool) -> bool {
342                self.replace_byte(handle, val as u8)
343            }
344
345            /* char */
346
347            /// Pushes a `char`. Returns its handle on success.
348            ///
349            /// # Errors
350            /// Returns `None` if the value does not fit in the remaining arena capacity.
351            $hvis const fn push_char(&mut self, val: char) -> Option<$Handle> {
352                self.push_u32(val as u32)
353            }
354            /// Reads a `char` from the given `handle`.
355            ///
356            /// # Errors
357            /// Returns `None` if the handle is invalid or incomplete.
358            $hvis const fn read_char(&self, handle: $Handle) -> Option<char> {
359                $crate::unwrap![some_map_into self.read_u32(handle), |c| char::from_u32(c)]
360            }
361            /// Replaces a `char` from the given `handle`. Returns `true` on success.
362            $hvis const fn replace_char(&mut self, handle: $Handle, val: char) -> bool {
363                self.replace_u32(handle, val as u32)
364            }
365        }

367        #[doc(hidden)]
368        #[allow(non_snake_case)]
369        mod $module {
370            /// Private helper to implement push, read & replace methods over primitives.
371            #[rustfmt::skip] // fixes rustfmt warnings
372            macro_rules! _impl_arena_methods_for_prims {
373                () => {
374                    $module::_impl_arena_methods_for_prims!(single-byte: u8, i8);
375                    $module::_impl_arena_methods_for_prims!(multi-byte:
376                        u16, u32, u64, u128, usize,
377                        i16, i32, i64, i128, isize,
378                        f32, f64,
379                    );
380                };
381                (single-byte: $_d($oprim:ty),+ $_d(,)?) => {
382                    $_d( $module::_impl_arena_methods_for_prims!(%single-byte: $oprim); )+
383                };
384                (%single-byte: $oprim:ty) => { $crate::paste! {
385                    #[doc = "Pushes a `" $oprim "`. Returns its handle on success."]
386                    ///
387                    /// # Errors
388                    /// Returns `None` if the value does not fit in the remaining arena capacity.
389                    $hvis const fn [<push_ $oprim>](&mut self, val: $oprim) -> Option<$Handle> {
390                        self.push_byte(val as u8)
391                    }
392                    #[doc = "Reads a `" $oprim "` from the given `handle`."]
393                    ///
394                    /// # Errors
395                    /// Returns `None` if the handle is invalid or incomplete.
396                    $hvis const fn [<read_ $oprim>](&self, handle: $Handle) -> Option<$oprim> {
397                        if let Some(b) = self.read_byte(handle) { Some(b as $oprim) } else { None }
398                    }
399                    #[doc = "Replaces a `" $oprim "`
400                    from the given `handle`. Returns `true` on success."]
401                    $hvis const fn [<replace_ $oprim>](&mut self, handle: $Handle, val: $oprim)
402                    -> bool {
403                        self.replace_byte(handle, val as u8)
404                    }
405                }};
406                (multi-byte: $_d($oprim:ty),+ $_d(,)?) => {
407                    $_d( $module::_impl_arena_methods_for_prims!(%multi-byte: $oprim); )+
408                };
409                (%multi-byte: $oprim:ty) => { $crate::paste! {
410                    #[doc = "Pushes a `" $oprim
411                    "` in little-endian order. Returns its handle on success."]
412                    ///
413                    /// # Errors
414                    /// Returns `None` if the value does not fit in the remaining arena capacity.
415                    $hvis const fn [<push_ $oprim>](&mut self, val: $oprim) -> Option<$Handle> {
416                        self.push_bytes(&val.to_le_bytes())
417                    }
418                    #[doc = "Reads a `" $oprim "` in little-endian order from the given `handle`."]
419                    ///
420                    /// # Errors
421                    /// Returns `None` if the handle is invalid or incomplete.
422                    $hvis const fn [<read_ $oprim>](&self, handle: $Handle) -> Option<$oprim> {
423                        const T_SIZE: usize = size_of::<$oprim>();
424                        let bytes = $crate::unwrap![some? self.read_bytes(handle)];
425                        if bytes.len() != T_SIZE { return None; }
426                        Some($oprim::from_le_bytes(
427                            *$crate::unwrap![some? bytes.first_chunk::<T_SIZE>()]))
428                    }
429                    #[doc = "Replaces a `" $oprim
430                    "` from the given `handle`. Returns `true` on success."]
431                    $hvis const fn [<replace_ $oprim>](&mut self, handle: $Handle, val: $oprim)
432                        -> bool {
433                        const T_SIZE: usize = size_of::<$oprim>();
434                        let bytes = $crate::unwrap![some_or? self.read_bytes_mut(handle), false];
435                        if bytes.len() != T_SIZE { return false; }
436                        let arr = $crate::unwrap![some_or?
437                            bytes.first_chunk_mut::<T_SIZE>(), false];
438                        *arr = val.to_le_bytes();
439                        true
440                    }
441                }};
442            }
443            pub(super) use _impl_arena_methods_for_prims;
444        }
445    };
