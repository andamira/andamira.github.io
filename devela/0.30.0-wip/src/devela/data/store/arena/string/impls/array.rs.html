3#[doc(hidden)]
4#[macro_export]
5macro_rules! __arena_string_impl_array· {
6    (
7        [index: $iprim:ident + $Index:ty;]
8        [cursor: $cprim:ident + $Cursor:ty;]
9        $(#[$arena_attr:meta])* $vis:vis $Arena:ident;
10        $hvis:vis $Handle:ident;
11        [mark: $($mvis:vis $Mark:ident)?]
12    ) => {
13        $(#[$arena_attr])*
14        #[derive(Clone, Debug)]
15        $vis struct $Arena<const STRINGS: usize, const BYTES: usize> {
16            /// Cumulative UTF-8 byte end of every retained string.
17            ends: [$crate::MaybeNiche<$Cursor>; STRINGS],
18
19            /// Packed UTF-8 bytes.
20            data: [$crate::MaybeByte; BYTES],
21
22            /// Current string insertion frontier.
23            len: $crate::MaybeNiche<$Index>,
24        }
25
26        impl<const STRINGS: usize, const BYTES: usize> $crate::ConstInit
27            for $Arena<STRINGS, BYTES> {
28            const INIT: Self = Self::new();
29        }
30        impl<const STRINGS: usize, const BYTES: usize> Default for $Arena<STRINGS, BYTES> {
31            fn default() -> Self { Self::new() }
32        }
33
34        #[allow(dead_code)]
35        impl<const STRINGS: usize, const BYTES: usize> $Arena<STRINGS, BYTES> {
36            const _VALID_CONFIG: () = {
37                const fn require_index<P: $crate::PrimIndex>() {}
38                require_index::<$iprim>();
39                require_index::<$cprim>();
40                assert!(!$crate::MaybeNiche::<$Index>::HAS_NEGATIVE
41                        && $crate::MaybeNiche::<$Index>::IS_CONTIGUOUS
42                        && $crate::MaybeNiche::<$Index>::ZERO.is_some(),
43                    "arena_string! index representation must be unsigned, contiguous, and contain zero",
44                );
45                assert!(!$crate::MaybeNiche::<$Cursor>::HAS_NEGATIVE
46                        && $crate::MaybeNiche::<$Cursor>::IS_CONTIGUOUS
47                        && $crate::MaybeNiche::<$Cursor>::ZERO.is_some(),
48                    "arena_string! cursor representation must be unsigned, contiguous, and contain zero",
49                );
50                // Also forces `$Cursor` to use `$cprim` as its primitive carrier.
51                let _ = $crate::MaybeNiche::<$Cursor>::try_from_prim(0 as $cprim);
52                assert!(STRINGS <= Self::MAX_CAPACITY,
53                    "arena_string! string capacity exceeds its index representation");
54                assert!(BYTES <= Self::MAX_BYTE_CAPACITY,
55                    "arena_string! byte capacity exceeds its cursor representation");
56            };
57
58            /// Maximum fixed string-entry capacity supported by the index representation.
59            $vis const MAX_CAPACITY: usize = {
60                $crate::unwrap![ok_or $crate::MaybeNiche::<$Index>::MAX.try_to_usize(), usize::MAX]
61            };
62            /// Maximum fixed packed-byte capacity supported by the cursor representation.
63            $vis const MAX_BYTE_CAPACITY: usize = {
64                $crate::unwrap![ok_or $crate::MaybeNiche::<$Cursor>::MAX.try_to_usize(), usize::MAX]
65            };
66
67            /* construction */
68
69            /// Creates an empty fixed-capacity string arena.
70            #[must_use]
71            $vis const fn new() -> Self {
72                let () = Self::_VALID_CONFIG;
73                let zero =
74                    $crate::unwrap![some_guaranteed_or_ub $crate::MaybeNiche::<$Cursor>::ZERO];
75                Self {
76                    ends: [zero; STRINGS],
77                    data: $crate::__ArenaBytesArray::<BYTES>::new_array(),
78                    len: Self::_len_zero(),
79                }
80            }
81
82            /* string capacity */
83
84            /// Returns the fixed string-entry capacity.
85            #[must_use]
86            $vis const fn capacity(&self) -> usize { STRINGS }
87
88            /// Returns the number of retained strings.
89            #[must_use]
90            $vis const fn len(&self) -> usize { self._len_usize() }
91
92            /// Returns whether no strings are retained.
93            #[must_use]
94            $vis const fn is_empty(&self) -> bool { self.len() == 0 }
95
96            /// Returns whether the string-entry capacity is exhausted.
97            #[must_use]
98            $vis const fn remaining(&self) -> usize { STRINGS - self.len() }
99
100            /// Returns the remaining string-entry capacity.
101            #[must_use]
102            $vis const fn is_full(&self) -> bool { self.len() == STRINGS }
103
104            /* byte capacity */
105
106            /// Returns the fixed packed-byte capacity.
107            #[must_use]
108            $vis const fn byte_capacity(&self) -> usize { BYTES }
109
110            /// Returns the number of packed bytes occupied by retained strings.
111            #[must_use]
112            $vis const fn byte_len(&self) -> usize {
113                let len = self.len();
114                if len == 0 { 0 } else { self._end_usize(len - 1) }
115            }
116            /// Returns the remaining packed-byte capacity.
117            #[must_use]
118            $vis const fn byte_remaining(&self) -> usize {
119                BYTES - self.byte_len()
120            }
121            /// Returns whether `string` fits the remaining entry and byte capacities.
122            #[must_use]
123            $vis const fn can_insert(&self, string: &str) -> bool {
124                !self.is_full() && string.len() <= self.byte_remaining()
125            }
126
127            /* access */
128
129            /// Returns whether `handle` currently resolves to a retained string.
130            #[must_use]
131            $hvis const fn contains(&self, handle: $Handle) -> bool {
132                self._resolve_index(handle).is_some()
133            }
134            /// Returns the retained string currently resolved by `handle`.
135            #[must_use]
136            $hvis const fn get(&self, handle: $Handle) -> Option<&str> {
137                let index = $crate::unwrap![some? self._resolve_index(handle)];
138                Some(self._str_at(index))
139            }
140            /// Returns all packed UTF-8 bytes, without preserving entry boundaries.
141            #[must_use]
142            $vis const fn as_bytes(&self) -> &[u8] {
143                $crate::__ArenaBytesArray::<BYTES>::slice_bytes(&self.data, 0, self.byte_len())
144            }
145
146            /* mutation */
147
148            /// Inserts a string at the current frontier.
149            ///
150            /// Equal strings are retained independently and receive distinct handles.
151            ///
152            /// Returns `None` if either the string-entry capacity or byte capacity
153            /// is insufficient.
154            $hvis const fn insert(&mut self, string: &str) -> Option<$Handle> {
155                if self.is_full() { return None; }
156                let index_usize = self.len();
157                let start = self.byte_len();
158                let end = $crate::unwrap![some? start.checked_add(string.len())];
159                if end > BYTES { return None; }
160                // Check every representational constraint before mutating storage.
161                let end_repr =
162                    $crate::unwrap![ok_some? $crate::MaybeNiche::<$Cursor>::try_from_usize(end)];
163                let index = self.len;
164                let handle = $Handle::new(index.get());
165                let bytes = string.as_bytes();
166                $crate::whilst! { i in 0..bytes.len(); {
167                    $crate::__ArenaBytesArray::<BYTES>
168                        ::write_byte(&mut self.data, start + i, bytes[i]);
169                }}
170                self.ends[index_usize] = end_repr;
171                self.len = Self::_len_from_usize(index_usize + 1);
172                Some(handle)
173            }
174
175            /* marks and reclamation */
176
177            $(
178                /// Returns a mark at the current string insertion frontier.
179                #[must_use]
180                $mvis const fn mark(&self) -> $Mark { <$Mark>::new(self.len) }
181
182                /// Reclaims every string inserted after `mark`.
183                ///
184                /// The packed byte frontier follows automatically
185                /// from the restored string frontier.
186                ///
187                /// Returns `false` without changing the arena
188                /// if `mark` lies ahead of the current frontier.
189                $mvis const fn rollback(&mut self, mark: $Mark) -> bool {
190                    if mark.0.gt(self.len) { return false; }
191                    self.len = mark.0;
192                    true
193                }
194            )?
195            /// Reclaims every retained string.
196            ///
197            /// Previously issued handles may resolve again after later insertion.
198            $vis const fn clear(&mut self) {
199                self.len = Self::_len_zero();
200            }
201
202            /* iteration */
203
204            /// Iterates over retained strings in insertion order.
205            $vis fn iter(&self) -> impl Iterator<Item = &str> + '_ {
206                (0..self.len()).map(|index| self._str_at(index))
207            }
208            /// Iterates over retained string handles in insertion order.
209            $vis fn ids(&self) -> impl Iterator<Item = $Handle> + '_ {
210                (0..self.len()).filter_map(|index| <$Handle>::try_from_usize(index).ok())
211            }
212            /// Iterates over retained handles and strings in insertion order.
213            $vis fn entries(&self) -> impl Iterator<Item = ($Handle, &str)> + '_ {
214                (0..self.len()).filter_map(move |index| {
215                    let handle = <$Handle>::try_from_usize(index).ok()?;
216                    Some((handle, self._str_at(index)))
217                })
218            }
219
220            /* private */
221
222            const fn _len_zero() -> $crate::MaybeNiche<$Index> {
223                $crate::unwrap![some_guaranteed_or_ub $crate::MaybeNiche::<$Index>::ZERO]
224            }
225            const fn _len_from_usize(len: usize) -> $crate::MaybeNiche<$Index> {
226                $crate::unwrap![ok_guaranteed_or_ub
227                    $crate::MaybeNiche::<$Index>::try_from_usize(len)]
228            }
229            const fn _len_usize(&self) -> usize {
230                $crate::unwrap![ok_guaranteed_or_ub self.len.try_to_usize()]
231            }
232            const fn _end_usize(&self, index: usize) -> usize {
233                $crate::unwrap![ok_guaranteed_or_ub self.ends[index].try_to_usize()]
234            }
235            const fn _range_at(&self, index: usize) -> (usize, usize) {
236                let start = if index == 0 { 0 } else { self._end_usize(index - 1) };
237                let end = self._end_usize(index);
238                (start, end)
239            }
240            const fn _str_at(&self, index: usize) -> &str {
241                let (start, end) = self._range_at(index);
242                let bytes = $crate::__ArenaBytesArray::<BYTES>::slice_bytes(&self.data, start, end);
243                // Every stored span entered through `&str`.
244                $crate::Str::__utf8_bytes_to_str(bytes)
245            }
246            const fn _resolve_index(&self, handle: $Handle) -> Option<usize> {
247                let index = $crate::unwrap![ok_some? handle.get_index_usize()];
248                if index >= self.len() { return None; }
249                Some(index)
250            }
251        }
252    };
253}
254pub use __arena_string_impl_array· as __arena_string_impl_array;
