7#[doc = crate::_tags!(construction data_structure)]
8/// Defines a custom static hashmap.
9#[doc = crate::_doc_meta!{
10    location("data/store/key/map", macro map),
11}]
12/// # Arguments
13/// - `$NAME`:      the name of the new hashmap.
14/// - `$KEY`:       the primitive keys type.
15///
16/// optional:
17/// - `$EMPTY`:     the `$KEY` value for empty entries.
18/// - `$TOMB`:      the `$KEY` value for deleted entries.
19/// - `$HASH_ARG`:  the argument representing the byte slice.
20/// - `$HASH_EXPR`: the const hasher expression using `$HASH_ARG`.
21///
22/// # Notes
23/// - values `V` have to be `Copy` + `ConstInit`.
24/// - keys `$KEY` can be any primitive integers, floats or `char`.
25/// - Two specific `$KEY` values are reserved to indicate empty deleted keys.
26///   They default to `MIN` and `MAX`, respectively, but can be customized.
27/// - The default hasher is [`HasherFx`][crate::HasherFx].
28///
29/// # Variants
30/// The macro supports three construction modes:
31///
32/// | Kind | Invocation | Description |
33/// |------|-------------|--------------|
34#[doc = concat!["| **Const** | `map![const MyMap, KEY: u16]` |",
35"Generates a fully `const` hashmap with compile-time operations and const methods. |"]]
36#[doc = concat!["| **Runtime** | `map![MyMap, KEY: u16]` |",
37"Generates a non-const variant storing markers as struct fields, suitable for runtime mutation. |"]]
38#[doc = concat!["| **TypeId-based** | `map![typeid MyMap]` |",
39"Uses `TypeId` hashes as keys and provides type-oriented helper methods. |"]]
40///
41/// # Examples
42/// See [`MapFixedConstU8Example`], [`MapFixedU8Example`], [`MapFixedTypeIdExample`].
43///
44/// [`MapFixedConstU8Example`]: crate::MapFixedConstU8Example
45/// [`MapFixedU8Example`]: crate::MapFixedU8Example
46/// [`MapFixedTypeIdExample`]: crate::MapFixedTypeIdExample
47///
48/// Overview
49/// ```
50/// use devela::map;
51///
52/// // 1. Const hashmap
53/// map![
54///     #[doc(hidden)] // supports attributes
55///     pub const MapConst, KEY: u16
56/// ];
57///
58/// // 2. Runtime hashmap
59/// map![pub(crate) MapRuntime, KEY: u16];
60///
61/// // 3. TypeId-keyed hashmap
62/// map![typeid MapTypeId];
63/// ```
64///
65/// Basic usage
66/// ```
67/// use devela::map;
68///
69/// // Define a static hashmap with `u16` keys and default hasher
70/// map![const ExampleMap, KEY: u16];
71///
72/// let mut map = ExampleMap::<u16, u32, 8>::new();
73///
74/// // Insert key-value pairs
75/// map.insert(1, 100).unwrap();
76/// map.insert(2, 200).unwrap();
77///
78/// // Retrieve values
79/// assert_eq!(map.get(1), Some(100));
80/// assert_eq!(map.get(2), Some(200));
81/// assert_eq!(map.get(3), None); // Key not found
82///
83/// // Delete a key
84/// assert!(map.remove(1));
85/// assert_eq!(map.get(1), None);
86///
87/// // Check introspection methods
88/// assert_eq!(map.len(), 1);
89/// assert!(!map.is_empty());
90/// assert!(!map.is_full());
91///
92/// // Rebuild after deletions to optimize probing
93/// if map.should_rebuild() {
94///     map.rebuild();
95/// }
96/// ```
97///
98/// Custom hashers
99/// ```
100/// use devela::{map, HasherFx};
101///
102/// // Define a static hashmap using `HasherFx` with a custom seed
103/// map![const ExampleMapFxSeeded, KEY: u16,
104///     HASHER: |b| HasherFx::<usize>::hash_bytes_with_seed(123, b)
105/// ];
106/// let mut map = ExampleMapFxSeeded::<u16, u32, 8>::new();
107/// map.insert(1, 100).unwrap();
108/// assert_eq!(map.get(1), Some(100));
109///
110/// # #[cfg(feature = "hash")] {
111/// use devela::HasherPengy;
112///
113/// // Define a static hashmap using a stateful pengy hasher
114/// # #[cfg(feature = "hash")]
115/// map![const ExampleMapPengy, KEY: u16,
116///     HASHER: |b| {
117///         let mut p = HasherPengy::new();
118///         p.process(b);
119///         p.digest() as usize
120///     }
121/// ];
122/// let mut map = ExampleMapPengy::<u16, u32, 8>::new();
123/// map.insert(1, 100).unwrap();
124/// assert_eq!(map.get(1), Some(100));
125/// # }
126/// ```
127#[cfg_attr(cargo_primary_package, doc(hidden))]
128#[macro_export]
129macro_rules! map· {
130    (
131    // Const variant
132    // ----------------------------------------------------------------------------------------
133    // Default constructor:
134        $(#[$attr:meta])*
135        $vis:vis const $NAME:ident, KEY:$KEY:ty $(,)?
136    ) => {
137        $crate::map![
138            $(#[$attr])*
139            $vis const $NAME, KEY:$KEY,
140            EMPTY:<$KEY>::MIN, TOMB:<$KEY>::MAX,
141            HASHER:|bytes| $crate::HasherFx::<usize>::hash_primitive_bytes(bytes)
142        ];
143    };
144    (// Custom Empty/Tomb, Default Hasher:
145        $(#[$attr:meta])*
146        $vis:vis const $NAME:ident, KEY:$KEY:ty,
147        EMPTY:$EMPTY:expr, TOMB:$TOMB:expr $(,)?
148    ) => {
149        $crate::map![
150            $(#[$attr])*
151            $vis const $NAME, KEY:$KEY,
152            EMPTY:$EMPTY, TOMB:$TOMB,
153            HASHER:|bytes| $crate::HasherFx::<usize>::hash_primitive_bytes(bytes)
154        ];
155    };
156    (// Custom Hasher, Default Empty/Tomb:
157        $(#[$attr:meta])*
158        $vis:vis const $NAME:ident, KEY:$KEY:ty,
159        HASHER: | $HASH_ARG:ident | $HASH_EXPR:expr $(,)?
160    ) => {
161        $crate::map![
162            $(#[$attr])*
163            $vis const $NAME, KEY:$KEY,
164            EMPTY:<$KEY>::MIN, TOMB:<$KEY>::MAX,
165            HASHER: | $HASH_ARG | $HASH_EXPR
166        ];
167    };
168    (
169        $(#[$attr:meta])*
170        $vis:vis const $NAME:ident, KEY:$KEY:ty,
171        EMPTY:$EMPTY:expr, TOMB:$TOMB:expr,
172        HASHER: | $HASH_ARG:ident | $HASH_EXPR:expr $(,)?
173    ) => {
174        $crate::__map_impl_const! {
175            $(#[$attr])*
176            $vis $NAME, KEY:$KEY,
177            EMPTY:$EMPTY, TOMB:$TOMB,
178            HASHER: | $HASH_ARG | $HASH_EXPR
179        }
180    };
181    (
182    // Runtime variant
183    // ----------------------------------------------------------------------------------------
184    // Default constructor:
185        $(#[$attr:meta])*
186        $vis:vis $NAME:ident, KEY:$KEY:ty $(,)?
187    ) => {
188        $crate::map![
189            $(#[$attr])*
190            $vis $NAME, KEY:$KEY, EMPTY:<$KEY>::MIN, TOMB:<$KEY>::MAX,
191            HASHER:|bytes| $crate::HasherFx::<usize>::hash_primitive_bytes(bytes)
192        ];
193    };
194    (// Custom Empty/Tomb, Default Hasher:
195        $(#[$attr:meta])*
196        $vis:vis $NAME:ident, KEY:$KEY:ty,
197        EMPTY:$EMPTY:expr, TOMB:$TOMB:expr $(,)?
198    ) => {
199        $crate::map![
200            $(#[$attr])*
201            $vis $NAME, KEY:$KEY, EMPTY:$EMPTY, TOMB:$TOMB,
202            HASHER:|bytes| $crate::HasherFx::<usize>::hash_primitive_bytes(bytes)
203        ];
204    };
205    (// Custom Hasher, Default Empty/Tomb:
206        $(#[$attr:meta])*
207        $vis:vis $NAME:ident, KEY:$KEY:ty,
208        HASHER: | $HASH_ARG:ident | $HASH_EXPR:expr $(,)?
209    ) => {
210        $crate::map![
211            $(#[$attr])*
212            $vis $NAME, KEY:$KEY,
213            EMPTY:<$KEY>::MIN, TOMB:<$KEY>::MAX,
214            HASHER: | $HASH_ARG | $HASH_EXPR
215        ];
216    };
217    (
218        $(#[$attr:meta])*
219        $vis:vis $NAME:ident, KEY:$KEY:ty,
220        EMPTY:$EMPTY:expr, TOMB:$TOMB:expr,
221        HASHER: | $HASH_ARG:ident | $HASH_EXPR:expr $(,)?
222    ) => {
223        $crate::__map_impl_runtime! {
224            $(#[$attr])*
225            $vis $NAME, KEY:$KEY,
226            EMPTY:$EMPTY, TOMB:$TOMB,
227            HASHER: | $HASH_ARG | $HASH_EXPR
228        }
229    };
230    (
231    // TypeId runtime variant
232    // ----------------------------------------------------------------------------------------
233    // Uses 64-bit hashes of `TypeId`s for the keys:
234        $(#[$attr:meta])*
235        $vis:vis typeid $NAME:ident $(,)?) => {
236        $crate::map![
237            $(#[$attr])*
238            #[doc = "A `TypeId`-keyed static hashmap.\n\n\
239            This variant uses 64-bit hashes of Rust `TypeId`s as keys and adds \
240            type-oriented methods such as `insert_type`, `get_type`, and `remove_type`. \
241            It is built on the runtime hashmap variant, inheriting its stored `empty` \
242            and `tomb` markers and behavior.\n\n"]
243            $vis $NAME, KEY: u64,
244            EMPTY: type_id_hash::<Empty>(), TOMB: type_id_hash::<Tomb>(),
245            HASHER:|bytes| $crate::HasherFx::<usize>::hash_primitive_bytes(bytes)
246        ];
247
248        struct Empty;
249        struct Tomb;
250        fn type_id_hash<T: 'static>() -> u64 {
251            let mut hasher = $crate::HasherFx::<u64>::new();
252            let id = $crate::TypeId::of::<T>();
253            $crate::Hash::hash(&id, &mut hasher);
254            $crate::Hasher::finish(&hasher)
255        }
256
257        #[allow(unused)]
258        /// Convenience methods for when the keys are `TypeId`s.
259        impl<V, const N: usize> $NAME<u64, V, N> {
260            /// Returns the hash of `T`'s `TypeId`.
261            pub fn type_id_hash<T: 'static>() -> u64 { type_id_hash::<T>() }
262
263            /// Retrieves some exclusive reference to the value associated with the given type `T`.
264            ///
265            /// Calls [`get_ref`][Self::get_ref] with the hash of its type id.
266            pub fn get_ref_type<T: 'static>(&self) -> Option<&V> {
267                let key = Self::type_id_hash::<T>();
268                self.get_ref(key)
269            }
270            /// Retrieves some exclusive reference to the value associated with the given type `T`.
271            ///
272            /// Calls [`get_mut`][Self::get_mut] with the hash of its type id.
273            pub fn get_mut_type<T: 'static>(&mut self) -> Option<&mut V> {
274                let key = Self::type_id_hash::<T>();
275                self.get_mut(key)
276            }
277            /// Inserts a value paired with the given type `T`.
278            ///
279            /// Calls [`insert`][Self::insert] with the hash of its type id.
280            ///
281            /// # Returns
282            /// - `Ok(())` if the insertion succeeds.
283            /// - `Err(`[`NotEnoughSpace`][crate::NotEnoughSpace]`)` if no slots are available.
284            pub fn insert_type<T: 'static>(&mut self, value: V)
285                -> Result<(), $crate::NotEnoughSpace> {
286                let key = Self::type_id_hash::<T>();
287                self.insert(key, value)
288            }
289        }
290        #[allow(unused)]
291        impl<V: Copy, const N: usize> $NAME<u64, V, N> {
292            /// Retrieves some value associated with the given type `T`.
293            ///
294            /// Calls [`get`][Self::get] with the hash of its type id.
295            pub fn get_type<T: 'static>(&self) -> Option<V> {
296                let key = Self::type_id_hash::<T>();
297                self.get(key)
298            }
299            /// Removes the value paired with the given type `T`.
300            ///
301            /// Calls [`remove`][Self::remove] with the hash of its type id.
302            pub fn remove_type<T: 'static>(&mut self) -> bool {
303                let key = Self::type_id_hash::<T>();
304                self.remove(key)
305            }
306        }
307    };
308}
