3macro_rules! __pool_impl_array· {
4    (
5     [
6      index: $iprim:ident + $Index:ty;
7      generation: $gprim:ident + $Generation:ty;
8     ]
9     $(#[$pool_attr:meta])* $vis:vis $Pool:ident;
10     $hvis:vis $Handle:ident;
11    ) => {
12        $(#[$pool_attr])*
13        #[derive(Clone, Debug)]
14        $vis struct $Pool<T, const CAP: usize> {
15            values: [Option<T>; CAP],
16            generations: [$crate::MaybeNiche<$Generation>; CAP],
17            free: [$crate::MaybeNiche<$Index>; CAP],
18            len: usize,
19            free_len: usize,
20        }
21
22        /* misc. trait impls */
23
24        impl<T, const CAP: usize> $crate::ConstInit for $Pool<T, CAP> {
25            const INIT: Self = Self::new();
26        }
27        impl<T, const CAP: usize> Default for $Pool<T, CAP> {
28            fn default() -> Self { Self::new() }
29        }
30
31        // Fundamental methods
32        #[allow(dead_code)]
33        impl<T, const CAP: usize> $Pool<T, CAP> {
34            /// Verifies the representation laws required by this pool.
35            const _VALID_CONFIG: () = {
36                assert!(CAP <= Self::MAX_CAPACITY,
37                    "the pool capacity exceeds its index representation");
38                assert!(!$crate::MaybeNiche::<$Index>::HAS_NEGATIVE,
39                    "the pool index representation must be unsigned");
40                assert!($crate::MaybeNiche::<$Index>::IS_CONTIGUOUS,
41                    "the pool index representation must be contiguous");
42                assert!($crate::MaybeNiche::<$Index>::ZERO.is_some(),
43                    "the pool index representation must contain zero");
44                assert!($crate::MaybeNiche::<$Generation>::MIN.get_prim()
45                    != $crate::MaybeNiche::<$Generation>::MAX.get_prim(),
46                    "the pool generation representation needs at least two values");
47            };
48
49            /// Returns a new empty pool.
50            #[must_use]
51            $vis const fn new() -> Self {
52                let () = Self::_VALID_CONFIG;
53                Self {
54                    values: [const { None }; CAP],
55                    generations: [
56                        <$crate::MaybeNiche<$Generation> as $crate::ConstInit>::INIT; CAP ],
57                    free: [ <$crate::MaybeNiche<$Index> as $crate::ConstInit>::INIT; CAP ],
58                    len: 0,
59                    free_len: 0,
60                }
61            }
62
63            /* capacity */
64
65            /// Returns the total number of slots.
66            #[must_use]
67            $vis const fn capacity(&self) -> usize { CAP }
68
69            // MAX_CAPACITY, len, is_empty, remaining
70            $crate::__pool! {%impl_common_core const
71                [ index: $iprim + $Index; generation: $gprim + $Generation; ]
72                $vis $Pool; $hvis $Handle;
73            }
74
75            /// Returns whether every slot is occupied.
76            #[must_use]
77            $vis const fn is_full(&self) -> bool { self.len == CAP }
78
79            /* access */
80
81            /// Returns whether `handle` currently resolves to a value.
82            #[must_use]
83            $hvis const fn contains(&self, handle: $Handle) -> bool {
84                self._resolve_index(handle).is_some()
85            }
86            /// Returns a shared reference to the value resolved by `handle`.
87            #[must_use]
88            $hvis const fn get(&self, handle: $Handle) -> Option<&T> {
89                let index = $crate::unwrap![some? self._resolve_index(handle)];
90                self.values[index].as_ref()
91            }
92            /// Returns an exclusive reference to the value resolved by `handle`.
93            #[must_use]
94            $hvis const fn get_mut(&mut self, handle: $Handle) -> Option<&mut T> {
95                let index = $crate::unwrap![some? self._resolve_index(handle)];
96                self.values[index].as_mut()
97            }
98
99            /// Returns exclusive references to the values resolved by `a` and `b`.
100            ///
101            /// The references follow the order of the supplied handles.
102            ///
103            /// Returns `None` if either handle is invalid
104            /// or both handles resolve to the same slot.
105            #[must_use]
106            $hvis const fn get2_mut(&mut self, a: $Handle, b: $Handle) -> Option<(&mut T, &mut T)> {
107                let a_index = $crate::unwrap![some? self._resolve_index(a)];
108                let b_index = $crate::unwrap![some? self._resolve_index(b)];
109                if a_index == b_index { return None; }
110                if a_index < b_index {
111                    let (left, right) = self.values.split_at_mut(b_index);
112                    Some((
113                        $crate::unwrap![some? left[a_index].as_mut()],
114                        $crate::unwrap![some? right[0].as_mut()],
115                    ))
116                } else {
117                    let (left, right) = self.values.split_at_mut(a_index);
118                    Some((
119                        $crate::unwrap![some? right[0].as_mut()],
120                        $crate::unwrap![some? left[b_index].as_mut()],
121                    ))
122                }
123            }
124
125            /* mutation */
126
127            /// Inserts `value`, returning its handle.
128            ///
129            /// # Errors
130            /// Returns `value` unchanged when the pool is full.
131            $hvis fn insert(&mut self, value: T) -> Result<$Handle, T> {
132                let Some((index_usize, index)) = self._acquire_slot() else { return Err(value); };
133                let generation = self.generations[index_usize];
134                self.values[index_usize] = Some(value);
135                self.len += 1;
136                Ok($Handle::new(index.get(), generation.get()))
137            }
138            /// Inserts a copyable `value`, returning its handle.
139            ///
140            /// This is the const-capable variant of [`insert`][Self::insert].
141            ///
142            /// # Errors
143            /// Returns `value` unchanged when the pool is full.
144            $hvis const fn insert_copy(&mut self, value: T) -> Result<$Handle, T> where T: Copy {
145                let Some((index_usize, index)) = self._acquire_slot() else { return Err(value); };
146                let generation = self.generations[index_usize];
147                self.values[index_usize] = Some(value);
148                self.len += 1;
149                Ok($Handle::new(index.get(), generation.get()))
150            }
151
152            /// Replaces the value resolved by `handle`, returning the previous value.
153            ///
154            /// # Errors
155            /// Returns `value` unchanged if `handle` does not currently resolve.
156            $hvis const fn replace(&mut self, handle: $Handle, value: T) -> Result<T, T> {
157                let index = $crate::unwrap![some_ok_or? self._resolve_index(handle), value];
158                match self.values[index].as_mut() {
159                    Some(slot) => Ok($crate::Mem::replace(slot, value)),
160                    None => Err(value), // unreachable while pool invariants hold (IMPROVE?)
161                }
162            }
163            /// Removes and returns the value resolved by `handle`.
164            ///
165            /// The vacated slot advances its generation before it can be reused.
166            $hvis const fn remove(&mut self, handle: $Handle) -> Option<T> {
167                let index = $crate::unwrap![some? self._resolve_index(handle)];
168                let next_generation = Self::_next_generation(self.generations[index]);
169                let free_pos = self.free_len;
170                self.generations[index] = next_generation;
171                self.free[free_pos] = $crate::MaybeNiche(handle.get_index());
172                self.free_len = free_pos + 1;
173                self.len -= 1;
174                $crate::Mem::replace(&mut self.values[index], None)
175            }
176            /// Removes every value and invalidates every live handle.
177            $vis fn clear(&mut self) {
178                let frontier = self.len + self.free_len;
179                $crate::whilst! { index in 0..frontier; {
180                    if self.values[index].take().is_some() { // drop
181                        self.generations[index] = Self::_next_generation(self.generations[index]);
182                    }
183                }}
184                self.len = 0;
185                self.free_len = 0;
186            }
187            /// Removes every copyable value and invalidates every live handle.
188            ///
189            /// This is the const-capable variant of [`clear`][Self::clear].
190            $vis const fn clear_copy(&mut self) where T: Copy {
191                let frontier = self.len + self.free_len;
192                $crate::whilst! { index in 0..frontier; {
193                    if self.values[index].is_some() {
194                        self.values[index] = None;
195                        self.generations[index] = Self::_next_generation(self.generations[index]);
196                    }
197                }}
198                self.len = 0;
199                self.free_len = 0;
200            }
201
202            /* iteration */
203
204            /// Iterates over occupied values in ascending slot order.
205            $vis const fn iter(&self) -> $crate::PoolIter<&[Option<T>]> {
206                $crate::PoolIter::_new(&self.values, self.len)
207            }
208            /// Iterates mutably over occupied values in ascending slot order.
209            $vis const fn iter_mut(&mut self) -> $crate::PoolIter<&mut [Option<T>]> {
210                $crate::PoolIter::_new(&mut self.values, self.len)
211            }
212
213            // MAYBE: concrete const-capable handle/entry iterator when needed.
214            $crate::__pool! {%impl_common_iter
215                [ index: $iprim + $Index; generation: $gprim + $Generation; ]
216                $vis $Pool; $hvis $Handle;
217            }
218
219            /* private */
220
221            const fn _acquire_slot(&mut self) -> Option<(usize, $crate::MaybeNiche<$Index>)> {
222                if self.is_full() { return None; }
223                if self.free_len != 0 {
224                    let free_pos = self.free_len - 1;
225                    let index = self.free[free_pos];
226                    let index_usize = $crate::unwrap![ok_some? index.try_to_usize()];
227                    self.free_len = free_pos;
228                    Some((index_usize, index))
229                } else {
230                    let index_usize = self.len;
231                    let index = $crate::unwrap![ok_some?
232                        $crate::MaybeNiche::<$Index>::try_from_usize(index_usize)];
233                    Some((index_usize, index))
234                }
235            }
236        }
237
238        $crate::__pool! {%impl_common_iter_traits const
239            [ index: $iprim + $Index; generation: $gprim + $Generation; ]
240            $vis $Pool; $hvis $Handle;
241        }
242    };
243}
244pub use __pool_impl_array· as __pool_impl_array;
