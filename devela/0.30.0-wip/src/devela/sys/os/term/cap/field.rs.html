3//! Defines [`TermCaps`].
4//
5
6use crate::{ColorDepth, TermCap};
7
8crate::bitfield! {
9    #[must_use]
10    #[doc = crate::_tags!(term runtime set bit)]
11    /// Terminal capability bits.
12    #[doc = crate::_doc_meta!{
13        location("sys/os/term", struct TermCaps),
14        test_size_of(TermCaps = 4|32; niche !Option),
15    }]
16    /// Stores independent terminal feature flags plus the maximum known color depth.
17    ///
18    /// See also [`TermCap`].
19    pub struct TermCaps(u32) {
20        /* input */
21        /// Keyboard input.
22        KEYBOARD = 0;
23        /// Mouse press/release.
24        MOUSE = 1;
25        /// Mouse drag.
26        MOUSE_DRAG = 2;
27        /// Mouse motion.
28        MOUSE_MOTION = 3;
29        /// SGR mouse cell-coordinate mode.
30        MOUSE_SGR = 4;
31        /// SGR mouse pixel-coordinate mode.
32        MOUSE_SGR_PIXELS = 5;
33        /// Focus-in/focus-out.
34        FOCUS = 6;
35        /// Bracketed paste.
36        BRACKETED_PASTE = 7;
37        /// Terminal resize.
38        RESIZE = 8;
39
40        /* output */
41        /// ANSI/VT escape sequences.
42        ANSI = 9;
43        /// Cursor movement and visibility control.
44        CURSOR = 10;
45        /// SGR styling, such as bold, reset, and colors.
46        STYLE = 11;
47        /// Alternate screen buffer.
48        ALT_SCREEN = 12;
49        /// Synchronized output updates.
50        SYNC_UPDATE = 13;
51
52        /* image */
53        /// Sixel image output.
54        SIXEL = 14;
55        /// Kitty graphics protocol image output.
56        KITTY_IMAGE = 15;
57        /// iTerm2 inline image output.
58        ITERM_IMAGE = 16;
59
60        /* query replies */
61        /// Device-attributes query replies.
62        QUERY_DEVICE_ATTRS = 17;
63        /// Cursor-position query replies.
64        QUERY_CURSOR_POS = 18;
65        /// Color query replies.
66        QUERY_COLOR = 19;
67
68        /// Terminal color depth.
69        COLOR_DEPTH = 20..=22; // 3-bit
70    }
71    impl {
72        /// Returns an empty terminal capability set.
73        pub const EMPTY: Self = Self::new();
74
75        /// A conservative ANSI terminal baseline.
76        pub const ANSI_BASE: Self = Self::new()
77            .with_keyboard(1)
78            .with_ansi(1)
79            .with_cursor(1)
80            .with_style(1)
81            .with_color_depth_enum(ColorDepth::Ansi4);
82
83        /// Returns the maximum known terminal color depth.
84        #[must_use]
85        pub const fn color_depth(self) -> ColorDepth {
86            ColorDepth::from_bits_trunc(self.get_color_depth() as u8)
87        }
88        /// Returns a copy with the given color depth.
89        pub const fn with_color_depth_enum(self, depth: ColorDepth) -> Self {
90            self.with_color_depth(depth as u32)
91        }
92        /// Sets the maximum known terminal color depth.
93        pub const fn set_color_depth_enum(&mut self, depth: ColorDepth) {
94            self.set_color_depth(depth as u32);
95        }
96
97        /// Returns whether `cap` is present.
98        #[must_use]
99        pub const fn has(self, cap: TermCap) -> bool {
100            self.bits() & cap.bit() != 0
101        }
102        /// Returns a copy with `cap` enabled.
103        pub const fn set_cap(&mut self, cap: TermCap) {
104            *self = self.with_cap(cap);
105        }
106        /// Returns a copy with `cap` enabled.
107        pub const fn unset_cap(&mut self, cap: TermCap) {
108            *self = self.without_cap(cap);
109        }
110        /// Returns a copy with `cap` enabled.
111        pub const fn with_cap(self, cap: TermCap) -> Self {
112            Self::from_bits(self.bits() | cap.bit())
113        }
114        /// Returns a copy with `cap` disabled.
115        pub const fn without_cap(self, cap: TermCap) -> Self {
116            Self::from_bits(self.bits() & !cap.bit())
117        }
118    }
119}
