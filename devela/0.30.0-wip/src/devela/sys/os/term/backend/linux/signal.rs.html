3//! Implements signal handling methods for [`TermLinux`].
4//
5
6use crate::{AppControl, AppControlSet, Linux, LinuxSignalSet, TermLinux};
7use crate::{AtomicBool, AtomicOrdering, AtomicU16, c_int};
8#[cfg(doc)]
9use crate::{EventKind::Control, EventWindow::Resized};
10
11static PENDING_APP_CONTROLS: AtomicU16 = AtomicU16::new(0);
12static PENDING_RESIZE: AtomicBool = AtomicBool::new(false);
13
14/// # Signal handling
15impl TermLinux {
16    /// Registers Linux signals used by this terminal event source.
17    ///
18    /// This enables application control notices from `controls` and terminal resize
19    /// notices from `SIGWINCH`. Pending notices are drained into this backend's event
20    /// queue as `EventKind::`[`Control`] and `EventWindow::`[`Resized`].
21    ///
22    /// This installs process-wide signal handlers. Registering another handler for
23    /// the same Linux signals may replace this path.
24    pub fn listen_signals(&mut self, controls: AppControlSet) {
25        self.listen_app_controls(controls);
26        self.listen_resize();
27    }
28
29    /// Registers application control notices as terminal events.
30    ///
31    /// This installs process-wide Linux signal handlers for the corresponding
32    /// controls. Pending notices are drained into this backend's event queue as
33    /// `EventKind::`[`Control`].
34    ///
35    /// Registering another handler for the same Linux signals may replace this path.
36    pub fn listen_app_controls(&mut self, controls: AppControlSet) {
37        Self::install_app_controls(controls);
38    }
39
40    /// Registers terminal resize notices as window resize events.
41    ///
42    /// This installs a process-wide `SIGWINCH` handler. Pending resize notices are
43    /// drained into this backend's event queue as `EventWindow::`[`Resized`].
44    ///
45    /// Registering another handler for `SIGWINCH` may replace this path.
46    pub fn listen_resize(&mut self) {
47        Self::install_resize();
48    }
49}
50
51// Internal helpers
52impl TermLinux {
53    /* app controls */
54
55    fn install_app_controls(controls: AppControlSet) {
56        Linux::app_control_handler(Self::app_control_handler, controls, None);
57    }
58    fn app_control_handler(control: AppControl) {
59        let bit = control.to_set().bits();
60        PENDING_APP_CONTROLS.fetch_or(bit, AtomicOrdering::SeqCst);
61    }
62    pub(super) fn take_app_controls() -> AppControlSet {
63        AppControlSet::from_bits(PENDING_APP_CONTROLS.swap(0, AtomicOrdering::SeqCst))
64    }
65
66    /* resize */
67
68    fn install_resize() {
69        Linux::sig_handler(Self::resize_handler, LinuxSignalSet::SIGWINCH, None);
70    }
71    fn resize_handler(_sig: c_int) {
72        PENDING_RESIZE.store(true, AtomicOrdering::SeqCst);
73    }
74    pub(super) fn take_resize() -> bool {
75        PENDING_RESIZE.swap(false, AtomicOrdering::SeqCst)
76    }
77}
