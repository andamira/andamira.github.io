12#[cfg(feature = "time")]
13const TERM_ESC_BLOCKING_TIMEOUT: Duration = Duration::from_millis(10);
14
15/// # Event polling
16impl TermLinux {
17    /// Polls for one terminal event using the default low-latency policy.
18    ///
19    /// This is non-blocking. It drains ready bytes and resolves a pending
20    /// lone Escape when no more bytes are immediately available.
21    pub fn poll_event(&mut self) -> LinuxResult<Option<EventKind>> {
22        self.poll_event_immediate_impl()
23    }
24    /// Polls for one terminal event using `policy`.
25    ///
26    /// # Features
27    /// With the `time` feature, this also supports [`TermPollPolicy::Timeout`].
28    pub fn poll_event_with(&mut self, policy: TermPollPolicy) -> LinuxResult<Option<EventKind>> {
29        match policy {
30            TermPollPolicy::Pending => self.poll_event_pending(),
31            TermPollPolicy::Immediate => self.poll_event(),
32            #[cfg(feature = "time")]
33            TermPollPolicy::Timeout(timeout) => self.poll_event_timeout(timeout),
34            TermPollPolicy::Blocking => self.wait_event().map(Some),
35        }
36    }
37    /// Non-blocking. Preserves pending partial input sequences.
38    pub fn poll_event_pending(&mut self) -> LinuxResult<Option<EventKind>> {
39        loop {
40            is! { let Some(ev) = self.poll_buffered_event(), return Ok(Some(ev)) }
41            is! { self.input_buf.refill_from_stdin()? == 0, return Ok(None) }
42        }
43    }
44    /// Non-blocking. Resolves pending lone Escape after ready bytes are drained.
45    fn poll_event_immediate_impl(&mut self) -> LinuxResult<Option<EventKind>> {
46        loop {
47            is! { let Some(ev) = self.poll_buffered_event(), return Ok(Some(ev)) }
48            is! { self.input_buf.refill_from_stdin()? == 0, return Ok(self.parser.flush_escape()) }
49        }
50    }
51    /// Non-blocking. Resolves pending lone Escape after `timeout`.
52    ///
53    /// This only waits when the parser is holding a pending `ESC`.
54    #[cfg(feature = "time")]
55    pub fn poll_event_timeout(&mut self, timeout: Duration) -> LinuxResult<Option<EventKind>> {
56        loop {
57            is! { let Some(ev) = self.poll_buffered_event(), return Ok(Some(ev)) }
58            is! { self.input_buf.refill_from_stdin()? != 0, continue }
59
60            // No ready bytes. If there is no pending ESC, return immediately.
61            is! { !self.parser.is_pending_escape(), return Ok(None) }
62
63            // Pending ESC: give a real escape sequence a short chance to arrive.
64            Linux::sleep(timeout)?;
65
66            is! { self.input_buf.refill_from_stdin()? != 0, continue }
67            return Ok(self.parser.flush_escape());
68        }
69    }
70    /// Blocks until a terminal event is available.
71    ///
72    /// Buffered events are returned first. Newly read bytes are parsed through
73    /// the semantic event queue, preserving the same construction
74    /// and timestamping path used by non-blocking polling.
75    ///
76    /// With `time`, a pending lone `ESC` is briefly disambiguated
77    /// before being emitted as an Escape event.
78    pub fn wait_event(&mut self) -> LinuxResult<EventKind> {
79        loop {
80            is! { let Some(ev) = self.poll_buffered_event(), return Ok(ev) }
81            let byte = Linux::get_byte()?;
82            let _ = self.feed_byte_to_event_queue(byte);
83            is! { let Some(ev) = self.pop_event_kind(), return Ok(ev) }
84            // Drain ready bytes so multibyte input and escape sequences can finish.
85            while self.input_buf.refill_from_stdin()? != 0 {
86                self.fill_event_queue_from_input();
87                is! { let Some(ev) = self.pop_event_kind(), return Ok(ev) }
88            }
89            // Resolve a standalone ESC without bypassing the semantic event queue.
90            if self.parser.is_pending_escape() {
91                #[cfg(feature = "time")]
92                {
93                    Linux::sleep(TERM_ESC_BLOCKING_TIMEOUT)?;
94                    if self.input_buf.refill_from_stdin()? != 0 {
95                        self.fill_event_queue_from_input();
96                        is! { let Some(ev) = self.pop_event_kind(), return Ok(ev) }
97                    }
98                }
99                if let Some(ev) = self.parser.flush_escape() {
100                    let queued = self.queue_event(ev);
101                    debug_assert!(queued);
102                    is! { let Some(ev) = self.pop_event_kind(), return Ok(ev) }
103                }
104            }
105        }
106    }
107}

109// Internal helpers
110impl TermLinux {
111    #[cfg(feature = "time")]
112    fn event_from_kind(&self, kind: EventKind) -> Event {
113        let ns = <LinuxInstant as TimeSource<u64>>::time_now();
114        let ms = (ns / 1_000_000).min(u32::MAX as u64) as u32;
115        Event::new(kind, Some(EventTimestamp::from_millis_u32(ms)))
116    }
117    #[cfg(not(feature = "time"))]
118    fn event_from_kind(&self, kind: EventKind) -> Event {
119        Event::new(kind, None)
120    }
121
122    /// Feeds one byte into the parser and queues a completed event, if any.
123    ///
124    /// Returns `true` if an event was queued.
125    fn feed_byte_to_event_queue(&mut self, byte: u8) -> bool {
126        is! { let Some(ev) = self.parser.feed(byte), self.queue_event(ev), false }
127    }
128    /// Feeds buffered bytes into the parser
129    /// until the semantic event queue is full or the byte buffer is empty.
130    fn fill_event_queue_from_input(&mut self) {
131        while !self.events.is_full() {
132            let Some(byte) = self.input_buf.pop_front() else { break };
133            let queued = self.feed_byte_to_event_queue(byte);
134            debug_assert!(!queued || !self.events.is_empty());
135        }
136    }
137    fn fill_event_queue_from_app_controls(&mut self) {
138        let controls = Self::take_app_controls();
139        controls.for_each_while(|control| {
140            let queued = self.queue_event(EventKind::Control(control));
141            debug_assert!(queued || self.events.is_full());
142            !self.events.is_full()
143        });
144    }
145    fn fill_event_queue_from_resize(&mut self) {
146        is! { !Self::take_resize(), return }
147        if let Ok(Some(size)) = self.refresh_size() {
148            let extent = ext![size.cols as u32, size.rows as u32];
149            let queued = self.queue_event(EventKind::Window(EventWindow::Resized(Some(extent))));
150            debug_assert!(queued || self.events.is_full());
151        } else {
152            let queued = self.queue_event(EventKind::Window(EventWindow::Resized(None)));
153            debug_assert!(queued || self.events.is_full());
154        }
155    }
156    /// Returns one queued event, filling the queue from buffered input if needed.
157    fn poll_buffered_event(&mut self) -> Option<EventKind> {
158        // events already queued have top priority
159        is! { let Some(ev) = self.pop_event_kind(), return Some(ev) }
160        // control notices have priority over newly parsed input
161        self.fill_event_queue_from_app_controls();
162        is! { let Some(ev) = self.pop_event_kind(), return Some(ev) }
163        // resize notices are also external semantic events
164        self.fill_event_queue_from_resize();
165        is! { let Some(ev) = self.pop_event_kind(), return Some(ev) }
166        self.fill_event_queue_from_input();
167        self.pop_event_kind()
168    }
169    /// Queues one terminal event kind as a global event.
170    ///
171    /// Returns `false` if the semantic event queue is full.
172    pub(super) fn queue_event(&mut self, kind: EventKind) -> bool {
173        let ev = self.event_from_kind(kind);
174        self.events.try_push(ev).is_ok()
175    }
176    /// Pops the oldest queued terminal event kind.
177    fn pop_event_kind(&mut self) -> Option<EventKind> {
178        self.events.pop().map(|ev| ev.kind)
179    }
