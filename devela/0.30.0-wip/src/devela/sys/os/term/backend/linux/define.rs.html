35/// Owns terminal input parsing, scoped session control, cached terminal and
36/// runtime capabilities, and the last known terminal size over the lower-level
37/// [`Linux`] terminal helpers.
38#[derive(Clone, Debug)]
39pub struct TermLinux {
40    #[cfg(feature = "event")]
41    pub(super) parser: TermInputParser,
42    #[cfg(feature = "event")]
43    pub(super) input_buf: TermLinuxInputBuf<TERM_INPUT_BUF_CAP>,
44    #[cfg(feature = "event")]
45    pub(super) events: EventQueue<TERM_EVENT_QUEUE_CAP>,
46
47    pub(super) term_caps: TermCaps,
48    pub(super) run_cap: RunCap,
49    pub(super) size: Option<TermSize>,
50}
51
52/// # Core frontend control
53impl TermLinux {
54    /// Opens a Linux terminal frontend.
55    ///
56    /// This does not enter raw mode and does not install terminal cleanup guards.
57    /// Session discipline belongs to the future terminal session layer.
58    pub fn open() -> LinuxResult<Self> {
59        let size = Linux::terminal_size().ok();
60        let mut term = Self {
61            #[cfg(feature = "event")]
62            parser: TermInputParser::new(),
63            #[cfg(feature = "event")]
64            input_buf: TermLinuxInputBuf::new(),
65            #[cfg(feature = "event")]
66            events: EventQueue::new(),
67
68            term_caps: TermCaps::ANSI_BASE,
69            run_cap: RunCap::default(),
70            size,
71        };
72        term.run_cap = Self::derive_run_capabilities(term.term_caps, term.size);
73        Ok(term)
74    }
75
76    /// Enters a scoped terminal session.
77    ///
78    /// Applies the requested terminal `mode` changes and
79    /// returns a guard that restores the changed state when dropped.
80    ///
81    /// The requested mode is translated by the backend restore payload,
82    /// which records only the changes that must be undone on drop.
83    ///
84    /// Only successfully applied changes are registered for restoration.
85    /// This makes partial setup failures safe: if entering a later mode fails,
86    /// earlier changes are still undone as the restore payload is dropped.
87    ///
88    /// The returned guard does not borrow `self`, so the terminal
89    /// frontend can continue to be used while the session is active.
90    pub fn session(&mut self, mode: TermMode) -> LinuxResult<TermSession<TermLinuxRestore>> {
91        Ok(TermSession::new(TermLinuxRestore::enter(mode)?))
92    }
93    /// Enters a scoped raw terminal session.
94    ///
95    /// This is a convenience wrapper around [`session`][Self::session] with [`TermMode::raw`].
96    pub fn session_raw(&mut self) -> LinuxResult<TermSession<TermLinuxRestore>> {
97        Ok(TermSession::new(TermLinuxRestore::raw()?))
98    }
99
100    /// Writes bytes to stdout.
101    pub fn print(&mut self, bytes: &[u8]) -> LinuxResult<()> {
102        Linux::print_bytes(bytes)
103    }
104
105    /// Returns the last known terminal size.
106    pub const fn size(&self) -> Option<TermSize> {
107        self.size
108    }
109    /// Refreshes and returns the terminal size.
110    pub fn refresh_size(&mut self) -> LinuxResult<Option<TermSize>> {
111        self.size = Linux::terminal_size().ok();
112        Ok(self.size)
113    }
114}
