10/// A cell-form UI adapter for terminal grids.

12    location("sys/os/term", struct TermCellUi),

15/// Projects backend-neutral UI drawing records into a [`TermGrid`] of
16/// [`Termel<char>`] values.
17///
18/// Logical UI geometry is converted into discrete terminal cells using the
19/// configured [`UiCellMetric`].
20///
21/// The style resolver returns a terminal-cell template. Rectangle operations
22/// use its character and appearance, while text operations replace its
23/// character with the corresponding text scalar.
24///
25/// Transparent foreground and background colors preserve the previously drawn
26/// cell colors. Other unresolved composition modes remain available for later
27/// processing.
28#[must_use]
29#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
30pub struct TermCellUi {
31    metric: UiCellMetric,
32}
33
34#[rustfmt::skip]
35impl TermCellUi {
36    /// Constructs a terminal-cell UI adapter.
37    pub const fn new(metric: UiCellMetric) -> Self {
38        Self { metric }
39    }
40    /// Returns the logical cell metric.
41    pub const fn metric(&self) -> UiCellMetric { self.metric }
42}

44impl TermCellUi {
45    /// Renders a painter-ordered UI draw list into `grid`.
46    pub fn render_with<S, T, D>(
47        &self,
48        draws: &UiDrawListView<'_, S, T>,
49        grid: &mut TermGrid<Termel<char>, D>,
50        mut resolve: impl FnMut(&S) -> Termel<char>,
51    ) where
52        T: AsRef<str>,
53        D: AsRef<[Termel<char>]> + AsMut<[Termel<char>]>,
54    {
55        for draw in draws.iter() {
56            match draw.kind() {
57                UiDrawKind::RectFill { style } => {
58                    let rect = self.metric.rect_to_cells(draw.rect(), UiRound::Outward);
59                    render_fill(grid, rect, resolve(style));
60                }
61                UiDrawKind::RectStroke { width, style } => {
62                    let rect = self.metric.rect_to_cells(draw.rect(), UiRound::Outward);
63                    let cols = self.metric.x_to_col(*width, UiRound::Ceil);
64                    let rows = self.metric.y_to_row(*width, UiRound::Ceil);
65                    render_inner_stroke(grid, rect, cols, rows, resolve(style));
66                }
67                UiDrawKind::Text { text, style } => {
68                    let rect = self.metric.rect_to_cells(draw.rect(), UiRound::Nearest);
69                    render_text(grid, rect, text.as_ref(), resolve(style));
70                }
71            }
72        }
73    }
74}

80fn render_fill<D>(grid: &mut TermGrid<Termel<char>, D>, rect: CellRect, cell: Termel<char>)
81where
82    D: AsRef<[Termel<char>]> + AsMut<[Termel<char>]>,
83{
84    paint_region(grid, rect, cell);
85}

87/// Renders a stroke entirely inside `rect`.
88///
89/// Horizontal and vertical thicknesses may differ because terminal cells need
90/// not represent square logical regions.
91fn render_inner_stroke<D>(
92    grid: &mut TermGrid<Termel<char>, D>,
93    rect: CellRect,
94    cols: i32,
95    rows: i32,
96    cell: Termel<char>,
97) where
98    D: AsRef<[Termel<char>]> + AsMut<[Termel<char>]>,
99{
100    let (x, y, w, h) = (rect.x(), rect.y(), rect.w(), rect.h());
101    is! { cols <= 0 || rows <= 0 || w <= 0 || h <= 0, return }
102    let double_cols = cols.saturating_mul(2);
103    let double_rows = rows.saturating_mul(2);
104    if double_cols >= w || double_rows >= h {
105        paint_region(grid, rect, cell);
106        return;
107    }
108    let (middle_y, middle_h) = (y.saturating_add(rows), h.saturating_sub(double_rows));
109    let bottom_y = y.saturating_add(h.saturating_sub(rows));
110    let right_x = x.saturating_add(w.saturating_sub(cols));
111    // Horizontal strips
112    paint_region(grid, CellRect::from_xy_wh(x, y, w, rows), cell);
113    paint_region(grid, CellRect::from_xy_wh(x, bottom_y, w, rows), cell);
114    // Vertical strips, excluding already-painted corners
115    paint_region(grid, CellRect::from_xy_wh(x, middle_y, cols, middle_h), cell);
116    paint_region(grid, CellRect::from_xy_wh(right_x, middle_y, cols, middle_h), cell);
117}

119/// Renders one unwrapped width-one text run.
120///
121/// Text is clipped to the rectangle and grid. Line-break characters terminate
122/// the run.
123fn render_text<D>(
124    grid: &mut TermGrid<Termel<char>, D>,
125    rect: CellRect,
126    text: &str,
127    template: Termel<char>,
128) where
129    D: AsRef<[Termel<char>]> + AsMut<[Termel<char>]>,
130{
131    is! { rect.w() <= 0 || rect.h() <= 0, return }
132    let y = rect.y();
133    is! { y < 0 || y as usize >= grid.height(), return }
134    let max_cols = rect.w() as usize;
135    for (column, ch) in text.chars().enumerate() {
136        is! { column >= max_cols || matches!(ch, '\r' | '\n'), break }
137        let x = rect.x().saturating_add(column as i32);
138        is! { x < 0 || x as usize >= grid.width(), continue }
139        let (x, y) = (x as usize, y as usize);
140        let under = grid.get_xy_copy(x, y).unwrap();
141        let over = template.with_value(ch);
142        let _ = grid.set_xy(x, y, compose_cell(under, over));
143    }
144}

146/// Paints `cell` over every visible position in `rect`.
147fn paint_region<D>(grid: &mut TermGrid<Termel<char>, D>, rect: CellRect, cell: Termel<char>)
148where
149    D: AsRef<[Termel<char>]> + AsMut<[Termel<char>]>,
150{
151    let Some(region) = clip_region(rect, grid.width(), grid.height()) else {
152        return;
153    };
154    // Fully opaque cells need no read-modify-write composition.
155    if cell.fg().is_opaque() && cell.bg().is_opaque() {
156        grid.fill_region(region, cell);
157        return;
158    }
159    let (x0, y0) = (region.x(), region.y());
160    let (x1, y1) = (x0 + region.w(), y0 + region.h());
161    for y in y0..y1 {
162        for x in x0..x1 {
163            let under = grid.get_xy_copy(x, y).unwrap();
164            let _ = grid.set_xy(x, y, compose_cell(under, cell));
165        }
166    }
167}

169/// Composes one terminal cell over another.
170///
171/// Transparent colors preserve their corresponding underlying color.
172/// Character and style come from the upper cell.
173fn compose_cell(under: Termel<char>, over: Termel<char>) -> Termel<char> {
174    let colors =
175        TermColors::new(compose_color(under.fg(), over.fg()), compose_color(under.bg(), over.bg()));
176    Termel::from_value(*over.value(), *over.style(), colors)
177}
178const fn compose_color(under: TermColor, over: TermColor) -> TermColor {
179    if over.is_transparent() { under } else { over }
180}

182/// Clips a signed cell rectangle to a grid and converts it to unsigned cells.
183fn clip_region(rect: CellRect, width: usize, height: usize) -> Option<RegionS2<usize>> {
184    is! { rect.w() <= 0 || rect.h() <= 0, return None }
185    let x0 = rect.x().max(0) as usize;
186    let y0 = rect.y().max(0) as usize;
187    let x1 = rect.x().saturating_add(rect.w()).max(0) as usize;
188    let y1 = rect.y().saturating_add(rect.h()).max(0) as usize;
189    let x1 = x1.min(width);
190    let y1 = y1.min(height);
191    is! { x0 >= x1 || y0 >= y1, return None }
192    Some(RegionS2::from_xy_wh(x0, y0, x1 - x0, y1 - y0))
193}

198    use crate::{Extent2, Lunit, TermColorMode, TermStyle, UiDraw, UiOutputView, UiRect};

200    type Cell = Termel<char>;

202    #[derive(Clone, Copy, Debug, PartialEq, Eq)]
203    enum Style {
204        Fill,
205        Stroke,
206        Text,
207    }
208
209    const FILL_BG: TermColor = TermColor::rgb([10, 20, 30]);
210    const TEXT_FG: TermColor = TermColor::rgb([220, 230, 240]);
211    const TRANSPARENT: TermColor = TermColor::default_with_mode(TermColorMode::Transparent);
212
213    const fn rect(x: i32, y: i32, width: i32, height: i32) -> UiRect {
214        UiRect::from_xy_wh(Lunit::px(x), Lunit::px(y), Lunit::px(width), Lunit::px(height))
215    }
216    const fn resolve(style: Style) -> Cell {
217        match style {
218            Style::Fill => Termel::from_value(
219                'F',
220                TermStyle::new(),
221                TermColors::new(TermColor::DEFAULT, FILL_BG),
222            ),
223            Style::Stroke => Termel::from_value('S', TermStyle::new(), TermColors::DEFAULT),
224            Style::Text => {
225                Termel::from_value(' ', TermStyle::new(), TermColors::new(TEXT_FG, TRANSPARENT))
226            }
227        }
228    }

230    #[test]
231    fn projects_non_square_cell_metrics() {
232        const W: usize = 5;
233        const H: usize = 4;
234        let mut grid =
235            TermGrid::new([Cell::plain_const('.'); W * H], Extent2::new([W, H])).unwrap();
236        let ui = TermCellUi::new(UiCellMetric::from_px(2, 3).unwrap());
237        let draws: [UiDraw<Style>; 1] = [UiDraw::rect_fill(rect(2, 3, 4, 6), Style::Fill)];
238        let output = UiOutputView::from_slices(&[], &[], &draws);
239        ui.render_with(output.draw_list(), &mut grid, |style| resolve(*style));
240        // Logical (2, 3, 4, 6) becomes cells (1, 1, 2, 2).
241        assert_eq!(*grid.get_xy(1, 1).unwrap().value(), 'F',);
242        assert_eq!(*grid.get_xy(2, 2).unwrap().value(), 'F',);
243        assert_eq!(*grid.get_xy(0, 0).unwrap().value(), '.',);
244        assert_eq!(*grid.get_xy(3, 2).unwrap().value(), '.',);
245    }
246    #[test]
247    fn stroke_remains_inside_its_rectangle() {
248        const W: usize = 6;
249        const H: usize = 6;
250        let mut grid =
251            TermGrid::new([Cell::plain_const('.'); W * H], Extent2::new([W, H])).unwrap();
252        let ui = TermCellUi::new(UiCellMetric::ONE);
253        let draws: [UiDraw<Style>; 1] =
254            [UiDraw::rect_stroke(rect(1, 1, 4, 4), Lunit::px(1), Style::Stroke)];
255        let output = UiOutputView::from_slices(&[], &[], &draws);
256        ui.render_with(output.draw_list(), &mut grid, |style| resolve(*style));
257        for y in 0..H {
258            for x in 0..W {
259                let value = *grid.get_xy(x, y).unwrap().value();
260                let on_edge = ((x == 1 || x == 4) && (1..=4).contains(&y))
261                    || ((y == 1 || y == 4) && (1..=4).contains(&x));
262                assert_eq!(value, if on_edge { 'S' } else { '.' }, "unexpected cell at ({x}, {y})",);
263            }
264        }
265        // The inner region remains untouched.
266        assert_eq!(*grid.get_xy(2, 2).unwrap().value(), '.',);
267    }
