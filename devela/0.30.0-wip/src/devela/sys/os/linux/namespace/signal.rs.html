18/// # Signal handling
19///
20/// These methods install process-wide Linux signal actions. Signal disposition is
21/// global to the process, so registering a handler for the same signal may
22/// replace an earlier registration.
23impl Linux {
24    /// Registers Linux handlers for application control notices.
25    ///
26    /// This is a semantic wrapper over [`sig_handler`][Self::sig_handler]. It translates
27    /// matching Linux signals into [`AppControl`] values before calling `handler`.
28    ///
29    /// This installs process-wide signal handlers for the signals represented by `controls`.
30    /// Registering another handler for the same signals may replace this path, and calling
31    /// this method again replaces the single app-control callback used by this adapter.
32    ///
33    /// # Notes
34    /// - Only controls with a Linux signal mapping are registered.
35    /// - `SIGKILL` and `SIGSTOP` cannot be caught.
36    /// - The callback runs from a signal handler; keep it small and prefer atomics.
37    pub fn app_control_handler(
38        handler: fn(AppControl),
39        controls: AppControlSet,
40        flags: impl Into<LinuxSigactionFlags>,
41    ) {
42        static HANDLER: AtomicPtr<()> = AtomicPtr::new(Ptr::null_mut());
43        fn adapter(sig: c_int) {
44            let Some(control) = LinuxSignal::from_c_int(sig).and_then(LinuxSignal::to_app_control)
45            else {
46                return;
47            };
48            let handler = HANDLER.load(SeqCst);
49            if !handler.is_null() {
50                // SAFETY: stored from a `fn(AppControl)` and checked for null.
51                let handler: fn(AppControl) = unsafe { transmute(handler) };
52                handler(control);
53            }
54        }
55        HANDLER.store(handler as *mut (), SeqCst);
56        Self::sig_handler(adapter, controls.to_linux_signals(), flags.into());
57    }

59    /// Registers multiple signals using a simple handler function.
60    ///
61    /// This is a convenience wrapper over [`sig_handler_info`][Self::sig_handler_info].
62    /// Internally it installs an `SA_SIGINFO` handler and discards the extra signal
63    /// information before calling `handler`.
64    ///
65    /// # Arguments
66    /// - `handler`: Function called with the received signal number.
67    /// - `signals`: Signals to handle.
68    /// - `flags`: Additional signal action flags.
69    ///
70    /// # Notes
71    /// - `SA_RESTORER` and `SA_SIGINFO` are always included internally.
72    /// - `SIGKILL` and `SIGSTOP` are ignored because they cannot be caught.
73    /// - Signal handlers run asynchronously; keep them small and prefer atomics.
74    ///
75    /// # Examples
76    /// ```no_run
77    /// # #[cfg(feature = "std")] {
78    /// use devela::{AtomicBool, AtomicI32, AtomicOrdering::SeqCst, Thread, ThreadExt};
79    /// use devela::{Linux, LinuxSignal, LinuxSignalSet};
80    ///
81    /// static GOT_SIGNAL: AtomicBool = AtomicBool::new(false);
82    /// static LAST_SIGNAL: AtomicI32 = AtomicI32::new(0);
83    ///
84    /// fn handler(sig: i32) {
85    ///     LAST_SIGNAL.store(sig, SeqCst);
86    ///     GOT_SIGNAL.store(true, SeqCst);
87    /// }
88    ///
89    /// fn main() {
90    ///     Linux::sig_handler(handler, LinuxSignalSet::SIGINT | LinuxSignalSet::SIGWINCH, None);
91    ///
92    ///     println!("press Ctrl+C or resize the terminal");
93    ///     loop {
94    ///         if GOT_SIGNAL.swap(false, SeqCst) {
95    ///             let sig = LAST_SIGNAL.load(SeqCst);
96    ///             println!("received signal {sig}");
97    ///             if sig == LinuxSignal::SIGINT.as_c_int() {
98    ///                 break;
99    ///             }
100    ///         }
101    ///         Thread::sleep_ms(50);
102    ///     }
103    ///     println!("bye");
104    /// }
105    /// # }
106    /// ```
107    pub fn sig_handler(
108        handler: fn(c_int),
109        signals: LinuxSignalSet,
110        flags: impl Into<LinuxSigactionFlags>,
111    ) {
112        fn simple_adapter(sig: c_int, _info: &LinuxSiginfo, _context: *mut c_void) {
113            is! { LinuxSignal::from_c_int(sig).is_none(), return }
114            let handler = LINUX_SIG_HANDLERS[sig as usize].load(SeqCst);
115            if !handler.is_null() {
116                // SAFETY: we control both storage and retrieval and check for null.
117                let handler: fn(c_int) = unsafe { transmute(handler) };
118                handler(sig);
119            }
120        }
121        signals.for_each_catchable(|signal| {
122            LINUX_SIG_HANDLERS[signal.as_c_int() as usize].store(handler as *mut (), SeqCst);
123        });
124        Self::sig_handler_info(simple_adapter, signals, flags);
125    }

127    /// Registers multiple signals using a handler that receives signal information.
128    ///
129    /// The handler receives the signal number, a borrowed [`LinuxSiginfo`], and the
130    /// raw signal context pointer supplied by the kernel.
131    ///
132    /// # Arguments
133    /// - `handler`: Function called with the signal number, signal info, and context.
134    /// - `signals`: Signals to handle.
135    /// - `flags`: Additional signal action flags.
136    ///
137    /// # Notes
138    /// - `SA_RESTORER` and `SA_SIGINFO` are always included internally.
139    /// - `SIGKILL` and `SIGSTOP` are ignored because they cannot be caught.
140    /// - Signal handlers run asynchronously; keep them small and prefer atomics.
141    /// - The `LinuxSiginfo` borrow is valid only during the handler call.
142    ///
143    /// # Examples
144    /// ```no_run
145    /// # #[cfg(feature = "std")] {
146    /// use devela::{AtomicBool, AtomicI32, AtomicOrdering::SeqCst, Thread, ThreadExt, c_void};
147    /// use devela::{Linux, LinuxSigactionFlags, LinuxSiginfo, LinuxSignalSet};
148    ///
149    /// static GOT_SIGNAL: AtomicBool = AtomicBool::new(false);
150    /// static LAST_SIGNAL: AtomicI32 = AtomicI32::new(0);
151    /// static LAST_PID: AtomicI32 = AtomicI32::new(0);
152    ///
153    /// fn handler(sig: i32, info: &LinuxSiginfo, _context: *mut c_void) {
154    ///     LAST_SIGNAL.store(sig, SeqCst);
155    ///     LAST_PID.store(info.pid(), SeqCst);
156    ///     GOT_SIGNAL.store(true, SeqCst);
157    /// }
158    ///
159    /// fn main() {
160    ///     Linux::sig_handler_info(
161    ///         handler,
162    ///         LinuxSignalSet::SIGINT,
163    ///         LinuxSigactionFlags::SA_RESTART,
164    ///     );
165    ///
166    ///     println!("press Ctrl+C");
167    ///     loop {
168    ///         if GOT_SIGNAL.swap(false, SeqCst) {
169    ///             let sig = LAST_SIGNAL.load(SeqCst);
170    ///             let pid = LAST_PID.load(SeqCst);
171    ///             println!("received signal {sig} from pid {pid}");
172    ///             break;
173    ///         }
174    ///         Thread::sleep_ms(50);
175    ///     }
176    ///     println!("bye");
177    /// }
178    /// # }
179    /// ```
180    pub fn sig_handler_info(
181        handler: fn(i32, &LinuxSiginfo, *mut c_void),
182        signals: LinuxSignalSet,
183        flags: impl Into<LinuxSigactionFlags>,
184    ) {
185        extern "C" fn c_handler_siginfo(sig: c_int, info: *mut LinuxSiginfo, context: *mut c_void) {
186            is! { LinuxSignal::from_c_int(sig).is_none() || info.is_null(), return }
187            let handler = LINUX_SIGINFO_HANDLERS[sig as usize].load(SeqCst);
188            if !handler.is_null() {
189                // SAFETY: we control both storage and retrieval and check for null.
190                let handler: fn(c_int, &LinuxSiginfo, *mut c_void) = unsafe { transmute(handler) };
191
192                // SAFETY: kernel-provided for this SA_SIGINFO handler call.
193                let info = unsafe { &*info };
194                handler(sig, info, context);
195            }
196        }
197        // let signals = signals.catchable_only();
198        let flags = flags.into().with_sa_restorer().with_sa_siginfo();
199        let sigaction = LinuxSigaction::new_siginfo(
200            c_handler_siginfo,
201            flags,
202            LinuxSigset::empty(),
203            Some(__devela_linux_restore_rt),
204        );
205        signals.for_each_catchable(|signal| {
206            let raw = signal.as_c_int();
207            LINUX_SIGINFO_HANDLERS[raw as usize].store(handler as *mut (), SeqCst);
208            let _res = unsafe {
209                Linux::sys_rt_sigaction(raw, &sigaction, Ptr::null_mut(), LinuxSigset::size())
210            };
211        });
212    }
