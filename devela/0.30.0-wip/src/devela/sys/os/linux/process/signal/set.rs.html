15    /// The enum discriminants are zero-based semantic indices, not raw Linux
16    /// signal numbers. Use [`as_c_int`][Self::as_c_int] for the raw signal number.
17    //
18    // For the terminal, the key signals are:
19    // - SIGINT    // Ctrl+C
20    // - SIGQUIT   // Ctrl+\
21    // - SIGTSTP   // Ctrl+Z
22    // - SIGTTIN   // background terminal read
23    // - SIGTTOU   // background terminal write
24    // - SIGWINCH  // terminal resize
25    // - SIGHUP    // terminal/session hangup
26    #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
27    pub enum LinuxSignal(
28        #[doc = crate::_tags!(linux signal set)]
29        /// A compact semantic set of standard Linux signals.
30        #[doc = crate::_doc_meta!{
31            location("sys/os/linux/process", struct LinuxSignalSet),
32            test_size_of(LinuxSignalSet = 4|32; niche !Option),
33        }]
34        /// This is not the raw Linux `sigset_t` mask. Signal bits are stored
35        /// compactly from zero: `SIGHUP` is bit `0`, `SIGINT` is bit `1`, …
36        pub LinuxSignalSet: u32
37    ) {
38        /// Terminal hangup or controlling process ended.
39        SIGHUP // = 0
40        /// Keyboard interrupt, usually `Ctrl+C`.
41        SIGINT, // = 1
42        /// Keyboard quit, usually `Ctrl+\`; often produces a core dump.
43        SIGQUIT, // = 2
44        /// Illegal CPU instruction.
45        SIGILL, // = 3
46        /// Trace or breakpoint trap, commonly used by debuggers.
47        SIGTRAP, // = 4
48        /// Process abort, usually from `abort()`.
49        SIGABRT, // = 5
50        /// Bus error, usually invalid or misaligned memory access.
51        SIGBUS, // = 6
52        /// Floating-point exception, such as division by zero in some contexts.
53        SIGFPE, // = 7
54        /// Uncatchable forced termination.
55        SIGKILL, // = 8
56        /// User-defined signal 1.
57        SIGUSR1, // = 9
58        /// Invalid memory reference, usually a segmentation fault.
59        SIGSEGV, // = 10
60        /// User-defined signal 2.
61        SIGUSR2, // = 11
62        /// Broken pipe, usually writing to a pipe or socket with no reader.
63        SIGPIPE, // = 12
64        /// Alarm timer expired, usually from `alarm()`.
65        SIGALRM, // = 13
66        /// Polite termination request, commonly sent by `kill` or service managers.
67        SIGTERM, // = 14
68        /// Stack fault signal, obsolete on most modern Linux systems.
69        SIGSTKFLT, // = 15
70        /// Child process stopped, continued, or exited.
71        SIGCHLD, // = 16
72        /// Continue a stopped process.
73        SIGCONT, // = 17
74        /// Uncatchable stop signal.
75        SIGSTOP, // = 18
76        /// Terminal stop, usually `Ctrl+Z`.
77        SIGTSTP, // = 19
78        /// Background process tried to read from the terminal.
79        SIGTTIN, // = 20
80        /// Background process tried to write to the terminal.
81        SIGTTOU, // = 21
82        /// Urgent condition on a socket.
83        SIGURG, // = 22
84        /// CPU time limit exceeded.
85        SIGXCPU, // = 23
86        /// File size limit exceeded.
87        SIGXFSZ, // = 24
88        /// Virtual interval timer expired.
89        SIGVTALRM, // = 25
90        /// Profiling timer expired.
91        SIGPROF, // = 26
92        /// Terminal or window size changed.
93        SIGWINCH, // = 27
94        /// Asynchronous I/O event is available.
95        SIGIO, // = 28
96        /// Power failure or power-management event.
97        SIGPWR, // = 29
98        /// Bad system call, often from seccomp filtering.
99        SIGSYS, // = 30
100    }

170        /// Returns whether this signal can be caught by a user handler.
171        #[must_use]
172        pub const fn is_catchable(self) -> bool {
173            !matches!(self, Self::SIGKILL | Self::SIGSTOP)
174        }
175        /// Returns whether the raw signal number is supported.
176        #[must_use]
177        pub const fn is_standard_number(signal: c_int) -> bool {
178            Self::from_c_int(signal).is_some()
179        }
180        /// Returns whether the raw signal number can be caught by a user handler.
181        #[must_use]
182        pub const fn is_catchable_number(signal: c_int) -> bool {
183            match Self::from_c_int(signal) {
184                Some(signal) => signal.is_catchable(),

204    impl set {
205        /// Returns a singleton set containing `signal`.
206        #[must_use]
207        pub const fn from_signal(signal: LinuxSignal) -> Self {
208            signal.to_set()
209        }
210        /// Returns the only signal in this set, if it contains exactly one.
211        #[must_use]
212        pub const fn to_signal(self) -> Option<LinuxSignal> {
213            let bits = self.bits();
214            if bits != 0 && bits.is_power_of_two() {
215                LinuxSignal::from_c_int(bits.trailing_zeros() as c_int + 1)
216            } else {
217                None
218            }
219        }
220        /// Calls `f` for each catchable signal in this set.
221        pub fn for_each_catchable(self, f: impl FnMut(LinuxSignal)) {
222            self.catchable_only().for_each(f);
223        }
224
225        /// Returns whether all signals in this set are catchable.
226        #[must_use]
227        pub const fn all_catchable(self) -> bool {
228            !self.has(Self::SIGKILL) && !self.has(Self::SIGSTOP)
229        }
230        /// Returns this set without without signals that cannot be caught.
231        #[must_use]
232        pub const fn catchable_only(self) -> Self {
233            self.without(Self::SIGKILL).without(Self::SIGSTOP)
234        }
235    }

257/// This is the mask representation passed to raw Linux signal syscalls.
258///
259/// Signal numbers start at `1`, while mask bits start at `0`,
260/// so signal `n` is stored in bit `n - 1`.
261///
262/// For the semantic devela set, use [`LinuxSignalSet`].
263//
264// This one resembles the [`sigset_t`] structure from libc.
265// [`sigset_t`]: https://man7.org/linux/man-pages/man7/system_data_types.7.html
266#[repr(C)]
267#[must_use]
268#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
269pub struct LinuxSigset {
270    /// Raw signal mask words.
271    pub sig: [c_size_t; Self::WORDS],
272}
273#[rustfmt::skip]
274impl LinuxSigset {
275    /// Number of bits in one raw mask word.
276    pub const WORD_BITS: usize = usize::BITS as usize;
277    /// Highest supported standard Linux signal number.
278    pub const MAX_SIGNAL: usize = LinuxSignal::MAX_NUMBER as usize;
279    /// Number of raw mask words needed for the supported signal range.
280    pub const WORDS: usize = Self::MAX_SIGNAL.div_ceil(Self::WORD_BITS);
281
282    /// Returns an empty signal mask.
283    pub const fn empty() -> Self { Self { sig: [0; Self::WORDS] } }
284    /// Returns the size in bytes passed to raw Linux signal syscalls.
285    #[must_use]
286    pub const fn size() -> c_size_t { size_of::<Self>() }
287
288
289    /// Returns a raw mask from a semantic signal set.
290    pub const fn from_signal_set(signals: LinuxSignalSet) -> Self {
291        Self { sig: [signals.bits() as usize; Self::WORDS] }
292    }
293    /// Returns this raw mask as a semantic signal set.
294    #[must_use]
295    pub const fn to_signal_set(self) -> LinuxSignalSet {
296        LinuxSignalSet::from_bits(self.sig[0] as u32)
297    }
298    /// Returns a copy with `signal` added.
299    pub const fn with_signal(mut self, signal: LinuxSignal) -> Self {
300        let bit = signal.as_index();
301        let word = bit / Self::WORD_BITS;
302        let shift = bit % Self::WORD_BITS;
303        self.sig[word] |= 1usize << shift;
304        self
305    }
306    /// Returns a copy with `signal` removed.
307    ///
308    /// Ignores unsupported signal numbers.
309    pub const fn without_signal(mut self, signal: LinuxSignal) -> Self {
310        let bit = signal.as_index();
311        let word = bit / Self::WORD_BITS;
312        let shift = bit % Self::WORD_BITS;
313        self.sig[word] &= !(1usize << shift);
314        self
315    }
316    /// Returns whether this raw mask contains `signal`.
317    #[must_use]
318    pub const fn has_signal(self, signal: LinuxSignal) -> bool {
319        let bit = signal.as_index();
320        let word = bit / Self::WORD_BITS;
321        let shift = bit % Self::WORD_BITS;
322        self.sig[word] & (1usize << shift) != 0
323    }
324    /// Adds `signal` to this raw mask.
325    pub fn insert_signal(&mut self, signal: LinuxSignal) {
326        *self = self.with_signal(signal);
327    }
328    /// Removes `signal` from this raw mask.
329    ///
330    /// # Panics
331    /// Panics if `signal` is not a supported standard Linux signal.
332    pub fn remove_signal(&mut self, signal: LinuxSignal) {
333        *self = self.without_signal(signal);
334    }
335    /// Returns a copy with the raw signal number added, if supported.
336    pub const fn with_signal_number(self, signal: c_int) -> Self {
337        match LinuxSignal::from_c_int(signal) {
338            Some(signal) => self.with_signal(signal),
339            None => self,
340        }
341    }
