13/// [`Linux`][crate::Linux] clock identifiers.
14///
15/// Used for [`sys_clock_gettime`][crate::Linux::sys_clock_gettime] and related time functions.
16#[doc = crate::_doc_meta!{
17    location("sys/os/linux/thread", enum LinuxClock),
18    test_size_of(LinuxClock = 4|32; niche Option),
19}]
20/// See [clock_gettime(2)] for detailed information about each clock type.
21///
22/// [clock_gettime(2)]: https://www.man7.org/linux/man-pages/man2/clock_gettime.2.html
23#[repr(i32)] // == c_int
24#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
25pub enum LinuxClock {
26    /// System-wide realtime clock (wall clock). Adjustable via NTP.
27    Realtime = 0,
28
29    /// Monotonic clock (steady since boot). Unaffected by NTP.
30    Monotonic = 1,
31
32    /// A clock that measures CPU time consumed by this process.
33    // since Linux 2.6.12
34    ProcessCpuTimeId = 2,
35
36    /// A clock that measures CPU time consumed by this thread.
37    // since Linux 2.6.12
38    ThreadCpuTimeId = 3,
39
40    /// Monotonic, raw hardware time (TSC-based where available).
41    // since Linux 2.6.28
42    MonotonicRaw = 4,
43
44    /// A faster but less precise version of [`Realtime`][Self::Realtime].
45    // since Linux 2.6.32
46    RealtimeCoarse = 5,
47
48    /// A faster but less precise version of [`Monotonic`][Self::Monotonic].
49    // since Linux 2.6.32
50    MonotonicCoarse = 6,
51
52    /// Like [`Monotonic`][Self::Monotonic] but includes suspend time.
53    // since Linux 2.6.39
54    Boottime = 7,
55
56    /// Like [`Realtime`][Self::Realtime], but not settable.
57    // since Linux 3.0
58    RealtimeAlarm = 8,
59
60    /// Like [`Boottime`][Self::Boottime], but not settable.
61    BoottimeAlarm = 9,
62
63    // The driver implementing this got removed
64    // SgiCycle = 10,
65    /// International Atomic Time (ignores leap seconds).
66    // since Linux 3.10
67    Tai = 11,
68}

70impl TryFrom<c_int> for LinuxClock {
71    type Error = LinuxError;
72    fn try_from(value: c_int) -> Result<Self> {
73        Self::from_raw(value).ok_or(LinuxError::Sys(LINUX_ERRNO::EINVAL))
74    }
75}
76impl From<LinuxClock> for c_int {
77    fn from(clock: LinuxClock) -> Self {
78        clock.as_raw()
79    }
80}

82impl LinuxClock {
83    /// Converts to the raw integer value for syscalls.
84    pub const fn as_raw(self) -> c_int {
85        self as c_int
86    }
87    /// Converts from a raw integer (for FFI or error handling).
88    pub const fn from_raw(value: c_int) -> Option<Self> {
89        match value {
90            0 => Some(Self::Realtime),
91            1 => Some(Self::Monotonic),
92            2 => Some(Self::ProcessCpuTimeId),
93            3 => Some(Self::ThreadCpuTimeId),
94            4 => Some(Self::MonotonicRaw),
95            5 => Some(Self::RealtimeCoarse),
96            6 => Some(Self::MonotonicCoarse),
97            7 => Some(Self::Boottime),
98            8 => Some(Self::RealtimeAlarm),
99            9 => Some(Self::BoottimeAlarm),
100            11 => Some(Self::Tai),
101            _ => None,
102        }
103    }
104}
105
106impl LinuxClock {
107    /// Is this clock adjusted by NTP or system time changes?
108    pub fn is_ntp_adjusted(self) -> bool {
109        matches!(self, Self::Realtime | Self::RealtimeCoarse)
110        // Note: TAI is not NTP-adjusted but can be manually set
111    }
112
113    /// Is this clock based on raw hardware (completely unadjusted)?
114    pub fn is_raw_hardware(self) -> bool {
115        matches!(self, Self::MonotonicRaw)
116        // TAI is also "raw" in a sense but can be set, so excluded
117    }
118
119    /// Is this clock settable via `clock_settime()`?
120    ///
121    /// Note that only alarm clocks are specifically NOT settable.
122    pub fn is_settable(self) -> bool {
123        !self.is_alarm()
124    }
125
126    /// Is this a monotonic clock? (never goes backwards)
127    ///
128    /// Note that CPU time clocks are monotonic for their process/thread.
129    pub fn is_monotonic(self) -> bool {
130        matches!(
131            self,
132            Self::Monotonic
133                | Self::MonotonicRaw
134                | Self::MonotonicCoarse
135                | Self::Boottime
136                | Self::BoottimeAlarm
137                | Self::ProcessCpuTimeId
138                | Self::ThreadCpuTimeId
139        )
140    }
141
142    /// Is this a realtime (wall clock) type? (represents calendar time)
143    pub fn is_realtime(self) -> bool {
144        matches!(self, Self::Realtime | Self::RealtimeCoarse | Self::RealtimeAlarm | Self::Tai)
145    }
146
147    /// Is this a CPU time measurement clock? (measures actual CPU usage)
148    pub fn is_cpu_time(self) -> bool {
149        matches!(self, Self::ProcessCpuTimeId | Self::ThreadCpuTimeId)
150    }
151
152    /// Is this an alarm clock? (for timerfd/signals, not settable)
153    pub fn is_alarm(self) -> bool {
154        matches!(self, Self::RealtimeAlarm | Self::BoottimeAlarm)
155    }
156
157    /// Does this clock include time during system suspend?
158    pub fn includes_suspend_time(self) -> bool {
159        matches!(self, Self::Boottime | Self::BoottimeAlarm)
160    }
161}

163#[crate::macro_apply(crate::_unsafe_syscall_not_miri)]
164impl LinuxClock {
165    /// Gets the current time for this clock
166    pub fn get_time(self) -> Result<LinuxTimespec> {
167        Linux::clock_gettime(self)
168    }
169    /// Gets the current time for this clock
170    pub fn get_duration(self) -> Result<Duration> {
171        Linux::clock_gettime(self).and_then(|ts| ts.try_to_duration().map_err(Into::into))
172    }
173
174    /// Gets the resolution for this clock
175    pub fn get_resolution(self) -> Result<LinuxTimespec> {
176        Linux::clock_getres(self)
177    }
178}
