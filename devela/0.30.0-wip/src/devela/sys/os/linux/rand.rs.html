8#[doc = crate::_tags!(rand linux)]
9/// Linux `getrandom` randomness mode.
10#[doc = crate::_doc_meta!{
11    location("sys/os/linux", enum LinuxRandomMode),
12    test_size_of(LinuxRandomMode = 1|8; niche Option),
13}]
14/// This selects the Linux `getrandom` policy
15/// used by the configurable `Linux::random_*_with` methods.
16///
17/// The default mode is [`SecureNonblock`][Self::SecureNonblock], exposed as
18/// [`Linux::RANDOM_MODE`]. It requests cryptographic random bytes without
19/// blocking; if the kernel randomness source is not ready, the operation
20/// returns an error instead.
21///
22///  [`Linux::RANDOM_MODE`]: crate::Linux::RANDOM_MODE
23#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
24pub enum LinuxRandomMode {
25    /// Cryptographic randomness, allowing the syscall to block.
26    ///
27    /// This uses no `getrandom` flags.
28    Secure,
29
30    #[doc = crate::_tags!(init)]
31    /// Cryptographic randomness without blocking.
32    ///
33    /// This uses `GRND_NONBLOCK`. If the kernel randomness source is not ready,
34    /// the syscall may fail with `EAGAIN`.
35    #[default]
36    SecureNonblock,
37
38    /// Non-cryptographic random bytes.
39    ///
40    /// This uses `GRND_INSECURE` and is suitable only
41    /// when cryptographic quality is not required.
42    Insecure,
43}
44impl LinuxRandomMode {
45    // from `sys/random.h`
46    const GRND_NONBLOCK: c_uint = 0x0001;
47    const GRND_INSECURE: c_uint = 0x0004;
48
49    /* queries */
50
51    /// Returns the Linux `getrandom` flags for this mode.
52    pub const fn flags(self) -> c_uint {
53        match self {
54            Self::Secure => 0,
55            Self::SecureNonblock => Self::GRND_NONBLOCK,
56            Self::Insecure => Self::GRND_INSECURE,
57        }
58    }
59
60    /// Returns whether this mode is intended for cryptographic randomness.
61    pub const fn is_cryptographic(self) -> bool {
62        matches!(self, Self::Secure | Self::SecureNonblock)
63    }
64    /// Returns whether this mode is known to be weak.
65    pub const fn is_weak(self) -> bool {
66        matches!(self, Self::Insecure)
67    }
68    /// Returns whether this mode may block while waiting for randomness.
69    pub const fn may_block(self) -> bool {
70        matches!(self, Self::Secure)
71    }
72}
