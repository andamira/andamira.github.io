24/// Represents the [`termios`] structure from libc, used to control terminal I/O.
25#[doc = crate::_doc_meta!{
26    location("sys/os/linux/io", struct LinuxTermios),
27    test_size_of(LinuxTermios = 36|288; niche !Option),

29/// It has fields for input, output, control, and local modes,
30/// as well as a line discipline and control characters.
31///
32/// [`termios`]: https://man7.org/linux/man-pages/man3/termios.3.html
33#[repr(C)]
34#[derive(Clone, Copy, Debug, Default, PartialEq)]
35pub struct LinuxTermios {
36    /// Input flags.
37    pub c_iflag: c_uint,
38
39    /// Output flags.
40    pub c_oflag: c_uint,
41
42    /// Control flags.
43    pub c_cflag: c_uint,
44
45    /// Local flags.
46    pub c_lflag: c_uint,
47
48    /// …
49    pub c_line: u8,
50
51    /// Special control characters.
52    pub c_cc: [u8; 19],
53}

81/// # Linux terminal state syscalls
82#[cfg(any_target_arch_linux)]
83#[crate::macro_apply(crate::_unsafe_syscall_not_miri)]
84impl LinuxTermios {
85    /// Reads the current terminal termios state.
86    pub fn read_state() -> Result<LinuxTermios> {
87        let mut state = LinuxTermios::new();
88        let res = unsafe {
89            Linux::sys_ioctl(LINUX_FILENO::STDIN, LINUX_IOCTL::TCGETS, state.as_mut_bytes_ptr())
90        };
91        is![res >= 0, Ok(state), Err(LinuxError::Sys(res))]
92    }
93    /// Writes this termios state as the current terminal state.
94    pub fn write_state(mut self) -> Result<()> {
95        let res = unsafe {
96            Linux::sys_ioctl(LINUX_FILENO::STDIN, LINUX_IOCTL::TCSETS, self.as_mut_bytes_ptr())
97        };
98        is![res >= 0, Ok(()), Err(LinuxError::Sys(res))]
99    }
100    /// Reads the current termios state, mutates it, and writes it back.
101    pub fn update_state(f: impl FnOnce(&mut Self)) -> Result<()> {
102        let mut state = Self::read_state()?;
103        f(&mut state);
104        state.write_state()
105    }
106
107    /// Returns `true` if standard input is attached to a terminal.
108    #[must_use]
109    pub fn is_terminal() -> bool {
110        match Self::read_state() {
111            Ok(_) => true,
112            Err(LinuxError::Sys(err)) => err != -LINUX_ERRNO::ENOTTY && err != -LINUX_ERRNO::EINVAL,
113            Err(_) => false, // Other errors are not related to terminal checks
114        }
115    }
116    /// Reads the terminal window size, in cells and pixels.
117    pub fn read_window_size() -> Result<TermSize> {
118        let mut winsize = TermSize::default();
119        let res = unsafe {
120            Linux::sys_ioctl(
121                LINUX_FILENO::STDIN,
122                LINUX_IOCTL::TIOCGWINSZ,
123                &mut winsize as *mut TermSize as *mut u8,
124            )
125        };
126        is![res >= 0, Ok(winsize), Err(LinuxError::Sys(res))]
127    }
128
129    /// Enables event-oriented terminal mode.
130    pub fn enable_event_mode() -> Result<()> {
131        Self::update_state(|state| state.make_event())
132    }
133    /// Enables raw terminal mode.
134    pub fn enable_raw_mode() -> Result<()> {
135        Self::update_state(|state| state.make_raw())
136    }
137    /// Best-effort reset to ordinary line-buffered terminal behavior.
138    ///
139    /// This does not restore the exact previous terminal state.
140    /// Prefer a scoped terminal session for exact restoration.
141    pub fn reset_cooked_mode() -> Result<()> {
142        Self::update_state(|state| state.make_cooked_reset())
143    }
144}

146/// # Terminal state transforms
147#[rustfmt::skip]
148impl LinuxTermios {
149    /// Makes this state event-oriented.
150    ///
151    /// Event mode disables canonical line buffering and echo, allowing input to
152    /// be read immediately while preserving terminal-generated signals.
153    pub const fn make_event(&mut self) {
154        self.remove_input_flags(I::IXON);
155        self.remove_local_flags(L::ECHO);
156        self.remove_local_flags(L::ICANON);
157        // self.remove_local_flags(L::IEXTEN);
158        self.set_read_min_timeout(1, 0);
159    }
160    /// Makes this state raw.
161    ///
162    /// Raw mode disables canonical input, echo, software flow control, input
163    /// translations, output processing, extensions, and terminal-generated
164    /// signal characters.
165    ///
166    /// In this mode, control keys such as Ctrl-C and Ctrl-Q are delivered as
167    /// input bytes instead of being handled by the terminal line discipline.
168    pub const fn make_raw(&mut self) {
169        self.remove_input_flags(I::BRKINT);
170        self.remove_input_flags(I::ICRNL);
171        self.remove_input_flags(I::INPCK);
172        self.remove_input_flags(I::ISTRIP);
173        self.set_software_flow(false);
174
175        self.remove_output_flags(O::OPOST);
176
177        self.remove_local_flags(L::ECHO);
178        self.remove_local_flags(L::ICANON);
179        self.remove_local_flags(L::IEXTEN);
180        self.remove_local_flags(L::ISIG);
181
182        self.set_char_size(LinuxTermiosCharSize::Bits8);
183        self.set_read_min_timeout(1, 0);
184    }
185    /// Makes this state conventionally line-buffered.
186    ///
187    /// This is a best-effort convenience reset, not an exact restoration of the
188    /// previous terminal state. For exact restoration, use a scoped terminal session.
189    pub const fn make_cooked_reset(&mut self) {
190        // input
191        self.set_software_flow(true);
192        self.set_input_cr_to_lf(true);
193        // local
194        self.set_canonical(true);
195        self.set_echo(true);
196        self.set_signals(true);
197        self.set_extensions(true);
198        self.set_output_processing(true);
199        self.set_read_min_timeout(1, 0);
200    }
201
202    /* local */
203    /// Enables or disables echoing typed input.
204    pub const fn set_echo(&mut self, enabled: bool) {
205        if enabled { self.insert_local_flags(L::ECHO); }
206        else { self.remove_local_flags(L::ECHO); }
207    }
208    /// Enables or disables canonical line buffering.
209    pub const fn set_canonical(&mut self, enabled: bool) {
210        if enabled { self.insert_local_flags(L::ICANON); }
211        else { self.remove_local_flags(L::ICANON); }
212    }
213    /// Enables or disables terminal-generated signal characters.
214    ///
215    /// When enabled, characters such as Ctrl-C and Ctrl-Z are handled by the
216    /// terminal line discipline instead of being delivered as ordinary input.
217    pub const fn set_signals(&mut self, enabled: bool) {
218        if enabled { self.insert_local_flags(L::ISIG); }
219        else { self.remove_local_flags(L::ISIG); }
220    }
221    /// Enables or disables implementation-defined line discipline extensions.
222    pub const fn set_extensions(&mut self, enabled: bool) {
223        if enabled { self.insert_local_flags(L::IEXTEN); }
224        else { self.remove_local_flags(L::IEXTEN); }
225    }
226
227    /* input */
228    /// Enables or disables interrupt generation on BREAK.
229    pub const fn set_break_interrupt(&mut self, enabled: bool) {
230        if enabled { self.insert_input_flags(I::BRKINT); }
231        else { self.remove_input_flags(I::BRKINT); }
232    }
233    /// Enables or disables translating carriage return to newline on input.
234    pub const fn set_input_cr_to_lf(&mut self, enabled: bool) {
235        if enabled { self.insert_input_flags(I::ICRNL); }
236        else { self.remove_input_flags(I::ICRNL); }
237    }
238    /// Enables or disables XON/XOFF software flow control.
239    pub const fn set_software_flow(&mut self, enabled: bool) {
240        if enabled {
241            // decides whether Ctrl-S/Ctrl-Q are intercepted for output flow control
242            self.insert_input_flags(I::IXON);
243            // sends start/stop control characters to the other side when input queues fill
244            self.insert_input_flags(I::IXOFF);
245        } else {
246            self.remove_input_flags(I::IXON);
247            self.remove_input_flags(I::IXOFF);
248        }
249    }
250
251    /* output */
252    /// Enables or disables terminal output post-processing.
253    pub const fn set_output_processing(&mut self, enabled: bool) {
254        if enabled { self.insert_output_flags(O::OPOST); }
255        else { self.remove_output_flags(O::OPOST); }
256    }
257
258    /* control characters */
259    /// Sets noncanonical read blocking behavior.
260    ///
261    /// - `min` is the minimum number of bytes required to satisfy a read.
262    /// - `timeout_deciseconds` is the inter-byte timeout in tenths of a second.
263    pub const fn set_read_min_timeout(&mut self, min: u8, timeout_deciseconds: u8) {
264        self.set_cc(Cc::VMIN, min);
265        self.set_cc(Cc::VTIME, timeout_deciseconds);
266    }
267}

337/// # Typed mode-field accessors
338impl LinuxTermios {
339    /// Returns the configured character size.
340    #[must_use]
341    pub const fn char_size(&self) -> LinuxTermiosCharSize {
342        let bits = self.c_cflag & C::CSIZE.as_c_uint();
343        if bits == C::CS6.as_c_uint() {
344            LinuxTermiosCharSize::Bits6
345        } else if bits == C::CS7.as_c_uint() {
346            LinuxTermiosCharSize::Bits7
347        } else if bits == C::CS8.as_c_uint() {
348            LinuxTermiosCharSize::Bits8
349        } else {
350            LinuxTermiosCharSize::Bits5
351        }
352    }
353    /// Sets the configured character size.
354    pub const fn set_char_size(&mut self, size: LinuxTermiosCharSize) {
355        let mask = C::CSIZE.as_c_uint();
356        self.c_cflag = (self.c_cflag & !mask) | size.to_control_flags().as_c_uint();
357    }
358}

377/// A Linux termios character size.
378#[doc = crate::_doc_meta!{
379    location("sys/os/linux/io/term", enum LinuxTermiosCharSize),
380    test_size_of(LinuxTermiosCharSize = 1|8; niche Option),

382#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
383#[repr(u8)]
384pub enum LinuxTermiosCharSize {
385    /// 5-bit characters.
386    Bits5 = 5,
387    /// 6-bit characters.
388    Bits6 = 6,
389    /// 7-bit characters.
390    Bits7 = 7,
391    #[doc = crate::_tags!(init)]
392    /// 8-bit characters.
393    #[default]
394    Bits8 = 8,
395}
396impl LinuxTermiosCharSize {
397    /// Returns the corresponding control-flag field value.
398    #[must_use]
399    pub const fn to_control_flags(self) -> C {
400        match self {
401            Self::Bits5 => C::CS5,
402            Self::Bits6 => C::CS6,
403            Self::Bits7 => C::CS7,
404            Self::Bits8 => C::CS8,
405        }
406    }
