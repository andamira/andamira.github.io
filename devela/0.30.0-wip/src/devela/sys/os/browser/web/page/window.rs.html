141#[doc = crate::_tags!(ui web)]
142/// Aggregates the live state of a [`WebWindow`], including geometry and screen context.
143#[doc = crate::_doc_meta!{
144    location("sys/os/browser/web", struct WebWindowState),
145    test_size_of(WebWindowState = 48|384; niche !Option),
146}]
147/// ### Performance
148/// All fields are fetched in a single JS→Rust call.
149#[repr(C)]
150#[derive(Clone, Copy, Default, PartialEq)] // manual: Debug
151pub struct WebWindowState {
152    /* window */
153    #[doc = _js_doc!("Window", "innerWidth")]
154    #[doc = _js_doc!("Window", "innerHeight")]
155    /// The extent in pixels of the content of the browser window including any rendered scrollbars.
156    pub inner_size: Extent<u32, 2>,
157
158    #[doc = _js_doc!("Window", "outerWidth")]
159    #[doc = _js_doc!("Window", "outerHeight")]
160    /// The extent in pixels of the outside of the browser window.
161    pub outer_size: Extent<u32, 2>,
162
163    /* screen */
164    #[doc = _js_doc!("Window", "screenLeft")]
165    #[doc = _js_doc!("Window", "screenTop")]
166    /// The window's offset in pixels from the screen's top-left origin.
167    pub screen_offset: Distance<i32, 2>,
168
169    #[doc = _js_doc!("Screen", "width")]
170    #[doc = _js_doc!("Screen", "height")]
171    /// The extent of the screen in pixels.
172    pub screen_size: Extent<u32, 2>,
173
174    #[doc = _js_doc!("Screen", "availWidth")]
175    #[doc = _js_doc!("Screen", "availHeight")]
176    /// The extent of the screen in pixels, minus user interface features displayed.
177    pub screen_usable_size: Extent<u32, 2>,
178
179    /* misc. */
180    #[doc = _js_doc!("Window", "devicePixelRatio")]
181    /// The device pixel ratio of the resolution in physical pixels to the resolution in CSS pixels.
182    ///
183    /// The value changes with the zoom on desktops yet remains static on mobile devices.
184    pub dpr: f32,
185
186    #[doc = _js_doc!("Screen", "colorDepth")]
187    /// The screen color depth, in bits per single pixel. It could be 8, 16, 24, 32 or 64.
188    pub bpp: u8,
189
190    // TODO: add bitpacked flags (is_popup, is_secure, etc.)
191    //
192    /// Explicit padding to align.
193    _pad: [u8; 3],
194}
195impl WebWindowState {
196    const __ASSERT_FIELD_OFFSETS: () = const {
197        assert!(offset_of!(Self, inner_size) == 0);
198        assert!(offset_of!(Self, outer_size) == 8);
199        assert!(offset_of!(Self, screen_offset) == 16);
200        assert!(offset_of!(Self, screen_size) == 24);
201        assert!(offset_of!(Self, screen_usable_size) == 32);
202        assert!(offset_of!(Self, dpr) == 40);
203        assert!(offset_of!(Self, bpp) == 44);
204    };
205
206    /// Returns a new up-to-date `WebWindowState`.
207    ///
208    /// # Safety
209    /// - JavaScript must write all non-padding fields at correct offsets.
210    pub fn new() -> WebWindowState {
211        let mut state = WebWindowState::default();
212        unsafe {
213            window_state(&mut state as *mut WebWindowState as *mut u8);
214        }
215        state
216    }
217
218    /// Overwrites this `WebWindowState` with the latest live metrics.
219    pub fn update(&mut self) {
220        unsafe { window_state(self as *mut Self as *mut u8) };
221    }
222
223    /// Validates the internal consistency of window metrics.
224    ///
225    /// Returns `true` if all these conditions hold:
226    /// - No dimensions are zero (invalid window state)
227    /// - Inner size ≤ outer size (logical constraint)
228    /// - Outer size ≤ screen size (unless multi-monitor)
229    /// - Device pixel ratio is sane (0.2 <= dpr <= 10.0)
230    /// - Screen color depth is plausible (8 <= depth <= 64)
231    // - Popup flags don't contradict window dimensions
232    pub const fn is_valid(&self) -> bool {
233        // 1. Non-zero dimensions
234        let non_zero = self.inner_size.x() > 0
235            && self.inner_size.y() > 0
236            && self.outer_size.x() > 0
237            && self.outer_size.y() > 0;
238
239        // 2. Inner <= Outer
240        let inner_le_outer = self.inner_size.dim[0] <= self.outer_size.dim[0]
241            && self.inner_size.dim[1] <= self.outer_size.dim[1];
242
243        // 3. Outer <= Screen (with 10px tolerance for window chrome)
244        let outer_le_screen = (self.outer_size.dim[0] <= self.screen_size.dim[0] + 10)
245            && (self.outer_size.dim[1] <= self.screen_size.dim[1] + 10);
246
247        // 4. Sane DPR range
248        let sane_dpr = self.dpr >= 0.2 && self.dpr <= 10.0;
249
250        // 5. Plausible color depth
251        let sane_bpp = self.bpp >= 8 && self.bpp <= 64;
252
253        // // 6. Popup consistency
254        // let valid_popup = !self.is_popup() || (
255        //     // Popups shouldn't fill the screen
256        //     self.outer_size.dim[0] < self.screen_usable_size.dim[0] - 10 &&
257        //     self.outer_size.dim[1] < self.screen_usable_size.dim[1] - 10
258        // );
259
260        non_zero && inner_le_outer && outer_le_screen && sane_dpr && sane_bpp // && valid_popup
261    }
262
263    /* derived metrics */
264
265    /// Returns the thickness of the window chrome (frame, scrollbars, etc.) in logical pixels.
266    ///
267    /// This is the difference between the outer and inner window sizes.
268    pub const fn chrome_size(&self) -> Extent<u32, 2> {
269        Extent::new([
270            self.outer_size.x() - self.inner_size.x(),
271            self.outer_size.y() - self.inner_size.y(),
272        ])
273    }
274    /// Checks if the window is approximately maximized (fills the available screen space).
275    ///
276    /// Tolerance: The window must be within 1 pixel of the screen's usable size.
277    pub const fn is_maximized(&self) -> bool {
278        self.outer_size.x() >= self.screen_usable_size.x()
279            && self.outer_size.y() >= self.screen_usable_size.y()
280    }
281    /// Checks if the window is in a portrait orientation (height > width).
282    pub const fn is_portrait(&self) -> bool {
283        self.inner_size.y() > self.inner_size.x()
284    }
285
286    /// Returns the physical size of the window in hardware pixels, truncating fractional values.
287    ///
288    /// Computed as `(inner_size * device_pixel_ratio)`.
289    ///
290    /// For rounded values, use [`physical_size_rounded()`][Self::physical_size_rounded].
291    pub const fn physical_size(&self) -> Extent<u32, 2> {
292        Extent::new([
293            (self.inner_size.x() as f32 * self.dpr) as u32,
294            (self.inner_size.y() as f32 * self.dpr) as u32,
295        ])
296    }
297    /// Returns the physical size of the window in hardware pixels, rounded to the nearest integer.
298    ///
299    /// Computed as `(inner_size * device_pixel_ratio).round()`.
300    ///
301    /// It's more accurate and expensive than [`physical_size()`][Self::physical_size].
302    pub const fn physical_size_rounded(&self) -> Extent<u32, 2> {
303        Extent::new([
304            Float(self.inner_size.x() as f32 * self.dpr).const_round().0 as u32,
305            Float(self.inner_size.y() as f32 * self.dpr).const_round().0 as u32,
306        ])
307    }
308
309    /// Returns the window's distance to each screen edge in logical pixels.
310    ///
311    /// Order: `[left, top, right, bottom]`. Negative values mean the window is outside the screen.
312    pub const fn screen_margins(&self) -> [i32; 4] {
313        [
314            self.screen_offset.dim[0],
315            self.screen_offset.dim[1],
316            (self.screen_size.x() as i32)
317                - (self.screen_offset.dim[0] + self.outer_size.x() as i32),
318            (self.screen_size.y() as i32)
319                - (self.screen_offset.dim[1] + self.outer_size.y() as i32),
320        ]
321    }
322}

324impl crate::Debug for WebWindowState {
325    fn fmt(&self, f: &mut crate::Formatter<'_>) -> crate::FmtResult<()> {
326        let mut state = f.debug_struct("WebWindowState");
327        /* fields */
328        state
329            .field("inner_size", &self.inner_size)
330            .field("outer_size", &self.outer_size)
331            .field("screen_offset", &self.screen_offset)
332            .field("screen_size", &self.screen_size)
333            .field("screen_usable_size", &self.screen_usable_size)
334            .field("dpr", &self.dpr)
335            .field("bpp", &self.bpp)
336            /* derived */
337            .field("chrome_size()", &self.chrome_size())
338            .field("is_maximized()", &self.is_maximized())
339            .field("is_portrait()", &self.is_portrait())
340            .field("is_valid()", &self.is_valid())
341            .field("physical_size()", &self.physical_size());
342        state.field("physical_size_rounded()", &self.physical_size_rounded());
343        state.field("screen_margins()", &self.screen_margins()).finish_non_exhaustive() // .finish()
344    }
