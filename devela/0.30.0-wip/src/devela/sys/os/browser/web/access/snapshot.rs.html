3use crate::{AsyncPoll, WebPermission, WebPermissionSet, is};
4use crate::{PermissionError, PermissionQuery, PermissionState};

6#[doc = crate::_tags!(web)]
7/// A point-in-time classification of cached browser permission queries.
8#[doc = crate::_doc_meta!{
9    location("sys/os/browser/web", struct WebPermissionSnapshot),
10    test_size_of(WebPermissionSnapshot = 24|192),
11}]
12/// The contained sets form a partial partition of the queried permission kinds:
13/// each represented permission belongs to exactly one query-result class, while
14/// permissions absent from every class are not represented by the snapshot.
15#[must_use]
16#[derive(Copy, Clone, Debug, Default, PartialEq, Eq, Hash)]
17pub struct WebPermissionSnapshot {
18    /// Permissions whose protected operations are authorized.
19    pub granted: WebPermissionSet,
20
21    /// Permissions whose authorization remains undecided or requestable.
22    pub prompt: WebPermissionSet,
23
24    /// Permissions whose protected operations are not authorized.
25    pub denied: WebPermissionSet,
26
27    /// Permissions whose queries have not resolved yet.
28    pub pending: WebPermissionSet,
29
30    /// Permissions that cannot be queried by the current browser.
31    pub unsupported: WebPermissionSet,
32
33    /// Permissions whose queries failed for another reason.
34    pub failed: WebPermissionSet,
35}
36impl WebPermissionSnapshot {
37    /// Returns an empty snapshot containing no cached queries.
38    pub const fn new() -> Self {
39        Self {
40            granted: WebPermissionSet::new(),
41            prompt: WebPermissionSet::new(),
42            denied: WebPermissionSet::new(),
43            pending: WebPermissionSet::new(),
44            unsupported: WebPermissionSet::new(),
45            failed: WebPermissionSet::new(),
46        }
47    }
48    /// Returns the query classification of `permission`, if present.
49    ///
50    /// Snapshots produced by devela are always valid. For a manually
51    /// constructed invalid snapshot, the first matching category is returned.
52    #[must_use]
53    pub const fn get(self, permission: WebPermission) -> Option<PermissionQuery> {
54        if permission.is_in(self.granted) {
55            Some(AsyncPoll::Ready(Ok(PermissionState::Granted)))
56        } else if permission.is_in(self.prompt) {
57            Some(AsyncPoll::Ready(Ok(PermissionState::Prompt)))
58        } else if permission.is_in(self.denied) {
59            Some(AsyncPoll::Ready(Ok(PermissionState::Denied)))
60        } else if permission.is_in(self.pending) {
61            Some(AsyncPoll::Pending)
62        } else if permission.is_in(self.unsupported) {
63            Some(AsyncPoll::Ready(Err(PermissionError::Unsupported)))
64        } else if permission.is_in(self.failed) {
65            Some(AsyncPoll::Ready(Err(PermissionError::Failed)))
66        } else {
67            None
68        }
69    }
70    /// Returns every permission represented by this snapshot.
71    pub const fn classified(self) -> WebPermissionSet {
72        self.granted
73            .union(self.prompt)
74            .union(self.denied)
75            .union(self.pending)
76            .union(self.unsupported)
77            .union(self.failed)
78    }
79    /// Returns permissions that resolved to an authorization state.
80    pub const fn resolved(self) -> WebPermissionSet {
81        self.granted.union(self.prompt).union(self.denied)
82    }
83    /// Returns permissions whose query completed, successfully or otherwise.
84    pub const fn settled(self) -> WebPermissionSet {
85        self.resolved().union(self.unsupported).union(self.failed)
86    }
87    /// Returns permissions from `scope` absent from this snapshot.
88    pub const fn unclassified_in(self, scope: WebPermissionSet) -> WebPermissionSet {
89        scope.difference(self.classified())
90    }
91    /// Returns permissions whose classifications differ from `previous`.
92    ///
93    /// The snapshots should normally have been produced from the same scope.
94    pub const fn changed_since(self, previous: Self) -> WebPermissionSet {
95        self.granted
96            .symmetric_difference(previous.granted)
97            .union(self.prompt.symmetric_difference(previous.prompt))
98            .union(self.denied.symmetric_difference(previous.denied))
99            .union(self.pending.symmetric_difference(previous.pending))
100            .union(self.unsupported.symmetric_difference(previous.unsupported))
101            .union(self.failed.symmetric_difference(previous.failed))
102    }
103    /// Returns whether the classification sets are mutually disjoint.
104    pub const fn is_valid(self) -> bool {
105        let mut classified = self.granted;
106        is! { classified.intersects(self.prompt), return false }
107        classified.insert(self.prompt);
108        is! { classified.intersects(self.denied), return false }
109        classified.insert(self.denied);
110        is! { classified.intersects(self.pending), return false }
111        classified.insert(self.pending);
112        is! { classified.intersects(self.unsupported), return false }
113        classified.insert(self.unsupported);
114        is! { classified.intersects(self.failed), return false }
115        true
116    }
117
118    /* private helpers */
119
120    pub(super) fn query(permissions: WebPermissionSet) -> Self {
121        let mut snapshot = Self::default();
122        permissions.for_each(|permission| {
123            snapshot.classify(permission, Some(permission.query()));
124        });
125        snapshot
126    }
127    pub(super) fn cached(permissions: WebPermissionSet) -> Self {
128        let mut snapshot = Self::default();
129        permissions.for_each(|permission| {
130            snapshot.classify(permission, permission.cached());
131        });
132        snapshot
133    }
134    /// Reclassifies `permission` according to `query`.
135    ///
136    /// Passing `None` removes the permission from the snapshot.
137    const fn classify(&mut self, permission: WebPermission, query: Option<PermissionQuery>) {
138        let permission = permission.to_set();
139        // Remove the preceding classification, if any.
140        self.granted.remove(permission);
141        self.prompt.remove(permission);
142        self.denied.remove(permission);
143        self.pending.remove(permission);
144        self.unsupported.remove(permission);
145        self.failed.remove(permission);
146        // Insert the new classification.
147        match query {
148            None => {}
149            Some(AsyncPoll::Pending) => {
150                self.pending.insert(permission);
151            }
152            Some(AsyncPoll::Ready(Ok(PermissionState::Granted))) => {
153                self.granted.insert(permission);
154            }
155            Some(AsyncPoll::Ready(Ok(PermissionState::Prompt))) => {
156                self.prompt.insert(permission);
157            }
158            Some(AsyncPoll::Ready(Ok(PermissionState::Denied))) => {
159                self.denied.insert(permission);
160            }
161            Some(AsyncPoll::Ready(Err(PermissionError::Unsupported))) => {
162                self.unsupported.insert(permission);
163            }
164            Some(AsyncPoll::Ready(Err(PermissionError::Failed))) => {
165                self.failed.insert(permission);
166            }
167        }
168    }
169}

171#[cfg(test)]
172mod tests {
173    use super::*;
174
175    #[test]
176    fn permission_snapshot_reclassifies() {
177        let permission = WebPermission::Camera;
178        let permission_set = permission.to_set();
179        let mut snapshot = WebPermissionSnapshot::new();
180        snapshot.classify(permission, Some(AsyncPoll::Pending));
181        assert_eq!(snapshot.pending, permission_set);
182        assert_eq!(snapshot.get(permission), Some(AsyncPoll::Pending));
183        assert!(snapshot.is_valid());
184        snapshot.classify(permission, Some(AsyncPoll::Ready(Ok(PermissionState::Granted))));
185        assert!(snapshot.pending.is_empty());
186        assert_eq!(snapshot.granted, permission_set);
187        assert_eq!(snapshot.get(permission), Some(AsyncPoll::Ready(Ok(PermissionState::Granted))));
188        assert!(snapshot.is_valid());
189        snapshot.classify(permission, None);
190        assert!(snapshot.classified().is_empty());
191        assert_eq!(snapshot.get(permission), None);
192    }
193    #[test]
194    fn permission_snapshot_detects_overlap() {
195        let camera = WebPermission::Camera.to_set();
196        let snapshot = WebPermissionSnapshot {
197            granted: camera,
198            denied: camera,
199            ..WebPermissionSnapshot::new()
200        };
201        assert!(!snapshot.is_valid());
202    }
203    #[test]
204    fn permission_snapshot_tracks_changes() {
205        let permission = WebPermission::Camera;
206        let camera = permission.to_set();
207        let mut previous = WebPermissionSnapshot::new();
208        previous.classify(permission, Some(AsyncPoll::Pending));
209        let mut current = WebPermissionSnapshot::new();
210        current.classify(permission, Some(AsyncPoll::Ready(Ok(PermissionState::Prompt))));
211        assert_eq!(current.changed_since(previous), camera);
212    }
