8/// Compact key code sent by JavaScript keyboard callbacks.
9pub type WebKeyCode = js_uint32;
10
11pub const WEB_KEY_UNKNOWN: u32 = 0x0000_0000;
12pub const WEB_KEY_CHAR: u32 = 0x0100_0000;
13pub const WEB_KEY_NAMED: u32 = 0x0200_0000;
14pub const WEB_KEY_FN: u32 = 0x0300_0000;
15pub const WEB_KEY_PAD: u32 = 0x0400_0000;
16pub const WEB_KEY_MEDIA: u32 = 0x0500_0000;
17pub const WEB_KEY_MOD: u32 = 0x0600_0000;
18
19const WEB_KEY_TAG: u32 = 0xFF00_0000;
20const WEB_KEY_VALUE: u32 = 0x00FF_FFFF;
21
22#[rustfmt::skip]
23impl Key {
24    /// Converts a compact JavaScript `KeyboardEvent.key` token into `Key`.
25    #[must_use]
26    pub const fn from_web_compact_key(code: WebKeyCode, location: WebKeyLocation) -> Self {
27        let (tag, value) = (code & WEB_KEY_TAG, code & WEB_KEY_VALUE);
28        match tag {
29            WEB_KEY_UNKNOWN => Key::Unknown,
30            WEB_KEY_CHAR => is![let Some(c) = char::from_u32(value), Key::Char(c), Key::Unknown],
31            WEB_KEY_NAMED => Self::from_web_compact_named(value),
32            WEB_KEY_FN => Self::from_web_compact_fn(value),
33            WEB_KEY_PAD => Self::from_web_compact_pad(value),
34            WEB_KEY_MEDIA => Self::from_web_compact_media(value),
35            WEB_KEY_MOD => Self::from_web_compact_mod(value, location),
36            _ => Key::Unknown,
37        }
38    }
39    /// Converts a compact JavaScript `KeyboardEvent.code` token into `Key`.
40    #[must_use]
41    pub const fn from_web_compact_code(code: WebKeyCode, location: WebKeyLocation) -> Self {
42        let (tag, value) = (code & WEB_KEY_TAG, code & WEB_KEY_VALUE);
43        match tag {
44            WEB_KEY_UNKNOWN => Key::Unknown,
45            WEB_KEY_CHAR => Key::Unknown, // physical `code` should not produce text chars
46            WEB_KEY_NAMED => Self::from_web_compact_named(value),
47            WEB_KEY_FN => Self::from_web_compact_fn(value),
48            WEB_KEY_PAD => Self::from_web_compact_pad(value),
49            WEB_KEY_MEDIA => Self::from_web_compact_media(value),
50            WEB_KEY_MOD => Self::from_web_compact_mod(value, location),
51            _ => Key::Unknown,
52        }
53    }
54    const fn from_web_compact_fn(value: u32) -> Self {
55        if value >= 1 && value <= 48 { Key::Fn(value as u8) } else { Key::Unknown }
56    }

142    const fn from_web_compact_mod(value: u32, location: WebKeyLocation) -> Self {
143        use WebKeyLocation as L;
144        let key = match (value, location) {
145            (0, L::Left) => KeyMod::LeftShift, //
146            (1, L::Left) => KeyMod::LeftControl,
147            (2, L::Left) => KeyMod::LeftAlt,
148            (3, L::Left) => KeyMod::LeftSuper,
149            (0, L::Right) => KeyMod::RightShift, //
150            (1, L::Right) => KeyMod::RightControl,
151            (2, L::Right) => KeyMod::RightAlt,
152            (3, L::Right) => KeyMod::RightSuper,
153            (4, L::Standard) => KeyMod::AltGr, //
154            (5, L::Standard) => KeyMod::IsoLevel5Shift,
155            _ => return Key::Unknown,
156        };
157        Key::Mod(key)
158    }
