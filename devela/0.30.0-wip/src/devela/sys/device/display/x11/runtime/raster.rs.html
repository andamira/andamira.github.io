5use crate::{RasterViewBytes, RunFrame, RunRender, XError, XPresent};

7#[doc = crate::_tags!(unix runtime)]
8/// Projects a byte-backed raster scene into an X11 presentation artifact.
9#[doc = crate::_doc_meta!{
10    location("sys/device/display/x11", struct XRasterRenderer),
11}]
12/// The source storage is borrowed and packaged as an [`XPresent`] for the
13/// X11 presenter. No color-model or channel-encoding conversion is performed;
14/// the source pixels must already be compatible with the active X11 visual.
15#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
16pub struct XRasterRenderer {
17    /// Whether to clear the window redraw flag after presentation.
18    pub clear_redraw: bool,
19}
20impl XRasterRenderer {
21    /// Creates a new X11 byte-image renderer.
22    pub const fn new(clear_redraw: bool) -> Self {
23        Self { clear_redraw }
24    }
25}
26impl Default for XRasterRenderer {
27    fn default() -> Self {
28        Self::new(true)
29    }
30}
31#[rustfmt::skip]
32impl<S: RasterViewBytes + ?Sized, E, C> RunRender<S, E, C> for XRasterRenderer {
33    type Error = XError;
34    type Output<'a> = XPresent<'a> where Self: 'a, S: 'a, E: 'a;
35
36    fn run_render<'a>(
37        &'a mut self,
38        _frame: &mut RunFrame<'a, E, C>,
39        scene: &'a S,
40    ) -> Result<Self::Output<'a>, Self::Error> {
41        let [width, height] = scene.raster_extent_bytes().dim;
42        let width = u16::try_from(width)
43            .map_err(|_| XError::Other("raster width exceeds the X11 u16 range"))?;
44        let height = u16::try_from(height)
45            .map_err(|_| XError::Other("raster height exceeds the X11 u16 range"))?;
46        Ok(XPresent::_new(
47            width,
48            height,
49            scene.raster_depth(),
50            scene.raster_bytes_per_pixel_bytes(),
51            scene.raster_bytes_per_line(),
52            scene.raster_row_start_bytes(),
53            scene.raster_bytes(),
54            self.clear_redraw,
55        ))
56    }
57}
