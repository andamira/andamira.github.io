538    pub root_y: i16,
539    /// If same_screen is true, this is the X coordinate relative to

546    pub state: u16,

760    /// - Never reads from the server socket.
761    /// - Non-blocking: returns NULL if the queue is empyt.
762    pub(crate) fn xcb_poll_for_queued_event(c: *mut xcb_connection_t) -> *mut xcb_generic_event_t;

764    /// Returns the next event or error from the server.
