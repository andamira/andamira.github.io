13/// Borrowed byte-backed presentation artifact for X11.
14#[doc = crate::_doc_meta!{
15    location("sys/device/display/x11", struct XPresent),
16    #[cfg(target_pointer_width = "32")]
17    test_size_of(XPresent<'_> = 24|192),
18    #[cfg(target_pointer_width = "64")]
19    test_size_of(XPresent<'_> = 40|320),
20}]
21/// This carries an already encoded pixel raster together with the storage
22/// metadata needed to adapt its scanlines to the active X11 image layout.
23///
24/// It does not perform color-model, channel-order, transfer, alpha,
25/// or endianness conversion. The source pixels must already use
26/// an encoding compatible with the active X11 visual.
27#[derive(Debug)]
28pub struct XPresent<'a> {
29    width: u16,
30    height: u16,
31
32    /// Logical pixel depth.
33    depth: u8,
34    /// Stored source bytes occupied by one pixel.
35    bytes_per_pixel: usize,
36    /// Stored source bytes between consecutive row starts.
37    bytes_per_line: usize,
38    /// Logical boundary represented by the first stored row.
39    row_start: Boundary1d,
40
41    bytes: &'a [u8],
42    clear_redraw: bool,
43}
44#[rustfmt::skip]
45impl<'a> XPresent<'a> {
46    #[allow(clippy::too_many_arguments)]
47    const fn _new(width: u16, height: u16, depth: u8, bytes_per_pixel: usize, bytes_per_line: usize,
48        row_start: Boundary1d, bytes: &'a [u8], clear_redraw: bool) -> Self {
49        Self {
50            width, height, depth, bytes_per_pixel, bytes_per_line, row_start, bytes, clear_redraw
51        }
52    }
53    /// Returns whether each source row contains no trailing padding.
54    pub const fn is_tight_rows(&self) -> bool {
55        matches!(
56            (self.width as usize).checked_mul(self.bytes_per_pixel),
57            Some(row_bytes) if row_bytes == self.bytes_per_line
58        )
59    }
60}

63/// Validated plan for copying raster rows into an X11 surface.
64#[doc = crate::_doc_meta!{
65    location("sys/device/display/x11", struct XCopyLayout),
66}]
67/// This separates meaningful pixel bytes from source and destination row
68/// padding. Construction verifies compatible depth and stored pixel width,
69/// valid strides, representable lengths, and sufficient source storage.
70///
71/// Row orientation is kept by [`XPresent`] and applied while copying.
72#[derive(Clone, Copy, Debug, PartialEq, Eq)]
73struct XCopyLayout {
74    row_bytes: usize,
75    src_bytes_per_line: usize,
76    dst_bytes_per_line: usize,
77    height: usize,
78    dst_required: usize,
79}
80impl XCopyLayout {
81    fn new(
82        input: &XPresent<'_>,
83        dst_depth: u8,
84        dst_bits_per_pixel: u8,
85        dst_bytes_per_line: usize,
86    ) -> Result<Self, XError> {
87        if input.depth != dst_depth {
88            return Err(XError::Other("raster depth does not match the X11 display depth"));
89        }
90        if dst_bits_per_pixel == 0 || !dst_bits_per_pixel.is_multiple_of(8) {
91            return Err(XError::Other("X11 display pixel storage is not byte-aligned"));
92        }
93        let dst_bytes_per_pixel = usize::from(dst_bits_per_pixel / 8);
94        if input.bytes_per_pixel != dst_bytes_per_pixel {
95            return Err(XError::Other(
96                "raster stored pixel width does not match the X11 display format",
97            ));
98        }
99        let width = usize::from(input.width);
100        let height = usize::from(input.height);
101        let row_bytes = width
102            .checked_mul(dst_bytes_per_pixel)
103            .ok_or(XError::Other("X11 raster row byte length overflow"))?;
104        if input.bytes_per_line < row_bytes {
105            return Err(XError::Other("raster row stride is shorter than its stored pixel row"));
106        }
107        if dst_bytes_per_line < row_bytes {
108            return Err(XError::Other("X11 row stride is shorter than its stored pixel row"));
109        }
110        // Like RasterLayout::min_len_bytes(): final-row padding is optional.
111        let src_required = if width == 0 || height == 0 {
112            0
113        } else {
114            (height - 1)
115                .checked_mul(input.bytes_per_line)
116                .and_then(|n| n.checked_add(row_bytes))
117                .ok_or(XError::Other("raster byte length overflow"))?
118        };
119        if input.bytes.len() < src_required {
120            return Err(XError::Other("raster byte storage is too short"));
121        }
122        let dst_required = dst_bytes_per_line
123            .checked_mul(height)
124            .ok_or(XError::Other("X11 surface byte length overflow"))?;
125        Ok(Self {
126            row_bytes,
127            src_bytes_per_line: input.bytes_per_line,
128            dst_bytes_per_line,
129            height,
130            dst_required,
131        })
132    }
133}

136/// A minimal X11 presenter for byte-backed image artifacts.
137#[doc = crate::_doc_meta!{
138    location("sys/device/display/x11", struct XPresenter),
139}]
140/// `XPresenter` consumes [`XPresent`] artifacts
141/// and applies them to the current X11 frame context.
142#[derive(Debug)]
143pub(crate) struct XPresenter {
144    mode: XImageMode,
145    surface: Option<XSurface>,
146}
147impl XPresenter {
148    pub const fn new(mode: XImageMode) -> Self {
149        Self { mode, surface: None }
150    }
151    pub const fn mode(&self) -> XImageMode {
152        self.mode
153    }
154    pub const fn active_mode(&self) -> Option<XImageMode> {
155        if let Some(ref s) = self.surface { Some(s.mode()) } else { None }
156    }
157    fn ensure_surface(
158        &mut self,
159        display: &XDisplay,
160        width: u16,
161        height: u16,
162        depth: u8,
163    ) -> Result<&mut XSurface, XError> {
164        let needs_new = match &self.surface {
165            None => true,
166            Some(s) => s.size() != (width, height) || s.depth() != depth,
167        };
168        if needs_new {
169            let surface = match self.mode {
170                XImageMode::Auto => XSurface::new(display, width, height, depth)?,
171                XImageMode::Cpu => XSurface::new_cpu(display, width, height, depth)?,
172                #[cfg(ffi_xcb_shm··)]
173                XImageMode::Shm => XSurface::new_shm(display, width, height, depth)?,
174            };
175            self.surface = Some(surface);
176        }
177        Ok(self.surface.as_mut().unwrap())
178    }
179    fn copy_rows(
180        dst: &mut [u8],
181        dst_bytes_per_line: usize,
182        src: &[u8],
183        src_bytes_per_line: usize,
184        row_bytes: usize,
185        height: usize,
186        row_start: Boundary1d,
187    ) {
188        for dst_y in 0..height {
189            let src_y = match row_start {
190                Boundary1d::Upper => dst_y,
191                Boundary1d::Lower => height - 1 - dst_y,
192            };
193            let dst_start = dst_y * dst_bytes_per_line;
194            let src_start = src_y * src_bytes_per_line;
195            let dst_row = &mut dst[dst_start..dst_start + row_bytes];
196            let src_row = &src[src_start..src_start + row_bytes];
197            dst_row.copy_from_slice(src_row);
198        }
199    }
200    pub(crate) fn surface_frame<'a>(
201        &'a mut self,
202        display: &XDisplay,
203        width: u16,
204        height: u16,
205        depth: u8,
206    ) -> Result<XSurfaceFrame<'a>, XError> {
207        let image_format = display.image_format;
208        let visual_format = display.visual_format;
209        let surface = self.ensure_surface(display, width, height, depth)?;
210        Ok(XSurfaceFrame::_new(surface, image_format, visual_format))
211    }
212    pub(crate) fn present_surface(
213        &mut self,
214        display: &mut XDisplay,
215        window: &mut XWindow,
216        clear_redraw: bool,
217    ) -> Result<(), XError> {
218        let surface = self.surface.as_mut().ok_or(XError::Other("no surface in XPresenter"))?;
219        surface.present(display, window)?;
220        is! { clear_redraw, window.clear_redraw(display) }
221        display.flush();
222        Ok(())
223    }
224}
225impl<'ctx> RunPresent<Event, XFrameCtx<'ctx>> for XPresenter {
226    type Input<'a>
227        = XPresent<'a>
228    where
229        Self: 'a;
230    type Output = ();
231    type Error = XError;
232
233    fn run_present<'a>(
234        &'a mut self,
235        frame: &mut RunFrame<'a, Event, XFrameCtx<'ctx>>,
236        input: Self::Input<'a>,
237    ) -> Result<Self::Output, Self::Error> {
238        let ctx = frame.context_mut();
239        let dst_bytes_per_line = usize::try_from(ctx.display.bytes_per_line(input.width))
240            .map_err(|_| XError::Other("X11 row stride exceeds usize"))?;
241        let layout = XCopyLayout::new(
242            &input,
243            ctx.display.depth(),
244            ctx.display.bits_per_pixel(),
245            dst_bytes_per_line,
246        )?;
247        let surface = self.ensure_surface(ctx.display, input.width, input.height, input.depth)?;
248        {
249            let dst = surface.bytes_mut();
250            if dst.len() < layout.dst_required {
251                return Err(XError::Other("X11 surface storage is too short"));
252            }
253            let dst = &mut dst[..layout.dst_required];
254            // Whole-buffer fast path. It is valid only when source rows are
255            // upper-first and both layouts have identical strides.
256            if input.row_start == Boundary1d::Upper
257                && layout.src_bytes_per_line == layout.dst_bytes_per_line
258                && input.bytes.len() >= layout.dst_required
259            {
260                dst.copy_from_slice(&input.bytes[..layout.dst_required]);
261            } else {
262                Self::copy_rows(
263                    dst,
264                    layout.dst_bytes_per_line,
265                    input.bytes,
266                    layout.src_bytes_per_line,
267                    layout.row_bytes,
268                    layout.height,
269                    input.row_start,
270                );
271            }
272        }
273        surface.present(ctx.display, ctx.window)?;
274        is! { input.clear_redraw, ctx.window.clear_redraw(ctx.display) }
275        ctx.display.flush();
276        Ok(())
277    }
278}

281/// Projects a byte-backed X11 image scene into a borrowed presentation artifact.
282#[doc = crate::_doc_meta!{
283    location("sys/device/display/x11", struct XRasterRenderer),
284}]
285/// It borrows image bytes from the scene and packages them
286/// as an [`XPresent`] for an `XPresenter` to upload.
287#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
288pub struct XRasterRenderer {
289    /// Whether to clear the window redraw flag after presentation.
290    pub clear_redraw: bool,
291}
292impl XRasterRenderer {
293    /// Creates a new X11 byte-image renderer.
294    pub const fn new(clear_redraw: bool) -> Self {
295        Self { clear_redraw }
296    }
297}
298impl Default for XRasterRenderer {
299    fn default() -> Self {
300        Self::new(true)
301    }
302}
303#[rustfmt::skip]
304impl<S: RasterViewBytes + ?Sized, E, C> RunRender<S, E, C> for XRasterRenderer {
305    type Error = XError;
306    type Output<'a> = XPresent<'a> where Self: 'a, S: 'a, E: 'a;
307
308    fn run_render<'a>(
309        &'a mut self,
310        _frame: &mut RunFrame<'a, E, C>,
311        scene: &'a S,
312    ) -> Result<Self::Output<'a>, Self::Error> {
313        let [width, height] = scene.raster_extent_bytes().dim;
314        let width = u16::try_from(width)
315            .map_err(|_| XError::Other("raster width exceeds the X11 u16 range"))?;
316        let height = u16::try_from(height)
317            .map_err(|_| XError::Other("raster height exceeds the X11 u16 range"))?;
318        Ok(XPresent::_new(
319            width,
320            height,
321            scene.raster_depth(),
322            scene.raster_bytes_per_pixel_bytes(),
323            scene.raster_bytes_per_line(),
324            scene.raster_row_start_bytes(),
325            scene.raster_bytes(),
326            self.clear_redraw,
327        ))
328    }
329}

331#[cfg(test)]
332mod _test {
333    use super::*;
334
335    #[test]
336    fn xpresent_tightness_uses_stored_width() {
337        let bytes = [0_u8; 24];
338
339        let present = XPresent::_new(3, 2, 24, 4, 12, Boundary1d::Upper, &bytes, true);
340
341        assert!(present.is_tight_rows());
342    }
343
344    #[test]
345    fn copy_rows_reverses_lower_first_storage() {
346        // Two 2-byte rows with two bytes of source padding per row.
347        //
348        // Stored first: lower logical row.
349        // Stored next:  upper logical row.
350        let src = [1, 2, 0, 0, 3, 4, 0, 0];
351        let mut dst = [0_u8; 4];
352
353        XPresenter::copy_rows(&mut dst, 2, &src, 4, 2, 2, Boundary1d::Lower);
354
355        assert_eq!(dst, [3, 4, 1, 2,]);
356    }
357
358    #[test]
359    fn copy_layout_rejects_stored_width_mismatch() {
360        let bytes = [0_u8; 12];
361
362        let input = XPresent::_new(
363            2,
364            2,
365            24,
366            3, // RGB8 storage
367            6,
368            Boundary1d::Upper,
369            &bytes,
370            true,
371        );
372
373        // Typical X11 root format: depth 24, stored bpp 32.
374        assert!(XCopyLayout::new(&input, 24, 32, 8).is_err());
375    }
