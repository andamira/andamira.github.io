23#[doc = crate::_tags!(unix runtime)]
24/// A single-window X11 runtime backend.
25#[doc = crate::_doc_meta!{
26    location("sys/device/display/x11", struct XBackend),
27}]
28/// `XBackend` owns the X11 display connection and window used for event
29/// collection and per-frame backend access.
30#[derive(Debug)]
31pub(crate) struct XBackend {
32    display: XDisplay,
33    window: XWindow,
34}
35impl XBackend {
36    /// Opens a single-window X11 backend.
37    pub fn open(x: i16, y: i16, width: u16, height: u16) -> Result<Self, XError> {
38        let mut display = XDisplay::open()?;
39        let window = XWindow::new(&mut display, x, y, width, height, 1)?;
40        Ok(Self { display, window })
41    }
42}
43impl RunBackend for XBackend {
44    type Event = Event;
45    type Error = XError;
46    type Context<'a>
47        = XFrameCtx<'a>
48    where
49        Self: 'a;
50    fn collect_events(&mut self, out: &mut [Self::Event]) -> Result<usize, Self::Error> {
51        whilst! { written in 0..out.len(); {
52            let ev = self.display.poll_event();
53            is! { ev.is_none(), break }
54            out[written] = ev;
55        }}
56        Ok(written)
57    }
58    fn context(&mut self) -> Self::Context<'_> {
59        XFrameCtx {
60            display: &mut self.display,
61            window: &mut self.window,
62        }
63    }
64}

66#[doc = crate::_tags!(unix runtime)]
67/// An X11 frontend for frame-driven runtimes.
68#[doc = crate::_doc_meta!{
69    location("sys/device/display/x11", struct XFrontend),
70}]
71/// Owns the current X11 runtime backend and its presentation finalizer,
72/// while leaving runtime control, app logic, and rendering policy external.
73///
74/// The current implementation manages one window.
75/// That is an implementation detail, not the intended long-term boundary.
76#[derive(Debug)]
77pub struct XFrontend {
78    backend: XBackend,
79    presenter: XPresenter,
80}
81#[rustfmt::skip]
82impl XFrontend {
83    const fn _new(backend: XBackend, presenter: XPresenter) -> Self { Self { backend, presenter } }
84
85    /// Opens an X11 frontend.
86    pub fn open(x: i16, y: i16, width: u16, height: u16) -> Result<Self, XError> {
87        Self::open_with(XImageMode::Auto, x, y, width, height)
88    }
89    /// Opens an X11 frontend with the given image `mode`.
90    pub fn open_with(mode: XImageMode, x: i16, y: i16, width: u16, height: u16)
91        -> Result<Self, XError> {
92        Ok(Self::_new(XBackend::open(x, y, width, height)?, XPresenter::new(mode)))
93    }
94
95    /// Returns a shared reference to the X11 Display server.
96    pub const fn display(&self) -> &XDisplay { &self.backend.display }
97    /// Returns an exclusive reference to the X11 Display server.
98    pub const fn display_mut(&mut self) -> &mut XDisplay { &mut self.backend.display }
99
100    /// Returns whether MIT-SHM is available on this display.
101    #[cfg(ffi_xcb_shm··)]
102    pub const fn has_shm(&self) -> bool { self.backend.display.has_shm() }
103
104    /// Returns the configured X11 image presentation mode.
105    ///
106    /// This is the mode requested when opening the frontend.
107    /// It may still be [`XImageMode::Auto`] before any presentation occurs.
108    pub const fn mode(&self) -> XImageMode { self.presenter.mode() }
109
110    /// Returns the currently resolved X11 image presentation mode, if any.
111    ///
112    /// This resolves the active backing chosen after presentation begins.
113    ///
114    /// It returns `None` while no presentation surface has been created yet.
115    pub fn active_mode(&self) -> Option<XImageMode> { self.presenter.active_mode() }
116
117    /// Polls the next event without blocking.
118    ///
119    /// See [`XDisplay::poll_event`] for more information.
120    pub fn poll_event(&mut self) -> Event { self.backend.display.poll_event() }
121
122    /// Waits for the next event, blocking until one is available.
123    ///
124    /// See [`XDisplay::wait_event`] for more information.
125    pub fn wait_event(&mut self) -> Event { self.backend.display.wait_event() }
126}
127impl XFrontend {
128    /// Drives one X11 runtime iteration including rendering and presentation.
129    #[allow(private_bounds, reason = "private XFrameCtx")]
130    pub fn step_frame<T, A, R, S, RE>(
131        &mut self,
132        driver: &mut RunDriver<T>,
133        app: &mut A,
134        renderer: &mut R,
135        scene: &S,
136        events: &mut [A::Event],
137    ) -> Result<RunControl, RunDriverFrameError<XError, A::Error, RE, XError>>
138    where
139        A: RunApp<Event = Event>,
140        for<'a> R: RunRender<S, Event, XFrameCtx<'a>, Error = RE, Output<'a> = XPresent<'a>>,
141    {
142        // NOTE: because of compiler limitations we can't do just this:
143        // driver.step_frame(&mut self.backend, app, renderer, &mut self.presenter, scene, events)
144        _run_driver_step_run_frame_body!(@check
145            driver, self.backend, app, renderer, self.presenter, scene(scene), events, control,
146            driver.runtime_mut().tick_once(), driver.runtime_mut().transition(RunPhase::Stopped));
147        Ok(control)
148    }
149
150    /// Runs until the app stops or an error occurs,
151    /// including rendering and presentation.
152    #[allow(private_bounds, reason = "private XFrameCtx")]
153    pub fn run_frame<T, A, R, S, RE>(
154        &mut self,
155        driver: &mut RunDriver<T>,
156        app: &mut A,
157        renderer: &mut R,
158        scene: &S,
159        events: &mut [A::Event],
160    ) -> Result<(), RunDriverFrameError<XError, A::Error, RE, XError>>
161    where
162        A: RunApp<Event = Event>,
163        for<'a> R: RunRender<S, Event, XFrameCtx<'a>, Error = RE, Output<'a> = XPresent<'a>>,
164    {
165        // NOTE: because of compiler limitations we can't do just this:
166        // driver.run_frame(&mut self.backend, app, renderer, &mut self.presenter, scene, events)
167        driver.start();
168        while driver.can_advance() {
169            _run_driver_step_run_frame_body!(
170                driver,
171                self.backend,
172                app,
173                renderer,
174                self.presenter,
175                scene(scene),
176                events,
177                control,
178                driver.runtime_mut().tick_once(),
179                driver.runtime_mut().transition(RunPhase::Stopped),
180                break
181            );
182        }
183        Ok(())
184    }
185}
186impl XFrontend {
187    /// Renders directly into the retained X11 surface and presents it.
188    ///
189    /// This bypasses [`XRasterRenderer`][crate::XRasterRenderer]
190    /// and avoids the scene-to-surface copy.
191    pub fn with_surface_frame<F, T>(
192        &mut self,
193        width: u16,
194        height: u16,
195        depth: u8,
196        clear_redraw: bool,
197        render: F,
198    ) -> Result<T, XError>
199    where
200        F: FnOnce(&mut XSurfaceFrame<'_>) -> Result<T, XError>,
201    {
202        let result = {
203            let mut frame =
204                self.presenter.surface_frame(&self.backend.display, width, height, depth)?;
205            render(&mut frame)?
206        };
207        self.presenter.present_surface(
208            &mut self.backend.display,
209            &mut self.backend.window,
210            clear_redraw,
211        )?;
212        Ok(result)
213    }
214
215    /// Advances one runtime frame and renders directly into the retained X11 surface.
216    ///
217    /// This is the X11-specific zero-copy rendering path. It bypasses
218    /// [`XRasterRenderer`][crate::XRasterRenderer] and exposes the
219    /// retained surface bytes for the duration of one frame.
220    ///
221    /// This avoids the intermediate artifact-to-surface copy,
222    /// but it also gives up backend independence.
223    //
224    // BENCH NOTE: On local X11 SHM tests at 2560x1440-like scale, direct surface
225    // rendering was ~5-20% faster than the artifact path depending on workload.
226    // Presentation remained the dominant cost. Keeping both paths.
227    #[allow(clippy::too_many_arguments)]
228    #[allow(private_bounds, reason = "private XFrameCtx")]
229    pub fn step_frame_surface<R, A, F, RE>(
230        &mut self,
231        driver: &mut RunDriver<R>,
232        app: &mut A,
233        width: u16,
234        height: u16,
235        depth: u8,
236        clear_redraw: bool,
237        events: &mut [Event],
238        render: F,
239    ) -> Result<RunControl, RunDriverFrameError<XError, A::Error, RE, XError>>
240    where
241        A: RunApp<Event = Event>,
242        F: FnOnce(&mut XSurfaceFrame<'_>) -> Result<(), RE>,
243    {
244        // 1. Collect backend events.
245        let written = self.backend.collect_events(events).map_err(RunDriverFrameError::Backend)?;
246        let events = &events[..written];
247        // 2. Build the runtime step.
248        let step = RunStep::new(driver.runtime_mut().tick(), driver.phase(), events);
249        // 3. Advance app logic.
250        let control = app.run_step(step).map_err(RunDriverFrameError::App)?;
251        // 4B. Stop before rendering/presenting.
252        if matches!(control, RunControl::Stop) {
253            driver.runtime_mut().transition(RunPhase::Stopped);
254            return Ok(control);
255        }
256        // 5B. Ensure the retained surface and render directly into it.
257        let mut surface = self
258            .presenter
259            .surface_frame(&self.backend.display, width, height, depth)
260            .map_err(RunDriverFrameError::Present)?;
261        render(&mut surface).map_err(RunDriverFrameError::Render)?;
262        // 6B. Present after the surface borrow ends.
263        self.presenter
264            .present_surface(&mut self.backend.display, &mut self.backend.window, clear_redraw)
265            .map_err(RunDriverFrameError::Present)?;
266        // 7B. Advance runtime.
267        driver.runtime_mut().tick_once();
268        Ok(control)
269    }
