3//! Defines [`XSurfaceFrame`].

5
6use crate::{Boundary1d, Position2, RasterLayout, ext, is, unwrap};
7use crate::{XImageFormat, XImageMode, XImageStore, XSurface, XVisualFormat};
8
9#[doc = crate::_tags!(unix runtime)]
10/// Borrowed mutable X11 surface for direct frame rendering.
11#[doc = crate::_doc_meta!{
12    location("sys/device/display/x11", struct XSurfaceFrame),
13    #[cfg(target_pointer_width = "64")]
14    test_size_of(XSurfaceFrame<'_> = 32|256; niche Option),
15}]
16/// This exposes the retained X11 presentation surface for one frame.
17///
18/// Drawing into this surface avoids the intermediate scene-to-surface copy
19/// used by [`XRasterRenderer`][crate::XRasterRenderer].
20///
21/// `XSurfaceFrame` is the X11 direct-surface path.
22///
23/// It exposes the retained X11 surface for one frame, allowing callers to render
24/// directly into the CPU/SHM presentation buffer.
25///
26/// Use it when X11-specific performance or surface control matters.
27pub struct XSurfaceFrame<'a> {
28    surface: &'a mut XSurface,
29    image_format: XImageFormat,
30    visual_format: XVisualFormat,
31}
32#[rustfmt::skip]
33impl<'a> XSurfaceFrame<'a> {
34    pub(crate) const fn _new(
35        surface: &'a mut XSurface,
36        image_format: XImageFormat,
37        visual_format: XVisualFormat,
38    ) -> Self {
39        Self { surface, image_format, visual_format }
40    }
41
42    /* geometry / storage */
43
44    /// Returns the surface width in pixels.
45    #[must_use]
46    pub const fn width(&self) -> u16 { self.surface.width }
47
48    /// Returns the surface height in pixels.
49    #[must_use]
50    pub const fn height(&self) -> u16 { self.surface.height }
51
52    /// Returns the surface pixel depth in bits.
53    #[must_use]
54    pub const fn depth(&self) -> u8 { self.surface.depth }
55
56    /// Returns the number of stored bits per pixel.
57    #[must_use]
58    pub const fn bits_per_pixel(&self) -> u8 { self.image_format.bits_per_pixel }
59
60    /// Returns the number of stored bytes per pixel, when byte-aligned.
61    #[must_use]
62    pub const fn bytes_per_pixel(&self) -> Option<u8> {
63        let bpp = self.bits_per_pixel();
64        is! { bpp.is_multiple_of(8), Some(bpp / 8), None }
65    }
66
67    /// Returns the byte stride between consecutive rows.
68    #[must_use]
69    pub const fn bytes_per_line(&self) -> u32 { self.image_format.bytes_per_line(self.width()) }
70
71    /// Returns the active surface backing mode.
72    pub const fn mode(&self) -> XImageMode { self.surface.mode() }
73
74    #[must_use]
75    /// Returns whether each row has no backend padding.
76    pub const fn is_tight_rows(&self) -> bool {
77        let bits = self.width() as u32 * self.bits_per_pixel() as u32;
78        bits.div_ceil(8) == self.bytes_per_line()
79    }
80
81    /// Returns this surface's byte-addressable raster layout.
82    ///
83    ///
84    /// The layout describes its pixel extent, stored bytes per pixel,
85    /// row stride, and upper-first row orientation.
86    /// It does not describe the pixel color or channel encoding.
87    ///
88    /// Returns `None` when the stored pixel width is not byte-aligned.
89    #[must_use]
90    pub const fn raster_layout(&self) -> Option<RasterLayout> {
91        let bytes_per_pixel = unwrap![some? self.bytes_per_pixel()];
92        Some(RasterLayout::interleaved(
93            ext![self.width() as u32, self.height() as u32],
94            bytes_per_pixel,
95            self.bytes_per_line(),
96            Boundary1d::Upper,
97        ))
98    }
99
100    /* pixels */
101
102    /// Returns whether RGB8 colors can be encoded and written directly to this surface.
103    pub const fn supports_rgb8(&self) -> bool {
104        self.supports_native_pixel() && self.visual_format.supports_rgb8()
105    }
106    /// Encodes an RGB8 color as this X11 surface's native pixel value.
107    ///
108    /// Returns `None` when this surface does not support direct RGB8 encoding.
109    pub const fn encode_rgb8(&self, rgb: [u8; 3]) -> Option<u32> {
110        is! { self.supports_rgb8(), Some(self.visual_format.encode_rgb8(rgb)), None }
111    }
112    /// Encodes and writes an RGB8 color at `coord`.
113    ///
114    /// Returns whether the pixel was written.
115    pub fn write_rgb8(&mut self, coord: Position2<u32>, rgb: [u8; 3]) -> bool {
116        let Some(pixel) = self.encode_rgb8(rgb) else { return false };
117        self.write_native_pixel(coord, pixel)
118    }
119
120    /// Returns whether native pixel values can be written directly to this surface.
121    pub const fn supports_native_pixel(&self) -> bool {
122        self.image_format.supports_native_pixel()
123    }
124    /// Writes an already encoded native X11 pixel at `coord`.
125    ///
126    /// Returns whether the pixel was written.
127    pub fn write_native_pixel(&mut self, coord: Position2<u32>, pixel: u32) -> bool {
128        let image_format = self.image_format;
129        let Some(layout) = self.raster_layout() else { return false };
130        let Some(offset) = layout.pixel_offset_bytes(coord) else { return false };
131        let Some(dst) = self.bytes_mut().get_mut(offset..) else { return false };
132        image_format.write_native_pixel(dst, pixel)
133    }
134
135    /// Returns the mutable surface bytes for direct rendering.
136    pub fn bytes_mut(&mut self) -> &mut [u8] { self.surface.bytes_mut() }
137}
