19/// # Availability
20/// Methods that call ALSA are available
21/// when the native `asound` library is discoverable at build time.
22#[derive(Debug)]
23pub struct Alsa;
24
25impl Alsa {
26    /// Returns whether the native ALSA library was found at build time.
27    pub const fn is_available() -> bool {
28        cfg!(ffi_alsa··)
29    }
30    /// Returns `Ok` if native ALSA support was compiled in.
31    pub const fn require_available() -> Result<(), AlsaError> {
32        if Self::is_available() { Ok(()) } else { Err(AlsaError::Unavailable) }
33    }
34}

36#[cfg(all(ffi_alsa··, not(miri)))]
37impl Alsa {
38    /// Opens an ALSA PCM device.
39    pub fn open_pcm(id: &CStr, dir: AudioStreamDir) -> Result<AlsaPcmHandle, AlsaError> {
40        unsafe {
41            let mut handle = Ptr::null_mut();
42            AlsaError::result(_raw::snd_pcm_open(&mut handle, id.as_ptr(), dir.to_alsa() as _, 0))?;
43            Ok(AlsaPcmHandle::from_raw(handle))
44        }
45    }
46    /// Opens the default playback device.
47    pub fn open_default_playback() -> Result<AlsaPcmHandle, AlsaError> {
48        Self::open_pcm(c"default", AudioStreamDir::Playback)
49    }
50    /// Opens the default capture device.
51    pub fn open_default_capture() -> Result<AlsaPcmHandle, AlsaError> {
52        Self::open_pcm(c"default", AudioStreamDir::Capture)
53    }
54
55    /// Returns the ALSA PCM device hints as owned device descriptions.
56    ///
57    /// This is the allocating counterpart to [`for_each_pcm_device`](Self::for_each_pcm_device).
58    /// It copies ALSA's temporary hint strings into Rust-owned storage before
59    /// returning.
60    ///
61    /// The returned device `id` can be passed back to [`open_pcm`](Self::open_pcm).
62    ///
63    /// # Notes
64    /// ALSA usually does not expose a separate human-readable display name through
65    /// this hint API, so `name` is commonly `None`. Use [`AudioDevice::label`] or
66    /// the device description when presenting devices to users.
67    ///
68    /// # Errors
69    /// Returns [`AlsaError`] if ALSA fails to enumerate device hints.
70    #[cfg(feature = "alloc")]
71    pub fn pcm_devices() -> Result<Vec<AudioDeviceCow<'static>>, AlsaError> {
72        let mut out = Vec::new();
73        Self::for_each_pcm_device(|dev| {
74            out.push(AudioDeviceCow {
75                id: Cow::Owned(dev.id.to_string()),
76                name: dev.name.map(|s| Cow::Owned(s.to_string())),
77                desc: dev.desc.map(|s| Cow::Owned(s.to_string())),
78                dir: dev.dir,
79            });
80            Ok(())
81        })?;
82        Ok(out)
83    }
84
85    /// Calls `f` for each ALSA PCM device hint.
86    ///
87    /// The callback receives a backend-neutral [`AudioDevice`] whose `id`
88    /// can be passed back to [`open_pcm`](Self::open_pcm).
89    ///
90    /// This method does not allocate Rust-owned storage.
91    /// ALSA still allocates temporary hint strings internally,
92    /// which are released before the method returns.
93    ///
94    /// # Notes
95    /// [`AudioDevice::name`] is commonly `None` for ALSA devices.
96    ///
97    /// # Errors
98    /// Returns [`AlsaError`] if ALSA fails to enumerate device hints
99    /// or if the callback returns an error.
100    pub fn for_each_pcm_device(
101        mut f: impl FnMut(AudioDevice<'_>) -> Result<(), AlsaError>,
102    ) -> Result<(), AlsaError> {
103        unsafe {
104            let mut hints = Ptr::null_mut();
105            AlsaError::result(_raw::snd_device_name_hint(-1, c"pcm".as_ptr(), &mut hints))?;
106            let hints = AlsaHintList { raw: hints };
107            let mut p = hints.raw;
108            while !p.is_null() && !(*p).is_null() {
109                let hint = *p;
110                let name = AlsaHintValue::get(hint, c"NAME");
111                let desc = AlsaHintValue::get(hint, c"DESC");
112                let ioid = AlsaHintValue::get(hint, c"IOID");
113                if let Some(id) = name.as_str() {
114                    f(AudioDevice {
115                        id,
116                        name: None,
117                        desc: desc.as_str(),
118                        dir: AudioDeviceDir::from_alsa_ioid(ioid.as_cstr()),
119                    })?;
120                }
121                p = p.add(1);
122            }
123            Ok(())
124        }
125    }
126}
