12/// Borrowed description of an audio endpoint.
13#[doc = crate::_doc_meta!{
14    location("media/audio", struct AudioDevice),
15    #[cfg(target_pointer_width = "32")]
16    test_size_of(AudioDevice = 28|224; niche Option),

20#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
21pub struct AudioDevice<'a> {
22    /// Backend device identifier.
23    ///
24    /// Examples include `default`, `hw:0,0`, PulseAudio sink names,
25    /// CPAL device names, or SDL device names.
26    pub id: &'a str,
27    /// Optional human-readable display name.
28    pub name: Option<&'a str>,
29    /// Optional human-readable description.
30    pub desc: Option<&'a str>,
31    /// Endpoint direction capability.
32    pub dir: AudioDeviceDir,
33}
34impl ConstInit for AudioDevice<'_> {
35    const INIT: Self = Self {
36        id: "",
37        name: None,
38        desc: None,
39        dir: AudioDeviceDir::INIT,
40    };
41}
42impl<'a> AudioDevice<'a> {
43    /// Returns the preferred human-facing label for this device.
44    pub const fn label(self) -> &'a str {
45        match self.name {
46            Some(name) => name,
47            None => self.id,
48        }
49    }
50    /// Returns the description when available, otherwise the identifier.
51    pub const fn desc_or_id(self) -> &'a str {
52        match self.desc {
53            Some(desc) => desc,
54            None => self.id,
55        }
56    }
57}

81#[doc = crate::_tags!(audio dir)]
82/// Direction capability of an audio endpoint.
83#[doc = crate::_doc_meta!{
84    location("media/audio", enum AudioDeviceDir),
85    test_size_of(AudioDeviceDir = 1|8; niche Option),
86}]
87/// An endpoint may support output as [`Playback`](Self::Playback),
88/// input as [`Capture`](Self::Capture), both directions as
89/// [`Duplex`](Self::Duplex), or have an unknown capability.
90#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
91pub enum AudioDeviceDir {
92    /// Output-capable endpoint.
93    Playback,
94    /// Input-capable endpoint.
95    Capture,
96    #[doc = crate::_tags!(init)]
97    /// Input/output-capable endpoint.
98    #[default]
99    Duplex,
100    /// Unknown or backend-specific endpoint direction.
101    Unknown,
102}
103_impl_init![Self::Duplex => AudioDeviceDir];
104impl_trait![fmt::Display for AudioDeviceDir |self, f| f.write_str(self.as_code())];
105impl AudioDeviceDir {
106    /// Returns the short lowercase code.
107    pub const fn as_code(self) -> &'static str {
108        match self {
109            Self::Playback => "playback",
110            Self::Capture => "capture",
111            Self::Duplex => "duplex",
112            Self::Unknown => "unknown",
113        }
114    }
115    /// Returns whether this endpoint can be used for playback.
116    pub const fn has_playback(self) -> bool {
117        matches!(self, Self::Playback | Self::Duplex)
118    }
119    /// Returns whether this endpoint can be used for capture.
120    pub const fn has_capture(self) -> bool {
121        matches!(self, Self::Capture | Self::Duplex)
122    }
123    /// Returns the unique stream direction, if this endpoint is single-direction.
124    pub const fn single_stream_dir(self) -> Option<AudioStreamDir> {
125        match self {
126            Self::Playback => Some(AudioStreamDir::Playback),
127            Self::Capture => Some(AudioStreamDir::Capture),
128            Self::Duplex | Self::Unknown => None,
129        }
130    }
131    // Returns the direction from ALSA format.
132    #[cfg(feature = "alsa")]
133    #[cfg_attr(not(ffi_alsa··), allow(dead_code))]
134    pub(crate) const fn from_alsa_ioid(ioid: Option<&CStr>) -> Self {
135        match ioid {
136            Some(cstr) => match cstr.to_bytes() {
137                b"Output" => Self::Playback,
138                b"Input" => Self::Capture,
139                _ => Self::Unknown,
140            },
141            None => Self::Duplex,
142        }
143    }
144}

146#[doc = crate::_tags!(audio dir)]
147/// Direction of an audio stream.
148#[doc = crate::_doc_meta!{
149    location("media/audio", enum AudioStreamDir),
150    test_size_of(AudioStreamDir = 1|8; niche Option),
151}]
152/// A stream can move audio toward an output endpoint as [`Playback`](Self::Playback),
153/// or from an input endpoint as [`Capture`](Self::Capture).
154#[derive(Copy, Clone, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
155pub enum AudioStreamDir {
156    #[doc = crate::_tags!(init)]
157    /// Output stream direction.
158    ///
159    /// Audio flows from the program to an output device.
160    #[default]
161    Playback,
162
163    /// Input stream direction.
164    ///
165    /// Audio flows from an input device to the program.
166    Capture,
167}
168_impl_init![Self::Playback => AudioStreamDir];
169impl_trait![fmt::Display for AudioStreamDir |self, f| f.write_str(self.as_code())];
170
171impl AudioStreamDir {
172    /// Returns the short lowercase code.
173    pub const fn as_code(self) -> &'static str {
174        match self {
175            Self::Playback => "playback",
176            Self::Capture => "capture",
177        }
178    }
179    /// Returns a human-readable name.
180    pub const fn as_name(self) -> &'static str {
181        match self {
182            Self::Playback => "Playback",
183            Self::Capture => "Capture",
184        }
185    }
186    /// Returns whether this is [`Playback`](Self::Playback).
187    pub const fn is_playback(self) -> bool {
188        matches!(self, Self::Playback)
189    }
190    /// Returns whether this is [`Capture`](Self::Capture).
191    pub const fn is_capture(self) -> bool {
192        matches!(self, Self::Capture)
193    }
194    // Converts the direction to ALSA format.
195    #[cfg(feature = "alsa")]
196    #[cfg_attr(not(ffi_alsa··), allow(dead_code))]
197    pub(crate) const fn to_alsa(self) -> _alsa_raw::snd_pcm_stream_t {
198        match self {
199            Self::Playback => _alsa_raw::SND_PCM_STREAM_PLAYBACK,
200            Self::Capture => _alsa_raw::SND_PCM_STREAM_CAPTURE,
201        }
202    }
