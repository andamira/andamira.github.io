28/// # Safe methods.
29impl Mem {
30    #[must_use]
31    /// Aligns `value` downward to the nearest multiple of `align`.
32    ///
33    /// This is equivalent to `value & !(align - 1)` but uses `wrapping_neg()`
34    /// which may generate better code on some architectures.
35    ///
36    /// # Requirements
37    /// - `align` must be a power of two (guaranteed when using [`MemLayout`][crate::MemLayout])
38    ///
39    /// # Examples
40    /// ```
41    /// use devela::Mem;
42    ///
43    /// assert_eq!(Mem::align_down(13, 8), 8);
44    /// assert_eq!(Mem::align_down(16, 8), 16);
45    /// ```
46    #[doc = crate::_doc_vendor!("mini-alloc")]
47    pub const fn align_down(value: usize, align: usize) -> usize {
48        value & align.wrapping_neg()
49    }
50
51    #[must_use]
52    /// Aligns `value` upward to the nearest multiple of `align`.
53    pub const fn align_up(value: usize, align: usize) -> usize {
54        (value + align - 1) & !(align - 1)
55    }
56
57    #[must_use]
58    /// Checks if `value` is aligned to `align`.
59    pub const fn is_aligned(value: usize, align: usize) -> bool {
60        value & (align - 1) == 0
61    }
62
63    /// Runtime version of
64    /// [`MemAligned::is_compatible`][super::MemAligned::is_compatible] for raw pointers.
65    pub fn is_aligned_to<T>(ptr: *const T, requirement: usize) -> bool {
66        let align = Mem::align_of::<T>();
67        requirement >= align && Mem::is_aligned(ptr as usize, align)
68    }
69
70    /// Returns the minimum alignment of the type in bytes.
71    ///
72    /// See `core::mem::`[`align_of`].
73    #[must_use]
74    pub const fn align_of<T>() -> usize {
75        align_of::<T>()
76    }
77
78    /// Returns the alignment of the pointed-to value in bytes.
79    ///
80    /// See `core::mem::`[`align_of_val`].
81    #[must_use]
82    pub const fn align_of_val<T: ?Sized>(val: &T) -> usize {
83        align_of_val(val)
84    }
85
86    /// Bitwise-copies a value.
87    ///
88    /// It is useful when you want to pass a function pointer to a combinator,
89    /// rather than defining a new closure.
90    ///
91    /// # Examples
92    /// ```
93    /// use devela::Mem;
94    ///
95    /// let result_from_ffi_fn: Result<(), &i32> = Err(&1);
96    /// let result_copied: Result<(), i32> = result_from_ffi_fn.map_err(Mem::copy);
97    /// ```
98    // WAIT: [core::mem::copy](https://github.com/rust-lang/rust/issues/98262)
99    #[must_use]
100    pub const fn copy<T: Copy>(x: &T) -> T {
101        *x
102    }
103
104    /// Returns a value uniquely identifying the enum variant in v.
105    #[must_use]
106    pub const fn discriminant<T>(v: &T) -> Discriminant<T> {
107        discriminant(v)
108    }
109
110    /// Disposes of a value.
111    ///
112    /// See `core::mem::`[`drop`].
113    pub fn drop<T>(_x: T) {
114        drop(_x);
115    }
116
117    /// Takes ownership and "forgets" about `t` *without running its destructor*.
118    ///
119    /// See `core::mem::`[`forget`].
120    pub fn forget<T>(t: T) {
121        forget(t);
122    }
123
124    /// Returns true if dropping values of type T matters.
125    ///
126    /// See `core::mem::`[`needs_drop`].
127    #[must_use]
128    pub const fn needs_drop<T: ?Sized>() -> bool {
129        needs_drop::<T>()
130    }
131
132    /// Moves `src` into `dest`, returning the previous `dest` value.
133    ///
134    /// See `core::mem::`[`replace`].
135    #[must_use]
136    pub const fn replace<T>(dest: &mut T, src: T) -> T {
137        replace::<T>(dest, src)
138    }
139
140    /// Returns the size of a type in bytes.
141    ///
142    /// See `core::mem::`[`size_of`].
143    #[must_use]
144    pub const fn size_of<T>() -> usize {
145        size_of::<T>()
146    }
147
148    /// Returns the size of the pointed-to value in bytes.
149    /// See `core::mem::`[`size_of_val`].
150    #[must_use]
151    pub const fn size_of_val<T: ?Sized>(val: &T) -> usize {
152        size_of_val(val)
153    }
154
155    /// Swaps the values at two locations, without deinitializing either one.
156    ///
157    /// See `core::mem::`[`swap`].
158    pub const fn swap<T>(x: &mut T, y: &mut T) {
159        swap::<T>(x, y);
160    }
161
162    /// Replaces `dest` with `T::default()`, returning the previous `dest` value.
163    ///
164    /// See `core::mem::`[`take`].
165    #[must_use]
166    pub fn take<T: Default>(dest: &mut T) -> T {
167        take::<T>(dest)
168    }
169}

171/// # Extra methods
172#[rustfmt::skip]
173impl Mem {
174    /// Returns the rounded-up byte count for a bit size.
175    ///
176    /// Fast path suitable for typical inputs.
177    /// For extremely large values (`> usize::MAX - 7`)
178    /// the internal wraparound can produce an incorrect result.
179    ///
180    /// Use [`Mem::bytes_from_bits_saturating`] when correctness
181    /// must be guaranteed even for impractical edge cases.
182    pub const fn bytes_from_bits(bit_size: usize) -> usize { (bit_size + 7) >> 3 }
183
184    /// Returns the rounded-up byte count for a bit size, saturating on overflow.
185    ///
186    /// If the addition overflows, this returns the maximum representable
187    /// rounded-up value: `usize::MAX >> 3`.
188    ///
189    /// Always produces the mathematically correct upper bound, even for inputs
190    /// near `usize::MAX`, at the cost of a small performance overhead compared
191    /// to [`Mem::bytes_from_bits`].
192    #[must_use]
193    pub const fn bytes_from_bits_saturating(bit_size: usize) -> usize {
194        #[cold] const fn bytes_from_bits_cold() -> usize { usize::MAX >> 3 }
195        is![let Some(t) = bit_size.checked_add(7), t >> 3, bytes_from_bits_cold()]
196    }
197}

201/// Mirrors unsafe operations from [`core::mem`].

206impl Mem {
207    // NOTE: can't compile, errors with: error[E0512]:

210    // /// Reinterprets the bits of a value of one type as another type.

217    /// Reads `src` as having type `&Dst` without moving the contained value.
218    ///
219    /// # Safety
220    /// See `core::mem::`[`transmute_copy`].
221    #[must_use]
222    pub const unsafe fn transmute_copy<Src, Dst>(src: &Src) -> Dst {
223        // SAFETY: Caller must uphold the safety contract.
224        unsafe { transmute_copy::<Src, Dst>(src) }
225    }

227    /// Returns the value of type `T` represented by the all-zero byte-pattern.
228    ///
229    /// # Safety
230    /// See `core::mem::`[`zeroed`].
231    #[must_use]
232    pub const unsafe fn zeroed<T>() -> T {
233        // SAFETY: Caller must uphold the safety contract.
234        unsafe { zeroed::<T>() }
235    }

238/// # Unsafe methods gated by `unsafe_slice`
239///
240/// ## Features
241/// They depend on enabling `unsafe_slice` feature, and not enabling `safe_mem`.
242#[cfg(all(not(feature = "safe_mem"), feature = "unsafe_slice"))]
243#[cfg_attr(nightly_doc, doc(cfg(feature = "unsafe_slice")))]
244impl Mem {
245    /// View any `T: Sync + Unpin + ?Sized` as `&[u8]`.
246    ///
247    /// This is a safer interface to `core::slice::`[`from_raw_parts`].
248    /// # Examples
249    /// ```
250    /// use devela::Mem;
251    ///
252    /// #[repr(C)]
253    /// struct Data(u32);
254    ///
255    /// let data = Data(1234);
256    /// let bytes = Mem::as_bytes(&data);
257    ///
258    /// if cfg!(target_endian = "little") {
259    ///     assert!(bytes == &[210, 4, 0, 0]);
260    /// } else {
261    ///     assert!(bytes == &[0, 0, 4, 210]);
262    /// }
263    /// ```
264    #[doc = crate::_doc_vendor!("rawbytes")]
265    #[must_use]
266    pub fn as_bytes<'t, T: Sync + Unpin + ?Sized + 't>(v: &T) -> &'t [u8] {
267        // SAFETY: `v` is valid; u8 has alignment 1, size_of_val(v) gives the exact byte length.
268        unsafe { from_raw_parts(v as *const _ as *const u8, size_of_val(v)) }
269    }
270
271    /// View any `T: Sync + Unpin + ?Sized` as `&mut [u8]`.
272    ///
273    /// This is a safer interface to `core::slice::`[`from_raw_parts_mut`].
274    /// # Examples
275    /// ```
276    /// use devela::Mem;
277    ///
278    /// #[repr(C)]
279    /// struct Data(u32);
280    ///
281    /// let mut data = Data(1234);
282    /// let bytes = Mem::as_bytes_mut(&mut data);
283    ///
284    /// if cfg!(target_endian = "little") {
285    ///     bytes[1] = 0;
286    ///     assert!(bytes == &[210, 0, 0, 0] && data.0 == 210);
287    /// } else {
288    ///     bytes[1] = 0;
289    ///     assert!(bytes == &[0, 0, 0, 210] && data.0 == 210);
290    /// }
291    /// ```
292    #[doc = crate::_doc_vendor!("rawbytes")]
293    #[must_use]
294    pub fn as_bytes_mut<'t, T: Sync + Unpin + ?Sized + 't>(v: &mut T) -> &'t mut [u8] {
295        // SAFETY: `v` is a valid, exclusive reference;
296        // u8's alignment is 1, and size_of_val(v) bounds the mutable slice.
297        unsafe { from_raw_parts_mut(v as *mut _ as *mut u8, size_of_val(v)) }
298    }
299
300    /// View any `T: Sync + Unpin + Sized` as `&[u8]` *compile-time* friendly.
301    ///
302    /// This is a safer interface to `core::slice::`[`from_raw_parts`], for `Sized` types.
303    /// # Examples
304    /// ```
305    /// use devela::Mem;
306    ///
307    /// const DATA: u32 = 1234;
308    /// const BYTES: &[u8] = Mem::as_bytes_sized(&DATA);
309    ///
310    /// if cfg!(target_endian = "little") {
311    ///     assert_eq!(BYTES, &[210, 4, 0, 0]);
312    /// } else {
313    ///     assert_eq!(BYTES, &[0, 0, 4, 210]);
314    /// }
315    /// ```
316    #[must_use]
317    pub const fn as_bytes_sized<T: Sync + Unpin>(v: &T) -> &[u8] {
318        // SAFETY: `v` is valid; casting to *const u8 is safe (u8 has alignment 1)
319        // and size_of::<T>() exactly covers the object.
320        unsafe { from_raw_parts(v as *const T as *const u8, size_of::<T>()) }
321    }
