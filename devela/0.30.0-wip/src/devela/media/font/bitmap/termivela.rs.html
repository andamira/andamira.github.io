3//! Defines the embedded Termivela bitmap-font faces.
4//
5
6use crate::{Dvbf, FontBitmapView, Fonts, unwrap};
7
8const REGULAR_BYTES: &[u8] =
9    include_bytes!("../../../../assets/font/termivela/termivela-8x16-regular.dvbf");
10const BOLD_BYTES: &[u8] =
11    include_bytes!("../../../../assets/font/termivela/termivela-8x16-bold.dvbf");
12
13const fn read_embedded(bytes: &'static [u8]) -> FontBitmapView<'static> {
14    unwrap![ok_expect Dvbf::read(bytes), "invalid embedded Termivela DVBF data"]
15}
16
17/// # Termivela
18///
19/// These embedded strikes are derived from
20/// [Terminus Font][crate::_doc::reference::vendored#terminus-font].
21///
22/// See the recorded [Termivela modifications][crate::_doc::reference::vendored::termivela].
23impl Fonts {
24    /// The regular 8×16 Termivela monochrome bitmap font.
25    pub const TERMIVELA_8_16: FontBitmapView<'static> = read_embedded(REGULAR_BYTES);
26
27    /// The bold 8×16 Termivela monochrome bitmap font.
28    pub const TERMIVELA_8_16_BOLD: FontBitmapView<'static> = read_embedded(BOLD_BYTES);
29}
30
31#[cfg(test)]
32mod _test {
33    use super::*;
34
35    #[test]
36    fn faces_have_expected_metrics() {
37        for font in [Fonts::TERMIVELA_8_16, Fonts::TERMIVELA_8_16_BOLD] {
38            assert_eq!(font.glyph_count(), 1356);
39            assert_eq!((font.width(), font.height()), (8, 16));
40            assert_eq!(font.bounds().pos.dim, [0, -4]);
41            assert_eq!(font.advance_x(), 8);
42            assert_eq!((font.ascent(), font.descent(), font.line_advance()), (12, 4, 16));
43            assert_eq!(font.default_character(), Some('\u{FFFD}'));
44        }
45    }
46    #[test]
47    fn regular_and_bold_are_distinct() {
48        let regular = Fonts::TERMIVELA_8_16.glyph('A').unwrap();
49        let bold = Fonts::TERMIVELA_8_16_BOLD.glyph('A').unwrap();
50        assert_ne!(regular.bitmap(), bold.bitmap());
51    }
52}
