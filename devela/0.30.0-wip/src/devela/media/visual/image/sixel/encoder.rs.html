65    /// Sixel encoding with RLE compression for repeated characters.
66    ///
67    /// - RLE encoding is used when the same sixel character repeats 4 or more times.
68    #[rustfmt::skip]
69    pub const fn encode_rgb(&mut self, rgb: &[u8], extent: Extent2<u32>, buf: &mut [u8])
70        -> Result<usize, NotEnoughSpace> {
71        __dbg![slog!{clear sixel_encoder:64+64}]; // slog point of entry
72        let mut off = 0;
73        off += unwrap![ok? Self::write_sixel_start_simple(slice![mut buf, off,..])];
74        unwrap![ok? self.build_palette(rgb, extent)];
75        off += self.palette.write_definitions(slice![mut buf, off, ..]);
76        let [width, height] = extent.dim;
77
78        // process each horizontal band
79        let mut band_y = 0;
80        while band_y < height {
81            let band_height = Cmp(height - band_y).min(6);
82            __dbg![slog!{sixel_encoder:64+64
83                "Processing band at row ", %band_y, " (height: ", %band_height, ")"}];
84
85            // for each color, render with RLE compression
86            let mut iter = self.palette.defined_colors();
87            while let Some((idx, palette_color)) = iter.next() {
88                is![idx == 0, continue]; // Skip background
89
90                __dbg![slog!{sixel_encoder:64+64 "  Rendering color #", %idx, " in band."}];
91                off += unwrap![ok? Self::write_color_reference(slice![mut buf, off,..], idx)];
92
93                // build and compress sixel pattern for this color
94                lets![mut current_char = None, mut repeat_count = 0];
95                whilst! { x in 0..width; {
96                    let sixel_bits = self.build_sixel_bits_for_color(rgb, width, x, band_y,
97                        band_height, palette_color);
98                    let sixel_char = SixelChar::from_bitmask(sixel_bits);
99
100                    // RLE compression logic
101                    // Handles lines longer than 255 by splitting long runs,
102                    // outputting the current run and starting a new one with the same character
103                    let is_same_char = is![let Some(c) = current_char, sixel_char.eq(c), false];
104                    match (is_same_char, repeat_count == 255) {
105                        (true, false) => repeat_count += 1,
106                        (true, true) => { // Split run: output but keep same character
107                            if let Some(char) = current_char {
108                                off += unwrap![ok? Self::write_rle_run(slice![mut buf, off,..],
109                                    char, repeat_count)];
110                            }
111                            repeat_count = 1;
112                        }
113                        (false, _) => { // New run: output and switch character
114                            if let Some(char) = current_char {
115                                off += unwrap![ok? Self::write_rle_run(slice![mut buf, off,..],
116                                    char, repeat_count)];
117                            }
118                            current_char = Some(sixel_char);
119                            repeat_count = 1;
120                        }
121                    }
122                }}
123                // output the final run
124                if let Some(char) = current_char {
125                    off += unwrap![ok?
126                        Self::write_rle_run(slice![mut buf, off,..], char, repeat_count)];
127                }
128                // return to start for next color
129                is! { idx != self.palette.len() as u16 - 1, write_at![buf, +=off, b'$'];}
130            }
131            band_y += band_height;
132            is! { band_y < height, write_at![buf, +=off, b'-'];} // move to next band?
133        }
134        off += unwrap![ok? Self::write_sixel_end(slice![mut buf, off,..])];
135        Ok(off)
136    }
137}

147impl<const MAX_COLORS: usize> SixelEncoder<MAX_COLORS> {
148    /// Builds the palette from the given rgb byte buffer.
149    const fn build_palette(&mut self, rgb: &[u8], extent: Extent2<u32>) -> Result<(), NotEnoughSpace> {
150        let [w, h] = extent.dim;
151        self.palette = SixelPalette::new();
152        unwrap![ok? self.palette.add_color(SixelColor::BLACK)];
153        whilst! { i in 0..(w * h); {
154            // is![self.palette.is_full(), break]; // early termination MAYBE for another version
155            let idx = (i * 3) as usize;
156            let color = SixelColor::from_rgb888(rgb[idx], rgb[idx + 1], rgb[idx + 2]);
157            if !color.is_black() { let _ = self.palette.find_or_add(color); }
158        }}
159        Ok(())
160    }
161
162    /// Build sixel bits for a specific color in a column.
163    const fn build_sixel_bits_for_color(&self, rgb: &[u8], width: u32, x: u32, band_y: u32,
164        band_height: u32, target_color: SixelColor) -> u8 {
165        let mut sixel_bits = 0u8;
166        whilst! { dy in 0..band_height; {
167            let y = band_y + dy;
168            let idx = ((y * width + x) * 3) as usize;
169            // only process if within bounds
170            if idx + 2 < rgb.len() {
171                let pixel_color = SixelColor::from_rgb888(rgb[idx], rgb[idx + 1], rgb[idx + 2]);
172                if pixel_color.is_similar_to(target_color) { sixel_bits |= 1 << dy }
173            }
174        }}
175        sixel_bits
176    }
177}

185impl<const MAX_COLORS: usize> SixelEncoder<MAX_COLORS> {
186    /// Write sixel sequence start with full raster attributes.
187    ///
188    /// Returns an error if the buffer doesn't have at least 21 bytes, for good measure.
189    pub const fn write_sixel_start(
190        buf: &mut [u8],
191        extent: Extent2<u32>,
192    ) -> Result<usize, NotEnoughSpace> {
193        let [width, height] = extent.dim;
194        let (wd, hd) = (Digits(width), Digits(height));
195        let needed = 12 + wd.count_digits10() as usize + hd.count_digits10() as usize;
196        is![buf.len() < needed, return Err(NotEnoughSpace(Some(needed)))];
197        let mut offset = 0;
198        // initial sequence (6 bytes)
199        write_at!(
200            buf, +=offset, b'\x1b', b'P', // CIS
201            b';', b'1', b';', // p2 = 1 (transparent background)
202            b'q', // sixel ID
203        );
204        // raster attributes opening sequence (5 bytes)
205        // https://vt100.net/docs/vt3xx-gp/chapter14.html#S14.3.2
206        write_at!(buf, +=offset, b'"', b'1', b';', b'1', b';'); // pixel aspect ratio
207        offset += wd.write_digits10(buf, offset);
208        write_at!(buf, +=offset, b';'); // + 1 byte
209        offset += hd.write_digits10(buf, offset);
210        Ok(offset)
211    }
212    /// Write sixel sequence start with full raster attributes
213    pub const fn write_sixel_start_simple(buf: &mut [u8]) -> Result<usize, NotEnoughSpace> {
214        is![buf.len() < 6, return Err(NotEnoughSpace(Some(6)))];
215        write_at!(
216            buf, 0, b'\x1b', b'P', // CIS
217            b';', b'1', b';', // p2 = 1 (transparent background)
218            b'q', // sixel ID
219        );
220        Ok(6)
221    }
222
223    /// Write sixel sequence end.
224    pub const fn write_sixel_end(buf: &mut [u8]) -> Result<usize, NotEnoughSpace> {
225        is![buf.len() < 2, return Err(NotEnoughSpace(Some(2)))];
226        write_at!(buf, 0, b'\x1b', b'\\'); // string terminator C0 sequence
227        Ok(2)
228    }
229
230    /// Write color reference (#NN)
231    const fn write_color_reference(buffer: &mut [u8], index: u16) -> Result<usize, NotEnoughSpace> {
232        is![buffer.len() < 2, return Err(NotEnoughSpace(Some(2)))];
233        buffer[0] = b'#';
234        let digits_written = Digits(index).write_digits10_nonzero(buffer, 1);
235        is![digits_written == 0, return Err(NotEnoughSpace(Some(2)))];
236        Ok(1 + digits_written)
237    }
238
239    /// Write an RLE run: either compressed `!NNN` or repeated characters.
240    const fn write_rle_run(
241        buffer: &mut [u8],
242        sc: SixelChar,
243        count: u8,
244    ) -> Result<usize, NotEnoughSpace> {
245        let mut off = 0;
246        if count >= 4 {
247            let digits = Digits(count);
248            let dcount = digits.count_digits10();
249            let needed = 2 + dcount as usize;
250            is![buffer.len() < needed, return Err(NotEnoughSpace(Some(needed)))];
251
252            write_at![buffer, +=off, b'!'];
253            off += digits.write_digits10(buffer, 1);
254            write_at![buffer, +=off, sc.as_byte()];
255            __dbg![slog! {sixel_encoder:64+64 "  RLE: !", %count, ":", @sc.to_string_box(),
256            " (saved ", %count-(2+dcount), " bytes)"}];
257        } else {
258            __dbg![slog! {sixel_encoder:64+64
259            "  NO RLE: ", %count, " times ", @sc.to_string_box()}];
260            whilst! { _n in 0..count; {
261                is![off >= buffer.len(), return Err(NotEnoughSpace(Some(buffer.len())))];
262                write_at![buffer, +=off, sc.as_byte()];
263            }}
264        }
265        Ok(off)
266    }
