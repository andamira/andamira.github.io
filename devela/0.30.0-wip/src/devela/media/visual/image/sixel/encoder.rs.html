64    /// Sixel encoding with RLE compression for repeated characters.
65    ///
66    /// - RLE encoding is used when the same sixel character repeats 4 or more times.
67    #[rustfmt::skip]
68    pub const fn encode_rgb(&mut self, rgb: &[u8], extent: Extent2<u32>, buf: &mut [u8])
69        -> Result<usize, NotEnoughSpace> {
70        __dbg![slog!{clear sixel_encoder:64+64}]; // slog point of entry
71        let mut off = 0;
72        off += unwrap![ok? Self::write_sixel_start_simple(slice![mut buf, off,..])];
73        unwrap![ok? self.build_palette(rgb, extent)];
74        off += self.palette.write_definitions(slice![mut buf, off, ..]);
75        let [width, height] = extent.dim;
76
77        // process each horizontal band
78        let mut band_y = 0;
79        while band_y < height {
80            let band_height = Cmp(height - band_y).min(6);
81            __dbg![slog!{sixel_encoder:64+64
82                "Processing band at row ", %band_y, " (height: ", %band_height, ")"}];
83
84            // for each color, render with RLE compression
85            let mut iter = self.palette.defined_colors();
86            while let Some((idx, palette_color)) = iter.next() {
87                is![idx == 0, continue]; // Skip background
88
89                __dbg![slog!{sixel_encoder:64+64 "  Rendering color #", %idx, " in band."}];
90                off += unwrap![ok? Self::write_color_reference(slice![mut buf, off,..], idx)];
91
92                // build and compress sixel pattern for this color
93                lets![mut current_char = None, mut repeat_count = 0];
94                whilst! { x in 0..width; {
95                    let sixel_bits = self.build_sixel_bits_for_color(rgb, width, x, band_y,
96                        band_height, palette_color);
97                    let sixel_char = SixelChar::from_bitmask(sixel_bits);
98
99                    // RLE compression logic
100                    // Handles lines longer than 255 by splitting long runs,
101                    // outputting the current run and starting a new one with the same character
102                    let is_same_char = is![let Some(c) = current_char, sixel_char.eq(c), false];
103                    match (is_same_char, repeat_count == 255) {
104                        (true, false) => repeat_count += 1,
105                        (true, true) => { // Split run: output but keep same character
106                            if let Some(char) = current_char {
107                                off += unwrap![ok? Self::write_rle_run(slice![mut buf, off,..],
108                                    char, repeat_count)];
109                            }
110                            repeat_count = 1;
111                        }
112                        (false, _) => { // New run: output and switch character
113                            if let Some(char) = current_char {
114                                off += unwrap![ok? Self::write_rle_run(slice![mut buf, off,..],
115                                    char, repeat_count)];
116                            }
117                            current_char = Some(sixel_char);
118                            repeat_count = 1;
119                        }
120                    }
121                }}
122                // output the final run
123                if let Some(char) = current_char {
124                    off += unwrap![ok?
125                        Self::write_rle_run(slice![mut buf, off,..], char, repeat_count)];
126                }
127                // return to start for next color
128                is! { idx != self.palette.len() as u16 - 1, write_at![buf, +=off, b'$'];}
129            }
130            band_y += band_height;
131            is! { band_y < height, write_at![buf, +=off, b'-'];} // move to next band?
132        }
133        off += unwrap![ok? Self::write_sixel_end(slice![mut buf, off,..])];
134        Ok(off)
135    }
136}

146impl<const MAX_COLORS: usize> SixelEncoder<MAX_COLORS> {
147    /// Builds the palette from the given rgb byte buffer.
148    const fn build_palette(&mut self, rgb: &[u8], extent: Extent2<u32>) -> Result<(), NotEnoughSpace> {
149        let [w, h] = extent.dim;
150        self.palette = SixelPalette::new();
151        unwrap![ok? self.palette.add_color(SixelColor::BLACK)];
152        whilst! { i in 0..(w * h); {
153            // is![self.palette.is_full(), break]; // early termination MAYBE for another version
154            let idx = (i * 3) as usize;
155            let color = SixelColor::from_rgb888(rgb[idx], rgb[idx + 1], rgb[idx + 2]);
156            if !color.is_black() { let _ = self.palette.find_or_add(color); }
157        }}
158        Ok(())
159    }
160
161    /// Build sixel bits for a specific color in a column.
162    const fn build_sixel_bits_for_color(&self, rgb: &[u8], width: u32, x: u32, band_y: u32,
163        band_height: u32, target_color: SixelColor) -> u8 {
164        let mut sixel_bits = 0u8;
165        whilst! { dy in 0..band_height; {
166            let y = band_y + dy;
167            let idx = ((y * width + x) * 3) as usize;
168            // only process if within bounds
169            if idx + 2 < rgb.len() {
170                let pixel_color = SixelColor::from_rgb888(rgb[idx], rgb[idx + 1], rgb[idx + 2]);
171                if pixel_color.is_similar_to(target_color) { sixel_bits |= 1 << dy }
172            }
173        }}
174        sixel_bits
175    }
176}

184impl<const MAX_COLORS: usize> SixelEncoder<MAX_COLORS> {
185    /// Write sixel sequence start with full raster attributes.
186    ///
187    /// Returns an error if the buffer doesn't have at least 21 bytes, for good measure.
188    pub const fn write_sixel_start(
189        buf: &mut [u8],
190        extent: Extent2<u32>,
191    ) -> Result<usize, NotEnoughSpace> {
192        let [width, height] = extent.dim;
193        let (wd, hd) = (Digits(width), Digits(height));
194        let needed = 12 + wd.count_digits10() as usize + hd.count_digits10() as usize;
195        is![buf.len() < needed, return Err(NotEnoughSpace(Some(needed)))];
196        let mut offset = 0;
197        // initial sequence (6 bytes)
198        write_at!(
199            buf, +=offset, b'\x1b', b'P', // CIS
200            b';', b'1', b';', // p2 = 1 (transparent background)
201            b'q', // sixel ID
202        );
203        // raster attributes opening sequence (5 bytes)
204        // https://vt100.net/docs/vt3xx-gp/chapter14.html#S14.3.2
205        write_at!(buf, +=offset, b'"', b'1', b';', b'1', b';'); // pixel aspect ratio
206        offset += wd.write_digits10(buf, offset);
207        write_at!(buf, +=offset, b';'); // + 1 byte
208        offset += hd.write_digits10(buf, offset);
209        Ok(offset)
210    }
211    /// Write sixel sequence start with full raster attributes
212    pub const fn write_sixel_start_simple(buf: &mut [u8]) -> Result<usize, NotEnoughSpace> {
213        is![buf.len() < 6, return Err(NotEnoughSpace(Some(6)))];
214        write_at!(
215            buf, 0, b'\x1b', b'P', // CIS
216            b';', b'1', b';', // p2 = 1 (transparent background)
217            b'q', // sixel ID
218        );
219        Ok(6)
220    }
221
222    /// Write sixel sequence end.
223    pub const fn write_sixel_end(buf: &mut [u8]) -> Result<usize, NotEnoughSpace> {
224        is![buf.len() < 2, return Err(NotEnoughSpace(Some(2)))];
225        write_at!(buf, 0, b'\x1b', b'\\'); // string terminator C0 sequence
226        Ok(2)
227    }
228
229    /// Write color reference (#NN)
230    const fn write_color_reference(buffer: &mut [u8], index: u16) -> Result<usize, NotEnoughSpace> {
231        is![buffer.len() < 2, return Err(NotEnoughSpace(Some(2)))];
232        buffer[0] = b'#';
233        let digits_written = Digits(index).write_digits10_nonzero(buffer, 1);
234        is![digits_written == 0, return Err(NotEnoughSpace(Some(2)))];
235        Ok(1 + digits_written)
236    }
237
238    /// Write an RLE run: either compressed `!NNN` or repeated characters.
239    const fn write_rle_run(
240        buffer: &mut [u8],
241        sc: SixelChar,
242        count: u8,
243    ) -> Result<usize, NotEnoughSpace> {
244        let mut off = 0;
245        if count >= 4 {
246            let digits = Digits(count);
247            let dcount = digits.count_digits10();
248            let needed = 2 + dcount as usize;
249            is![buffer.len() < needed, return Err(NotEnoughSpace(Some(needed)))];
250
251            write_at![buffer, +=off, b'!'];
252            off += digits.write_digits10(buffer, 1);
253            write_at![buffer, +=off, sc.as_byte()];
254            __dbg![slog! {sixel_encoder:64+64 "  RLE: !", %count, ":", @sc.to_string_box(),
255            " (saved ", %count-(2+dcount), " bytes)"}];
256        } else {
257            __dbg![slog! {sixel_encoder:64+64
258            "  NO RLE: ", %count, " times ", @sc.to_string_box()}];
259            whilst! { _n in 0..count; {
260                is![off >= buffer.len(), return Err(NotEnoughSpace(Some(buffer.len())))];
261                write_at![buffer, +=off, sc.as_byte()];
262            }}
263        }
264        Ok(off)
265    }
