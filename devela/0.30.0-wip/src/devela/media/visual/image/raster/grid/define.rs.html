7use crate::is;
8use crate::{ArrayShape, Extent2, Overflow, Position2, RasterCoordIter, RegionS2};

10#[doc = crate::_tags!(image layout)]
11/// A finite logical grid of canonical raster cells.
12#[doc = crate::_doc_meta!{
13    location("media/visual/image/raster/grid", struct RasterGrid),
14    test_size_of(RasterGrid = 8|64; niche !Option),
15}]
16/// `RasterGrid` defines the logical geometry of a two-dimensional raster,
17/// independently of its samples and physical storage.
18///
19/// Its canonical coordinate system has:
20/// - the origin at the upper-left boundary;
21/// - positive `x` extending rightward;
22/// - positive `y` extending downward;
23/// - an extent ordered as `[width, height]`;
24/// - `u32` raster-cell coordinates;
25/// - valid cell coordinates satisfying `x < width` and `y < height`.
26///
27/// The cell at coordinate `[x, y]` occupies the half-open boundary-space region:
28/// ```text
29/// [x, x + 1) × [y, y + 1)
30/// ```
31///
32/// Its conceptual center is:
33/// ```text
34/// [x + 1/2, y + 1/2]
35/// ```
36///
37/// Signed raster-lattice positions used by rasterization and clipping are
38/// distinct from contained raster-cell coordinates.
39/// [`checked_coord`][Self::checked_coord] validates
40/// and projects such positions into the grid.
41///
42/// Raster geometry is also independent of machine-addressable array geometry.
43/// [`try_array_shape`][Self::try_array_shape] provides an explicit projection
44/// into an [`ArrayShape`] when its dimensions are representable there.
45#[must_use]
46#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
47pub struct RasterGrid {
48    extent: Extent2<u32>,
49}
50
51impl RasterGrid {
52    /* construction */
53
54    /// Creates a logical raster grid with the given extent.
55    ///
56    /// Empty grids are valid.
57    pub const fn new(extent: Extent2<u32>) -> Self {
58        Self { extent }
59    }
60
61    /* dimensions */
62
63    /// Returns the logical raster extent as `[width, height]`.
64    ///
65    /// Axis `0` is `x`/width and axis `1` is `y`/height.
66    pub const fn extent(&self) -> Extent2<u32> {
67        self.extent
68    }
69    /// Returns the raster width.
70    #[must_use]
71    pub const fn width(&self) -> u32 {
72        self.extent.dim[0]
73    }
74    /// Returns the raster height.
75    #[must_use]
76    pub const fn height(&self) -> u32 {
77        self.extent.dim[1]
78    }
79    /// Returns the exact number of logical raster cells.
80    #[must_use]
81    pub const fn cell_count(&self) -> u64 {
82        self.width() as u64 * self.height() as u64
83    }
84    /// Returns whether the grid contains no raster cells.
85    #[must_use]
86    pub const fn is_empty(&self) -> bool {
87        self.width() == 0 || self.height() == 0
88    }
89
90    /* regions and coordinates */
91
92    /// Returns the complete logical raster bounds.
93    ///
94    /// The returned region begins at `[0, 0]` and has the grid's extent.
95    pub const fn bounds(&self) -> RegionS2<u32> {
96        RegionS2::new(Position2::new([0, 0]), self.extent)
97    }
98    /// Returns whether `coord` identifies a cell in this grid.
99    #[must_use]
100    pub const fn contains(&self, coord: Position2<u32>) -> bool {
101        coord.dim[0] < self.width() && coord.dim[1] < self.height()
102    }
103    /// Converts a signed raster-lattice position into a contained raster coordinate.
104    ///
105    /// Negative and external positions are rejected.
106    #[must_use]
107    pub const fn checked_coord(&self, position: Position2<i64>) -> Option<Position2<u32>> {
108        let [x, y] = position.dim;
109        is! { x < 0 || y < 0 || x > u32::MAX as i64 || y > u32::MAX as i64, return None }
110        let coord = Position2::new([x as u32, y as u32]);
111        is! { self.contains(coord), Some(coord), None }
112    }
113    /// Returns the unit boundary-space region occupied by `coord`.
114    ///
115    /// Returns `None` when `coord` is outside the grid.
116    #[must_use]
117    pub const fn cell_bounds(&self, coord: Position2<u32>) -> Option<RegionS2<u32>> {
118        is! { !self.contains(coord), return None }
119        Some(RegionS2::new(coord, Extent2::new([1, 1])))
120    }
121
122    /* indexing */
123
124    /// Returns the canonical logical index of `coord`.
125    ///
126    /// Raster indices are x-fast and independent of physical storage:
127    ///
128    /// ```text
129    /// index = y * width + x
130    /// ```
131    ///
132    /// The returned `u64` can represent every cell of every `u32 × u32`
133    /// raster grid.
134    ///
135    /// Returns `None` when `coord` lies outside the grid.
136    #[must_use]
137    pub const fn cell_index(&self, coord: Position2<u32>) -> Option<u64> {
138        is! { !self.contains(coord), return None }
139        let [x, y] = coord.dim;
140        Some(y as u64 * self.width() as u64 + x as u64)
141    }
142    /// Returns the coordinate at the canonical logical `index`.
143    ///
144    /// This is the inverse of [`cell_index`][Self::cell_index].
145    ///
146    /// Returns `None` when `index >= self.cell_count()`.
147    #[must_use]
148    pub const fn coord_at(&self, index: u64) -> Option<Position2<u32>> {
149        is! { index >= self.cell_count(), return None }
150        // `index < cell_count()` implies a non-zero width.
151        let width = self.width() as u64;
152        let (x, y) = (index % width, index / width);
153        Some(Position2::new([x as u32, y as u32]))
154    }
155
156    /* traversal */
157
158    /// Returns an exhaustive iterator over the raster-cell coordinates.
159    ///
160    /// Coordinates are yielded as [`Position2<u32>`] values in canonical
161    /// raster order, with `x` changing fastest:
162    ///
163    /// ```text
164    /// [0, 0], [1, 0], …, [width - 1, 0],
165    /// [0, 1], [1, 1], …
166    /// ```
167    ///
168    /// Traversal is independent of sample storage, physical layout,
169    /// and machine-addressable array coordinates.
170    pub const fn coords(&self) -> RasterCoordIter {
171        RasterCoordIter::new(*self)
172    }
173
174    /* projections */
175
176    /// Projects this raster grid into a two-dimensional [`ArrayShape`].
177    ///
178    /// Raster coordinates and extents use `u32`, independently of machine
179    /// pointer width. [`ArrayShape`] uses `usize` axis lengths, so this
180    /// conversion is fallible when either raster dimension cannot be
181    /// represented as a `usize`.
182    ///
183    /// Axis `0` corresponds to raster `x`/width and axis `1` to `y`/height.
184    ///
185    /// This projection describes only the logical array shape. It does not
186    /// choose a storage order or construct an [`ArrayLayout`][crate::ArrayLayout].
187    /// In particular, the resulting shape's total [`element_count`][ArrayShape::element_count]
188    /// may still be unrepresentable as a `usize`.
189    ///
190    /// # Errors
191    /// Returns [`Overflow`] if either raster dimension cannot be represented as a `usize`.
192    pub const fn try_array_shape(&self) -> Result<ArrayShape<2>, Overflow> {
193        let [width, height] = self.extent.dim;
194        if width as u64 > usize::MAX as u64 || height as u64 > usize::MAX as u64 {
195            return Err(Overflow(None));
196        }
197        Ok(ArrayShape::new([width as usize, height as usize]))
198    }
199}
