18/// `RasterLineIter` rasterizes a one-cell-wide line
19/// between two signed raster-lattice positions.
20///
21/// The line:
22/// - includes both endpoints;
23/// - is 8-connected;
24/// - emits at most one element per major-axis position;
25/// - emits only cells contained by its [`RasterGrid`];
26/// - gives every emitted element [`Coverage8::FULL`][crate::Coverage8::FULL].
27///
28/// Rasterization is independent of sample storage, color, paint, compositing,
29/// and physical raster layout.
30///
31/// # Cell-selection invariant
32///
33/// For any endpoints `a` and `b`, swapping the endpoints reverses the emitted
34/// sequence without changing the selected cells:
35///
36/// ```text
37/// line(a, b) == reverse(line(b, a))
38/// ```
39///
40/// This includes exact half-cell ties and cells skipped by grid clipping.
41///
42/// At exact ties, this invariant is preferred over complete reflection symmetry:
43/// reflecting a line across an axis may select the complementary equally-near cells.
44///
45/// # Algorithm
46///
47/// Uses a canonicalized major-axis integer error accumulator in the
48/// Bresenham/DDA family. One major-axis step is taken per candidate cell, while
49/// a bounded unsigned error recurrence decides when to advance the minor axis.
50///
51/// The recurrence avoids floating-point arithmetic, division during iteration,
52/// and doubled deltas that could overflow at the full `i64` raster-lattice range.
53///
54/// Constructor-time clipping uses widened integer arithmetic to jump directly
55/// to the exact recurrence state at a selected major-axis step.
56/// Iteration itself remains incremental and division-free.
57///
58/// # Grid clipping
59///
60/// The candidate major-axis interval is restricted to the portion overlapping the raster grid.
61///
62/// The iterator reconstructs the exact integer-rasterization state at the
63/// clipped interval boundaries, so clipping does not restart or alter the
64/// underlying line recurrence. The emitted sequence therefore remains identical
65/// to rasterizing the complete line and discarding cells outside the grid.
66///
67/// Candidates outside the grid along the minor axis are still skipped during iteration.
68///
69/// After construction, candidate work is bounded by the grid extent along the
70/// line's major axis: at most `width` candidates for an x-major line or
71/// `height` candidates for a y-major line.
72#[must_use]
73#[derive(Clone, Debug, PartialEq, Eq, Hash)]
74pub struct RasterLineIter {
75    grid: RasterGrid,
76    position: Position2<i64>,
77    end: Position2<i64>,
78
79    major_delta: u64,
80    minor_delta: u64,
81    error: u64,
82
83    major_axis: u8,
84    minor_step: i8,
85    forward: bool,
86    finished: bool,
87}
88
89impl RasterLineIter {
90    /* constructors */
91
92    /// Creates an aliased raster-line iterator.
93    ///
94    /// `start` and `end` are signed raster-lattice positions and may lie outside `grid`.
95    /// Emitted [`RasterElement`] coordinates are always contained `u32` raster-cell coordinates.
96    pub const fn new(grid: RasterGrid, start: Position2<i64>, end: Position2<i64>) -> Self {
97        let dx = start.dim[0].abs_diff(end.dim[0]);
98        let dy = start.dim[1].abs_diff(end.dim[1]);
99        let (major_axis, major_delta, minor_delta) = is![dx >= dy, (0, dx, dy), (1, dy, dx)];
100        lets! { major = major_axis as usize, minor = 1 - major };
101        // Canonical traversal always increases along the major axis.
102        let forward = start.dim[major] <= end.dim[major];
103        let (canonical_start, canonical_end) = is![forward, (start, end), (end, start)];
104        let minor_step = is![
105            canonical_start.dim[minor] < canonical_end.dim[minor],
106            1,
107            is![canonical_start.dim[minor] > canonical_end.dim[minor], -1, 0]
108        ];
109        lets! { mut position = start, mut iter_end = end, mut error = major_delta / 2 }
110        let mut finished = !line_bounds_intersect(grid, start, end);
111        if !finished {
112            if let Some((first_step, last_step)) =
113                grid_major_step_range(grid, canonical_start, canonical_end, major_axis)
114            {
115                let (first_position, first_error) = state_at(
116                    canonical_start,
117                    major_axis,
118                    major_delta,
119                    minor_delta,
120                    minor_step,
121                    first_step,
122                );
123                let (last_position, last_error) = state_at(
124                    canonical_start,
125                    major_axis,
126                    major_delta,
127                    minor_delta,
128                    minor_step,
129                    last_step,
130                );
131                if forward {
132                    position = first_position;
133                    iter_end = last_position;
134                    error = first_error;
135                } else {
136                    position = last_position;
137                    iter_end = first_position;
138                    error = last_error;
139                }
140            } else {
141                finished = true;
142            }
143        }
144        Self {
145            grid,
146            position,
147            end: iter_end,
148            major_delta,
149            minor_delta,
150            error,
151            major_axis,
152            minor_step,
153            forward,
154            finished,
155        }
156    }
157
158    /* queries */
159
160    /// Returns the raster grid that clips this line.
161    pub const fn grid(&self) -> RasterGrid {
162        self.grid
163    }
164    /// Returns whether the iterator has reached its finished state.
165    #[must_use]
166    pub const fn is_finished(&self) -> bool {
167        self.finished
168    }
169
170    /* iteration */
171
172    /// Advances the iterator and returns the next contained raster element.
173    #[must_use]
174    pub const fn next(&mut self) -> Option<RasterElement> {
175        loop {
176            is! { self.finished, return None }
177            let position = self.position;
178            is! { self.at_end(), self.finished = true, self.advance() }
179            if let Some(coord) = self.grid.checked_coord(position) {
180                return Some(RasterElement::full(coord));
181            }
182        }
183    }
184
185    /* private */
186
187    const fn at_end(&self) -> bool {
188        self.position.dim[0] == self.end.dim[0] && self.position.dim[1] == self.end.dim[1]
189    }
190    const fn advance(&mut self) {
191        is! { self.forward, self.advance_forward(), self.advance_backward() }
192    }
193
194    /// Advances one step in canonical positive-major-axis order.
195    const fn advance_forward(&mut self) {
196        let major = self.major_axis as usize;
197        // Safe: current major coordinate is strictly before the endpoint.
198        self.position.dim[major] += 1;
199        is! { self.minor_delta == 0, return }
200        let gap = self.major_delta - self.minor_delta;
201        if self.error >= gap {
202            self.error -= gap;
203            self.step_minor(self.minor_step);
204        } else {
205            self.error += self.minor_delta;
206        }
207    }
208    /// Advances one step against canonical positive-major-axis order.
209    const fn advance_backward(&mut self) {
210        let major = self.major_axis as usize;
211        // Safe: current major coordinate is strictly after the endpoint.
212        self.position.dim[major] -= 1;
213        is! { self.minor_delta == 0, return }
214        if self.error < self.minor_delta {
215            self.error += self.major_delta - self.minor_delta;
216            self.step_minor(-self.minor_step);
217        } else {
218            self.error -= self.minor_delta;
219        }
220    }
221    const fn step_minor(&mut self, step: i8) {
222        let minor = 1 - self.major_axis as usize;
223        is![step > 0, self.position.dim[minor] += 1, is![step < 0, self.position.dim[minor] -= 1]];
224    }
225    const fn candidate_upper_bound(&self) -> Option<usize> {
226        is! { self.finished, return Some(0) }
227        let major = self.major_axis as usize;
228        let delta = self.position.dim[major].abs_diff(self.end.dim[major]);
229        let count = unwrap![some? delta.checked_add(1)];
230        is! { count > usize::MAX as u64, None, Some(count as usize) }
231    }
232}

234/* helpers */

236/// Returns whether the line's bounding box intersects the grid.
237const fn line_bounds_intersect(
238    grid: RasterGrid,
239    start: Position2<i64>,
240    end: Position2<i64>,
241) -> bool {
242    is! { grid.is_empty(), return false }
243    let min_x = if start.dim[0] <= end.dim[0] { start.dim[0] } else { end.dim[0] };
244    let max_x = if start.dim[0] >= end.dim[0] { start.dim[0] } else { end.dim[0] };
245    let min_y = if start.dim[1] <= end.dim[1] { start.dim[1] } else { end.dim[1] };
246    let max_y = if start.dim[1] >= end.dim[1] { start.dim[1] } else { end.dim[1] };
247    is! { max_x < 0 || max_y < 0, return false }
248    is! { min_x >= grid.width() as i64, return false }
249    is! { min_y >= grid.height() as i64, return false }
250    true
251}
252/// Returns the canonical major-step interval that can overlap the grid.
253const fn grid_major_step_range(
254    grid: RasterGrid,
255    canonical_start: Position2<i64>,
256    canonical_end: Position2<i64>,
257    major_axis: u8,
258) -> Option<(u64, u64)> {
259    let major = major_axis as usize;
260    let extent = if major == 0 { grid.width() } else { grid.height() };
261    is! { extent == 0, return None }
262    let start = canonical_start.dim[major];
263    let end = canonical_end.dim[major];
264    let grid_end = extent as i64 - 1;
265    is! { end < 0 || start > grid_end, return None } // Canonical major coords always increase
266    let first = is! { start < 0, start.abs_diff(0), 0 };
267    let last = is! { end > grid_end, start.abs_diff(grid_end), start.abs_diff(end) };
268    Some((first, last))
269}
270/// Reconstructs the exact canonical line state after `step` major steps.
271const fn state_at(
272    canonical_start: Position2<i64>,
273    major_axis: u8,
274    major_delta: u64,
275    minor_delta: u64,
276    minor_step: i8,
277    step: u64,
278) -> (Position2<i64>, u64) {
279    is! { major_delta == 0, return (canonical_start, 0) }
280    lets! { major = major_axis as usize, minor = 1 - major }
281    // `u128` makes the full i64 lattice safe.
282    let total = (major_delta / 2) as u128 + step as u128 * minor_delta as u128;
283    let minor_steps = total / major_delta as u128;
284    let error = (total % major_delta as u128) as u64;
285    let mut dim = canonical_start.dim;
286    dim[major] = (dim[major] as i128 + step as i128) as i64;
287    dim[minor] = (dim[minor] as i128 + minor_step as i128 * minor_steps as i128) as i64;
288    (Position2::new(dim), error)
289}
