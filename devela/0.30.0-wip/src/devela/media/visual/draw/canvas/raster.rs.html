7#[doc = crate::_tags!(image)]
8/// A [`Canvas`] whose spatial units are canonical raster cells.
9#[doc = crate::_doc_meta!{
10    location("media/visual/draw", trait CanvasRaster),
11}]
12/// Implementing `CanvasRaster` declares that the canvas extent defines
13/// a [`RasterGrid`] with `u32` cell coordinates.
14///
15/// Raster cells use an upper-left origin, with positive `x` extending rightward
16/// and positive `y` extending downward.
17///
18/// This trait describes raster geometry only. It does not imply pixel storage,
19/// byte layout, color encoding, readback, compositing, or presentation.
20///
21/// Rasterization algorithms may use signed `i64` lattice positions before
22/// clipping and projecting them into the raster grid.
23pub trait CanvasRaster: Canvas<Unit = u32> {}
24
25#[doc = crate::_tags!(image)]
26/// Extension methods for a canonical raster [`CanvasRaster`].
27#[doc = crate::_doc_meta!{
28    location("media/visual/draw", trait CanvasRasterExt),
29}]
30/// This trait is implemented for every [`CanvasRaster`].
31///
32/// Its methods derive canonical software-raster operations from the minimal [`Canvas`] contract,
33/// without requiring or exposing any particular raster storage representation.
34pub trait CanvasRasterExt: CanvasRaster {
35    /// Returns the canonical raster grid covering this canvas.
36    fn canvas_raster_grid(&self) -> RasterGrid {
37        RasterGrid::new(self.canvas_extent())
38    }
39
40    /// Draws a canonical aliased one-cell-wide raster line.
41    ///
42    /// `start` and `end` are signed raster-lattice positions and may lie outside
43    /// the canvas. The line is clipped to the canvas grid using [`RasterLineIter`].
44    fn canvas_draw_line(
45        &mut self,
46        start: Position2<i64>,
47        end: Position2<i64>,
48        color: Self::Color,
49    ) -> Result<(), Self::Error>
50    where
51        Self::Color: Copy,
52    {
53        let mut iter = RasterLineIter::new(self.canvas_raster_grid(), start, end);
54        while let Some(element) = iter.next() {
55            self.canvas_set_color(element.coord(), color)?;
56        }
57        Ok(())
58    }
59
60    /// Draws the outline of a raster region.
61    fn canvas_draw_region(
62        &mut self,
63        region: RegionS2<u32>,
64        color: Self::Color,
65    ) -> Result<(), Self::Error>
66    where
67        Self::Color: Copy,
68    {
69        let (x, y) = (region.x() as i64, region.y() as i64);
70        let (w, h) = (region.w() as i64, region.h() as i64);
71        is! { w == 0 || h == 0, return Ok(()) }
72        let (right, bottom) = (x + w - 1, y + h - 1);
73        is! { h == 1, return self.canvas_draw_line(pos![x, y], pos![right, y], color) }
74        is! { w == 1, return self.canvas_draw_line(pos![x, y], pos![x, bottom], color) }
75        self.canvas_draw_line(pos![x, y], pos![right, y], color)?;
76        self.canvas_draw_line(pos![right, y], pos![right, bottom], color)?;
77        self.canvas_draw_line(pos![right, bottom], pos![x, bottom], color)?;
78        self.canvas_draw_line(pos![x, bottom], pos![x, y], color)
79    }
80}
