6#[doc = crate::_tags!(color rework)]
7/// RGB color with 3 channels.
8#[doc = crate::_doc_meta!{
9    location("media/visual/color", struct Rgb),
10}]
11#[repr(C)]
12#[must_use]
13#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
14pub struct Rgb<T, const LINEAR: bool = false> {
15    /// Color channels in order: [red, green, blue].
16    pub c: [T; 3],
17}
18
19#[doc = crate::_tags!(color rework)]
20/// RGB+A color with 4 channels.
21#[doc = crate::_doc_meta!{
22    location("media/visual/color", struct Rgba),
23}]
24#[repr(C)]
25#[must_use]
26#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
27pub struct Rgba<T, const LINEAR: bool = false, const PREMUL: bool = false> {
28    /// Color channels in order: [red, green, blue, alpha].
29    pub c: [T; 4],
30}
