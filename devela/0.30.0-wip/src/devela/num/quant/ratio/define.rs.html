30///
31/// The generic form does not define arithmetic semantics by itself.
32/// Primitive aliases such as [`RatioI32`] and [`RatioU32`] use nonzero
33/// denominators and provide checked constructors.
34///
35/// # Examples
36/// ```
37/// use devela::RatioU8;
38///
39/// let aspect_ratio = RatioU8::new(16, 9).unwrap();
40/// ```
41#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
42pub struct Ratio<N, D> {
43    /// The numerator.
44    n: N,
45    /// The denominator.
46    d: D,
47}
48
49#[rustfmt::skip]
50impl<N, D> Ratio<N, D> {
51    /// Constructs a ratio from its stored parts.
52    pub const fn from_parts(n: N, d: D) -> Self { Self { n, d } }
53    /// Returns a reference to the stored numerator.
54    pub const fn num_ref(&self) -> &N { &self.n }
55    /// Returns a reference to the stored denominator.
56    pub const fn den_ref(&self) -> &D { &self.d }
57    /// Decomposes the ratio into its stored parts.
58    pub fn into_parts(self) -> (N, D) { (self.n, self.d) }
59}

168            pub const fn is_proper(self) -> RatioResult<bool> {
169                if unlikely(self._has_min_term()) { return Err(Overflow(None)); }
170                let (n, d) = self._num_den_abs();
171                Ok(n < d)
172            }
173            #[compile(same($is_signed, true))]
174            /// Returns whether the ratio is reduced.
175            pub const fn is_reduced(self) -> RatioResult<bool> {
176                match self.gcd() { Ok(g) => Ok(g == 1), Err(e) => Err(e) }
177            }

179            /// Returns the relative ordering of two ratios.
180            pub const fn cmp_value(self, other: Self) -> RatioResult<Ordering> {
181                let Some(left) = (self.n as $Upcasted).checked_mul(other.d.get() as $Upcasted)
182                    else { return Err(Overflow(None)); };
183                let Some(right) = (other.n as $Upcasted).checked_mul(self.d.get() as $Upcasted)
184                    else { return Err(Overflow(None)); };
185                let ord = if left < right { Less } else if left > right { Greater } else { Equal };
186                // if denominators have opposite signs, cross-multiplication reverses order
187                if cif!(same($is_signed, true)) && (self.d.get() > 0) != (other.d.get() > 0) {
188                    return Ok(ord.reverse());
189                }
190                Ok(ord)
191            }

227            /// Returns a copy with a positive denominator.
228            ///
229            /// This moves the denominator sign to the numerator. It does not reduce the terms.
230            ///
231            /// Returns `Overflow` if changing the sign is not representable.
232            pub const fn with_pos_den(self) -> RatioResult<Self> {
233                if self.den() > 0 { return Ok(self); }
234                // den is negative here. Both signs must be flipped.
235                if unlikely(self.den() == <$N>::MIN || self.num() == <$N>::MIN) {
236                    cold_path(); return Err(Overflow(None)); }
237                let n = -self.num();
238                let d = -self.den();
239                let d = unwrap![some_guaranteed_or_ub <$D>::new(d)];
240                Ok(Self { n, d })
241            }

243            /// Returns a copy with a positive denominator, or `self` if the sign move overflows.
244            pub const fn with_pos_den_or_self(self) -> Self {

274            /// Composes two ratios by multiplication.
275            ///
276            /// This reduces cross terms to reduce the chance of overflowing.
277            pub const fn compose(self, other: Self) -> RatioResult<Self> {
278                if cif!(same($is_signed, true)) && (self._has_min_term() || other._has_min_term()) {
279                    cold_path(); return Err(Overflow(None));
280                }
281                let g1 = Self::_gcd_pair_unchecked(self.n, other.den());
282                let g2 = Self::_gcd_pair_unchecked(other.n, self.den());
283                let (a, b) = (self.n / g1, self.den() / g2);
284                let (c, d) = (other.n / g2, other.den() / g1);
285                let n = unwrap![ok? Self::_mul_prim(a, c)];
286                let d_raw = unwrap![ok? Self::_mul_prim(b, d)];
287                let Some(d) = <$D>::new(d_raw) else {
288                    cold_path();
289                    return Err(Overflow(None));
290                };
291                Ok(Self { n, d })
292            }
293            /// Composes two ratios by multiplying their stored terms directly.
294            ///
295            /// This preserves the stored term structure.
296            pub const fn compose_raw(self, other: Self) -> RatioResult<Self> {
297                let n = unwrap![ok? Self::_mul_prim(self.n, other.n)];
298                let d_raw = unwrap![ok? Self::_mul_prim(self.den(), other.den())];
299                let Some(d) = <$D>::new(d_raw) else { cold_path(); return Err(Overflow(None)); };
300                Ok(Self { n, d })
301            }

457            /// Writes the ratio into `buf` and returns it as `str`.
458            ///
459            /// Can return `MismatchedSizes` if `buf` is too small.
460            ///
461            /// # Features
462            /// Uses `unsafe_str`, when enabled, to avoid duplicated UTF-8 validation.
463            pub const fn write_str_with_sep(self, buf: &mut [u8], sep: char) -> RatioResult<&str> {
464                let len = unwrap![ok? self.write_with_sep(buf, sep)];
465                cfg_select! { all(feature = "unsafe_str", not(feature = "safe_num")) => {
466                    // SAFETY: only decimal ASCII digits, `-`,
467                    // and valid UTF-8 separator written by `Char::write_utf8_to`.
468                    unsafe { Ok(Str::from_utf8_unchecked(slice![mut buf, ..len])) }
469                } _ => {
470                    Ok(unwrap![ok_guaranteed_or_ub Str::from_utf8(slice![mut buf, ..len])])
471                }}
472            }

507            // panics for signed ratios if a stored term is the minimum value.
508            const fn _gcd_unchecked(self) -> $N {
509                let (mut a, mut b) = self._num_den_abs();
510                is![a == 0, return b];
511                is![b == 0, return a];
512                let k = (a | b).trailing_zeros();
513                a >>= a.trailing_zeros();
514                b >>= b.trailing_zeros();
515                while b != 0 {
516                    b >>= b.trailing_zeros();
517                    is![a > b, cswap![mut: a, b], b -= a];
518                }
519                a << k
520            }
521            const fn _gcd_pair_unchecked(a: $N, b: $N) -> $N {
522                let mut a = Self::_abs_prim_unchecked(a);
523                let mut b = Self::_abs_prim_unchecked(b);
524                is![a == 0, return b];
525                is![b == 0, return a];
526                let k = (a | b).trailing_zeros();
527                a >>= a.trailing_zeros();
528                b >>= b.trailing_zeros();
529                while b != 0 {
530                    b >>= b.trailing_zeros();
531                    is![a > b, cswap![mut: a, b], b -= a];
532                }
533                a << k
534            }

536            const fn _lcm(self) -> RatioResult<$N> {
537                let (a, b) = self._num_den_abs();
538                if a == 0 { return Ok(0); }
539                let g = self._gcd_unchecked();
540                let aup = (a / g) as $Upcasted;
541                let bup = b as $Upcasted;
542                let Some(res) = aup.checked_mul(bup) else {
543                    cold_path(); return Err(Overflow(None)); };
544                is![res <= <$N>::MAX as $Upcasted, Ok(res as $N), Err(Overflow(None))]
545            }
546
547            #[compile(same($is_signed, false))]
548            const fn _is_integer_checked(self) -> bool { self.n % self.d.get() == 0 }
549            #[compile(same($is_signed, true))]
550            const fn _is_integer_checked(self) -> bool {
551                // Avoid signed overflow in MIN % -1.
552                if self.n == <$N>::MIN && self.d.get() == -1 { cold_path(); return true; }
553                self.n % self.d.get() == 0
554            }

558            const fn _reduced_unchecked(self) -> Self {
559                let g = self._gcd_unchecked();
560                let n = self.n / g;
561                let d = self.d.get() / g;
562                let d = unwrap![some_guaranteed_or_ub <$D>::new(d)]; // original den is nonzero
563                Self { n, d }
564            }

566            #[compile(same($is_signed, false))]
567            const fn _abs_prim_unchecked(x: $N) -> $N { x }
568            #[compile(same($is_signed, true))]
569            const fn _abs_prim_unchecked(x: $N) -> $N { x.abs() }
570            #[compile(same($is_signed, true))]
571            const fn abs_up(x: $Upcasted) -> RatioResult<$Upcasted> {
572                if x == <$Upcasted>::MIN { cold_path(); Err(Overflow(None)) }
573                else if x < 0 { Ok(-x) } else { Ok(x) }
574            }

577            const fn up_to_prim(x: $Upcasted) -> RatioResult<$N> {
578                is![x <= <$N>::MAX as $Upcasted, Ok(x as $N), Err(Self::_err_overflow_pos())]
579            }
580            #[compile(same($is_signed, true))]
581            const fn up_to_prim(x: $Upcasted) -> RatioResult<$N> {
582                is![x < <$N>::MIN as $Upcasted || x > <$N>::MAX as $Upcasted,
583                    Err(Self::_err_overflow_pos()), Ok(x as $N)]
584            }

588            const fn _apply_step(self, value: $N, step: $N, mode: u8) -> RatioResult<$N> {
589                let step = unwrap![ok? Self::_checked_step(step)];
590                // (value * n / d) / step = value * n / (d * step)
591                let d_raw = unwrap![ok? Self::_mul_prim(self.den(), step)];
592                let Some(d) = <$D>::new(d_raw) else { cold_path(); return Err(Overflow(None)); };
593                let scaled = Self { n: self.n, d };
594                let units = match mode {
595                    0 => unwrap![ok? scaled.apply_floor(value)],
596                    1 => unwrap![ok? scaled.apply_ceil(value)],
597                    _ => unwrap![ok? scaled.apply_round(value)],
598                };
599                Self::_mul_prim(units, step)
600            }

602            #[compile(same($is_signed, false))]
603            const fn _checked_step(step: $N) -> RatioResult<$N> {
604                if step == 0 { cold_path(); Err(NonZeroRequired) } else { Ok(step) }
605            }
606            #[compile(same($is_signed, true))]
607            const fn _checked_step(step: $N) -> RatioResult<$N> {
608                if step == 0 { cold_path(); Err(NonZeroRequired) }
609                else if step < 0 { cold_path(); Err(NonNegativeRequired) }
610                else { Ok(step) }
611            }
612
613            const fn _mul_up(a: $N, b: $N) -> RatioResult<$Upcasted> {
614                unwrap![some_ok_or (a as $Upcasted).checked_mul(b as $Upcasted), Overflow(None)]
615            }
616            const fn _mul_prim(a: $N, b: $N) -> RatioResult<$N> {
617                let x = unwrap![ok? Self::_mul_up(a, b)];
618                Self::up_to_prim(x)
619            }

621            #[compile(same($is_signed, false))]
622            const fn _div_floor_up(a: $Upcasted, b: $Upcasted) -> RatioResult<$Upcasted> {
623                Ok(a / b)
624            }
625            #[compile(same($is_signed, true))]
626            const fn _div_floor_up(a: $Upcasted, b: $Upcasted) -> RatioResult<$Upcasted> {
627                if a == <$Upcasted>::MIN && b == -1 {
628                    cold_path(); return Err(Self::_err_overflow_pos()); }
629                let (q, r) = (a / b, a % b);
630                Ok(is![r != 0 && ((r > 0) != (b > 0)), q - 1, q])
631            }
632
633            #[compile(same($is_signed, false))]
634            const fn _div_ceil_up(a: $Upcasted, b: $Upcasted) -> RatioResult<$Upcasted> {
635                let (q, r) = (a / b, a % b);
636                is![r != 0, Ok(q + 1), Ok(q)]
637            }
638            #[compile(same($is_signed, true))]
639            const fn _div_ceil_up(a: $Upcasted, b: $Upcasted) -> RatioResult<$Upcasted> {
640                // Avoid signed overflow in MIN % -1.
641                if a == <$Upcasted>::MIN && b == -1 { cold_path(); return Err(Overflow(None)); }
642                let (q, r) = (a / b, a % b);
643                is![r != 0 && ((r > 0) == (b > 0)), Ok(q + 1), Ok(q)]
644            }
645
646            #[compile(same($is_signed, false))]
647            const fn _div_round_away_up(a: $Upcasted, b: $Upcasted) -> RatioResult<$Upcasted> {
648                let (q, r) = (a / b, a % b);
649                let half = b / 2;
650                let tie = b % 2 == 0 && r == half;
651                let above = r > half;
652                if above || tie { unwrap![some_ok_or q.checked_add(1), Overflow(None)] }
653                else { Ok(q) }
654            }
655            #[compile(same($is_signed, true))]
656            const fn _div_round_away_up(a: $Upcasted, b: $Upcasted) -> RatioResult<$Upcasted> {
657                if a == <$Upcasted>::MIN && b == -1 {
658                    cold_path(); return Err(Self::_err_overflow_pos()); }
659                let (q, r) = (a / b, a % b);
660                if r == 0 { return Ok(q); }
661                let Ok(ar) = Self::abs_up(r) else { return Err(Overflow(None)); };
662                let Ok(ab) = Self::abs_up(b) else { return Err(Overflow(None)); };
663                let half = ab / 2;
664                let tie = ab % 2 == 0 && ar == half;
665                let above = ar > half;
666                if above || tie {
667                    unwrap![some_ok_or
668                        if (a > 0) == (b > 0) { q.checked_add(1) } else { q.checked_sub(1) },
669                        Overflow(None)]
670                } else {
671                    Ok(q)
672                }
673            }

689            const fn _write_prim10(buf: &mut [u8], mut offset: usize, value: $N)

692                if value < 0 {
693                    offset = unwrap![ok? Self::_write_char(buf, offset, '-')];
694                    mag = value.unsigned_abs() as $Unsigned;
695                } else { mag = value as $Unsigned; }

698                else { Ok(offset + written) }
699            }
