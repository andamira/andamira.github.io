8#[doc = crate::_tags!(lin)]
9/// A fixed `D`-dimensional coordinate vector.
10#[doc = crate::_doc_meta!{
11    location("num/alg", struct Vector),
12}]
13/// Its components are expressed in a chosen basis. Vector addition and scalar
14/// multiplication are available when supported by the component type.
15///
16/// Geometric displacement, Euclidean length, normalization, and orientation
17/// are interpretations or additional structures, not representation invariants.
18#[must_use]
19#[repr(transparent)]
20pub struct Vector<T, const D: usize> {
21    /// The vector coordinates in some basis.
22    pub coords: [T; D],
23}
24
25#[doc = crate::_tags!(lin)]
26/// A static 2-dimensional vector.
27#[doc = crate::_doc_meta!{
28    location("num/alg", type Vector2),
29}]
30pub type Vector2<T> = Vector<T, 2>;
31
32#[doc = crate::_tags!(lin)]
33/// A static 3-dimensional vector.
34#[doc = crate::_doc_meta!{
35    location("num/alg", type Vector3),
36}]
37pub type Vector3<T> = Vector<T, 3>;

39mod impl_traits {
40    use crate::{ConstInit, Debug, FmtResult, Formatter, Hash, Hasher, Vector, init_array};
41
42    impl<T: Clone, const D: usize> Clone for Vector<T, D> {
43        fn clone(&self) -> Self {
44            Self { coords: self.coords.clone() }
45        }
46    }
47
48    impl<T: Copy, const D: usize> Copy for Vector<T, D> {}
49
50    impl<T: Default, const D: usize> Default for Vector<T, D> {
51        /// Returns a vector filled with each component's default value.
52        fn default() -> Self {
53            Self::new(init_array![default [T; D], "safe_num", "unsafe_array"])
54        }
55    }
56
57    impl<T: ConstInit, const D: usize> ConstInit for Vector<T, D> {
58        /// A vector filled with each component's initial value.
59        const INIT: Self = Self::new(init_array![INIT in ConstInit [T; D]]);
60    }
61
62    impl<T: Debug, const D: usize> Debug for Vector<T, D> {
63        fn fmt(&self, f: &mut Formatter<'_>) -> FmtResult<()> {
64            f.debug_struct("Vector").field("D", &D).field("coords", &self.coords).finish()
65        }
66    }
67
68    impl<T: Eq, const D: usize> Eq for Vector<T, D> {}
69    impl<T: PartialEq, const D: usize> PartialEq for Vector<T, D> {
70        fn eq(&self, other: &Self) -> bool {
71            self.coords == other.coords
72        }
73    }
74
75    impl<T: Hash, const D: usize> Hash for Vector<T, D> {
76        fn hash<HR: Hasher>(&self, state: &mut HR) {
77            self.coords.hash(state);
78        }
79    }
80}
