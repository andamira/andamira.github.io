9fn zip_array<A, B, O, F, const LEN: usize>(
10    lhs: [A; LEN],
11    rhs: [B; LEN],
12    mut operation: F,
13) -> [O; LEN]
14where
15    F: FnMut(A, B) -> O,
16{
17    let (mut lhs, mut rhs) = (lhs.into_iter(), rhs.into_iter());
18    array_from_fn(|_| {
19        let Some(lhs) = lhs.next() else {
20            unreachable!("array iterator has exactly LEN elements")
21        };
22        let Some(rhs) = rhs.next() else {
23            unreachable!("array iterator has exactly LEN elements")
24        };
25        operation(lhs, rhs)
26    })
27}

29/* entry-wise addition and subtraction */
30
31impl<T: Add<U>, U, const R: usize, const C: usize, const LEN: usize> Add<Matrix<U, R, C, LEN>>
32    for Matrix<T, R, C, LEN>
33{
34    type Output = Matrix<<T as Add<U>>::Output, R, C, LEN>;
35
36    fn add(self, rhs: Matrix<U, R, C, LEN>) -> Self::Output {
37        Matrix::new(zip_array(self.data, rhs.data, |lhs, rhs| lhs + rhs))
38    }
39}
40impl<T: Sub<U>, U, const R: usize, const C: usize, const LEN: usize> Sub<Matrix<U, R, C, LEN>>
41    for Matrix<T, R, C, LEN>
42{
43    type Output = Matrix<<T as Sub<U>>::Output, R, C, LEN>;
44
45    fn sub(self, rhs: Matrix<U, R, C, LEN>) -> Self::Output {
46        Matrix::new(zip_array(self.data, rhs.data, |lhs, rhs| lhs - rhs))
47    }
48}
49impl<T: Neg, const R: usize, const C: usize, const LEN: usize> Neg for Matrix<T, R, C, LEN> {
50    type Output = Matrix<<T as Neg>::Output, R, C, LEN>;
51
52    fn neg(self) -> Self::Output {
53        Matrix::new(self.data.map(|element| -element))
54    }
55}

57/* entry-wise assignment */
58
59impl<T: AddAssign<U>, U, const R: usize, const C: usize, const LEN: usize>
60    AddAssign<Matrix<U, R, C, LEN>> for Matrix<T, R, C, LEN>
61{
62    fn add_assign(&mut self, rhs: Matrix<U, R, C, LEN>) {
63        for (lhs, rhs) in self.data.iter_mut().zip(rhs.data) {
64            *lhs += rhs;
65        }
66    }
67}
68impl<T: SubAssign<U>, U, const R: usize, const C: usize, const LEN: usize>
69    SubAssign<Matrix<U, R, C, LEN>> for Matrix<T, R, C, LEN>
70{
71    fn sub_assign(&mut self, rhs: Matrix<U, R, C, LEN>) {
72        for (lhs, rhs) in self.data.iter_mut().zip(rhs.data) {
73            *lhs -= rhs;
74        }
75    }
76}

78/* primitive scalar and vector operators */

80macro_rules! impl_matrix_primitive_ops {
81    ($($t:ty),+ $(,)?) => {
82        $(
83            impl<const R: usize, const C: usize, const LEN: usize> Mul<$t>
84                for Matrix<$t, R, C, LEN>
85            {
86                type Output = Self;
87                fn mul(self, scalar: $t) -> Self::Output { self.mul_scalar(scalar) }
88            }
89
90            impl<const R: usize, const C: usize, const LEN: usize>
91                Mul<Matrix<$t, R, C, LEN>> for $t
92            {
93                type Output = Matrix<$t, R, C, LEN>;
94                fn mul(self, matrix: Matrix<$t, R, C, LEN>) -> Self::Output {
95                    matrix.mul_scalar(self)
96                }
97            }
98
99            impl<const R: usize, const C: usize, const LEN: usize> Div<$t>
100                for Matrix<$t, R, C, LEN>
101            {
102                type Output = Self;
103                fn div(self, scalar: $t) -> Self::Output { self.div_scalar(scalar) }
104            }
105
106            impl<const R: usize, const C: usize, const LEN: usize> MulAssign<$t>
107                for Matrix<$t, R, C, LEN>
108            {
109                fn mul_assign(&mut self, scalar: $t) {
110                    for element in &mut self.data { *element *= scalar; }
111                }
112            }
113
114            impl<const R: usize, const C: usize, const LEN: usize> DivAssign<$t>
115                for Matrix<$t, R, C, LEN>
116            {
117                fn div_assign(&mut self, scalar: $t) {
118                    for element in &mut self.data { *element /= scalar; }
119                }
120            }
121
122            impl<const R: usize, const C: usize, const LEN: usize> Mul<Vector<$t, C>>
123                for Matrix<$t, R, C, LEN>
124            {
125                type Output = Vector<$t, R>;
126                fn mul(self, vector: Vector<$t, C>) -> Self::Output { self.mul_vector(&vector) }
127            }
128
129            impl<const R: usize, const C: usize, const LEN: usize> Mul<&Vector<$t, C>>
130                for &Matrix<$t, R, C, LEN>
131            {
132                type Output = Vector<$t, R>;
133                fn mul(self, vector: &Vector<$t, C>) -> Self::Output { self.mul_vector(vector) }
134            }
135
136            /* matrix product */
137
138            impl<const N: usize, const LEN: usize> Mul for Matrix<$t, N, N, LEN> {
139                type Output = Self;
140                fn mul(self, other: Self) -> Self::Output { self.mul_square(&other) }
141            }
142            impl<const N: usize, const LEN: usize> Mul<&Matrix<$t, N, N, LEN>>
143                for &Matrix<$t, N, N, LEN>
144            {
145                type Output = Matrix<$t, N, N, LEN>;
146                fn mul(self, other: &Matrix<$t, N, N, LEN>) -> Self::Output {
147                    self.mul_square(other)
148                }
149            }
150            impl<const N: usize, const LEN: usize> MulAssign for Matrix<$t, N, N, LEN> {
151                fn mul_assign(&mut self, other: Self) { *self = self.mul_square(&other); }
152            }
153        )+
154    };
155}
