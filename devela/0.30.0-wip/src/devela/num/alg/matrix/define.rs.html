3//! Defines [`Matrix`].
4//
5
6#[doc = crate::_tags!(lin)]
7/// A static `R × C` matrix backed by a contiguous row-major array.
8#[doc = crate::_doc_meta! {
9    location("num/alg", struct Matrix)
10}]
11/// `R` is the number of rows and `C` is the number of columns.
12///
13/// Elements are stored in row-major order: all columns of the first row,
14/// followed by all columns of the second row, and so on.
15///
16/// `LEN` is the number of stored elements and must equal `R * C`. It is an
17/// explicit parameter because stable Rust cannot yet generally use `R * C`
18/// as an array length involving outer const parameters.
19///
20/// The shape relationship is validated by its constructors.
21///
22/// # Coordinates
23///
24/// Element coordinates use `(row, column)` order.
25/// ```
26/// use devela::Matrix;
27///
28/// let matrix = Matrix::<_, 2, 3, 6>::new([
29///     1, 2, 3,
30///     4, 5, 6,
31/// ]);
32/// assert_eq!(matrix[(0, 2)], 3);
33/// assert_eq!(matrix[(1, 0)], 4);
34/// ```
35#[must_use]
36#[non_exhaustive]
37#[repr(transparent)]
38pub struct Matrix<T, const R: usize, const C: usize, const LEN: usize> {
39    /// The matrix elements in contiguous row-major order.
40    pub data: [T; LEN],
41}
42

45mod impl_traits {
46    use crate::{ConstInit, Debug, FmtResult, Formatter, Hash, Hasher, Matrix, init_array};

48    impl<T: Clone, const R: usize, const C: usize, const LEN: usize> Clone for Matrix<T, R, C, LEN> {
49        fn clone(&self) -> Self {
50            Self::new(self.data.clone())
51        }
52    }
53    impl<T: Copy, const R: usize, const C: usize, const LEN: usize> Copy for Matrix<T, R, C, LEN> {}
54    impl<T: Default, const R: usize, const C: usize, const LEN: usize> Default
55        for Matrix<T, R, C, LEN>
56    {
57        /// Returns a matrix filled with each element's default value.
58        fn default() -> Self {
59            Self::new(init_array![default [T; LEN], "safe_num", "unsafe_array"])
60        }
61    }
62    impl<T: ConstInit, const R: usize, const C: usize, const LEN: usize> ConstInit
63        for Matrix<T, R, C, LEN>
64    {
65        /// A matrix filled with each element's initial value.
66        const INIT: Self = Self::new(init_array![INIT in ConstInit [T; LEN]]);
67    }
68    impl<T: Debug, const R: usize, const C: usize, const LEN: usize> Debug for Matrix<T, R, C, LEN> {
69        fn fmt(&self, f: &mut Formatter<'_>) -> FmtResult<()> {
70            f.debug_struct("Matrix")
71                .field("rows", &R)
72                .field("columns", &C)
73                .field("data", &self.data) // IMPROVE: omit the middle for large LEN
74                .finish()
75        }
76    }
77    impl<T: PartialEq, const R: usize, const C: usize, const LEN: usize> PartialEq
78        for Matrix<T, R, C, LEN>
79    {
80        fn eq(&self, other: &Self) -> bool {
81            self.data == other.data
82        }
83    }
84    impl<T: Eq, const R: usize, const C: usize, const LEN: usize> Eq for Matrix<T, R, C, LEN> {}
85    impl<T: Hash, const R: usize, const C: usize, const LEN: usize> Hash for Matrix<T, R, C, LEN> {
86        fn hash<H: Hasher>(&self, state: &mut H) {
87            self.data.hash(state);
88        }
89    }
90    impl<T, const R: usize, const C: usize, const LEN: usize> AsRef<[T]> for Matrix<T, R, C, LEN> {
91        fn as_ref(&self) -> &[T] {
92            &self.data
93        }
94    }
95    impl<T, const R: usize, const C: usize, const LEN: usize> AsMut<[T]> for Matrix<T, R, C, LEN> {
96        fn as_mut(&mut self) -> &mut [T] {
97            &mut self.data
98        }
99    }
