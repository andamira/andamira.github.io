6#[doc = crate::_tags!(construction num)]
7/// Defines a bounded integer wrapper with embedded boundary metadata.
8#[doc = crate::_doc_meta!{
9    location("num/grain/lim", macro bound_int),
10}]
11/// The generated type stores a primitive carrier through [`MaybeNiche`][crate::MaybeNiche].
12///
13/// The `value_bits(...)` argument chooses how many low bits encode the payload.
14/// The remaining high bits encode boundary metadata:
15/// - one direction bit, reporting the last lower/upper boundary event,
16/// - the remaining metadata bits as a saturating event counter.
17///
18/// For signed carriers, the raw carrier minimum value is reserved as invalid.
19/// Non-canonical raw values with `count == 0` and a set direction bit are
20/// accepted by `from_raw` but canonicalized by clearing the direction bit.
21///
22/// `range(...)` chooses the decoded payload interval:
23///
24/// - `full`: uses every payload value representable by `value_bits`.
25/// - `symmetric`: for signed carriers, excludes the negative extra endpoint
26///   so the range is symmetric around zero.
27///
28/// The default is `full`.
29///
30/// `symmetric` is only valid for signed carriers.
31///
32/// # Invariants
33///
34/// - `value_bits` must be at least `1` and must leave at least two metadata bits:
35///   one count bit and one direction bit.
36/// - Methods ending in `_meta` preserve or merge existing boundary metadata.
37///   Unsuffixed methods are value-first and only record boundary events caused
38///   by the operation itself.
39///
40/// # Operation groups
41///
42/// - `sat`: saturating arithmetic.
43/// - `che`: checked arithmetic.
44/// - `mod`: explicit-modulus arithmetic.
45/// - `up`: exact arithmetic returning the upcasted primitive.
46/// - `all`: all supported operation groups.
47/// - `default`: `sat` and `che`.
48///
49/// # Syntax
50/// ```ignore
51/// bound_int! {
52///     pub struct Name: repr(Representation => carrier);
53///
54///     value_bits(N);
55///     range(full); // optional: full | symmetric
56///     ops(sat, che, mod, up);
57/// }
58/// ```
59///
60/// # Examples
61///
62/// See [`BoundI8Example`][crate::BoundI8Example].

272            /// Number of bits in the primitive carrier.
273            $vis const CARRIER_BITS: u32 = <$Carrier>::BITS;
274            /// Number of bits used by the signed payload.
275            $vis const VALUE_BITS: u32 = ($VALUE_BITS) as u32;
276            /// Number of high bits used by boundary metadata.
277            $vis const META_BITS: u32 = Self::CARRIER_BITS - Self::VALUE_BITS;
278            /// Number of metadata bits used by the boundary direction marker.
279            $vis const DIR_BITS: u32 = 1;
280            /// Number of metadata bits used by the saturating boundary-event counter.
281            $vis const COUNT_BITS: u32 = Self::META_BITS - 1;
282            /// Maximum representable boundary-event count.
283            $vis const MAX_COUNT: $Unsigned = (1 as $Unsigned << Self::COUNT_BITS) - 1;
284
285            /// Whether the payload carrier is signed.
286            $vis const IS_SIGNED: bool = $crate::cif!(same($signed, true));
287            /// Whether the signed payload range is symmetric around zero.

289
290            /// Minimum representable payload value.

292            /// Maximum representable payload value.
293            $vis const MAX_VALUE: $Carrier = Self::_MAX_VALUE;

302            /// Maximum payload value, with empty metadata.

307            /// Creates a value, saturating to the payload range.
308            ///
309            /// If `value` is below [`MIN_VALUE`](#associatedconstant.MIN_VALUE),
310            /// the result is clamped to `MIN` and records one lower-boundary event.
311            ///
312            /// If `value` is above [`MAX_VALUE`](#associatedconstant.MAX_VALUE),
313            /// the result is clamped to `MAX` and records one upper-boundary event.
314            $vis const fn new(value: $Carrier) -> Self {
315                if value < Self::MIN_VALUE { Self::from_value_meta(Self::MIN_VALUE, 1, false) }
316                else if value > Self::MAX_VALUE { Self::from_value_meta(Self::MAX_VALUE, 1, true) }
317                else { Self::from_value_meta(value, 0, false) }
318            }
319            /// Creates a value if it fits in the payload range.
320            $vis const fn new_checked(value: $Carrier) -> Option<Self> {
321                if value < Self::MIN_VALUE || value > Self::MAX_VALUE { None }
322                else { Some(Self::from_value_meta(value, 0, false)) }
323            }
324
325            /// Creates a value saturated to the payload range.
326            $vis const fn new_saturated(value: $Carrier) -> Self {
327                let value = $crate::cmp!(clamp value, Self::MIN_VALUE, Self::MAX_VALUE);
328                Self::from_value_meta(value, 0, false)
329            }
330            /// Creates a value saturated to the payload range, from an upcasted carrier.
331            $vis const fn new_saturated_up(value: $Up) -> Self {
332                Self::new_saturated($crate::cast!(saturating value => $Carrier))
333            }
334
335            /// Creates a value from a raw encoded carrier.
336            ///
337            /// Returns `None` if `raw` is reserved, invalid for the underlying representation,
338            /// or decodes outside the payload range.
339            ///
340            /// Non-canonical raw encodings with `count == 0` are canonicalized so
341            /// the direction bit is cleared.
342            $vis const fn from_raw(raw: $Carrier) -> Option<Self> {
343                if raw == Self::_RESERVED_RAW { return None; }
344                let _ = $crate::unwrap![ok_some? $crate::MaybeNiche::<$Repr>::try_from_prim(raw)];
345                let (value, count) = (Self::decode_value(raw), Self::decode_count(raw));
346                // Only `symmetric` excludes a value that the signed payload decoder can produce.
347                // `full` accepts the complete decoded payload domain.
348                // Custom ranges must add their own validation arm.
349                if $crate::cif!(same($Range, symmetric)) && !Self::decoded_value_fits(value) {
350                    return None }
351                let dir_upper = count != 0 && Self::decode_dir_upper(raw);
352                let raw = Self::encode_raw(value, count, dir_upper);
353                let res = $crate::unwrap![ok_some? $crate::MaybeNiche::<$Repr>::try_from_prim(raw)];
354                Some(Self(res))
355            }

357            /* raw/meta methods */
358
359            /// Returns the raw encoded carrier.
360            $vis const fn raw(self) -> $Carrier { self.0.prim() }
361            /// Returns the decoded payload value.
362            $vis const fn get(self) -> $Carrier { Self::decode_value(self.raw()) }
363
364            /// Returns whether this value carries at least one boundary event.
365            $vis const fn is_bounded(self) -> bool { self.bound_count() != 0 }
366
367            /// Returns the saturating boundary-event count.
368            ///
369            /// A value of `0` means no boundary event is recorded. When this is `0`,
370            /// [`bound_dir`](#method.bound_dir) returns `None`.
371            $vis const fn bound_count(self) -> $Unsigned { Self::decode_count(self.raw()) }
372
373            /// Returns the last recorded boundary direction.
374            ///
375            /// Returns `None` when [`bound_count`](#method.bound_count) is `0`.
376            $vis const fn bound_dir(self) -> Option<$crate::Boundary1d> {
377                if self.bound_count() == 0 { None }
378                else if Self::decode_dir_upper(self.raw()) { Some($crate::Boundary1d::Upper) }
379                else { Some($crate::Boundary1d::Lower) }
380            }
381            /// Returns the same payload value with empty metadata.
382            $vis const fn clear_meta(self) -> Self {
383                Self::from_value_meta(self.get(), 0, false)
384            }
385            /// Returns the same payload value with explicit boundary metadata.
386            ///
387            /// `count` saturates to [`MAX_COUNT`](#associatedconstant.MAX_COUNT).
388            ///
389            /// When `count == 0`, `dir` is ignored
390            /// and the raw direction bit is canonicalized to zero.
391            $vis const fn with_bound_meta(self, count: $Unsigned, dir: Option<$crate::Boundary1d>)
392                -> Self {
393                let dir_upper = match dir { Some($crate::Boundary1d::Upper) => true, _ => false };
394                Self::from_value_meta(self.get(), count, dir_upper)
395            }

397            /* sign */
398
399            // TODO: allow lints when supporting unsigned
400            /// Whether the value is less than zero.
401            $vis const fn is_negative(self) -> bool { self.get() < 0 }
402            /// Whether the value is greater than zero.
403            $vis const fn is_positive(self) -> bool { self.get() > 0 }
404            /// Whether the value is zero.
405            $vis const fn is_zero(self) -> bool { self.get() == 0 }
406            /// Whether the value is greater than or equal to zero.
407            $vis const fn is_nonnegative(self) -> bool { self.get() >= 0 }
408            /// Whether the value is less than or equal to zero.
409            $vis const fn is_nonpositive(self) -> bool { self.get() <= 0 }
410
411            /* ordering */
412
413            /// Equality comparison over decoded payload values.
414            $vis const fn eq(self, other: Self) -> bool { self.get() == other.get() }
415
416            /// Compares decoded payload values.
417            $vis const fn cmp(self, other: Self) -> $crate::Ordering {
418                if self.get() < other.get() { $crate::Ordering::Less }
419                else if self.get() > other.get() { $crate::Ordering::Greater }
420                else { $crate::Ordering::Equal }
421            }

423            /// Returns the greater decoded value.
424            ///
425            /// Metadata is cleared because the result is recomputed.
426            $vis const fn max(self, other: Self) -> Self {
427                Self::from_value_nometa(Self::max_carrier(self.get(), other.get()))
428            }
429            /// Returns the greater value, preserving the selected operand metadata.
430            $vis const fn max_meta(self, other: Self) -> Self {
431                if self.get() >= other.get() { self } else { other }
432            }
433
434            /// Returns the decoded value floored at zero.
435            ///
436            /// Metadata is cleared because the result is recomputed.
437            $vis const fn max_zero(self) -> Self {
438                Self::from_value_nometa(Self::max_carrier(self.get(), 0))
439            }
440            /// Returns the decoded value floored at zero, preserving selected metadata.
441            $vis const fn max_zero_meta(self) -> Self { self.max_meta(Self::ZERO) }
442

444            ///
445            /// Metadata is cleared because the result is recomputed.

447                Self::from_value_nometa(Self::min_carrier(self.get(), other.get()))
448            }

452            }

469                Self::from_value_nometa(Self::clamp_carrier(self.get(), min.get(), max.get()))
470            }

475                if self.get() < min.get() { min }
476                else if self.get() > max.get() { max } else { self }
477            }

480        // Common private helpers
481        #[allow(dead_code)]
482        impl $Name {
483            /* constants */
484
485            const _CHECK_INVARIANTS: () = {
486                assert!(Self::VALUE_BITS >= 1, "bound_int!: value_bits must be at least 1");
487                assert!(Self::VALUE_BITS <= Self::CARRIER_BITS - 2,
488                    "bound_int!: value_bits must leave at least 2 metadata bits");
489                assert!(Self::META_BITS >= 2,
490                    "bound_int!: metadata requires at least 1 count bit and 1 direction bit");
491                assert!(Self::COUNT_BITS >= 1,
492                    "bound_int!: boundary count requires at least 1 bit");
493            };
494
495            const VALUE_MASK: $Unsigned = (1 as $Unsigned << Self::VALUE_BITS) - 1;
496            const COUNT_MASK: $Unsigned = (1 as $Unsigned << Self::COUNT_BITS) - 1;
497            const DIR_BIT: $Unsigned = 1 as $Unsigned << Self::COUNT_BITS;
498
499            /* constructors */
500
501            /// Creates a value from a canonical raw carrier.
502            ///
503            /// The caller must only pass a raw pattern emitted by [`encode_raw`].
504            const fn from_canonical_raw(raw: $Carrier) -> Self {
505                match $crate::MaybeNiche::<$Repr>::try_from_prim(raw) {
506                    Ok(repr) => Self(repr),
507                    Err(_) => unreachable!(),
508                }
509            }
510            /// Creates a metadata-empty value from an already in-range payload.
511            const fn from_value_nometa(value: $Carrier) -> Self {
512                Self::from_value_meta(value, 0, false)
513            }
514            const fn from_value_meta(value: $Carrier, count: $Unsigned, dir_upper: bool) -> Self {
515                let raw = Self::encode_raw(value, count, dir_upper);
516                Self::from_canonical_raw(raw)
517            }
518
519            /* meta */
520
521            const fn encode_raw(value: $Carrier, count: $Unsigned, dir_upper: bool) -> $Carrier {
522                let count = if count > Self::MAX_COUNT { Self::MAX_COUNT } else { count };
523                // Canonical rule: direction is absent when count == 0.
524                let dir = if count == 0 { 0 } else if dir_upper { Self::DIR_BIT } else { 0 };
525                let meta = ((count as $Unsigned) & Self::COUNT_MASK) | dir;
526                let bits = (meta << Self::VALUE_BITS) | ((value as $Unsigned) & Self::VALUE_MASK);
527                bits as $Carrier
528            }
529            const fn decode_count(raw: $Carrier) -> $Unsigned {
530                (((raw as $Unsigned) >> Self::VALUE_BITS) & Self::COUNT_MASK)
531            }
532            const fn decode_dir_upper(raw: $Carrier) -> bool {
533                ((((raw as $Unsigned) >> Self::VALUE_BITS) & Self::DIR_BIT) != 0)
534            }
535            // less branchy: `value < Self::MIN_VALUE || value > Self::MAX_VALUE`
536            const fn decoded_value_fits(value: $Carrier) -> bool {
537                ((value - Self::MIN_VALUE) as $Unsigned)
538                    <= ((Self::MAX_VALUE - Self::MIN_VALUE) as $Unsigned)
539            }
540
541            /* arithmetic ops */
542
543            // Safe from carrier overflow: COUNT_BITS <= CARRIER_BITS - 2, so
544            // MAX_COUNT + MAX_COUNT + MAX_COUNT fits in the unsigned carrier.
545            const fn count_inc(a: $Unsigned, b: $Unsigned, extra: $Unsigned) -> $Unsigned {
546                let sum = (a + b + extra);
547                if sum > Self::MAX_COUNT { Self::MAX_COUNT } else { sum }
548            }
549            const fn merged_dir_upper(a: Self, b: Self) -> bool {
550                if b.bound_count() != 0 { Self::decode_dir_upper(b.raw()) }
551                else if a.bound_count() != 0 { Self::decode_dir_upper(a.raw()) }
552                else { false }
553            }
554
555            /* ordering / clamping */
556
557            /// Returns the greater carrier value.
558            ///
559            /// Safe for the ranges produced by this bounded type.
560            const fn max_carrier(x: $Carrier, y: $Carrier) -> $Carrier {
561                (x + y + (x - y).abs()) / 2
562            }
563            /// Returns the lesser carrier value.
564            ///
565            /// Safe for the ranges produced by this bounded type.
566            const fn min_carrier(x: $Carrier, y: $Carrier) -> $Carrier {
567                (x + y - (x - y).abs()) / 2
568            }
569            /// Clamps `x` between `min` and `max`.
570            ///
571            /// If `min > max`, this returns `min`.
572            const fn clamp_carrier(x: $Carrier, min: $Carrier, max: $Carrier) -> $Carrier {
573                Self::max_carrier(min, Self::min_carrier(x, max))
574            }
575            /// Clamps a carrier value to the payload range.
576            ///
577            /// The input must be in the arithmetic range produced
578            /// by valid payload addition, subtraction, or distance.
579            const fn clamp_carrier_to_value(x: $Carrier) -> $Carrier {
580                Self::clamp_carrier(x, Self::MIN_VALUE, Self::MAX_VALUE)
581            }
582        }
583
584        /* common traits */
585
586        impl Default for $Name { fn default() -> Self { Self::ZERO } }
587        $crate::_impl_init![Self::ZERO => $Name];
588        $crate::impl_trait![fmt::Display for $Name |self, f| self.get().fmt(f)];
589
590        impl Eq for $Name {}
591        impl PartialEq for $Name {
592            fn eq(&self, other: &Self) -> bool { self.get().eq(&other.get()) }
593        }
594        // Enables symmetric comparisons with the primitive carrier.
595        impl PartialEq<$Carrier> for $Name {
596            fn eq(&self, other: &$Carrier) -> bool { self.get() == *other }
597        }
598        // NOTE: this can expose ambiguous inference in tests using untyped literals.
599        impl PartialEq<$Name> for $Carrier {
600            fn eq(&self, other: &$Name) -> bool { *self == other.get() }
601        }
602        impl PartialOrd for $Name {
603            fn partial_cmp(&self, other: &Self) -> Option<$crate::Ordering> {
604                Some(self.cmp(other))
605            }
606        }
607        impl Ord for $Name {
608            fn cmp(&self, other: &Self) -> $crate::Ordering { self.get().cmp(&other.get()) }
609        }
610        impl $crate::Hash for $Name {
611            fn hash<H: $crate::Hasher>(&self, state: &mut H) { self.get().hash(state); }
612        }
613    };
614
615    // emit: user impl blocks
616    (%emit_user_impls $Name:ident;) => {};
617    (%emit_user_impls
618        $Name:ident;
619        $(#[$impl_attr:meta])*
620        impl { $($user_impl:item)* }
621        $($rest:tt)*
622    ) => {
623        $(#[$impl_attr])*
624        impl $Name { $($user_impl)* }
625        $crate::bound_int!(%emit_user_impls $Name; $($rest)*);
626    };
627    (%emit_user_impls $Name:ident; $($bad:tt)+) => {
628        compile_error!("bound_int!: expected optional `impl { ... }` block after `ops(...)`");
629    };
630}
