9/// Construction from explicit seed material or another random source.
10#[doc = crate::_doc_meta!{
11    location("num/prob/rand", trait RandSeedable),
12}]
13/// `RandSeedable` defines a portable byte-backed seed representation and
14/// provides construction from explicit seed material or another random source.
15///
16/// Multi-byte values encoded in seeds use little-endian order, preserving
17/// reproducibility across target endianness.
18pub trait RandSeedable: Sized {
19    /// Byte-backed seed material accepted by this type.
20    type RandSeed: Clone + Default + AsRef<[u8]> + AsMut<[u8]>;
21    // type RandSeed: Default + AsMut<[u8]>;
22
23    /// Constructs this type from explicit seed material.
24    fn rand_from_seed(seed: Self::RandSeed) -> Self;
25
26    /// Attempts to construct this type by drawing seed material from `source`.
27    fn rand_try_from_source<R>(source: &mut R) -> Result<Self, R::Error>
28    where
29        R: RandTry + ?Sized,
30    {
31        let mut seed = Self::RandSeed::default();
32        source.rand_try_fill_bytes(seed.as_mut())?;
33        Ok(Self::rand_from_seed(seed))
34    }
35
36    /// Constructs this type by drawing seed material from an infallible source.
37    #[must_use]
38    fn rand_from_source<R>(source: &mut R) -> Self
39    where
40        R: Rand + ?Sized,
41    {
42        match Self::rand_try_from_source(source) {
43            Ok(value) => value,
44            Err(error) => match error {},
45        }
46    }
47}

50/// Fallible source of raw random data.
51#[doc = crate::_doc_meta!{
52    location("num/prob/rand", trait RandTry),
53}]
54/// `RandTry` is the core trait for random sources that may fail, such as
55/// OS calls, hardware RNGs, host APIs, finite buffers, and fallible adapters.
56///
57/// Source behavior and suitability are described by [`RandQualities`].
58///
59/// Infallible sources use [`Infallible`][crate::Infallible] as their error type.
60/// and automatically implement [`Rand`].
61pub trait RandTry {
62    /* required */
63
64    /// Error returned when complete random data could not be produced.
65    type Error;
66
67    /// Number of bits produced per output step.
68    ///
69    /// Adapters or composite sources may report a conservative value.
70    const RAND_OUTPUT_BITS: u32;
71
72    /// Size of the source state owned by this value, in bits.
73    ///
74    /// External entropy sources that do not store generator state
75    /// in the value itself should use `0`.
76    const RAND_STATE_BITS: u32;
77
78    /// Semantic qualities of this random source.
79    const RAND_QUALITIES: RandQualities;
80
81    /// Attempts to produce the next 64 bits of randomness.
82    fn rand_try_next_u64(&mut self) -> Result<u64, Self::Error>;
83
84    /*** provided ***/
85
86    /// Returns the semantic qualities of this random source.
87    #[must_use]
88    fn rand_qualities(&self) -> RandQualities {
89        Self::RAND_QUALITIES
90    }
91
92    /* derived primitives and byte access */
93
94    /// Attempts to produce the next random `u128`.
95    fn rand_try_next_u128(&mut self) -> Result<u128, Self::Error> {
96        let pair = [self.rand_try_next_u64()?, self.rand_try_next_u64()?];
97        Ok(Cast::<u128>::from_u64_le(pair))
98    }
99    /// Attempts to produce the next random `u32`.
100    fn rand_try_next_u32(&mut self) -> Result<u32, Self::Error> {
101        Ok(self.rand_try_next_u64()? as u32)
102    }
103    /// Attempts to produce the next random `u16`.
104    fn rand_try_next_u16(&mut self) -> Result<u16, Self::Error> {
105        Ok(self.rand_try_next_u64()? as u16)
106    }
107    /// Attempts to produce the next random `u8`.
108    fn rand_try_next_u8(&mut self) -> Result<u8, Self::Error> {
109        Ok(self.rand_try_next_u16()? as u8)
110    }
111    /// Attempts to produce a random boolean.
112    fn rand_try_next_bool(&mut self) -> Result<bool, Self::Error> {
113        Ok((self.rand_try_next_u64()? & 1) != 0)
114    }
115
116    /// Attempts to fill the buffer with generated bytes.
117    ///
118    /// This method must either fill the whole buffer or return an error.
119    fn rand_try_fill_bytes(&mut self, buffer: &mut [u8]) -> Result<(), Self::Error> {
120        let mut i = 0;
121        let len = buffer.len();
122        while i + 8 <= len {
123            buffer[i..i + 8].copy_from_slice(&self.rand_try_next_u64()?.to_ne_bytes());
124            i += 8;
125        }
126        if i < len {
127            let tail = self.rand_try_next_u64()?.to_ne_bytes();
128            buffer[i..].copy_from_slice(&tail[..len - i]);
129        }
130        Ok(())
131    }
132
133    /* bounded sampling */
134
135    /// Attempts to return a uniformly random integer in `0..upper`.
136    ///
137    /// Uses rejection sampling to avoid modulo bias.
138    ///
139    /// # Panics
140    /// Panics if `upper == 0`.
141    fn rand_try_below(&mut self, upper: u64) -> Result<u64, Self::Error> {
142        assert!(upper > 0);
143        let threshold = upper.wrapping_neg() % upper;
144        loop {
145            let v = self.rand_try_next_u64()?;
146            is! { v >= threshold, return Ok(v % upper) }
147        }
148    }
149    /// Attempts to return a uniformly random integer in `low..high`.
150    ///
151    /// # Panics
152    /// Panics if `low >= high`.
153    #[track_caller]
154    fn rand_try_range(&mut self, low: u64, high: u64) -> Result<u64, Self::Error> {
155        assert!(low < high);
156        Ok(low + self.rand_try_below(high - low)?)
157    }
158
159    /* common discrete helpers */
160
161    /// Attempts to roll a fair die with a number of `sides`.
162    ///
163    /// Returns a value in `1..=sides`.
164    fn rand_try_roll(&mut self, sides: u64) -> Result<u64, Self::Error> {
165        Ok(self.rand_try_below(sides)? + 1)
166    }
167    /// Attempts to shuffle the elements of the slice in place.
168    ///
169    /// Uses the Fisher–Yates algorithm.
170    fn rand_try_shuffle<T>(&mut self, slice: &mut [T]) -> Result<(), Self::Error> {
171        /*
172        let mut i = slice.len();
173        while i > 1 {
174            i -= 1;
175            let j = self.rand_try_below((i + 1) as u64)? as usize;
176            slice.swap(i, j);
177        }
178        */
179
180        whilst! { i in rev 1..slice.len(); {
181            let j = self.rand_try_below((i + 1) as u64)? as usize;
182            slice.swap(i, j);
183        }}
184        Ok(())
185    }
186
187    /* selection helpers */
188
189    /// Attempts to choose one valid item using reservoir sampling.
190    ///
191    /// This is uniform over all valid items and uses one pass.
192    fn rand_try_choose_reservoir<I, F>(
193        &mut self,
194        iter: I,
195        mut valid: F,
196    ) -> Result<Option<I::Item>, Self::Error>
197    where
198        I: IntoIterator,
199        F: FnMut(&I::Item) -> bool,
200    {
201        let (mut seen, mut picked) = (0u64, None);
202        for item in iter {
203            if valid(&item) {
204                seen = seen.checked_add(1).expect("too many valid random candidates");
205                is! { self.rand_try_below(seen)? == 0, picked = Some(item) }
206            }
207        }
208        Ok(picked)
209    }
210    /// Attempts to choose one valid item using random scores.
211    ///
212    /// Each valid item receives a random score. The highest score wins.
213    /// Ties are resolved uniformly.
214    fn rand_try_choose_scored<I, F>(
215        &mut self,
216        iter: I,
217        mut valid: F,
218    ) -> Result<Option<I::Item>, Self::Error>
219    where
220        I: IntoIterator,
221        F: FnMut(&I::Item) -> bool,
222    {
223        let (mut picked, mut best_score, mut ties) = (None, 0u64, 0u64);
224        for item in iter {
225            if valid(&item) {
226                let score = self.rand_try_next_u64()?;
227                if picked.is_none() || score > best_score {
228                    picked = Some(item);
229                    best_score = score;
230                    ties = 1;
231                } else if score == best_score {
232                    ties = ties.checked_add(1).expect("too many tied random candidates");
233                    is! { self.rand_try_below(ties)? == 0, picked = Some(item) }
234                }
235            }
236        }
237        Ok(picked)
238    }
239}

248/// `Rand` is implemented automatically for every [`RandTry`] source
249/// whose error type is [`Infallible`].
250///
251/// Determinism, reproducibility, cryptographic suitability, weakness,
252/// external state, and blocking behavior are described by the source
253/// characteristics inherited from [`RandTry`].
254#[rustfmt::skip]
255pub trait Rand: RandTry<Error = Infallible> {

300    /// # Panics

303    fn rand_range(&mut self, low: u64, high: u64) -> u64 {

305    }

311    /// Returns a value in `1..=sides`.

314    }

317    /// Uses the Fisher–Yates algorithm.

319        match self.rand_try_shuffle(slice) { Ok(()) => {} Err(e) => match e {} }
320    }

325    ///
326    /// This is uniform over all valid items and uses one pass.
327    fn rand_choose_reservoir<I, F>(&mut self, iter: I, valid: F) -> Option<I::Item>
328    where I: IntoIterator, F: FnMut(&I::Item) -> bool {
329        match self.rand_try_choose_reservoir(iter, valid) { Ok(v) => v, Err(e) => match e {} }
330    }
331    /// Chooses one valid item using random scores.
332    ///
333    /// Each valid item receives a random score. The highest score wins.
334    /// Ties are resolved uniformly.
335    fn rand_choose_scored<I, F>(&mut self, iter: I, valid: F) -> Option<I::Item>
336    where I: IntoIterator, F: FnMut(&I::Item) -> bool {
337        match self.rand_try_choose_scored(iter, valid) { Ok(v) => v, Err(e) => match e {} }
338    }
