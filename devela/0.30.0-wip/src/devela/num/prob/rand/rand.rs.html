8/// Construction from explicit seed material or another random source.
9#[doc = crate::_doc_meta!{
10    location("num/prob/rand", trait RandSeedable),
11}]
12/// `RandSeedable` defines a portable byte-backed seed representation and
13/// provides construction from explicit seed material or another random source.
14///
15/// Multi-byte values encoded in seeds use little-endian order, preserving
16/// reproducibility across target endianness.
17pub trait RandSeedable: Sized {
18    /// Byte-backed seed material accepted by this type.
19    type RandSeed: Clone + Default + AsRef<[u8]> + AsMut<[u8]>;
20    // type RandSeed: Default + AsMut<[u8]>;
21
22    /// Constructs this type from explicit seed material.
23    fn rand_from_seed(seed: Self::RandSeed) -> Self;
24
25    /// Attempts to construct this type by drawing seed material from `source`.
26    fn rand_try_from_source<R>(source: &mut R) -> Result<Self, R::Error>
27    where
28        R: RandTry + ?Sized,
29    {
30        let mut seed = Self::RandSeed::default();
31        source.rand_try_fill_bytes(seed.as_mut())?;
32        Ok(Self::rand_from_seed(seed))
33    }
34
35    /// Constructs this type by drawing seed material from an infallible source.
36    #[must_use]
37    fn rand_from_source<R>(source: &mut R) -> Self
38    where
39        R: Rand + ?Sized,
40    {
41        match Self::rand_try_from_source(source) {
42            Ok(value) => value,
43            Err(error) => match error {},
44        }
45    }
46}

49/// Fallible source of raw random data.
50#[doc = crate::_doc_meta!{
51    location("num/prob/rand", trait RandTry),
52}]
53/// `RandTry` is the core trait for random sources that may fail, such as
54/// OS calls, hardware RNGs, host APIs, finite buffers, and fallible adapters.
55///
56/// Source behavior and suitability are described by [`RandQualities`].
57///
58/// Infallible sources use [`Infallible`][crate::Infallible] as their error type.
59/// and automatically implement [`Rand`].
60pub trait RandTry {
61    /* required */
62
63    /// Error returned when complete random data could not be produced.
64    type Error;
65
66    /// Number of bits produced per output step.
67    ///
68    /// Adapters or composite sources may report a conservative value.
69    const RAND_OUTPUT_BITS: u32;
70
71    /// Size of the source state owned by this value, in bits.
72    ///
73    /// External entropy sources that do not store generator state
74    /// in the value itself should use `0`.
75    const RAND_STATE_BITS: u32;
76
77    /// Semantic qualities of this random source.
78    const RAND_QUALITIES: RandQualities;
79
80    /// Attempts to produce the next 64 bits of randomness.
81    fn rand_try_next_u64(&mut self) -> Result<u64, Self::Error>;
82
83    /*** provided ***/
84
85    /// Returns the semantic qualities of this random source.
86    #[must_use]
87    fn rand_qualities(&self) -> RandQualities {
88        Self::RAND_QUALITIES
89    }
90
91    /* derived primitives and byte access */
92
93    /// Attempts to produce the next random `u128`.
94    fn rand_try_next_u128(&mut self) -> Result<u128, Self::Error> {
95        let pair = [self.rand_try_next_u64()?, self.rand_try_next_u64()?];
96        Ok(Cast::<u128>::from_u64_le(pair))
97    }
98    /// Attempts to produce the next random `u32`.
99    fn rand_try_next_u32(&mut self) -> Result<u32, Self::Error> {
100        Ok(self.rand_try_next_u64()? as u32)
101    }
102    /// Attempts to produce the next random `u16`.
103    fn rand_try_next_u16(&mut self) -> Result<u16, Self::Error> {
104        Ok(self.rand_try_next_u64()? as u16)
105    }
106    /// Attempts to produce the next random `u8`.
107    fn rand_try_next_u8(&mut self) -> Result<u8, Self::Error> {
108        Ok(self.rand_try_next_u16()? as u8)
109    }
110    /// Attempts to produce a random boolean.
111    fn rand_try_next_bool(&mut self) -> Result<bool, Self::Error> {
112        Ok((self.rand_try_next_u64()? & 1) != 0)
113    }
114
115    /// Attempts to fill the buffer with generated bytes.
116    ///
117    /// This method must either fill the whole buffer or return an error.
118    fn rand_try_fill_bytes(&mut self, buffer: &mut [u8]) -> Result<(), Self::Error> {
119        let mut i = 0;
120        let len = buffer.len();
121        while i + 8 <= len {
122            buffer[i..i + 8].copy_from_slice(&self.rand_try_next_u64()?.to_ne_bytes());
123            i += 8;
124        }
125        if i < len {
126            let tail = self.rand_try_next_u64()?.to_ne_bytes();
127            buffer[i..].copy_from_slice(&tail[..len - i]);
128        }
129        Ok(())
130    }
131
132    /* bounded sampling */
133
134    /// Attempts to return a uniformly random integer in `0..upper`.
135    ///
136    /// Uses rejection sampling to avoid modulo bias.
137    ///
138    /// # Panics
139    /// Panics if `upper == 0`.
140    fn rand_try_below(&mut self, upper: u64) -> Result<u64, Self::Error> {
141        assert!(upper > 0);
142        let threshold = upper.wrapping_neg() % upper;
143        loop {
144            let v = self.rand_try_next_u64()?;
145            is! { v >= threshold, return Ok(v % upper) }
146        }
147    }
148    /// Attempts to return a uniformly random integer in `low..high`.
149    ///
150    /// # Panics
151    /// Panics if `low >= high`.
152    #[track_caller]
153    fn rand_try_range(&mut self, low: u64, high: u64) -> Result<u64, Self::Error> {
154        assert!(low < high);
155        Ok(low + self.rand_try_below(high - low)?)
156    }
157
158    /* common discrete helpers */
159
160    /// Attempts to roll a fair die with a number of `sides`.
161    ///
162    /// Returns a value in `1..=sides`.
163    fn rand_try_roll(&mut self, sides: u64) -> Result<u64, Self::Error> {
164        Ok(self.rand_try_below(sides)? + 1)
165    }
166    /// Attempts to shuffle the elements of the slice in place.
167    ///
168    /// Uses the Fisher–Yates algorithm.
169    fn rand_try_shuffle<T>(&mut self, slice: &mut [T]) -> Result<(), Self::Error> {
170        /*
171        let mut i = slice.len();
172        while i > 1 {
173            i -= 1;
174            let j = self.rand_try_below((i + 1) as u64)? as usize;
175            slice.swap(i, j);
176        }
177        */
178
179        whilst! { i in rev 1..slice.len(); {
180            let j = self.rand_try_below((i + 1) as u64)? as usize;
181            slice.swap(i, j);
182        }}
183        Ok(())
184    }
185
186    /* selection helpers */
187
188    /// Attempts to choose one valid item using reservoir sampling.
189    ///
190    /// This is uniform over all valid items and uses one pass.
191    fn rand_try_choose_reservoir<I, F>(
192        &mut self,
193        iter: I,
194        mut valid: F,
195    ) -> Result<Option<I::Item>, Self::Error>
196    where
197        I: IntoIterator,
198        F: FnMut(&I::Item) -> bool,
199    {
200        let (mut seen, mut picked) = (0u64, None);
201        for item in iter {
202            if valid(&item) {
203                seen = seen.checked_add(1).expect("too many valid random candidates");
204                is! { self.rand_try_below(seen)? == 0, picked = Some(item) }
205            }
206        }
207        Ok(picked)
208    }
209    /// Attempts to choose one valid item using random scores.
210    ///
211    /// Each valid item receives a random score. The highest score wins.
212    /// Ties are resolved uniformly.
213    fn rand_try_choose_scored<I, F>(
214        &mut self,
215        iter: I,
216        mut valid: F,
217    ) -> Result<Option<I::Item>, Self::Error>
218    where
219        I: IntoIterator,
220        F: FnMut(&I::Item) -> bool,
221    {
222        let (mut picked, mut best_score, mut ties) = (None, 0u64, 0u64);
223        for item in iter {
224            if valid(&item) {
225                let score = self.rand_try_next_u64()?;
226                if picked.is_none() || score > best_score {
227                    picked = Some(item);
228                    best_score = score;
229                    ties = 1;
230                } else if score == best_score {
231                    ties = ties.checked_add(1).expect("too many tied random candidates");
232                    is! { self.rand_try_below(ties)? == 0, picked = Some(item) }
233                }
234            }
235        }
236        Ok(picked)
237    }
238}

247/// `Rand` is implemented automatically for every [`RandTry`] source
248/// whose error type is [`Infallible`].

250/// Determinism, reproducibility, cryptographic suitability, weakness,
251/// external state, and blocking behavior are described by the source
252/// characteristics inherited from [`RandTry`].

254pub trait Rand: RandTry<Error = Infallible> {

319    }

324    ///
325    /// This is uniform over all valid items and uses one pass.
326    fn rand_choose_reservoir<I, F>(&mut self, iter: I, valid: F) -> Option<I::Item>
327    where I: IntoIterator, F: FnMut(&I::Item) -> bool {
328        match self.rand_try_choose_reservoir(iter, valid) { Ok(v) => v, Err(e) => match e {} }
329    }

332    /// Each valid item receives a random score. The highest score wins.

335    where I: IntoIterator, F: FnMut(&I::Item) -> bool {
336        match self.rand_try_choose_scored(iter, valid) { Ok(v) => v, Err(e) => match e {} }
