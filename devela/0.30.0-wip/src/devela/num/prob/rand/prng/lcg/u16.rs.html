15/// Based on original code from Ken Musgrave, 1985, in Graphics Gems II.
16#[derive(Clone, Copy, Debug, PartialEq, Eq)]
17pub struct Lcg16(u16);

30// Constant defaults for the Lcg16
31impl Lcg16 {
32    #[doc(hidden)]
33    pub const DEFAULT_SEED: u16 = 0xDEFA;
34
35    /// Multiplier.
36    const MUL: u16 = 25173;
37    /// Increment.
38    const INC: u16 = 13849;
39    /// Modulus.
40    const MOD: u16 = 65535;
41}
42
43impl Lcg16 {
44    /// Creates a new `Lcg16` instance with the given seed.
45    #[must_use]
46    pub const fn new(seed: u16) -> Self {
47        Self(seed)
48    }
49    /// Reseeds the generator with a new seed.
50    pub const fn reseed(&mut self, seed: u16) {
51        self.0 = seed;
52    }
53    #[must_use]
54    /// Returns the PRNG's inner state as a raw snapshot.
55    pub const fn inner_state(self) -> u16 {
56        self.0
57    }
58    /// Restores the PRNG from the given state.
59    pub const fn from_state(state: u16) -> Self {
60        Self(state)
61    }
62    /// Returns the current seed value.
63    #[must_use]
64    pub const fn current_u16(&self) -> u16 {
65        self.0
66    }
67    /// Advances to the next random `u16` value.
68    #[must_use]
69    pub const fn next_u16(&mut self) -> u16 {
70        self.0 = (Self::MUL.wrapping_mul(self.0).wrapping_add(Self::INC)) & Self::MOD;
71        self.0
72    }
73    /// Returns a copy of the next state of the generator.
74    #[must_use]
75    pub const fn peek_next_state(&self) -> Self {
76        let x = (Self::MUL.wrapping_mul(self.0).wrapping_add(Self::INC)) & Self::MOD;
77        Self(x)
78    }
79    /// Fills the buffer with generated bytes.
80    pub const fn fill_bytes(&mut self, buffer: &mut [u8]) {
81        let mut i = 0;
82        while i < buffer.len() {
83            let random_u16 = self.next_u16();
84            let bytes = random_u16.to_le_bytes();
85            let remaining = buffer.len() - i;
86            if remaining >= 2 {
87                buffer[i] = bytes[0];
88                buffer[i + 1] = bytes[1];
89                i += 2;
90            } else {
91                buffer[i] = bytes[0];
92                i += 1;
93            }
94        }
95    }
96}
97/// # Extras
98impl Lcg16 {
99    /// Returns a seeded `Lcg16` generator from the given 16-bit seed.
100    ///
101    /// This is an alias of [`new`][Self#method.new].
102    pub const fn new1_u16(seed: u16) -> Self {
103        Self::new(seed)
104    }
105    /// Returns a seeded `Lcg16` generator from the given 2 × 8-bit seeds.
106    ///
107    /// The seeds will be joined in little endian order.
108    #[must_use]
109    pub const fn new2_u8(seeds: [u8; 2]) -> Self {
110        Self::new(u16::from_le_bytes(seeds))
111    }
112
113    /// Returns the next 2 × random `u16` combined as a single `u32`.
114    pub const fn next_u32(&mut self) -> u32 {
115        Cast::<u32>::from_u16_le([self.next_u16(), self.next_u16()])
116    }
117    /// Returns the next 4 × random `u16` combined as a single `u64`.
118    pub const fn next_u64(&mut self) -> u64 {
119        Cast::<u64>::from_u16_le([
120            self.next_u16(),
121            self.next_u16(),
122            self.next_u16(),
123            self.next_u16(),
124        ])
125    }
126}
