14/// Based on original code from Ken Musgrave, 1985, in Graphics Gems II.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub struct Lcg16(u16);

29// Constant defaults for the Lcg16
30impl Lcg16 {
31    #[doc(hidden)]
32    pub const DEFAULT_SEED: u16 = 0xDEFA;
33
34    /// Multiplier.
35    const MUL: u16 = 25173;
36    /// Increment.
37    const INC: u16 = 13849;
38    /// Modulus.
39    const MOD: u16 = 65535;
40}
41
42impl Lcg16 {
43    /// Creates a new `Lcg16` instance with the given seed.
44    #[must_use]
45    pub const fn new(seed: u16) -> Self {
46        Self(seed)
47    }
48    /// Reseeds the generator with a new seed.
49    pub const fn reseed(&mut self, seed: u16) {
50        self.0 = seed;
51    }
52    #[must_use]
53    /// Returns the PRNG's inner state as a raw snapshot.
54    pub const fn inner_state(self) -> u16 {
55        self.0
56    }
57    /// Restores the PRNG from the given state.
58    pub const fn from_state(state: u16) -> Self {
59        Self(state)
60    }
61    /// Returns the current seed value.
62    #[must_use]
63    pub const fn current_u16(&self) -> u16 {
64        self.0
65    }
66    /// Advances to the next random `u16` value.
67    #[must_use]
68    pub const fn next_u16(&mut self) -> u16 {
69        self.0 = (Self::MUL.wrapping_mul(self.0).wrapping_add(Self::INC)) & Self::MOD;
70        self.0
71    }
72    /// Returns a copy of the next state of the generator.
73    #[must_use]
74    pub const fn peek_next_state(&self) -> Self {
75        let x = (Self::MUL.wrapping_mul(self.0).wrapping_add(Self::INC)) & Self::MOD;
76        Self(x)
77    }
78    /// Fills the buffer with generated bytes.
79    pub const fn fill_bytes(&mut self, buffer: &mut [u8]) {
80        let mut i = 0;
81        while i < buffer.len() {
82            let random_u16 = self.next_u16();
83            let bytes = random_u16.to_le_bytes();
84            let remaining = buffer.len() - i;
85            if remaining >= 2 {
86                buffer[i] = bytes[0];
87                buffer[i + 1] = bytes[1];
88                i += 2;
89            } else {
90                buffer[i] = bytes[0];
91                i += 1;
92            }
93        }
94    }
95}
96/// # Extras
97impl Lcg16 {
98    /// Returns a seeded `Lcg16` generator from the given 16-bit seed.
99    ///
100    /// This is an alias of [`new`][Self#method.new].
101    pub const fn new1_u16(seed: u16) -> Self {
102        Self::new(seed)
103    }
104    /// Returns a seeded `Lcg16` generator from the given 2 × 8-bit seeds.
105    ///
106    /// The seeds will be joined in little endian order.
107    #[must_use]
108    pub const fn new2_u8(seeds: [u8; 2]) -> Self {
109        Self::new(u16::from_le_bytes(seeds))
110    }
111
112    /// Returns the next 2 × random `u16` combined as a single `u32`.
113    pub const fn next_u32(&mut self) -> u32 {
114        Cast::<u32>::from_u16_le([self.next_u16(), self.next_u16()])
115    }
116    /// Returns the next 4 × random `u16` combined as a single `u64`.
117    pub const fn next_u64(&mut self) -> u64 {
118        Cast::<u64>::from_u16_le([
119            self.next_u16(),
120            self.next_u16(),
121            self.next_u16(),
122            self.next_u16(),
123        ])
124    }
125}
