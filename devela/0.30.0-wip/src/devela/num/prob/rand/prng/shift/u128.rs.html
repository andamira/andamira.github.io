39
40impl XorShift128 {
41    /// Returns a seeded `XorShift128` generator from the given 4 × 32-bit seeds.
42    ///
43    /// If the seeds are all `0`, the default seed is used instead.
44    pub const fn new(seeds: [u32; 4]) -> Self {
45        if (seeds[0] | seeds[1] | seeds[2] | seeds[3]) == 0 {
46            Self::cold_path_default()
47        } else {
48            Self([seeds[0], seeds[1], seeds[2], seeds[3]])
49        }
50    }
51    /// Returns a seeded `XorShift128` generator from the given 8-bit seed,
52    /// unchecked.
53    ///
54    /// The seeds must not be all `0`, otherwise every result will also be `0`.
55    ///
56    /// # Panics
57    /// Panics in debug if the seeds are all `0`.
58    pub const fn new_unchecked(seeds: [u32; 4]) -> Self {
59        debug_assert![(seeds[0] | seeds[1] | seeds[2] | seeds[3]) != 0, "Seeds must be non-zero"];
60        Self(seeds)
61    }
62    #[must_use]
63    /// Returns the PRNG's inner state as a raw snapshot.
64    pub const fn inner_state(self) -> [u32; 4] {
65        self.0
66    }
67    /// Restores the PRNG from the given state.
68    pub const fn from_state(state: [u32; 4]) -> Self {
69        Self(state)
70    }
71
72    #[must_use]
73    /// Returns the current random `u64`.
74    pub const fn current_u64(&self) -> u64 {
75        ((self.0[0] as u64) << 32) | (self.0[1] as u64)
76    }
77    #[must_use]
78    /// Returns the next random `u64`.
79    // Note how the output of the RNG is computed before updating the state,
80    // unlike on Xoroshiro128pp, for example.
81    pub const fn next_u64(&mut self) -> u64 {
82        let t = self.0[3];
83        let mut s = self.0[0];
84        self.0[3] = self.0[2];
85        self.0[2] = self.0[1];
86        self.0[1] = s;
87        s ^= s << 11;
88        s ^= s >> 8;
89        self.0[0] = s ^ t ^ (t >> 19);
90        ((self.0[0] as u64) << 32) | (self.0[1] as u64)
91    }
92    /// Returns a copy of the next new random state.
93    pub const fn peek_next_state(&self) -> Self {
94        let mut x = self.0;
95        let t = x[3];
96        let mut s = x[0];
97        x[3] = x[2];
98        x[2] = x[1];
99        x[1] = s;
100        s ^= s << 11;
101        s ^= s >> 8;
102        x[0] = s ^ t ^ (t >> 19);
103        Self(x)
104    }
105
106    /// Fills the buffer with generated bytes.
107    pub const fn fill_bytes(&mut self, buffer: &mut [u8]) {
108        let mut i = 0;
109        while i < buffer.len() {
110            let random_u64 = self.next_u64();
111            let bytes = random_u64.to_le_bytes();
112            let remaining = buffer.len() - i;
113            if remaining >= 8 {
114                // buffer[i..i + 8].copy_from_slice(&bytes);
115                Slice::copy(slice!(mut buffer, i, ..i + 8), &bytes);
116                i += 8;
117            } else {
118                // buffer[i..].copy_from_slice(&bytes[..remaining]);
119                Slice::copy(slice!(mut buffer, i, ..), slice!(&bytes, ..remaining));
120                break;
121            }
122        }
123    }
124}

126/// # Extra constructors
127impl XorShift128 {
128    /// Returns a seeded `XorShift128` generator from the given 128-bit seed.
129    ///
130    /// The seeds will be split in little endian order.
131    pub const fn new1_u128(seed: u128) -> Self {
132        Self::new(Cast(seed).into_u32_le())
133    }
134
135    /// Returns a seeded `XorShift128` generator from the given 2 × 64-bit seeds.
136    ///
137    /// The seeds will be split in little endian order.
138    pub const fn new2_u64(seeds: [u64; 2]) -> Self {
139        let [x, y] = Cast(seeds[0]).into_u32_le();
140        let [z, a] = Cast(seeds[1]).into_u32_le();
141        Self::new([x, y, z, a])
142    }
143
144    /// Returns a seeded `XorShift128` generator from the given 4 × 32-bit seeds.
145    ///
146    /// This is an alias of [`new`][Self#method.new].
147    pub const fn new4_u32(seeds: [u32; 4]) -> Self {
148        Self::new(seeds)
149    }
150
151    /// Returns a seeded `XorShift128` generator from the given 8 × 16-bit seeds.
152    ///
153    /// The seeds will be joined in little endian order.
154    pub const fn new8_u16(seeds: [u16; 8]) -> Self {
155        Self::new([
156            Cast::<u32>::from_u16_le([seeds[0], seeds[1]]),
157            Cast::<u32>::from_u16_le([seeds[2], seeds[3]]),
158            Cast::<u32>::from_u16_le([seeds[4], seeds[5]]),
159            Cast::<u32>::from_u16_le([seeds[6], seeds[7]]),
160        ])
161    }
162
163    /// Returns a seeded `XorShift128` generator from the given 16 × 8-bit seeds.
164    ///
165    /// The seeds will be joined in little endian order.
166    pub const fn new16_u8(seeds: [u8; 16]) -> Self {
167        Self::new([
168            u32::from_le_bytes([seeds[0], seeds[1], seeds[2], seeds[3]]),
169            u32::from_le_bytes([seeds[4], seeds[5], seeds[6], seeds[7]]),
170            u32::from_le_bytes([seeds[8], seeds[9], seeds[10], seeds[11]]),
171            u32::from_le_bytes([seeds[12], seeds[13], seeds[14], seeds[15]]),
172        ])
173    }
174}

182        fn rand_try_next_u64(&mut self) -> InfallibleResult<u64> { Ok(self.next_u64()) }

191    }
