4
5#[doc = crate::_tags!(construction rand)]
6#[doc = concat!["Defines a canonical ", crate::_ABBR_PCG!(), " (Permuted Congruential Generator) ",
7crate::_ABBR_PRNG!(), " type."]]
8#[doc = crate::_doc_meta!{location("num/prob/rand", macro rand_pcg)}]
9///
10/// This macro generates a concrete, fixed-width PCG PRNG
11/// with a canonical state transition and output permutation.
12///
13/// The output width is selected by specifying one of the supported integer
14/// sizes (`u8`, `u16`, `u32`, `u64`). The appropriate state size, multiplier,
15/// increment, and output permutation are chosen automatically.
16///
17/// The generator stores two internal state values (state and inc), each twice
18/// the output width, for a total internal state of four times the output width.
19///
20/// The 8- and 16-bit PCG variants are classified as weak PRNGs due to their limited
21/// per-stream state and period. The 32- and 64-bit variants use the general PRNG profile.
22///
23/// # Generated items
24/// For the defined PRNG type, this macro generates:
25/// - the PRNG struct with internal state.
26/// - constructors and state accessors.
27/// - step, peek, and jump-ahead operations.
28/// - bounded generation helpers.
29/// - an implementation of [`Rand`][crate::Rand] for generic usage.
30///
31/// # Examples
32/// ```
33/// use devela::rand_pcg;
34///
35/// rand_pcg! {
36///     /// Custom attributes
37///     pub(crate) Pcg8: (u8)
38/// }
39/// ```
40/// See also [`Pcg32`][crate::Pcg32].
41///
42/// # Reference
43/// PCG family reference: <https://www.pcg-random.org>
44#[macro_export]
45#[cfg_attr(cargo_primary_package, doc(hidden))]
46macro_rules! rand_pcg· {
47    (
48    /* public API */
49

52        $crate::rand_pcg!(% $(#[$attrs])* $vis $name, 32+u16, 8+u8, 1,
53            output_xsh_rr_16_8, 0xDEFA, 12829, 47989);
54    };
55    ( // Defines a 16-bit PCG PRNG (`u16` output, 32-bit state)
56      $(#[$attrs:meta])* $vis:vis $name:ident : (u16)) => {
57        $crate::rand_pcg!(% $(#[$attrs])* $vis $name, 64+u32, 16+u16, 2,
58            output_xsh_rr_32_16, 0xDEFA_0017, 747796405, 2891336453);
59    };
60    ( // Defines a 32-bit PCG PRNG (`u32` output, 64-bit state)
61      $(#[$attrs:meta])* $vis:vis $name:ident : (u32)) => {
62        $crate::rand_pcg!(% $(#[$attrs])* $vis $name, 128+u64, 32+u32, 4,
63            output_xsh_rr_64_32, 0xDEFA_0017_DEFA_0017, 6364136223846793005, 1442695040888963407);
64    };
65    ( // Defines a 64-bit PCG PRNG (`u64` output, 128-bit state)
66      $(#[$attrs:meta])* $vis:vis $name:ident : (u64)

69    ) => {

71            output_xsl_rr_128_64, 0xDEFA_0017_DEFA_0017_DEFA_0017_DEFA_0017,
72            47026247687942121848144207491837523525, 117397592171526113268558934119004209487);
73    };

104    /* private arms */
105    (%define $(#[$attrs:meta])*
106     $vis:vis $name:ident, $sbits:literal+$state:ty, $obits:literal+$output:ty,
107     $obytes:literal, $output_fn:ident,
108     $SEED:literal, $MUL:literal, $INC:literal
109     ) => { $crate::paste! {
110        $crate::rand_pcg!(%impls $output $name);
111
112        $(#[$attrs])*
113        #[doc = "It uses a " $sbits
114        "-bit linear congruential state with an independent stream selector, and generates "
115        $obits "-bit values."]
116        ///
117        /// Compared to simple xorshift-style generators of similar size, it
118        /// has stronger statistical properties while remaining small and fast.
119        ///
120        /// This implementation follows a canonical PCG output permutation
121        /// (XSH RR or XSL RR, depending on size), by Melissa O'Neill.
122        ///
123        /// This generator is deterministic and non-cryptographic.
124        ///
125        /// Reference: <https://www.pcg-random.org>
126        #[derive(Clone, Copy, Debug, PartialEq, Eq)]
127        $vis struct $name {
128            state: $state,
129            inc: $state, // must always be odd
130        }
131        /// Creates a new PRNG initialized with the default fixed seed.
132        impl Default for $name {
133            fn default() -> Self {
134                use $crate::ConstInit;
135                Self::INIT
136            }
137        }
138        /// Creates a new PRNG initialized with the default fixed seed.
139        impl $crate::ConstInit for $name {
140            const INIT: Self = Self::new(Self::DEFAULT_SEED, Self::DEFAULT_INC);
141        }
142
143        // Private associated item
144        impl $name {
145            #[doc(hidden)]
146            pub const DEFAULT_SEED: $state = $SEED;
147            #[doc(hidden)]
148            pub const DEFAULT_INC: $state = $INC;
149
150            const fn advance_lcg(state: $state,
151                mut delta: $state,
152                mut cur_mult: $state,
153                mut cur_plus: $state,
154            ) -> $state {
155                let mut acc_mult = 1 as $state;
156                let mut acc_plus = 0 as $state;
157                while delta > 0 {
158                    if (delta & 1) != 0 {
159                        acc_mult = acc_mult.wrapping_mul(cur_mult);
160                        acc_plus = acc_plus.wrapping_mul(cur_mult).wrapping_add(cur_plus);
161                    }
162                    cur_plus = cur_mult.wrapping_add(1).wrapping_mul(cur_plus);
163                    cur_mult = cur_mult.wrapping_mul(cur_mult);
164                    delta >>= 1;
165                }
166                acc_mult.wrapping_mul(state).wrapping_add(acc_plus)
167            }
168        }
169
170        impl $name {
171            #[doc = "Creates a new " $name " using the canonical PCG seeding scheme."]
172            ///
173            /// This performs a small warm-up so that closely related seeds produce
174            /// well-scrambled initial outputs.
175            pub const fn new(seed: $state, stream: $state) -> Self {
176                let mut rng = Self::new_fast(0, stream);
177                let _ = rng.[<next_ $output>]();
178                rng.state = rng.state.wrapping_add(seed);
179                let _ = rng.[<next_ $output>]();
180                rng
181            }
182
183            #[doc = "Creates a new " $name " with the given initial state and stream."]
184            ///
185            /// The stream value selects an independent random sequence.
186            pub const fn new_fast(state: $state, stream: $state) -> Self {
187                Self { state, inc: (stream << 1) | 1 }
188            }
189
190            #[doc = "Creates a new " $name " without enforcing invariants."]
191            ///
192            /// # Panics
193            /// Panics in debug if `inc` is not odd.
194            pub const fn new_unchecked(state: $state, inc: $state) -> Self {
195                debug_assert![!inc.is_multiple_of(2), "`inc` must be odd"];
196                Self { state, inc }
197            }
198
199            /// Returns the PRNG's inner state as a raw snapshot.
200            #[must_use]
201            pub const fn inner_state(self) -> ($state, $state) {
202                (self.state, self.inc)
203            }
204
205            /// Restores the PRNG from the given state.
206            #[must_use]
207            pub const fn from_state(state: ($state, $state)) -> Self {
208                Self { state: state.0, inc: state.1 }
209            }
210
211            #[doc = "Returns the current `" $output "` value."]
212            #[must_use]
213            pub const fn [<current_ $output>](&self) -> $output {
214                Self::$output_fn(self.state)
215            }
216
217            #[doc = "Advances to the next random `" $output "` value."]
218            #[must_use]
219            pub const fn [<next_ $output>](&mut self) -> $output {
220                let old = self.state;
221                self.state = old.wrapping_mul($MUL).wrapping_add(self.inc | 1);
222                Self::$output_fn(old)
223            }
224
225            /// Returns a copy of the next PRNG state.
226            #[must_use]
227            pub const fn peek_next_state(&self) -> Self {
228                let next_state = self.state.wrapping_mul($MUL).wrapping_add(self.inc | 1);
229                Self { state: next_state, inc: self.inc }
230            }
231            #[doc = "Returns the next random `" $output "` without advancing the state."]
232            #[must_use]
233            pub const fn [<peek_next_ $output>](&self) -> $output {
234                Self::$output_fn(self.state)
235            }
236
237            #[doc = "Returns both the next random state and the `" $output "` value."]
238            pub const fn [<step_ $output>](self) -> (Self, $output) {
239                let value = Self::$output_fn(self.state);
240                let next = self.peek_next_state();
241                (next, value)
242            }
243
244            /// Fills the buffer with generated bytes.
245            pub const fn fill_bytes(&mut self, buffer: &mut [u8]) {
246                use $crate::{Slice, slice};
247                let mut i = 0;
248                while i < buffer.len() {
249                    let r = self.[<next_ $output>]();
250                    let bytes = r.to_le_bytes();
251                    let remaining = buffer.len() - i;
252                    if remaining >= $obytes {
253                        // buffer[i..i + $obytes].copy_from_slice(&bytes);
254                        Slice::copy(slice!(mut buffer, i, ..i + $obytes), &bytes);
255                        i += $obytes;
256                    } else {
257                        // buffer[i..].copy_from_slice(&bytes[..remaining]);
258                        Slice::copy(slice!(mut buffer, i, ..), slice!(&bytes, ..remaining));
259                        break;
260                    }
261                }
262            }
263
264            /// Advances the generator by `delta` steps.
265            pub const fn advance(&mut self, delta: $state) {
266                self.state = Self::advance_lcg(self.state, delta, $MUL, self.inc | 1);
267            }
268
269            /// Returns a copy of the state advanced by `delta` steps.
270            pub const fn copy_advance(self, delta: $state) -> Self {
271                let mut s = self;
272                s.advance(delta);
273                s
274            }
275
276            /// Returns a uniformly distributed value in `0..bound`.
277            ///
278            /// Uses rejection sampling to avoid modulo bias.
279            ///
280            /// # Panics
281            /// Panics if `bound == 0`.
282            #[must_use]
283            pub const fn next_bounded(&mut self, bound: $output) -> $output {
284                assert!(bound > 0);
285                let threshold = bound.wrapping_neg() % bound;
286                loop {
287                    let r = self.[<next_ $output>]();
288                    if r >= threshold { return r % bound; }
289                }
290            }
291            /// Returns a uniformly distributed `u64` value in `0..bound`.
292            ///
293            /// Uses the generator's native output width when `bound` fits,
294            /// otherwise draws a full `u64`.
295            ///
296            /// Uses rejection sampling to avoid modulo bias.
297            ///
298            /// # Panics
299            /// Panics if `bound == 0`.
300            #[must_use]
301            pub const fn next_bounded_u64(&mut self, bound: u64) -> u64 {
302                assert!(bound > 0);
303                if bound <= <$output>::MAX as u64 {
304                    return self.next_bounded(bound as $output) as u64;
305                }
306                let threshold = bound.wrapping_neg() % bound;
307                loop {
308                    let r = self.next_u64();
309                    if r >= threshold { return r % bound; }
310                }
311            }
312        }

315    }};
