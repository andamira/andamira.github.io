8/// An online accumulator for the mean and variance.
9#[doc = crate::_doc_meta!{
10    location("num/prob/stats", struct StatsMoment),
11    test_size_of(StatsMoment = 24|192; niche !Option),
12}]
13/// Values can be added one at a time without retaining the observations.
14/// The accumulator keeps the observation count, running mean,
15/// and the accumulated squared deviation $M_2$.
16///
17/// Independent accumulators can also be combined with [`merge`](#method.merge),
18/// allowing the same statistics to be collected in separate batches.
19///
20/// Floating-point values follow normal `f64` arithmetic.
21/// Non-finite inputs are accepted and may propagate to the results.
22#[must_use]
23#[derive(Clone, Copy, Debug, PartialEq)]
24pub struct StatsMoment {
25    count: u64,
26    mean: f64,
27    m2: f64,
28}
29
30_impl_init! { Self::new() => StatsMoment }
31
32impl Default for StatsMoment {
33    fn default() -> Self {
34        Self::new()
35    }
36}
37
38impl StatsMoment {
39    /// Creates an empty accumulator.
40    pub const fn new() -> Self {
41        Self { count: 0, mean: 0.0, m2: 0.0 }
42    }
43    /// Returns the number of incorporated observations.
44    #[must_use]
45    pub const fn count(&self) -> u64 {
46        self.count
47    }
48    /// Returns whether no observations have been incorporated.
49    #[must_use]
50    pub const fn is_empty(&self) -> bool {
51        self.count == 0
52    }
53
54    /// Returns the arithmetic mean, or `None` when empty.
55    #[must_use]
56    pub const fn mean(&self) -> Option<f64> {
57        if self.count == 0 { None } else { Some(self.mean) }
58    }
59    /// Returns the accumulated squared deviation $M_2$.
60    ///
61    /// This is the unnormalized second central moment:
62    /// $$
63    /// M_2 = \sum_{i=1}^{n}(x_i-\mu)^2.
64    /// $$
65    ///
66    /// It is zero for an empty accumulator and after one finite observation.
67    #[must_use]
68    pub const fn m2(&self) -> f64 {
69        self.m2
70    }
71    /// Incorporates one observation.
72    ///
73    /// Uses Welford's online update, requiring no stored observations.
74    ///
75    /// # Formula
76    /// For a new value $x$, with current count $n$ and mean $\mu$:
77    /// $$
78    /// n' = n + 1,\quad
79    /// \delta = x-\mu,\quad
80    /// \mu' = \mu+\frac{\delta}{n'},\quad
81    /// M_2' = M_2+\delta(x-\mu').
82    /// $$
83    ///
84    /// Returns `false` without modifying the accumulator if the observation
85    /// count has already reached `u64::MAX`.
86    #[must_use]
87    pub const fn push(&mut self, value: f64) -> bool {
88        let count = unwrap![some_or self.count.checked_add(1), return false];
89        let delta = value - self.mean;
90        let mean = self.mean + delta / count as f64;
91        let delta2 = value - mean;
92        self.count = count;
93        self.mean = mean;
94        self.m2 += delta * delta2;
95        true
96    }
97    /// Returns the population variance, or `None` when empty.
98    ///
99    /// $$
100    /// \sigma^2 = \frac{M_2}{n}.
101    /// $$
102    #[must_use]
103    pub const fn variance_population(&self) -> Option<f64> {
104        is! { self.count == 0, None, Some(self.m2 / self.count as f64) }
105    }
106    /// Returns the unbiased sample variance, or `None` for fewer than two observations.
107    ///
108    /// $$
109    /// s^2 = \frac{M_2}{n-1}.
110    /// $$
111    #[must_use]
112    pub const fn variance_sample(&self) -> Option<f64> {
113        is! { self.count < 2, None, Some(self.m2 / (self.count - 1) as f64) }
114    }
115    /// Merges the observations accumulated in `other` into this accumulator.
116    ///
117    /// This combines the retained statistics directly, without replaying
118    /// individual observations. An empty accumulator acts as an identity.
119    ///
120    /// # Formula
121    /// For accumulators $a$ and $b$, let $\delta = \mu_b-\mu_a$ and $n=n_a+n_b$. Then:
122    /// $$
123    /// \mu = \mu_a + \delta\frac{n_b}{n},
124    /// \qquad
125    /// M_2 = M_{2,a} + M_{2,b} + \delta^2\frac{n_a n_b}{n}.
126    /// $$
127    ///
128    /// Returns `false` without modifying the accumulator if the combined
129    /// observation count would overflow `u64`.
130    #[must_use]
131    pub const fn merge(&mut self, other: Self) -> bool {
132        is! { other.count == 0, return true }
133        is! { self.count == 0, { *self = other; return true; } }
134        let new_count = unwrap![some_or self.count.checked_add(other.count), return false];
135        let count_a = self.count as f64;
136        let count_b = other.count as f64;
137        let count = new_count as f64;
138        let delta = other.mean - self.mean;
139        let mean = self.mean + delta * count_b / count;
140        let m2 = self.m2 + other.m2 + delta * delta * count_a * count_b / count;
141        self.count = new_count;
142        self.mean = mean;
143        self.m2 = m2;
144        true
145    }
146}

148#[cfg(test)]
149mod _test {
150    use crate::{StatsMoment, assert_approx_eq_all};
151
152    const CONST_MOMENT: StatsMoment = {
153        let mut stats = StatsMoment::new();
154        let _ = stats.push(2.0);
155        let _ = stats.push(4.0);
156        let _ = stats.push(4.0);
157        let _ = stats.push(4.0);
158        let _ = stats.push(5.0);
159        let _ = stats.push(5.0);
160        let _ = stats.push(7.0);
161        let _ = stats.push(9.0);
162        stats
163    };
164
165    #[test]
166    fn moment_starts_empty() {
167        let stats = StatsMoment::new();
168        assert_eq!(stats.count(), 0);
169        assert!(stats.is_empty());
170        assert_eq!(stats.mean(), None);
171        assert_eq!(stats.m2(), 0.0);
172        assert_eq!(stats.variance_population(), None);
173        assert_eq!(stats.variance_sample(), None);
174    }
175    #[test]
176    fn moment_accumulates_mean_and_second_moment() {
177        assert_eq!(CONST_MOMENT.count(), 8);
178        assert!(!CONST_MOMENT.is_empty());
179        assert_eq!(CONST_MOMENT.mean(), Some(5.0));
180        assert_eq!(CONST_MOMENT.m2(), 32.0);
181    }
182    #[test]
183    fn moment_derives_population_and_sample_variance() {
184        assert_eq!(CONST_MOMENT.variance_population(), Some(4.0));
185        assert_approx_eq_all![
186            tolerance: 1e-12_f64,
187            CONST_MOMENT.variance_sample().unwrap(),
188            32.0 / 7.0
189        ];
190    }
191    #[test]
192    fn moment_single_observation_has_zero_population_variance() {
193        let mut stats = StatsMoment::new();
194        assert!(stats.push(42.0));
195        assert_eq!(stats.mean(), Some(42.0));
196        assert_eq!(stats.m2(), 0.0);
197        assert_eq!(stats.variance_population(), Some(0.0));
198        assert_eq!(stats.variance_sample(), None);
199    }
200    #[test]
201    fn moment_merge_combines_independent_batches() {
202        let mut a = StatsMoment::new();
203        for value in [2.0, 4.0, 4.0, 4.0] {
204            assert!(a.push(value));
205        }
206        let mut b = StatsMoment::new();
207        for value in [5.0, 5.0, 7.0, 9.0] {
208            assert!(b.push(value));
209        }
210        assert!(a.merge(b));
211        assert_eq!(a.count(), CONST_MOMENT.count());
212        assert_eq!(a.mean(), CONST_MOMENT.mean());
213        assert_eq!(a.m2(), CONST_MOMENT.m2());
214        assert_eq!(a.variance_population(), CONST_MOMENT.variance_population());
215        assert_approx_eq_all![
216            tolerance: 1e-12_f64,
217            a.variance_sample().unwrap(),
218            CONST_MOMENT.variance_sample().unwrap()
219        ];
220    }
221    #[test]
222    fn moment_merge_treats_empty_as_identity() {
223        let mut empty = StatsMoment::new();
224        assert!(empty.merge(CONST_MOMENT));
225        assert_eq!(empty, CONST_MOMENT);
226        let mut stats = CONST_MOMENT;
227        assert!(stats.merge(StatsMoment::new()));
228        assert_eq!(stats, CONST_MOMENT);
229    }
