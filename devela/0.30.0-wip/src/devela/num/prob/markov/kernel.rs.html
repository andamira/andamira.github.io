9/// A finite-state Markov transition kernel.
10#[doc = crate::_doc_meta!{
11    location("num/prob/markov", struct MarkovKernel),
12    #[cfg(target_pointer_width = "32")]
13    test_size_of(MarkovKernel = 8|64; niche Option),

17/// Each state has one categorical distribution over the possible next
18/// states. All rows therefore contain the same number of categories as
19/// there are states.
20///
21/// The kernel borrows its transition distributions and performs no
22/// allocation.
23#[must_use]
24#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
25pub struct MarkovKernel<'a> {
26    transitions: &'a [DistCategorical<'a>],
27}
28
29impl<'a> MarkovKernel<'a> {
30    /// Constructs a transition kernel from one distribution per state.
31    ///
32    /// Returns `None` if there are no states or if any distribution has
33    /// a different number of categories from the number of states.
34    #[must_use]
35    pub const fn new(transitions: &'a [DistCategorical<'a>]) -> Option<Self> {
36        let state_count = transitions.len();
37        is! { state_count == 0, return None }
38        whilst! { state in 0..state_count; {
39            is! { transitions[state].category_count() != state_count, return None }
40        }}
41        Some(Self { transitions })
42    }
43    /// Returns the number of states.
44    #[must_use]
45    pub const fn state_count(&self) -> usize {
46        self.transitions.len()
47    }
48    /// Returns all transition distributions in state order.
49    pub const fn transitions(&self) -> &'a [DistCategorical<'a>] {
50        self.transitions
51    }
52    /// Returns the next-state distribution for `state`.
53    #[must_use]
54    pub const fn transitions_from(&self, state: usize) -> Option<DistCategorical<'a>> {
55        is! { state < self.transitions.len(), Some(self.transitions[state]), None }
56    }
57    /// Returns the exact transition probability from one state to another.
58    #[must_use]
59    pub const fn probability(&self, from: usize, to: usize) -> Option<Probability> {
60        let transitions = unwrap![some? self.transitions_from(from)];
61        transitions.probability_of(to)
62    }
63}

65#[cfg(test)]
66mod _test {
67    use crate::{DistCategorical, MarkovKernel, Probability};
68
69    #[test]
70    fn rejects_empty_and_non_square_kernels() {
71        assert_eq!(MarkovKernel::new(&[]), None);
72        let a = DistCategorical::new(&[1, 1]).unwrap();
73        let b = DistCategorical::new(&[1, 1, 1]).unwrap();
74        assert_eq!(MarkovKernel::new(&[a, b]), None);
75    }
76    #[test]
77    fn exposes_transition_structure() {
78        let a = DistCategorical::new(&[1, 3]).unwrap();
79        let b = DistCategorical::new(&[2, 2]).unwrap();
80        let transitions = [a, b];
81        let kernel = MarkovKernel::new(&transitions).unwrap();
82        assert_eq!(kernel.state_count(), 2);
83        assert_eq!(kernel.transitions(), &transitions);
84        assert_eq!(kernel.transitions_from(0), Some(a));
85        assert_eq!(kernel.transitions_from(1), Some(b));
86        assert_eq!(kernel.transitions_from(2), None);
87    }
88    #[test]
89    fn exposes_exact_transition_probabilities() {
90        let transitions =
91            [DistCategorical::new(&[1, 3]).unwrap(), DistCategorical::new(&[2, 2]).unwrap()];
92        let kernel = MarkovKernel::new(&transitions).unwrap();
93        assert_eq!(kernel.probability(0, 0), Probability::new(1, 4));
94        assert_eq!(kernel.probability(0, 1), Probability::new(3, 4));
95        assert_eq!(kernel.probability(1, 0), Probability::new(1, 2));
96        assert_eq!(kernel.probability(1, 1), Probability::new(1, 2));
97        assert_eq!(kernel.probability(2, 0), None);
98        assert_eq!(kernel.probability(0, 2), None);
99    }
