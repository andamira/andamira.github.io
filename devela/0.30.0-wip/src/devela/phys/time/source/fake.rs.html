16/// `TimeFake` provides a controlled, adjustable timestamp source for tests.
17/// This enables predictable behavior when testing time-dependent systems.
18///
19/// Although TimeFake is commonly advanced monotonically, it allows arbitrary
20/// time jumps and therefore does not provide a monotonic timeline guarantee.
21///
22/// This form is ideal for:
23/// - tests that need explicit control
24/// - deterministic state machines
25/// - simulations
26///
27/// # Example:
28/// ```
29/// use devela::{TimeFake, TimeScale};
30///
31/// // Create an independent fake clock starting at t = 100 ms
32/// let clock = TimeFake::new(100, TimeScale::Millis);
33///
34/// // Inspect the current fake time
35/// assert_eq!(clock.get_time(), 100);
36///
37/// // Advance time explicitly
38/// clock.advance_time(25);
39/// assert_eq!(clock.get_time(), 125);
40///
41/// // Set time to an exact value
42/// clock.set_time(1_000);
43/// assert_eq!(clock.get_time(), 1_000);
44/// ```
45#[derive(Debug, Default)]
46pub struct TimeFake {
47    now: AtomicU64,
48    scale: TimeScale,
49}
50impl TimeFake {
51    /// Creates a new `TimeFake` with the given starting fake time and scale.
52    pub const fn new(start: u64, scale: TimeScale) -> Self {
53        Self { now: AtomicU64::new(start), scale }
54    }
55    /// Creates a new `TimeFake` with the given starting fake time in milliseconds.
56    pub const fn new_millis(start: u64) -> Self {
57        Self {
58            now: AtomicU64::new(start),
59            scale: TimeScale::Millis,
60        }
61    }
62
63    /// Returns the time scale.
64    pub fn time_scale(&self) -> TimeScale {
65        self.scale
66    }
67
68    /// Gets the time value.
69    pub fn get_time(&self) -> u64 {
70        self.now.load(AtomicOrdering::SeqCst)
71    }
72
73    /// Manually sets the time to a specific value.
74    pub fn set_time(&self, value: u64) {
75        self.now.store(value, AtomicOrdering::SeqCst);
76    }
77
78    /// Advances the fake time by a given amount.
79    pub fn advance_time(&self, delta: u64) {
80        self.now.fetch_add(delta, AtomicOrdering::SeqCst);
81    }
82
83    /// Gets the current time value in milliseconds.
84    pub fn get_millis(&self) -> u64 {
85        self.scale.convert_simulated(self.get_time(), TimeScale::Milliseconds)
86    }
87
88    /// Sets the current time to the given value in milliseconds.
89    pub fn set_millis(&self, time: u64) {
90        let value = TimeScale::Milliseconds.convert_simulated(time, self.scale);
91        self.set_time(value);
92    }
93
94    /// Advances the current time by the given number of milliseconds.
95    pub fn advance_millis(&self, delta: u64) {
96        let value = TimeScale::Milliseconds.convert_simulated(delta, self.scale);
97        self.advance_time(value);
98    }
99
100    /// Returns a borrowed view of this fake time source.
101    pub fn view(&self) -> TimeFakeRef<'_> {
102        TimeFakeRef { src: self }
103    }
104}

107/// A borrowed configuration handle selecting a specific `TimeFake` timeline.
108#[doc = crate::_doc_meta!{
109    location("phys/time/source", struct TimeFakeRef),
110    #[cfg(target_pointer_width = "32")]
111    test_size_of(TimeFakeRef = 4|32; niche Option),
112    #[cfg(target_pointer_width = "64")]
113    test_size_of(TimeFakeRef = 8|64; niche Option),

115/// This type does not own time state; it exists solely to satisfy
116/// [`TimeSourceCfg`] and model timeline selection explicitly.
117///
118/// This form is ideal for:
119/// - code generic over `TimeSourceCfg`
120/// - schedulers and timers
121/// - components that must not own the clock
122///
123/// # Examples
124/// ```
125/// use devela::{TimeFake, TimeFakeRef, TimeSourceCfg};
126///
127/// // Create a fake clock
128/// let clock = TimeFake::new_millis(500);
129///
130/// // Obtain a configuration handle selecting this timeline
131/// let view = clock.view();
132///
133/// // Read time through the generic time-source interface
134/// let t1 = <TimeFakeRef as TimeSourceCfg<u64>>::time_now_millis(view);
135/// // or just:
136/// let t1 = TimeFakeRef::time_now_millis(view);
137/// // or:
138/// let t1 = view.now_millis();
139///
140/// assert_eq!(t1, 500);
141///
142/// // Advance the underlying clock
143/// clock.advance_time(200);
144///
145/// // Observe the updated time through the same config
146/// let t2 = TimeFakeRef::time_now_millis(view);
147/// assert_eq!(t2, 700);
148/// ```
149#[derive(Clone, Copy, Debug)]
150pub struct TimeFakeRef<'a> {
151    src: &'a TimeFake,
152}
153
154impl<'a> TimeFakeRef<'a> {
155    /// Returns the current time snapshot in milliseconds.
156    ///
157    /// Equivalent to calling `TimeSourceCfg::time_now_millis`.
158    pub fn now_millis(self) -> u64 {
159        <Self as TimeSourceCfg<u64>>::time_now_millis(self)
160    }
161}
162
163/// The trait is implemented for `TimeFakeRef` rather than `TimeFake` itself
164/// because `TimeSourceCfg::Config` represents a *selection of a timeline*,
165/// not ownership of the time source. This allows multiple independent fake
166/// clocks to coexist without relying on global state.
167#[rustfmt::skip]
168impl<'a> TimeSourceCfg<u64> for TimeFakeRef<'a> {
169    type Config = Self;
170    fn time_is_monotonic(_: Self::Config) -> bool { false }
171    fn time_is_absolute(_: Self::Config) -> bool { false }
172    fn time_scale(cfg: Self::Config) -> TimeScale { cfg.src.scale }
173
174    fn time_now(cfg: Self::Config) -> u64 { cfg.src.get_time() }
175    fn time_point_value(_: Self::Config, point: u64) -> u64 { point }
176    fn time_elapsed_value(_: Self::Config, elapsed: u64) -> u64 { elapsed }
177}
