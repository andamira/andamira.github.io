10/// A normalized pressable pointer/mouse button.
11#[doc = crate::_doc_meta!{
12    location("ui/event", enum EventButton),
13    test_size_of(EventButton = 2|16; niche Option),
14}]
15/// `Left`, `Middle`, and `Right` represent the three primary conventional buttons.
16/// `X1..X5` represent additional auxiliary button slots when a backend can report them.
17///
18/// These variants name normalized button slots, not guaranteed user actions.
19/// For example, `X1` is often used as “back”, `X2` as “forward”, and on
20/// Pointer Events a pen eraser may be reported through an auxiliary slot.
21#[repr(u8)]
22#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
23pub enum EventButton {
24    #[doc = crate::_tags!(init)]
25    /// Left mouse button.
26    #[default]
27    Left,
28    /// Right mouse button.
29    Right,
30    /// Middle mouse button (usually the scroll wheel button).
31    Middle,
32    /// The first auxiliary mouse button, commonly "back".
33    X1,
34    /// The second auxiliary mouse button, commonly "forward".
35    X2,
36    /// The third auxiliary mouse button.
37    X3,
38    /// The fourth auxiliary mouse button.
39    X4,
40    /// The fifth auxiliary mouse button.
41    X5,
42    /// Additional buttons (e.g., side buttons on advanced mice).
43    Other(u8),
44}
45_impl_init! { Self::Left => EventButton }
46impl EventButton {
47    /// Returns some button as long as it's not 0.
48    pub const fn new(number: u8) -> Option<Self> {
49        match number {
50            1 => Some(Self::Left),
51            2 => Some(Self::Middle),
52            3 => Some(Self::Right),
53            _ => Some(Self::Other(number)),
54        }
55    }
56
57    /// Returns the button represented by the mask, if exactly one button is set.
58    pub const fn from_one_bit_mask(mask: EventButtons) -> Option<EventButton> {
59        match mask.bits() {
60            1 => Some(EventButton::Left),
61            2 => Some(EventButton::Right),
62            4 => Some(EventButton::Middle),
63            8 => Some(EventButton::X1),
64            16 => Some(EventButton::X2),
65            32 => Some(EventButton::X3),
66            64 => Some(EventButton::X4),
67            128 => Some(EventButton::X5),
68            _ => None,
69        }
70    }
71    /// Returns this button as a held-button mask, if it has a normalized role.
72    pub const fn to_mask(self) -> EventButtons {
73        match self {
74            Self::Left => EventButtons::new().with(EventButtons::LEFT),
75            Self::Right => EventButtons::new().with(EventButtons::RIGHT),
76            Self::Middle => EventButtons::new().with(EventButtons::MIDDLE),
77            Self::X1 => EventButtons::new().with(EventButtons::X1),
78            Self::X2 => EventButtons::new().with(EventButtons::X2),
79            Self::X3 => EventButtons::new().with(EventButtons::X3),
80            Self::X4 => EventButtons::new().with(EventButtons::X4),
81            Self::X5 => EventButtons::new().with(EventButtons::X5),
82            Self::Other(_) => EventButtons::new(),
83        }
84    }
85}

106set! {
107    #[doc = crate::_tags!(event interaction set)]
108    /// A semantic bitmask of currently held pressable buttons.
109    #[doc = crate::_doc_meta!{
110        location("ui/event", struct EventButtons),
111        test_size_of(EventButtons = 1|8; niche !Option),
112    }]
113    /// The bits represent normalized button roles, not raw backend button numbers.
114    ///
115    /// This means:
116    /// - `LEFT`, `RIGHT`, and `MIDDLE` are the three primary buttons.
117    /// - `X1..X5` are additional auxiliary buttons when a backend can report them.
118    /// - wheel motion is **not** represented here; it belongs in
119    ///   [`EventWheel`][crate::EventWheel].
120    ///
121    /// Unsupported buttons are left cleared.
122    ///
123    /// # Backend notes
124    /// - **Web** can naturally carry `LEFT`, `RIGHT`, `MIDDLE`, `X1`, and `X2`.
125    /// - **X11 core** currently sets only `LEFT`, `RIGHT`, and `MIDDLE`;
126    ///   buttons 4 and 5 are treated as wheel input in the current backend.
127    /// - **Terminal SGR** reports primary buttons, wheel motion, and some
128    ///   extended button encodings. This library normalizes the first supported
129    ///   non-primary button encodings into `X1..X5` when unambiguous.
130    /// - Other backends may populate only the subset they can observe.
131    ///
132    /// This type is semantic and cross-platform.
133    /// Backend-specific numbering should be translated at the backend edge.
134    //
135    // Firefox may consume browser Back/Forward mouse buttons before dispatching
136    // DOM mouse events, so button=3/4 and buttons=8/16 may never reach web code.
137    // WAIT: [firefox-back-forward-buttons](https://bugzilla.mozilla.org/show_bug.cgi?id=1933746)
138    pub struct EventButtons(u8) {
139        /// The primary left button.
140        LEFT = 0;
141        /// The primary right button.
142        RIGHT = 1;
143        /// The primary middle button.
144        MIDDLE = 2;
145        /// The first auxiliary button, often “back”.
146        X1 = 3;
147        /// The second auxiliary button, often “forward”.
148        X2 = 4;
149        /// The third auxiliary button, when available.
150        X3 = 5;
151        /// The fourth auxiliary button, when available.
152        X4 = 6;
153        /// The fifth auxiliary button, when available.
154        X5 = 7;
155    }
156    traits(!Debug);
157}
