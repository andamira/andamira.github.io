9/// An FFI-safe, versioned representation of [`Key`].
10#[doc = crate::_doc_meta!{
11    location("ui/event", struct KeyFfi),
12    test_size_of(KeyFfi = 8|64; niche Option),
13}]
14///
15/// Used in [`EventKeyFfi`][crate::EventKeyFfi] and [`WebEventKey`][crate::WebEventKey].
16///
17/// Unlike [`Key`], this enum is exhaustive because it participates in
18/// FFI-compatible layouts. New variants require a coordinated API/ABI change.
19#[repr(C)]
20#[allow(missing_docs)] #[rustfmt::skip]
21#[derive(Copy, Clone, Default, PartialEq, Eq, Hash)]
22pub enum KeyFfi {
23    // common control keys
24    Backspace, Enter, Tab, Escape, Space,
25    // navigation keys
26    Left, Right, Up, Down,
27    Home, End, PageUp, PageDown,
28    // editing keys
29    Delete, Insert,
30    // lock keys
31    CapsLock, ScrollLock, NumLock,
32    // special system kesy
33    PrintScreen, Pause, Menu,
34
35    /// A keypad key.
36    Pad(KeyPad),
37    /// A multimedia key.
38    Media(KeyMedia),
39    /// A modifier key.
40    Mod(KeyMod),
41
42    /// Function keys (F1-F48).
43    Fn(u8),
44
45    #[doc = crate::_tags!(ffi)]
46    /// A unicode character (FFI-safe, stored as `u32`).
47    Char(u32),
48
49    /// A dead key (accent prefix, composition).
50    Dead(KeyDead),
51
52    // Letters
53    A, B, C, D, E, F, G, H, I, J,
54    K, L, M, N, O, P, Q, R, S, T,
55    U, V, W, X, Y, Z,
56
57    // Digits (E row)
58    Digit1, Digit2, Digit3, Digit4, Digit5,
59    Digit6, Digit7, Digit8, Digit9, Digit0,
60
61    // ANSI punctuation / symbol positions
62    // E row
63    /// Physical key left of '1'
64    Backquote,
65    /// Physical `-` key.
66    Minus,
67    /// Physical `=` key.
68    Equal,
69    // D row
70    /// Physical `[` key.
71    BracketLeft,
72    /// Physical `]` key.
73    BracketRight,
74    // C (home) row
75    /// Physical `;` key.
76    Semicolon,
77    /// Physical `'` key.
78    Quote,
79    // B row
80    /// Physical `\` key.
81    Backslash,
82    /// ISO-102 extra key.
83    IntlBackslash,
84    // A row
85    /// Physical `,` key.
86    Comma,
87    /// Physical `.` key.
88    Dot,
89    /// Physical `/` key.
90    Slash,
91
92    /// A physical scancode key.
93    Scancode(u16),
94
95    #[doc = crate::_tags!(init)]
96    #[default]
97    Unknown,
98}
99impl ConstInit for KeyFfi {
100    const INIT: Self = Self::Unknown;
101}
