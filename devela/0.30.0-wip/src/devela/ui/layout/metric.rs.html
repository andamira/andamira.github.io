43#[rustfmt::skip]
44impl UiRect {
45    /// Returns the left edge position.
46    pub const fn left(self) -> Lunit { self.x() }
47    /// Returns the top edge position.
48    pub const fn top(self) -> Lunit { self.y() }
49
50    /// Returns the right edge position, saturating on overflow.
51    pub const fn right(self) -> Lunit { self.x().sat_add(self.w()) }
52    /// Returns the bottom edge position, saturating on overflow.
53    pub const fn bottom(self) -> Lunit { self.y().sat_add(self.h()) }
54
55    /// Returns whether the rectangle has no positive area.
56    pub const fn is_empty(self) -> bool {
57        self.w().positive_part().eq(Lunit::ZERO)
58            || self.h().positive_part().eq(Lunit::ZERO)
59    }
60
61    /// Returns this rectangle inset equally on all sides.
62    ///
63    /// Negative insets are ignored. If the inset exceeds the rectangle extent,
64    /// the resulting extent is clamped to zero.
65    pub const fn inset(self, all: Lunit) -> Self {
66        self.inset_ltrb(all, all, all, all)
67    }
68    /// Returns this rectangle inset horizontally and vertically.
69    ///
70    /// Negative insets are ignored. If an inset exceeds the rectangle extent,
71    /// the resulting extent is clamped to zero.
72    pub const fn inset_xy(self, x: Lunit, y: Lunit) -> Self {
73        self.inset_ltrb(x, y, x, y)
74    }
75
76    /// Returns this rectangle inset by left, top, right, and bottom margins.
77    ///
78    /// Negative insets are ignored. The resulting rectangle is kept inside the
79    /// original rectangle and its extent is clamped to zero.
80    pub const fn inset_ltrb(self, left: Lunit, top: Lunit, right: Lunit, bottom: Lunit) -> Self {
81        let w = self.w().positive_part();
82        let h = self.h().positive_part();
83        let l = left.positive_part().min(w);
84        let t = top.positive_part().min(h);
85        let w_after_l = w.sub_floor_zero(l);
86        let h_after_t = h.sub_floor_zero(t);
87        let r = right.positive_part().min(w_after_l);
88        let b = bottom.positive_part().min(h_after_t);
89        Self::from_xy_wh(
90            self.x().sat_add(l),
91            self.y().sat_add(t),
92            w_after_l.sub_floor_zero(r),
93            h_after_t.sub_floor_zero(b),
94        )
95    }
96
97    /// Cuts a rectangle from the left side, returning `(taken, rest)`.
98    ///
99    /// Negative widths take nothing. Widths larger than the rectangle are
100    /// clamped to the rectangle width.
101    pub const fn cut_left(self, width: Lunit) -> (Self, Self) {
102        let w = self.w().positive_part();
103        let h = self.h().positive_part();
104        let used = width.positive_part().min(w);
105        let rest_w = w.sub_floor_zero(used);
106        let taken = Self::from_xy_wh(self.x(), self.y(), used, h);
107        let rest = Self::from_xy_wh(self.x().sat_add(used), self.y(), rest_w, h);
108        (taken, rest)
109    }
110    /// Cuts a rectangle from the right side, returning `(taken, rest)`.
111    ///
112    /// Negative widths take nothing. Widths larger than the rectangle are
113    /// clamped to the rectangle width.
114    pub const fn cut_right(self, width: Lunit) -> (Self, Self) {
115        let w = self.w().positive_part();
116        let h = self.h().positive_part();
117        let used = width.positive_part().min(w);
118        let rest_w = w.sub_floor_zero(used);
119        let rest = Self::from_xy_wh(self.x(), self.y(), rest_w, h);
120        let taken = Self::from_xy_wh(self.x().sat_add(rest_w), self.y(), used, h);
121        (taken, rest)
122    }
123    /// Cuts a rectangle from the top side, returning `(taken, rest)`.
124    ///
125    /// Negative heights take nothing. Heights larger than the rectangle are
126    /// clamped to the rectangle height.
127    pub const fn cut_top(self, height: Lunit) -> (Self, Self) {
128        let w = self.w().positive_part();
129        let h = self.h().positive_part();
130        let used = height.positive_part().min(h);
131        let rest_h = h.sub_floor_zero(used);
132        let taken = Self::from_xy_wh(self.x(), self.y(), w, used);
133        let rest = Self::from_xy_wh(self.x(), self.y().sat_add(used), w, rest_h);
134        (taken, rest)
135    }
136    /// Cuts a rectangle from the bottom side, returning `(taken, rest)`.
137    ///
138    /// Negative heights take nothing. Heights larger than the rectangle are
139    /// clamped to the rectangle height.
140    pub const fn cut_bottom(self, height: Lunit) -> (Self, Self) {
141        let w = self.w().positive_part();
142        let h = self.h().positive_part();
143        let used = height.positive_part().min(h);
144        let rest_h = h.sub_floor_zero(used);
145        let rest = Self::from_xy_wh(self.x(), self.y(), w, rest_h);
146        let taken = Self::from_xy_wh(self.x(), self.y().sat_add(rest_h), w, used);
147        (taken, rest)
148    }
149}

151#[cfg(test)]
152mod _test {
153    use super::*;
154
155    #[test]
156    fn edges() {
157        let r = UiRect::from_xy_wh(Lunit::px(10), Lunit::px(20), Lunit::px(30), Lunit::px(40));
158        assert_eq!(r.left(), Lunit::px(10));
159        assert_eq!(r.top(), Lunit::px(20));
160        assert_eq!(r.right(), Lunit::px(40));
161        assert_eq!(r.bottom(), Lunit::px(60));
162    }
163    #[test]
164    fn empty() {
165        assert!(UiRect::from_xy_wh(Lunit::ZERO, Lunit::ZERO, Lunit::ZERO, Lunit::px(1)).is_empty());
166        assert!(UiRect::from_xy_wh(Lunit::ZERO, Lunit::ZERO, Lunit::px(1), Lunit::ZERO).is_empty());
167        assert!(
168            !UiRect::from_xy_wh(Lunit::ZERO, Lunit::ZERO, Lunit::px(1), Lunit::px(1)).is_empty()
169        );
170    }
171    #[test]
172    fn inset() {
173        let r = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(100), Lunit::px(50));
174        let inner = r.inset(Lunit::px(10));
175        assert_eq!(inner.x(), Lunit::px(10));
176        assert_eq!(inner.y(), Lunit::px(10));
177        assert_eq!(inner.w(), Lunit::px(80));
178        assert_eq!(inner.h(), Lunit::px(30));
179    }
180    #[test]
181    fn over_inset_clamps_to_empty() {
182        let r = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(10), Lunit::px(10));
183        let inner = r.inset(Lunit::px(100));
184        assert!(inner.is_empty());
185        assert_eq!(inner.w(), Lunit::ZERO);
186        assert_eq!(inner.h(), Lunit::ZERO);
187    }
188    #[test]
189    fn cut_left() {
190        let r = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(100), Lunit::px(20));
191        let (left, rest) = r.cut_left(Lunit::px(30));
192        assert_eq!(left.x(), Lunit::px(0));
193        assert_eq!(left.w(), Lunit::px(30));
194        assert_eq!(rest.x(), Lunit::px(30));
195        assert_eq!(rest.w(), Lunit::px(70));
196    }
197    #[test]
198    fn cut_right() {
199        let r = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(100), Lunit::px(20));
200        let (right, rest) = r.cut_right(Lunit::px(30));
201        assert_eq!(right.x(), Lunit::px(70));
202        assert_eq!(right.w(), Lunit::px(30));
203        assert_eq!(rest.x(), Lunit::px(0));
204        assert_eq!(rest.w(), Lunit::px(70));
205    }
206    #[test]
207    fn over_cut_clamps() {
208        let r = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(10), Lunit::px(10));
209        let (taken, rest) = r.cut_left(Lunit::px(100));
210        assert_eq!(taken.w(), Lunit::px(10));
211        assert_eq!(rest.w(), Lunit::ZERO);
212    }
213    #[test]
214    fn cut_top() {
215        let r = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(100), Lunit::px(50));
216        let (top, rest) = r.cut_top(Lunit::px(20));
217        assert_eq!(top.y(), Lunit::px(0));
218        assert_eq!(top.h(), Lunit::px(20));
219        assert_eq!(rest.y(), Lunit::px(20));
220        assert_eq!(rest.h(), Lunit::px(30));
221    }
