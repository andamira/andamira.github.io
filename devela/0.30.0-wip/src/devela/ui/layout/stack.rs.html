1// devela/ui/layout/stack.rs

3//! Defines [`UiStack`].
4//
5// Boundary: UiStack is a local region-consumption helper,
6// not a layout tree or general layout engine.
7//
8
9use crate::{Boundary2d, Lunit, UiRect};
10
11#[doc = crate::_tags!(layout)]
12/// Cursor-like helper for sequentially taking rectangles from a region.
13#[doc = crate::_doc_meta!{
14    location("ui/layout", struct UiStack),
15    test_size_of(UiStack = 24|192; niche Option),
16}]
17#[must_use]
18#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
19pub struct UiStack {
20    rest: UiRect,
21    side: Boundary2d,
22    gap: Lunit,
23}
24#[rustfmt::skip]
25impl UiStack {
26    /* constructors */
27
28    /// Creates a stack that takes rectangles from `side`.
29    ///
30    /// Negative gaps are ignored.
31    pub const fn new(rest: UiRect, side: Boundary2d, gap: Lunit) -> Self {
32        Self { rest, side, gap: gap.positive_part() }
33    }
34    /// Creates a stack that takes from the top side and advances downward.
35    pub const fn down(rest: UiRect) -> Self {
36        Self::new(rest, Boundary2d::Top, Lunit::ZERO)
37    }
38    /// Creates a stack that takes from the bottom side and advances upward.
39    pub const fn up(rest: UiRect) -> Self {
40        Self::new(rest, Boundary2d::Bottom, Lunit::ZERO)
41    }
42    /// Creates a stack that takes from the left side and advances rightward.
43    pub const fn right(rest: UiRect) -> Self {
44        Self::new(rest, Boundary2d::Left, Lunit::ZERO)
45    }
46    /// Creates a stack that takes from the right side and advances leftward.
47    pub const fn left(rest: UiRect) -> Self {
48        Self::new(rest, Boundary2d::Right, Lunit::ZERO)
49    }
50
51    /* get */
52
53    /// Returns the remaining rectangle.
54    pub const fn rest(self) -> UiRect { self.rest }
55
56    /// Returns the side this stack takes from.
57    pub const fn side(self) -> Boundary2d { self.side }
58
59    /// Returns the gap consumed after successful takes.
60    pub const fn gap(self) -> Lunit { self.gap }
61
62    /// Returns this stack with a different gap.
63    ///
64    /// Negative gaps are ignored.
65    pub const fn with_gap(self, gap: Lunit) -> Self {
66        Self { rest: self.rest, side: self.side, gap: gap.positive_part() }
67    }
68
69    /* take */
70
71    /// Takes the next rectangle from the configured side.
72    pub const fn take(&mut self, extent: Lunit) -> UiRect {
73        match self.side {
74            Boundary2d::Left => self.take_left(extent),
75            Boundary2d::Right => self.take_right(extent),
76            Boundary2d::Top => self.take_top(extent),
77            Boundary2d::Bottom => self.take_bottom(extent),
78        }
79    }
80    /// Takes a rectangle from the left side.
81    pub const fn take_left(&mut self, width: Lunit) -> UiRect {
82        let (taken, rest) = self.rest.cut_left(width);
83        self.rest = rest;
84        if !taken.is_empty() { self.skip_left(self.gap); }
85        taken
86    }
87    /// Takes a rectangle from the right side.
88    pub const fn take_right(&mut self, width: Lunit) -> UiRect {
89        let (taken, rest) = self.rest.cut_right(width);
90        self.rest = rest;
91        if !taken.is_empty() { self.skip_right(self.gap); }
92        taken
93    }
94    /// Takes a rectangle from the top side.
95    pub const fn take_top(&mut self, height: Lunit) -> UiRect {
96        let (taken, rest) = self.rest.cut_top(height);
97        self.rest = rest;
98        if !taken.is_empty() { self.skip_top(self.gap); }
99        taken
100    }
101    /// Takes a rectangle from the bottom side.
102    pub const fn take_bottom(&mut self, height: Lunit) -> UiRect {
103        let (taken, rest) = self.rest.cut_bottom(height);
104        self.rest = rest;
105        if !taken.is_empty() { self.skip_bottom(self.gap); }
106        taken
107    }
108    /// Takes all remaining space.
109    pub const fn take_rest(&mut self) -> UiRect {
110        let rest = self.rest;
111        self.rest = UiRect::from_xy_wh(rest.x(), rest.y(), Lunit::ZERO, Lunit::ZERO);
112        rest
113    }
114
115    /* discard */
116
117    /// Discards space from the left side.
118    pub const fn skip_left(&mut self, width: Lunit) {
119        let (_, rest) = self.rest.cut_left(width);
120        self.rest = rest;
121    }
122    /// Discards space from the right side.
123    pub const fn skip_right(&mut self, width: Lunit) {
124        let (_, rest) = self.rest.cut_right(width);
125        self.rest = rest;
126    }
127    /// Discards space from the top side.
128    pub const fn skip_top(&mut self, height: Lunit) {
129        let (_, rest) = self.rest.cut_top(height);
130        self.rest = rest;
131    }
132    /// Discards space from the bottom side.
133    pub const fn skip_bottom(&mut self, height: Lunit) {
134        let (_, rest) = self.rest.cut_bottom(height);
135        self.rest = rest;
136    }
137}

139#[cfg(test)]
140mod _test {
141    use super::*;
142    #[test]
143
144    fn take_top_with_gap() {
145        let root = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(100), Lunit::px(100));
146        let mut stack = UiStack::down(root).with_gap(Lunit::px(5));
147        let title = stack.take_top(Lunit::px(20));
148        let rest = stack.rest();
149        assert_eq!(title.y(), Lunit::px(0));
150        assert_eq!(title.h(), Lunit::px(20));
151        assert_eq!(rest.y(), Lunit::px(25));
152        assert_eq!(rest.h(), Lunit::px(75));
153    }
154    #[test]
155    fn take_uses_configured_side() {
156        let root = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(100), Lunit::px(100));
157        let mut stack = UiStack::new(root, Boundary2d::Left, Lunit::px(2));
158        let sidebar = stack.take(Lunit::px(30));
159        let rest = stack.rest();
160        assert_eq!(sidebar.x(), Lunit::px(0));
161        assert_eq!(sidebar.w(), Lunit::px(30));
162        assert_eq!(rest.x(), Lunit::px(32));
163        assert_eq!(rest.w(), Lunit::px(68));
164    }
165    #[test]
166    fn empty_take_does_not_consume_gap() {
167        let root = UiRect::from_xy_wh(Lunit::px(0), Lunit::px(0), Lunit::px(100), Lunit::px(100));
168        let mut stack = UiStack::new(root, Boundary2d::Top, Lunit::px(5));
169        let taken = stack.take(Lunit::ZERO);
170        let rest = stack.rest();
171        assert!(taken.is_empty());
172        assert_eq!(rest.y(), Lunit::px(0));
173        assert_eq!(rest.h(), Lunit::px(100));
174    }
175}
