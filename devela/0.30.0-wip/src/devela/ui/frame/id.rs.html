3//! Defines [`UiId`], [`UiKey`], [`UiScope`].
4//!
5//! UI identities intentionally reserve `u64::MAX` so `Option<UiKey>` and
6//! `Option<UiId>` retain a niche. Their ergonomic constructors project that
7//! single input into the usable identity domain, while their [`WordTry`]
8//! representations remain exact and reject the reserved raw value.
9//
10// Boundary: keys are author-local inputs; resolved ids are the tokens shared with
11// layout, routing, semantics, widgets, and presentation.
12//
13
14use crate::{InvalidValue, NonMaxU64, SplitMix64, unwrap, word};
15
16#[doc = crate::_tags!(ui uid)]
17/// Stable author-provided UI identity seed.
18#[doc = crate::_doc_meta!{
19    location("ui/frame", struct UiKey),
20    test_size_of(UiKey = 8|64; niche Option),
21}]
22#[must_use]
23#[repr(transparent)]
24#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
25pub struct UiKey(NonMaxU64);
26impl UiKey {
27    /// The root identity seed.
28    pub const ROOT: Self = Self(unwrap![some_guaranteed_or_ub NonMaxU64::new(0)]);
29
30    /// Constructs a key from an integer seed.
31    ///
32    /// `u64::MAX` is reserved to preserve the `Option<UiKey>` niche
33    /// and is mapped to `u64::MAX - 1`.
34    pub const fn new(raw: u64) -> Self {
35        Self(NonMaxU64::new_lossy(raw))
36    }
37}
38word! {
39    impl UiKey => u64 {
40        type Error = InvalidValue;
41        raw(key) { key.0.get() }
42        try_from_raw(raw) { Ok(Self(unwrap![some_ok_or? NonMaxU64::new(raw), InvalidValue])) }
43    }
44}

76#[doc = crate::_tags!(ui uid)]
77/// Scoped UI identity namespace.
78#[doc = crate::_doc_meta!{
79    location("ui/frame", struct UiScope),
80    test_size_of(UiScope = 8|64; niche Option),
81}]
82#[must_use]
83#[repr(transparent)]
84#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
85pub struct UiScope {
86    id: UiId,
87}
88impl UiScope {
89    /// The root identity scope.
90    pub const ROOT: Self = Self { id: UiId::ROOT };
91
92    /// Constructs a scope from a resolved id.
93    pub const fn new(id: UiId) -> Self {
94        Self { id }
95    }
96    /// Returns the resolved id of this scope.
97    pub const fn id(self) -> UiId {
98        self.id
99    }
100    /// Resolves a key inside this scope.
101    pub const fn resolve(self, key: UiKey) -> UiId {
102        UiId::new(mix_u64(self.id.raw(), key.raw()))
103    }
104    /// Enters a child scope resolved from `key`.
105    pub const fn enter(self, key: UiKey) -> Self {
106        Self::new(self.resolve(key))
107    }
108}
109
110/// Mixes a scope id and key into a resolved id.
111//
112// NOTES
113// - `UiId::ROOT` stays the literal root id.
114// - `resolve()` derives a child id from `(scope, key)`, so resolving
115//   `UiKey::ROOT` inside `UiScope::ROOT` should not return `UiId::ROOT`.
116// - `GOLDEN_GAMMA` prevents the all-zero pair from mixing back to zero.
117const fn mix_u64(scope: u64, key: u64) -> u64 {
118    SplitMix64::mix64(scope ^ key.wrapping_add(SplitMix64::GOLDEN_GAMMA))
119}

121#[cfg(test)]
122mod _test {
123    use super::*;
124
125    #[test]
126    fn key_and_id_preserve_raw_values_except_max() {
127        assert_eq!(UiKey::new(0).raw(), 0);
128        assert_eq!(UiId::new(0).raw(), 0);
129        assert_eq!(UiKey::new(u64::MAX).raw(), u64::MAX - 1);
130        assert_eq!(UiId::new(u64::MAX).raw(), u64::MAX - 1);
131    }
132    #[test]
133    fn default_identity_values_are_root() {
134        assert_eq!(UiKey::default(), UiKey::ROOT);
135        assert_eq!(UiId::default(), UiId::ROOT);
136        assert_eq!(UiScope::default(), UiScope::ROOT);
137    }
138    #[test]
139    fn resolving_key_is_deterministic() {
140        let scope = UiScope::ROOT;
141        let key = UiKey::new(42);
142        assert_eq!(scope.resolve(key), scope.resolve(key));
143    }
144    #[test]
145    fn entering_scope_matches_resolved_id() {
146        let scope = UiScope::ROOT;
147        let key = UiKey::new(7);
148        assert_eq!(scope.enter(key).id(), scope.resolve(key));
149    }
150    #[test]
151    fn resolving_root_key_inside_root_scope_derives_child_id() {
152        assert_ne!(UiScope::ROOT.resolve(UiKey::ROOT), UiId::ROOT);
153    }
154}
