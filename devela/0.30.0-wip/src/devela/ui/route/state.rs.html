6use crate::UiId;

8macro_rules! define_ui_state {
9    ($($name:ident, $doc:literal, $det:literal, $adj:literal);+ $(;)?) => {
10        $( define_ui_state!(% $name, $doc, $det, $adj); )+
11    };
12    (% $name:ident, $doc:literal, $det:literal, $adj:literal) => {
13        #[doc = crate::_tags!(ui interaction)]
14        #[doc = $doc]
15        #[doc = crate::_doc_meta! {
16            location("ui/route", struct $name),
17            test_size_of($name = 8|64; niche !Option),
18        }]
19        #[must_use]
20        #[repr(transparent)]
21        #[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
22        pub struct $name(Option<UiId>);
23        impl $name {
24            #[doc = concat!("No ", $adj, " identity.")]
25            pub const NONE: Self = Self(None);
26
27            #[doc = concat!("Constructs ", $det, " ", $adj, " state from an optional identity.")]
28            pub const fn new(id: Option<UiId>) -> Self { Self(id) }
29            #[doc = concat!("Constructs ", $det, " ", $adj, " state with an identity.")]
30            pub const fn some(id: UiId) -> Self { Self(Some(id)) }
31
32            #[must_use]
33            #[doc = concat!("Returns the ", $adj, " identity.")]
34            pub const fn id(self) -> Option<UiId> { self.0 }
35
36            #[must_use]
37            #[doc = concat!("Returns whether `id` is the ", $adj, " identity.")]
38            pub const fn is(self, id: UiId) -> bool {
39                match self.0 {
40                    Some(hot) => hot.raw() == id.raw(),
41                    None => false,
42                }
43            }
44            #[must_use]
45            #[doc = concat!("Returns whether there is ", $det, " ", $adj, " identity.")]
46            pub const fn is_some(self) -> bool { self.0.is_some() }
47            #[doc = concat!("Returns whether there is no ", $adj, " identity.")]
48            #[must_use]
49            pub const fn is_none(self) -> bool { self.0.is_none() }
50        }
51    };
52}
53define_ui_state! {
54    RouteActive, "Current owner of an ongoing interaction.", "an", "active";
55    RouteCapture, "Current pointer-capture owner", "a", "captured";
56    RouteFocus, "Current keyboard or navigation focus owner.", "a", "focused";
57    RouteHot, "Current candidate under direct pointing interaction.", "a", "hot";
58}
