32/// The list is a backend-neutral display list in resolved logical UI space.
33///
34/// Records are presented from first to last. When records overlap, a later
35/// record appears in front of an earlier record. This painter ordering is
36/// authoritative: presenters must preserve it and must not independently
37/// sort records by operation, style, or backend representation.
38///
39/// `B` determines ownership and storage. It may contain a borrowed slice,
40/// fixed storage, an allocated vector, or another contiguous draw sequence.
41#[must_use]
42#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
43pub struct UiDrawList<S, T, B> {
44    draws: B,
45    _marker: PhantomData<fn() -> UiDraw<S, T>>,
46}

48impl<S, T, const LEN: usize> UiDrawList<S, T, [UiDraw<S, T>; LEN]> {
49    /// Constructs a draw list from an array.
50    pub const fn from_array(draws: [UiDraw<S, T>; LEN]) -> Self {
51        Self::from_storage(draws)
52    }
53}

55    /// Constructs a borrowed draw list from a slice.
56    pub const fn from_slice(draws: &'a [UiDraw<S, T>]) -> Self {

58    }

61#[rustfmt::skip]
62impl<S, T, B> UiDrawList<S, T, B> {
63    /// Constructs a draw list over the given storage.
64    pub const fn from_storage(draws: B) -> Self { Self { draws, _marker: PhantomData } }
65
66    /// Returns the underlying storage.
67    pub const fn storage(&self) -> &B { &self.draws }
68    /// Returns the underlying storage as an exclusive reference.
69    pub const fn storage_mut(&mut self) -> &mut B { &mut self.draws }
70
71    /// Consumes the list and returns its storage.
72    pub fn into_storage(self) -> B { self.draws }
73}

75#[rustfmt::skip]
76impl<S, T, B: AsRef<[UiDraw<S, T>]>> UiDrawList<S, T, B> {
77    /// Returns the number of draw records.
78    pub fn len(&self) -> usize { self.as_slice().len() }
79
80    /// Returns whether the draw list is empty.
81    pub fn is_empty(&self) -> bool { self.as_slice().is_empty() }
82
83    /// Returns the ordered draw records.
84    pub fn as_slice(&self) -> &[UiDraw<S, T>] { self.draws.as_ref() }
85
86    /// Returns a borrowed view preserving this list's draw order.
87    pub fn as_view(&self) -> UiDrawListView<'_, S, T> { UiDrawList::from_storage(self.as_slice()) }
88
89    /// Iterates over draw records in painter order, from back to front.
90    pub fn iter(&self) -> ::core::slice::Iter<'_, UiDraw<S, T>> {
91        self.as_slice().iter()
92    }
93}

97impl<S, T> UiDrawList<S, T, Vec<UiDraw<S, T>>> {
98    /// Constructs an empty allocating draw list.
99    pub const fn new() -> Self { Self::from_storage(Vec::new()) }
100
101    /// Constructs an empty allocating draw list with the given capacity.
102    pub fn with_capacity(capacity: usize) -> Self {
103        Self::from_storage(Vec::with_capacity(capacity))
104    }
105
106    /// Appends a draw record.
107    pub fn push(&mut self, draw: UiDraw<S, T>) { self.draws.push(draw); }
108
109    /// Clears all draw records while retaining allocated capacity.
110    pub fn clear(&mut self) { self.draws.clear(); }
111
112    /// Returns the draw capacity.
113    pub fn capacity(&self) -> usize { self.draws.capacity() }
114}

116#[cfg(test)]
117mod list_tests {
118    use super::*;
119    use crate::UiRect;
120
121    #[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
122    enum Style {
123        Back,
124        Middle,
125        Front,
126    }
127    fn draw(style: Style) -> UiDraw<Style, &'static str> {
128        UiDraw::rect_fill(UiRect::default(), style)
129    }
130
131    #[test]
132    fn preserves_painter_order() {
133        let draws = [draw(Style::Back), draw(Style::Middle), draw(Style::Front)];
134        let list = UiDrawList::from_storage(draws.as_slice());
135        let mut iter = list.iter();
136        assert_eq!(iter.next().unwrap().kind().style(), &Style::Back);
137        assert_eq!(iter.next().unwrap().kind().style(), &Style::Middle);
138        assert_eq!(iter.next().unwrap().kind().style(), &Style::Front);
139        assert!(iter.next().is_none());
140    }
141    #[test]
142    fn borrowed_view_preserves_storage_and_order() {
143        let draws = [draw(Style::Back), draw(Style::Middle), draw(Style::Front)];
144        let list = UiDrawList::from_storage(draws);
145        let view = list.as_view();
146        assert_eq!(view.len(), 3);
147        assert_eq!(view.as_slice(), list.as_slice());
148        // Confirms that the view points at the same underlying sequence.
149        assert!(core::ptr::eq(view.as_slice().as_ptr(), list.as_slice().as_ptr(),));
150    }
151    #[test]
152    fn empty_and_length_queries() {
153        let empty: &[UiDraw<Style, &str>] = &[];
154        let empty = UiDrawList::from_storage(empty);
155        assert!(empty.is_empty());
156        assert_eq!(empty.len(), 0);
157        let draws = [draw(Style::Back), draw(Style::Front)];
158        let list = UiDrawList::from_storage(draws.as_slice());
159        assert!(!list.is_empty());
160        assert_eq!(list.len(), 2);
161    }
162    #[test]
163    fn into_storage_returns_original_storage() {
164        let draws = [draw(Style::Back), draw(Style::Front)];
165        let list = UiDrawList::from_array(draws);
166        assert_eq!(list.into_storage(), draws);
167    }
