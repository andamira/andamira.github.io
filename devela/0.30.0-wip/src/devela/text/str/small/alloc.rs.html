10#[doc = crate::_tags!(string)]
11/// A UTF-8 string with inline storage and heap spillover.
12#[doc = crate::_doc_meta!{
13    location("text/str", struct StringSmallAlloc),
14    #[cfg(target_pointer_width = "32")]
15    test_size_of(StringSmallAlloc = 12|96; niche Option),
16    #[cfg(target_pointer_width = "64")]
17    test_size_of(StringSmallAlloc = 24|192; niche Option),

19/// Stores strings of up to `CAP` bytes inline,
20/// and spills to the heap when more capacity is needed.
21///
22/// The default capacity, `23`, matches the usual inline payload target for
23/// a `String`-sized compact string on 64-bit platforms.
24#[must_use]
25#[derive(Clone)]
26pub struct StringSmallAlloc<const CAP: usize = STRING_SMALL_ALLOC_DEFAULT_CAP> {
27    repr: StringSmallAllocRepr<CAP>,
28}
29#[derive(Clone)]
30enum StringSmallAllocRepr<const CAP: usize> {
31    Inline(StringU8<CAP>),
32    Heap(String),
33}
34
35impl<const CAP: usize> StringSmallAlloc<CAP> {

41    /// The default inline byte capacity.
42    ///
43    /// Chosen so the safe enum-backed `StringSmallAlloc`
44    /// occupies the same stack size as `String`.
45    pub const DEFAULT_CAPACITY: usize = STRING_SMALL_ALLOC_DEFAULT_CAP;

47    /// Inline capacity that makes `StringU8<CAP>` the same size as `String`.
48    // (23 in 64-bit, 11 in 32-bit, 5 in 16 bit)
49    pub const STRING_INLINE_CAPACITY: usize = size_of::<String>() - 1;

51    /// Inline capacity that usually makes this safe enum-backed type
52    /// the same size as `String`.
53    // (15 in 64-bit, 7 in 32-bit, 3 in 16 bit)
54    pub const STRING_SIZED_CAPACITY: usize = 2 * size_of::<usize>() - 1;

58    /// Creates a new empty string.
59    ///
60    /// # Panics
61    /// Panics if `CAP > StringU8::MAX_CAPACITY`.
62    pub const fn new() -> Self {
63        Self {
64            repr: StringSmallAllocRepr::Inline(StringU8::<CAP>::new()),
65        }
66    }
67    /// Creates a new empty string.
68    ///
69    /// # Errors
70    /// Returns [`MismatchedCapacity`] if `CAP > StringU8::MAX_CAPACITY`.
71    pub fn new_checked() -> Result<Self, MismatchedCapacity> {
72        Ok(Self {
73            repr: StringSmallAllocRepr::Inline(StringU8::<CAP>::new_checked()?),
74        })
75    }

77    /// Creates a new string from `string`.
78    ///
79    /// Uses inline storage when `string.len() <= CAP`, otherwise spills to the heap.
80    ///
81    /// # Errors
82    /// Returns [`MismatchedCapacity`] if `CAP > StringU8::MAX_CAPACITY`.
83    #[allow(clippy::should_implement_trait, reason = "it does as well")]
84    pub fn from_str(string: &str) -> Result<Self, MismatchedCapacity> {
85        let _ = StringU8::<CAP>::new_checked()?;
86        if string.len() <= CAP {
87            Ok(Self {
88                repr: StringSmallAllocRepr::Inline(StringU8::<CAP>::from_str(string)?),
89            })
90        } else {
91            Ok(Self {
92                repr: StringSmallAllocRepr::Heap(String::from(string)),
93            })
94        }
95    }
96
97    /// Creates a new string from `string`.
98    ///
99    /// Reuses `string` directly when it does not fit inline.
100    ///
101    /// # Errors
102    /// Returns [`MismatchedCapacity`] if `CAP > StringU8::MAX_CAPACITY`.
103    pub fn from_string(string: String) -> Result<Self, MismatchedCapacity> {
104        let _ = StringU8::<CAP>::new_checked()?;
105        if string.len() <= CAP {
106            Ok(Self {
107                repr: StringSmallAllocRepr::Inline(StringU8::<CAP>::from_str(string.as_str())?),
108            })
109        } else {
110            Ok(Self { repr: StringSmallAllocRepr::Heap(string) })
111        }
112    }

148    /// Returns the current active capacity in bytes.

168    /// Returns the current remaining active capacity in bytes.
169    #[must_use]
170    pub fn remaining_capacity(&self) -> usize {
171        self.capacity().saturating_sub(self.len())
172    }

194    /// Clears the string.
195    ///
196    /// If the string has spilled to the heap, the heap allocation is retained.
197    pub fn clear(&mut self) {
198        match &mut self.repr {
199            StringSmallAllocRepr::Inline(s) => s.clear(),
200            StringSmallAllocRepr::Heap(s) => s.clear(),
201        }
202    }

204    /// Truncates the string to `new_len` bytes.
205    ///
206    /// Does nothing if `new_len >= self.len()`.
207    ///
208    /// # Panics
209    /// Panics if `new_len` is not a UTF-8 boundary.
210    pub fn truncate(&mut self, new_len: usize) {
211        if new_len >= self.len() {
212            return;
213        }
214        assert![self.as_str().is_char_boundary(new_len), "not a UTF-8 boundary"];
215
216        match &mut self.repr {
217            StringSmallAllocRepr::Inline(s) => {
218                while s.len() > new_len {
219                    let _ = s.pop();
220                }
221            }
222            StringSmallAllocRepr::Heap(s) => s.truncate(new_len),
223        }
224    }
225
226    /// Reserves capacity for at least `additional` more bytes.
227    ///
228    /// Spills to the heap if the current inline storage is not enough.
229    pub fn reserve(&mut self, additional: usize) {
230        if additional <= self.remaining_capacity() {
231            return;
232        }
233
234        match &mut self.repr {
235            StringSmallAllocRepr::Inline(s) => {
236                let mut heap = String::with_capacity(s.len() + additional);
237                heap.push_str(s.as_str());
238                self.repr = StringSmallAllocRepr::Heap(heap);
239            }
240            StringSmallAllocRepr::Heap(s) => s.reserve(additional),
241        }
242    }
243
244    /// Moves the string back inline if it fits.
245    ///
246    /// Returns `true` if the string is inline after the call.
247    pub fn shrink_to_inline(&mut self) -> bool {
248        match &self.repr {
249            StringSmallAllocRepr::Inline(_) => true,
250            StringSmallAllocRepr::Heap(s) if s.len() <= CAP => {
251                let inline = StringU8::<CAP>::from_str(s.as_str())
252                    .expect("validated StringSmallAlloc inline capacity");
253                self.repr = StringSmallAllocRepr::Inline(inline);
254                true
255            }
256            StringSmallAllocRepr::Heap(_) => false,
257        }
258    }
259
260    /// Shrinks heap allocation if spilled, or moves back inline if possible.
261    pub fn shrink_to_fit(&mut self) {
262        if self.shrink_to_inline() {
263            return;
264        }
265        if let StringSmallAllocRepr::Heap(s) = &mut self.repr {
266            s.shrink_to_fit();
267        }
268    }

270    /// Removes the last character and returns it.
271    #[must_use]
272    pub fn pop(&mut self) -> Option<char> {
273        match &mut self.repr {
274            StringSmallAllocRepr::Inline(s) => s.pop(),
275            StringSmallAllocRepr::Heap(s) => s.pop(),
276        }
277    }
278
279    /// Tries to remove the last character and return it.
280    ///
281    /// # Errors
282    /// Returns [`NotEnoughElements`] if the string is empty.
283    pub fn try_pop(&mut self) -> Result<char, NotEnoughElements> {
284        match self.pop() {
285            Some(c) => Ok(c),
286            None => Err(NotEnoughElements(Some(1))),
287        }
288    }
289
290    /// Appends `character`, spilling to the heap if needed.
291    ///
292    /// Returns the number of bytes written.
293    pub fn push(&mut self, character: char) -> usize {
294        let len = character.len_utf8();
295
296        match &mut self.repr {
297            StringSmallAllocRepr::Inline(s) => {
298                if s.try_push(character).is_ok() {
299                    len
300                } else {
301                    let mut heap = String::with_capacity(s.len() + len);
302                    heap.push_str(s.as_str());
303                    heap.push(character);
304                    self.repr = StringSmallAllocRepr::Heap(heap);
305                    len
306                }
307            }
308            StringSmallAllocRepr::Heap(s) => {
309                s.push(character);
310                len
311            }
312        }
313    }
314
315    /// Appends `string`, spilling to the heap if needed.
316    ///
317    /// Returns the number of bytes written.
318    pub fn push_str(&mut self, string: &str) -> usize {
319        if string.is_empty() {
320            return 0;
321        }
322
323        match &mut self.repr {
324            StringSmallAllocRepr::Inline(s) => {
325                if let Ok(n) = s.try_push_str_complete(string) {
326                    n
327                } else {
328                    let mut heap = String::with_capacity(s.len() + string.len());
329                    heap.push_str(s.as_str());
330                    heap.push_str(string);
331                    self.repr = StringSmallAllocRepr::Heap(heap);
332                    string.len()
333                }
334            }
335            StringSmallAllocRepr::Heap(s) => {
336                s.push_str(string);
337                string.len()
338            }
339        }
340    }
