9const STRING_SMALL_ALLOC_DEFAULT_CAP: usize = 2 * size_of::<usize>() - 1;

11#[doc = crate::_tags!(string)]
12/// A UTF-8 string with inline storage and heap spillover.
13#[doc = crate::_doc_meta!{
14    location("text/str", struct StringSmallAlloc),
15    #[cfg(target_pointer_width = "32")]
16    test_size_of(StringSmallAlloc = 12|96; niche Option),
17    #[cfg(target_pointer_width = "64")]
18    test_size_of(StringSmallAlloc = 24|192; niche Option),
19}]
20/// Stores strings of up to `CAP` bytes inline,
21/// and spills to the heap when more capacity is needed.
22///
23/// The default capacity, `23`, matches the usual inline payload target for
24/// a `String`-sized compact string on 64-bit platforms.
25#[must_use]
26#[derive(Clone)]
27pub struct StringSmallAlloc<const CAP: usize = STRING_SMALL_ALLOC_DEFAULT_CAP> {
28    repr: StringSmallAllocRepr<CAP>,
29}
30#[derive(Clone)]
31enum StringSmallAllocRepr<const CAP: usize> {
32    Inline(StringU8<CAP>),
33    Heap(String),
34}
35
36impl<const CAP: usize> StringSmallAlloc<CAP> {

38
39    /// The inline byte capacity of this type.
40    pub const INLINE_CAPACITY: usize = CAP;

42    /// The default inline byte capacity.
43    ///
44    /// Chosen so the safe enum-backed `StringSmallAlloc`
45    /// occupies the same stack size as `String`.
46    pub const DEFAULT_CAPACITY: usize = STRING_SMALL_ALLOC_DEFAULT_CAP;
47
48    /// Inline capacity that makes `StringU8<CAP>` the same size as `String`.
49    // (23 in 64-bit, 11 in 32-bit, 5 in 16 bit)
50    pub const STRING_INLINE_CAPACITY: usize = size_of::<String>() - 1;
51
52    /// Inline capacity that usually makes this safe enum-backed type
53    /// the same size as `String`.
54    // (15 in 64-bit, 7 in 32-bit, 3 in 16 bit)
55    pub const STRING_SIZED_CAPACITY: usize = 2 * size_of::<usize>() - 1;

59    /// Creates a new empty string.
60    ///
61    /// # Panics
62    /// Panics if `CAP > StringU8::MAX_CAPACITY`.
63    pub const fn new() -> Self {
64        Self {
65            repr: StringSmallAllocRepr::Inline(StringU8::<CAP>::new()),
66        }
67    }
68    /// Creates a new empty string.
69    ///
70    /// # Errors
71    /// Returns [`MismatchedCapacity`] if `CAP > StringU8::MAX_CAPACITY`.
72    pub fn new_checked() -> Result<Self, MismatchedCapacity> {
73        Ok(Self {
74            repr: StringSmallAllocRepr::Inline(StringU8::<CAP>::new_checked()?),
75        })
76    }
77
78    /// Creates a new string from `string`.
79    ///
80    /// Uses inline storage when `string.len() <= CAP`, otherwise spills to the heap.
81    ///
82    /// # Errors
83    /// Returns [`MismatchedCapacity`] if `CAP > StringU8::MAX_CAPACITY`.
84    #[allow(clippy::should_implement_trait, reason = "it does as well")]
85    pub fn from_str(string: &str) -> Result<Self, MismatchedCapacity> {
86        let _ = StringU8::<CAP>::new_checked()?;
87        if string.len() <= CAP {
88            Ok(Self {
89                repr: StringSmallAllocRepr::Inline(StringU8::<CAP>::from_str(string)?),
90            })
91        } else {
92            Ok(Self {
93                repr: StringSmallAllocRepr::Heap(String::from(string)),
94            })
95        }
96    }
97
98    /// Creates a new string from `string`.
99    ///
100    /// Reuses `string` directly when it does not fit inline.
101    ///
102    /// # Errors
103    /// Returns [`MismatchedCapacity`] if `CAP > StringU8::MAX_CAPACITY`.
104    pub fn from_string(string: String) -> Result<Self, MismatchedCapacity> {
105        let _ = StringU8::<CAP>::new_checked()?;
106        if string.len() <= CAP {
107            Ok(Self {
108                repr: StringSmallAllocRepr::Inline(StringU8::<CAP>::from_str(string.as_str())?),
109            })
110        } else {
111            Ok(Self { repr: StringSmallAllocRepr::Heap(string) })
112        }
113    }

149    /// Returns the current active capacity in bytes.
150    ///
151    /// This is `CAP` while inline, and the heap capacity after spilling.

153    pub fn capacity(&self) -> usize {

155            StringSmallAllocRepr::Inline(_) => CAP,
156            StringSmallAllocRepr::Heap(s) => s.capacity(),

163        match &self.repr {
164            StringSmallAllocRepr::Inline(s) => s.len(),
165            StringSmallAllocRepr::Heap(s) => s.len(),

169    /// Returns the current remaining active capacity in bytes.
170    #[must_use]
171    pub fn remaining_capacity(&self) -> usize {
172        self.capacity().saturating_sub(self.len())
173    }

175    /// Returns `true` if the string is empty.

178        self.len() == 0

185    }

188    #[must_use]
189    pub const fn is_heap(&self) -> bool {
190        matches!(self.repr, StringSmallAllocRepr::Heap(_))
191    }

195    /// Clears the string.
196    ///
197    /// If the string has spilled to the heap, the heap allocation is retained.
198    pub fn clear(&mut self) {
199        match &mut self.repr {
200            StringSmallAllocRepr::Inline(s) => s.clear(),
201            StringSmallAllocRepr::Heap(s) => s.clear(),
202        }
203    }
204
205    /// Truncates the string to `new_len` bytes.
206    ///
207    /// Does nothing if `new_len >= self.len()`.
208    ///
209    /// # Panics
210    /// Panics if `new_len` is not a UTF-8 boundary.
211    pub fn truncate(&mut self, new_len: usize) {
212        if new_len >= self.len() {
213            return;
214        }
215        assert![self.as_str().is_char_boundary(new_len), "not a UTF-8 boundary"];
216
217        match &mut self.repr {
218            StringSmallAllocRepr::Inline(s) => {
219                while s.len() > new_len {
220                    let _ = s.pop();
221                }
222            }
223            StringSmallAllocRepr::Heap(s) => s.truncate(new_len),
224        }
225    }
226
227    /// Reserves capacity for at least `additional` more bytes.
228    ///
229    /// Spills to the heap if the current inline storage is not enough.
230    pub fn reserve(&mut self, additional: usize) {
231        if additional <= self.remaining_capacity() {
232            return;
233        }
234
235        match &mut self.repr {
236            StringSmallAllocRepr::Inline(s) => {
237                let mut heap = String::with_capacity(s.len() + additional);
238                heap.push_str(s.as_str());
239                self.repr = StringSmallAllocRepr::Heap(heap);
240            }
241            StringSmallAllocRepr::Heap(s) => s.reserve(additional),
242        }
243    }
244
245    /// Moves the string back inline if it fits.
246    ///
247    /// Returns `true` if the string is inline after the call.
248    pub fn shrink_to_inline(&mut self) -> bool {
249        match &self.repr {
250            StringSmallAllocRepr::Inline(_) => true,
251            StringSmallAllocRepr::Heap(s) if s.len() <= CAP => {
252                let inline = StringU8::<CAP>::from_str(s.as_str())
253                    .expect("validated StringSmallAlloc inline capacity");
254                self.repr = StringSmallAllocRepr::Inline(inline);
255                true
256            }
257            StringSmallAllocRepr::Heap(_) => false,
258        }
259    }
260
261    /// Shrinks heap allocation if spilled, or moves back inline if possible.
262    pub fn shrink_to_fit(&mut self) {
263        if self.shrink_to_inline() {
264            return;
265        }
266        if let StringSmallAllocRepr::Heap(s) = &mut self.repr {
267            s.shrink_to_fit();
268        }
269    }
270
271    /// Removes the last character and returns it.
272    #[must_use]
273    pub fn pop(&mut self) -> Option<char> {
274        match &mut self.repr {
275            StringSmallAllocRepr::Inline(s) => s.pop(),
276            StringSmallAllocRepr::Heap(s) => s.pop(),
277        }
278    }
279
280    /// Tries to remove the last character and return it.
281    ///
282    /// # Errors
283    /// Returns [`NotEnoughElements`] if the string is empty.
284    pub fn try_pop(&mut self) -> Result<char, NotEnoughElements> {
285        match self.pop() {
286            Some(c) => Ok(c),
287            None => Err(NotEnoughElements(Some(1))),
288        }
289    }
290
291    /// Appends `character`, spilling to the heap if needed.
292    ///
293    /// Returns the number of bytes written.
294    pub fn push(&mut self, character: char) -> usize {
295        let len = character.len_utf8();
296
297        match &mut self.repr {
298            StringSmallAllocRepr::Inline(s) => {
299                if s.try_push(character).is_ok() {
300                    len
301                } else {
302                    let mut heap = String::with_capacity(s.len() + len);
303                    heap.push_str(s.as_str());
304                    heap.push(character);
305                    self.repr = StringSmallAllocRepr::Heap(heap);
306                    len
307                }
308            }
309            StringSmallAllocRepr::Heap(s) => {
310                s.push(character);
311                len
312            }
313        }
314    }
315
316    /// Appends `string`, spilling to the heap if needed.
317    ///
318    /// Returns the number of bytes written.
319    pub fn push_str(&mut self, string: &str) -> usize {
320        if string.is_empty() {
321            return 0;
322        }
323
324        match &mut self.repr {
325            StringSmallAllocRepr::Inline(s) => {
326                if let Ok(n) = s.try_push_str_complete(string) {
327                    n
328                } else {
329                    let mut heap = String::with_capacity(s.len() + string.len());
330                    heap.push_str(s.as_str());
331                    heap.push_str(string);
332                    self.repr = StringSmallAllocRepr::Heap(heap);
333                    string.len()
334                }
335            }
336            StringSmallAllocRepr::Heap(s) => {
337                s.push_str(string);
338                string.len()
339            }
340        }
341    }

432        #[cfg(target_pointer_width = "64")]

434            assert_eq![size_of::<String>(), 24];
435            assert_eq![size_of::<Option<Option<Option<Option<Option<String>>>>>>(), 24];
