11/// Provides stateless, non-allocating operations over valid UTF-8 string slices.
12///
13/// - For scalar-level Unicode operations see [`Char`][crate::Char].
14/// - For cursor-based parsing see [`TextScanner`][crate::TextScanner].
15///
16/// # Methods
17///
18/// - [UTF-8 conversion methods](#utf-8-conversion-methods)
19///   - [from_utf8](#method.from_utf8) ([*mut*](#method.from_utf8_mut),
20///     [*unchecked*](#method.from_utf8_unchecked),
21///     [*unchecked_mut*](#method.from_utf8_unchecked_mut)).
22///
23/// - [text traversal methods](#text-traversal-methods)
24///   - [chars](#method.chars),
25///   - [char_count](#method.char_count),
26///   - [graphemes](#method.graphemes), ([*in*](#method.graphemes_in)),
27///     [*charu*](#method.graphemes_charu), [*charu_in*](#method.graphemes_charu_in)),
28///   - [grapheme_count](#method.grapheme_count).
29///
30/// - [equality and boundary methods](#equality-and-boundary-methods)
31///   - [eq](#method.eq),
32///   - [starts_with](#method.starts_with) ([*char*](#method.starts_with_char)),
33///   - [ends_with](#method.ends_with) ([*char*](#method.ends_with_char)),
34///   - [strip_prefix](#method.strip_prefix) ([*char*](#method.strip_prefix_char)),
35///   - [strip_suffix](#method.strip_suffix) ([*char*](#method.strip_suffix_char)),
36///   - [strip_circumfix](#method.strip_circumfix) ([*chars*](#method.strip_circumfix_chars)).
37///
38/// - [text writing methods](#text-writing-methods)
39///   - [repeat_into](#method.repeat_into) ([*slice*](#method.repeat_into_slice)),
40///   - [new_counter](#method.new_counter).

42///     ([*or*](#method.translit_ascii_into_or)).
43///
44/// - [`range*` API methods](#range-api-methods-for-returning-substrings):<br/>
45///   - [range](#method.range)
46///    ([*checked*](#method.range_checked),
47///     [_mut_](#method.range_mut),
48///     [*mut_checked*](#method.range_mut_checked)),                     ≈ `&str[start..end]`
49///   - [range_inclusive](#method.range_inclusive)
50///    ([*checked*](#method.range_inclusive_checked),
51///     [_mut_](#method.range_inclusive_mut),
52///     [*mut_checked*](#method.range_inclusive_mut_checked)),           ≈ `&str[start..=end]`
53///   - [range_from](#method.range_from),
54///    ([*checked*](#method.range_from_checked),
55///     [_mut_](#method.range_from_mut),
56///     [*mut_checked*](#method.range_from_mut_checked)),                ≈ `&str[start..]`
57///   - [range_to](#method.range_to)
58///    ([*checked*](#method.range_to_checked),
59///     [_mut_](#method.range_to_mut),
60///     [*mut_checked*](#method.range_to_mut_checked)),                  ≈ `&str[..end]`
61///   - [range_to_inclusive](#method.range_to_inclusive)
62///    ([*checked*](#method.range_to_inclusive_checked),
63///     [_mut_](#method.range_to_inclusive_mut),
64///     [*mut_checked*](#method.range_to_inclusive_mut_checked)).        ≈ `&str[..=end]`
65///
66/// - [`take*` API methods](#take-api-methods-for-subslicing):<br/>
67///   - [take_first](#method.take_first)
68///    ([*checked*](#method.take_first_checked),
69///     [_mut_](#method.take_first_mut),
70///     [*mut_checked*](#method.take_first_mut_checked)),                ≈ `&str[..n]`

75///   - [take_omit_last](#method.take_omit_last)
76///    ([*checked*](#method.take_omit_last_checked),
77///     [_mut_](#method.take_omit_last_mut),
78///     [*mut_checked*](#method.take_omit_last_mut_checked)).            ≈ `&str[..len - n]`
79///
80/// - [`*split*` API methods](#split-api-methods-for-returning-substrings):<br/>
81///   - [lsplit](#method.lsplit)
82///    ([*checked*](#method.lsplit_checked),
83///    [mut](#method.lsplit_mut),
84///    [*mut_checked*](#method.lsplit_mut_checked)),
85///   - [rsplit](#method.rsplit)
86///    ([*checked*](#method.rsplit_checked),
87///    [*mut*](#method.rsplit_mut),
88///    [*mut_checked*](#method.rsplit_mut_checked)),
89///   - [msplit_left](#method.msplit_left)
90///    ([*checked*](#method.msplit_left_checked),
91///    [*mut*](#method.msplit_left_mut),
92///    [*mut_checked*](#method.msplit_left_mut_checked)),
93///   - [msplit_right](#method.msplit_right)
94///    ([*checked*](#method.msplit_right_checked),
95///    [*mut*](#method.msplit_right_mut).
96///    [*mut_checked*](#method.msplit_right_mut_checked)).
