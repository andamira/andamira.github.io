20    /// Repeats a `string` a given number of times into the provided `buffer`.
21    /// and returns a reference to the new `&str`.
22    /// # Examples
23    /// ```
24    /// use devela::Str;
25    ///
26    /// let mut buf = [0_u8; 12];
27    /// let repeated = Str::repeat_into("ay", 3, &mut buf);
28    /// assert_eq![repeated, "ayayay"];
29    /// ```
30    /// # Features
31    /// Makes use of the `unsafe_str` feature if enabled.
32    ///
33    /// See also [`StrExt::repeat_into`], which should be faster, because it uses `copy_from_slice`.
34    ///
35    /// [`StrExt::repeat_into`]: crate::StrExt::repeat_into
36    #[must_use]
37    pub const fn repeat_into<'input, const CAP: usize>(
38        string: &str,
39        n: usize,
40        buffer: &'input mut [u8; CAP],
41    ) -> &'input str {
42        Self::repeat_into_slice(string, n, buffer)
43    }

45    /// Repeats `string` up to `n` times into `buffer`,
46    /// stopping early if it does not fit,
47    /// and returns the written prefix as `&str`.
48    ///
49    /// Like [`repeat_into`][Self::repeat_into], but accepts a dynamically sized buffer.
50    ///
51    /// # Features
52    /// Makes use of the `unsafe_str` feature if enabled.
53    #[must_use]
54    pub const fn repeat_into_slice<'input>(
55        string: &str,
56        n: usize,
57        buffer: &'input mut [u8],
58    ) -> &'input str {
59        let s_bytes = string.as_bytes();
60        let s_len = s_bytes.len();
61        let mut index = 0;
62        whilst! { count in 0..n; {
63            is![index + s_len > buffer.len(), break];
64            // buffer[index..index + s_len].copy_from_slice(s_bytes);
65            slice![mut buffer, index, ..index + s_len].copy_from_slice(s_bytes);
66            index += s_len;
67        }}
68        cfg_select! { all(feature = "unsafe_str", not(feature = "safe_text")) => {
69            // SAFETY: since `string` is a valid &str, checks are unneeded.
70            unsafe { Str::from_utf8_unchecked(slice![buffer, ..index]) }
71        } _ => { unwrap![ok Str::from_utf8(slice![buffer, ..index])] }}
72    }

76    /// Returns a [`&str`] backed by a `buffer`, where you always know each
77    /// character's position.
78    ///
79    /// A [*counter string*][0] is a graduated string of arbitrary `length`,
80    /// with a `separator` positioned after the immediately preceding number.
81    /// # Examples
82    /// ```
83    /// use devela::Str;
84    ///
85    /// let mut buf = [0; 15];
86    /// assert_eq!("2*4*6*8*11*14*", Str::new_counter(&mut buf, 14, '*'));
87    /// assert_eq!("_3_5_7_9_12_15_", Str::new_counter(&mut buf, 15, '_'));
88    /// ```
89    /// # Panics
90    /// Panics if `buffer.len() < length`, or if `!separator.is_ascii()`.
91    ///
92    /// # Features
93    /// Makes use of the `unsafe_str` feature if enabled.
94    ///
95    /// See also [`StrExt::new_counter`].
96    ///
97    /// [`StrExt::new_counter`]: crate::StrExt::new_counter
98    /// [0]: https://www.satisfice.com/blog/archives/22
99    pub const fn new_counter(buffer: &mut [u8], length: usize, separator: char) -> &str {
100        assert![buffer.len() >= length];
101        assert![separator.is_ascii()];
102        if length == 0 {
103            Str::new_cold_empty()
104        } else {
105            let separator = separator as u8;
106            let mut index = length - 1; // start writing from the end
107            let mut num = length; // the first number to write is the length
108            let mut separator_turn = true; // start writing the separator
109            let mut digits = Digits(num).count_digits10() as usize;
110            loop {
111                if separator_turn {
112                    buffer[index] = separator;
113                } else {
114                    let start = index + 1 - digits;
115                    let _ = Digits(num).write_digits10(buffer, start);
116                    index = start;
117                    num = index;
118                    digits = Digits(num).count_digits10() as usize;
119                }
120                is![index == 0, break, index -= 1];
121                separator_turn = !separator_turn;
122            }
123            cfg_select! { all(feature = "unsafe_str", not(feature = "safe_text")) => {
124                // SAFETY: only ASCII bytes are written
125                unsafe { Str::from_utf8_unchecked(slice![buffer, ..length]) }
126            } _ => { unwrap![ok Str::from_utf8(slice![buffer, ..length])] }}
127        }
128    }

132    /// Writes an ASCII transliteration of `string` into `dst`.
133    ///
134    /// Unmapped scalars are omitted.
135    ///
136    /// Returns the number of bytes written, or `None` if `dst` is too small.
137    #[cfg(feature = "translit")]
138    pub fn translit_ascii_into(string: &str, dst: &mut [u8]) -> Option<usize> {
139        Translit::write_ascii(string, dst)
140    }
