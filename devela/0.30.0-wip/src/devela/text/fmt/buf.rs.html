15#[doc = crate::_tags!(fmt)]
16/// Returns a formatted [`str`] slice backed by a buffer, non-allocating.
17#[doc = crate::_doc_meta!{
18    location("text/fmt", macro format_buf),
19}]
20/// This is implemented via [`FmtWriter::format`] and [`format_args!`][crate::format_args].
21///
22/// # Examples
23/// ```
24/// use devela::format_buf;
25///
26/// let mut buf = [0u8; 12];
27/// let s = format_buf![&mut buf, "Test: {} {}", "foo", 4];
28/// assert_eq!(Ok("Test: foo 4"), s);
29///
30/// // use the `?` arm to unwrap the result even if it's truncated
31/// let s = format_buf![? &mut buf, "Test: {} {}", "foo", 400_000];
32/// assert_eq!("Test: foo 40", s);
33/// ```
34/// # Features
35/// Makes use of the `unsafe_str` feature if enabled.
36#[macro_export]
37#[cfg_attr(cargo_primary_package, doc(hidden))]
38macro_rules! format_buf· {
39    ($buf:expr, $($arg:tt)*) => {
40        $crate::FmtWriter::format($buf, $crate::format_args![$($arg)*])
41    };
42    (? $buf:expr, $($arg:tt)*) => {
43        $crate::unwrap![ok_err $crate::FmtWriter::format($buf, $crate::format_args![$($arg)*])]
44    };
45}
46#[doc(inline)]
47pub use format_buf· as format_buf;

50/// A specialized formatter with a fixed byte buffer and truncation detection.
51#[doc = crate::_doc_meta!{
52    location("text/fmt", struct FmtWriter),
53    #[cfg(target_pointer_width = "32")]
54    test_size_of(FmtWriter = 16|128; niche Option),
55    #[cfg(target_pointer_width = "64")]
56    test_size_of(FmtWriter = 32|256; niche Option),
57}]
58/// It should be faster at runtime than the default formatter.
59#[derive(Debug)]
60pub struct FmtWriter<'a> {
61    buf: &'a mut [u8],
62    /// The number of bytes actually written.
63    len: usize,
64    /// Set to true if any call to write_str did not write the complete input.
65    truncated: bool,
66}

82#[rustfmt::skip]
83impl<'a> FmtWriter<'a> {
84    /// Creates a new writer for the given buffer.
85    pub const fn new(buf: &'a mut [u8]) -> Self { FmtWriter { buf, len: 0, truncated: false } }
86
87    /// Returns true if truncation occurred.
88    pub const fn is_truncated(&self) -> bool { self.truncated }
89
90    /// Returns the number of bytes written.
91    pub const fn written_len(&self) -> usize { self.len }
92
93    /// Writes formatted output into the given byte buffer, returning a string slice.
94    ///
95    /// Returns:
96    /// - `Ok(&str)` if all the formatted data fits into `buf`.
97    /// - `Err(&str)` containing the valid partial result if truncation occurred.
98    ///
99    /// # Examples
100    /// ```
101    /// use devela::FmtWriter;
102    ///
103    /// let mut buf = [0u8; 32]; // Big enough to fit everything
104    /// let s = FmtWriter::format(&mut buf, format_args!["Test: {} {}", "foo", 42]);
105    /// assert_eq!(Ok("Test: foo 42"), s);
106    ///
107    /// let mut buf = [0u8; 9]; // Can't fit everything
108    /// let s = FmtWriter::format(&mut buf, format_args!["Test: {} {}", "foo", 42]);
109    /// assert_eq!(Err("Test: foo"), s);
110    /// ```
111    pub fn format(buf: &'a mut [u8], args: FmtArguments) -> Result<&'a str, &'a str> {
112        let mut w = Self::new(buf);
113        let _ = ::core::fmt::write(&mut w, args);
114        is![w.is_truncated(), Err(w.into_str()), Ok(w.into_str())]
115    }
116
117    /// Writes formatted output into the given byte buffer, returning the number of written bytes.
118    ///
119    /// Returns:
120    /// - `Ok(usize)` if all the formatted data fits into `buf`.
121    /// - `Err(usize)` containing the number of valid UTF-8 written bytes if truncation ocurred,
122    ///
123    /// # Examples
124    /// ```
125    /// use devela::FmtWriter;
126    ///
127    /// let mut buf = [0u8; 32];
128    /// assert_eq![Ok(12), FmtWriter::format_len(&mut buf, format_args!["Test: {} {}", "foo", 42])];
129    ///
130    /// let mut buf = [0u8; 9];
131    /// assert_eq![Err(9), FmtWriter::format_len(&mut buf, format_args!["Test: {} {}", "foo", 42])];
132    /// ```
133    pub fn format_len(buf: &'a mut [u8], args: FmtArguments) -> Result<usize, usize> {
134        let mut w = Self::new(buf);
135        let _ = ::core::fmt::write(&mut w, args);
136        is![w.is_truncated(), Err(w.written_len()), Ok(w.written_len())]
137    }
138    /// Writes formatted output into the given byte buffer, returning the number of written bytes.
139    pub fn format_len_unchecked(buf: &'a mut [u8], args: FmtArguments) -> usize {
140        let mut w = Self::new(buf);
141        let _ = ::core::fmt::write(&mut w, args);
142        w.written_len()
143    }
144
145    /// Writes a string slice to the buffer, returning the number of bytes written.
146    ///
147    /// It copies as much of the string as will fit into the remaining buffer space.
148    ///
149    /// # Examples
150    /// ```
151    /// use devela::FmtWriter;
152    ///
153    /// const fn create_message(buf: &mut [u8]) -> &str {
154    ///     let mut writer = FmtWriter::new(buf);
155    ///     writer.write_str_truncate("Hello");
156    ///     writer.write_str_truncate(" world");
157    ///     writer.into_str_const()
158    /// }
159    /// ```
160    pub const fn write_str_truncate(&mut self, s: &str) -> usize {
161        let available = self.buf.len().saturating_sub(self.len);
162        let s_bytes = s.as_bytes();
163        let n = Cmp(s_bytes.len()).min(available);
164        if n > 0 {
165            slice![mut self.buf, self.len, ..self.len + n].copy_from_slice(slice![s_bytes, ..n]);
166        }
167        is![n < s_bytes.len(), self.truncated = true];
168        self.len += n;
169        n
170    }
171    /// Writes a string slice, returning the actually written part.
172    ///
173    /// Returns `Err` if truncation occurred, with the truncated result.
174    pub const fn write_str_truncate_checked(&mut self, s: &str) -> Result<&str, &str> {
175        self.write_str_truncate(s);
176        let written = self.as_str_const();
177        is![self.is_truncated(), Err(written), Ok(written)]
178    }
179    /// Writes a string slice, returning the number of the bytes actually written.
180    pub const fn write_str_truncate_checked_len(&mut self, s: &str) -> Result<usize, usize> {
181        let n = self.write_str_truncate(s);
182        is![self.is_truncated(), Err(n), Ok(n)]
183    }

185    #[must_use]
186    #[doc = _DOC_AS_INTO_STR!()]
187    pub fn as_str(&self) -> &str {
188        match from_utf8(&self.buf[..self.len]) { // == &(*self.buf).index(..self.len)
189            Ok(valid_str) => valid_str,
190            Err(e) => Self::get_str_from_slice(self.buf, e.valid_up_to()),
191        }
192    }
193    #[must_use]
194    #[doc = _DOC_AS_INTO_STR!()]
195    pub fn into_str(self) -> &'a str {
196        match from_utf8(&self.buf[..self.len]) {
197            Ok(valid_str) => valid_str,
198            Err(e) => Self::get_str_from_slice(self.buf, e.valid_up_to()),
199        }
200    }
201    #[must_use]
202    #[doc = _DOC_AS_INTO_STR_CONST!()]
203    pub const fn as_str_const(&'a self) -> &'a str {
204        match Str::from_utf8(slice![self.buf, ..self.len]) {
205            Ok(valid_str) => valid_str,
206            Err(e) => Self::get_str_from_slice_const(self.buf, e.valid_up_to),
207        }
208    }
209    #[must_use]
210    #[doc = _DOC_AS_INTO_STR_CONST!()]
211    pub const fn into_str_const(self) -> &'a str {
212        match Str::from_utf8(slice![self.buf, ..self.len]) {
213            Ok(valid_str) => valid_str,
214            Err(e) => Self::get_str_from_slice_const(self.buf, e.valid_up_to)
215        }
216    }

218    /* helper methods */
219
220    fn get_str_from_slice(slice: &[u8], valid_len: usize) -> &str {
221        let valid_range = &slice[..valid_len]; // could be faster in debug builds (non-const)
222        cfg_select! { all(feature = "unsafe_str", not(feature = "safe_text")) => {
223            // SAFETY: we only convert the confirmed valid utf-8 length
224            unsafe { Str::from_utf8_unchecked(valid_range) }
225        } _ => { from_utf8(valid_range).unwrap() }} // could use dep_simdutf8 (non-const)
226    }
227    const fn get_str_from_slice_const(slice: &[u8], valid_len: usize) -> &str {
228        let valid_range = slice![slice, ..valid_len];
229        cfg_select! { all(feature = "unsafe_str", not(feature = "safe_text")) => {
230            // SAFETY: we only convert the confirmed valid utf-8 length
231            unsafe { Str::from_utf8_unchecked(valid_range) }
232        } _ => { crate::unwrap![ok Str::from_utf8(valid_range)] }}
233    }
234}

236impl FmtWrite for FmtWriter<'_> {
237    fn write_str(&mut self, s: &str) -> FmtResult<()> {
238        self.write_str_truncate(s);
239        Ok(())
240    }
