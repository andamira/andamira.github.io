1use crate::{TextRange, TextScanner};
2use crate::{is, whilst};

4/// EOL and line-oriented scanning.
5impl<'a> TextScanner<'a> {
6    /// Consumes a single line ending.
7    ///
8    /// Accepts `\n`, `\r`, or `\r\n`.
9    ///
10    /// Returns `true` if a line ending was consumed.
11    #[must_use]
12    pub const fn eat_eol(&mut self) -> bool {
13        let pos = self.cursor.index.as_usize();
14        is! { pos >= self.bytes.len(), return false }
15        match self.bytes[pos] {
16            b'\n' => {
17                self._cursor_bump(1);
18                true
19            }
20            b'\r' => {
21                if pos + 1 < self.bytes.len() && self.bytes[pos + 1] == b'\n' {
22                    self._cursor_bump(2);
23                } else {
24                    self._cursor_bump(1);
25                }
26                true
27            }
28            _ => false,
29        }
30    }
31
32    /// Consumes and returns the range up to, but excluding, the next line ending.
33    ///
34    /// Stops before either `\\n` or `\\r`.
35    /// If no line ending is found, consumes to end of input.
36    pub const fn take_until_eol(&mut self) -> TextRange {
37        let start = self.mark();
38        whilst! { let Some(byte) = self.peek_byte(); {
39            match byte {
40                b'\n' | b'\r' => break,
41                _ => self._cursor_bump(1),
42            }
43        }}
44        self.range_from(start)
45    }
46
47    /// Consumes and returns the next line, excluding its line ending.
48    ///
49    /// Consumes a following `\n`, `\r`, or `\r\n` if present.
50    ///
51    /// Returns `None` at end of input.
52    #[must_use]
53    pub const fn next_line(&mut self) -> Option<TextRange> {
54        is! { self.is_eof(), return None }
55        let range = self.take_until_eol();
56        let _ = self.eat_eol();
57        Some(range)
58    }
59
60    /// Consumes and returns the next line trimmed of ASCII horizontal whitespace.
61    ///
62    /// Currently trims: space, tab.
63    ///
64    /// Returns `None` at end of input.
65    #[must_use]
66    pub const fn next_line_trimmed(&mut self) -> Option<TextRange> {
67        let Some(line) = self.next_line() else {
68            return None;
69        };
70        Some(self.trim_ascii_hws(line))
71    }
72
73    /// Consumes and returns the next line before `byte`, trimmed of ASCII horizontal whitespace.
74    ///
75    /// The delimiter byte is not consumed as part of the returned range, but the
76    /// full source line and its line ending are consumed.
77    ///
78    /// This is useful for simple line-comment formats, for example with `b'#'`.
79    /// It is byte-blind and does not understand quoted strings.
80    ///
81    /// Returns `None` at end of input.
82    #[must_use]
83    pub const fn next_line_trimmed_before(&mut self, byte: u8) -> Option<TextRange> {
84        let Some(line) = self.next_line() else {
85            return None;
86        };
87        let mut end = line.end.0;
88        whilst! { i in line.start.0,..line.end.0; {
89            if self.bytes[i as usize] == byte {
90                end = i;
91                break;
92            }
93        }}
94        Some(self.trim_ascii_hws(TextRange::from_prim(line.start.0, end)))
95    }
96}
