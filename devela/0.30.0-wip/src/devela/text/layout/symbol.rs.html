17/// This controls where text may become separate layout symbols before
18/// inline layout negotiation happens.
19#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
20pub enum TextBreakMode {
21    #[doc = crate::_tags!(init)]
22    /// Break only at explicit whitespace boundaries.
23    #[default]
24    Whitespace,
25
26    /// Break at word-like boundaries.
27    ///
28    /// This is intended for Unicode-aware word segmentation when available.
29    Word,
30
31    /// Break between grapheme clusters.
32    ///
33    /// This is useful for CJK-friendly wrapping and fine-grained cursor mapping.
34    Grapheme,
35}
36_impl_init![Self::Whitespace => TextBreakMode];

58/// Spatial cohesion rules for a text symbol during layout.
59#[doc = crate::_doc_meta!{
60    location("text/layout", enum TextCohesion),
61    test_size_of(TextCohesion = 1|8; niche Option),
62}]
63/// This enum defines what a layout engine is allowed to do with a symbol
64/// when negotiating limited inline space. It expresses layout permissions only;
65/// it does not encode meaning, style, or language semantics.
66#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
67pub enum TextCohesion {
68    /// Must be consumed as a whole or not at all.
69    ///
70    /// The symbol may not be split or removed.
71    /// If it does not fit, layout must stop or fail the fit decision.
72    Atomic,
73
74    /// May be split across layout steps.
75    ///
76    /// The symbol provides a soft break opportunity
77    /// and may be partially consumed if space runs out.
78    Breakable,
79
80    /// May be omitted to satisfy space constraints.
81    ///
82    /// The symbol may be entirely removed from the layout result
83    /// if needed to achieve a fit.
84    Elidable,
85}
86_impl_init![Self::Atomic => TextCohesion];

89/// Mapping between a contiguous text range and its consumed inline space.
90#[doc = crate::_doc_meta!{
91    location("text/layout", struct TextLayoutSpan),
92    test_size_of(TextLayoutSpan = 12|96; niche !Option),
93}]
94/// A `TextLayoutSpan` records that a contiguous range of symbols contributed
95/// a given amount of inline space during a layout step.
96///
97/// It is the primary structural output of text layout
98/// and enables stable mapping between text indices and spatial occupation.
99#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
100pub struct TextLayoutSpan {
101    /// Range between the first and last symbols included in this span.
102    pub range: TextRange,
103    /// Total inline space consumed by the symbols in this span.
104    pub units: TextUnit,
105}
106_impl_init![Self::from_prim(0, 0, 0) => TextLayoutSpan];
107
108impl TextLayoutSpan {
109    /// Creates a span from symbol indices and consumed units.
110    pub const fn new(start: TextIndex, end: TextIndex, units: TextUnit) -> Self {
111        Self { range: TextRange { start, end }, units }
112    }
113    /// Creates a span from symbol indices and consumed units.
114    pub const fn with_range(range: TextRange, units: TextUnit) -> Self {
115        Self { range, units }
116    }
117    /// Creates a span from symbol indices and consumed units.
118    pub const fn from_prim(start: TextUnit, end: TextUnit, units: TextUnit) -> Self {
119        Self::new(TextIndex(start), TextIndex(end), units)
120    }
121    /// Returns the start index.
122    pub const fn start(self) -> TextIndex {
123        self.range.start
124    }
125    /// Returns the end index.
126    pub const fn end(self) -> TextIndex {
127        self.range.end
128    }
129    /// Returns the interval from `start` to `end`.
130    pub const fn interval(self) -> Interval<TextIndex> {
131        Interval::closed_open(self.start(), self.end())
132    }
133}

141/// A `TextSymbol` describes how a unit of text participates in layout:
142/// how much inline space it consumes and what spatial constraints apply under limited space.
143///
144/// It carries no identity or semantic meaning beyond layout behavior.
145#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
146pub struct TextSymbol {
147    /// Inline space consumed by this symbol when laid out as a whole.
148    pub units: TextUnit,
149    /// Spatial cohesion rules applied to this symbol under limited space.
150    pub cohesion: TextCohesion,
151}
152_impl_init![Self::new(TextUnit::INIT, TextCohesion::INIT) => TextSymbol];
153
154impl TextSymbol {
155    /// Creates a span from symbol indices and consumed units.
156    pub const fn new(units: TextUnit, cohesion: TextCohesion) -> Self {
157        Self { units, cohesion }
158    }

161#[doc = crate::_tags!(text layout)]
162/// Policy bundle for deriving layout symbols from text.
163#[doc = crate::_doc_meta!{
164    location("text/layout", struct TextSymbolConfig),
165    test_size_of(TextSymbolConfig = 3|24; niche Option),

167/// `TextSymbolConfig` describes how text is converted into layout symbols:
168/// where boundaries may occur, how widths are measured, and which symbols
169/// may be omitted during layout negotiation.
170#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
171pub struct TextSymbolConfig {
172    /// Boundary policy for derived symbols.
173    pub break_mode: TextBreakMode,
174
175    /// Inline width policy for derived symbols.
176    pub width_mode: TextelWidthMode,
177
178    /// Elision policy for derived symbols.
179    pub elide_mode: TextElideMode,
180}
181_impl_init![Self::DEFAULT => TextSymbolConfig];
182impl Default for TextSymbolConfig {
183    fn default() -> Self {
184        Self::DEFAULT
185    }
186}
187impl TextSymbolConfig {
188    /// Default text symbol configuration.
189    pub const DEFAULT: Self = Self {
190        break_mode: TextBreakMode::Whitespace,
191        width_mode: TextelWidthMode::Mono,
192        elide_mode: TextElideMode::None,
193    };
194    /// Creates a new text symbol configuration.
195    #[must_use]
196    pub const fn new(
197        break_mode: TextBreakMode,
198        width_mode: TextelWidthMode,
199        elide_mode: TextElideMode,
200    ) -> Self {
201        Self { break_mode, width_mode, elide_mode }
202    }
203
204    /// Creates a whitespace-based monospace configuration.
205    #[must_use]
206    pub const fn whitespace_mono() -> Self {
207        Self::DEFAULT
208    }
209    /// Creates a grapheme-based monospace configuration.
210    #[must_use]
211    pub const fn grapheme_mono() -> Self {
212        Self::new(TextBreakMode::Grapheme, TextelWidthMode::Mono, TextElideMode::None)
213    }
214    /// Creates a word-based monospace configuration.
215    #[must_use]
216    pub const fn word_mono() -> Self {
217        Self::new(TextBreakMode::Word, TextelWidthMode::Mono, TextElideMode::None)
218    }
219    /// Returns this configuration with a different break mode.
220    #[must_use]
221    pub const fn with_break_mode(mut self, break_mode: TextBreakMode) -> Self {
222        self.break_mode = break_mode;
223        self
224    }
225    /// Returns this configuration with a different width mode.
226    #[must_use]
227    pub const fn with_width_mode(mut self, width_mode: TextelWidthMode) -> Self {
228        self.width_mode = width_mode;
229        self
230    }
231    /// Returns this configuration with a different elide mode.
232    #[must_use]
233    pub const fn with_elide_mode(mut self, elide_mode: TextElideMode) -> Self {
234        self.elide_mode = elide_mode;
235        self
236    }
