9#[doc = crate::_tags!(text layout)]
10/// Segment behavior during text wrapping.
11#[doc = crate::_doc_meta!{
12    location("text/layout", enum TextSegmentKind),
13    test_size_of(TextSegmentKind = 1|8; niche Option),
14}]
15#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
16pub enum TextSegmentKind {
17    #[doc = crate::_tags!(init)]
18    /// Visible content that normally stays together.
19    #[default]
20    Box,
21
22    /// Collapsible boundary space between boxes.
23    Glue,
24
25    /// Explicit forced line break.
26    HardBreak,
27}
28_impl_init![Self::Box => TextSegmentKind];
29#[rustfmt::skip]
30impl TextSegmentKind {
31    /// Returns whether this segment kind is a box.
32    #[must_use]
33    pub const fn is_box(self) -> bool { matches!(self, Self::Box) }
34
35    /// Returns whether this segment kind is glue.
36    #[must_use]
37    pub const fn is_glue(self) -> bool { matches!(self, Self::Glue) }
38
39    /// Returns whether this segment kind is a hard break.
40    #[must_use]
41    pub const fn is_hard_break(self) -> bool { matches!(self, Self::HardBreak) }
42
43    /// Returns whether both segment kinds are the same.
44    #[must_use]
45    pub const fn eq(self, other: Self) -> bool {
46        matches![(self, other),
47            (Self::Box, Self::Box) | (Self::Glue, Self::Glue) | (Self::HardBreak, Self::HardBreak)
48        ]
49    }
50}

52#[doc = crate::_tags!(text layout)]
53/// A semantic text segment used for wrapping.
54#[doc = crate::_doc_meta!{
55    location("text/layout", struct TextSegment),
56    test_size_of(TextSegment = 16|128; niche Option),
57}]
58#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
59pub struct TextSegment {
60    /// Covered symbol range and whole-segment inline extent.
61    pub span: TextLayoutSpan,
62
63    /// Segment behavior during wrapping.
64    pub kind: TextSegmentKind,
65}
66_impl_init![Self::INIT => TextSegment];
67
68impl TextSegment {
69    /// A zero-width empty box segment.
70    pub const INIT: Self = Self {
71        span: TextLayoutSpan::INIT,
72        kind: TextSegmentKind::Box,
73    };
74
75    /// Creates a text segment.
76    #[must_use]
77    pub const fn new(span: TextLayoutSpan, kind: TextSegmentKind) -> Self {
78        Self { span, kind }
79    }
80
81    /// Creates a box segment from primitive indices and units.
82    #[must_use]
83    pub const fn box_from_prim(start: u32, end: u32, units: TextUnit) -> Self {
84        Self::new(TextLayoutSpan::from_prim(start, end, units), Kind::Box)
85    }
86
87    /// Creates a glue segment from primitive indices and units.
88    #[must_use]
89    pub const fn glue_from_prim(start: u32, end: u32, units: TextUnit) -> Self {
90        Self::new(TextLayoutSpan::from_prim(start, end, units), Kind::Glue)
91    }
92
93    /// Creates a hard-break segment from primitive indices.
94    #[must_use]
95    pub const fn hard_break_from_prim(start: u32, end: u32) -> Self {
96        Self::new(TextLayoutSpan::from_prim(start, end, 0), Kind::HardBreak)
97    }
98
99    /// Returns whether this segment is a box.
100    #[must_use]
101    pub const fn is_box(self) -> bool {
102        matches!(self.kind, Kind::Box)
103    }
104
105    /// Returns whether this segment is glue.
106    #[must_use]
107    pub const fn is_glue(self) -> bool {
108        matches!(self.kind, Kind::Glue)
109    }
110
111    /// Returns whether this segment is a hard break.
112    #[must_use]
113    pub const fn is_hard_break(self) -> bool {
114        matches!(self.kind, Kind::HardBreak)
115    }
116
117    /// Fills `out` with simple paragraph-oriented segments derived from chars.
118    ///
119    /// Current policy:
120    /// - contiguous non-whitespace chars become [`TextSegmentKind::Box`]
121    /// - contiguous inline whitespace becomes one [`TextSegmentKind::Glue`]
122    /// - `\n` and `\r\n` become [`TextSegmentKind::HardBreak`]
123    ///
124    /// Returns the initialized prefix length.
125    pub const fn segments_from_chars(text: &str, out: &mut [TextSegment]) -> usize {
126        let (mut seg_len, mut char_i) = (0, 0);
127        let (mut run_start, mut run_units) = (0, 0);
128        let mut run_kind: Option<TextSegmentKind> = None;
129        let mut chars = CharIter::<&str>::new(text);
130        while let Some(ch) = chars.next_char() {
131            if ch == '\n' || ch == '\r' {
132                if let Some(kind) = run_kind.take() {
133                    let span = TextLayoutSpan::from_prim(run_start, char_i, run_units);
134                    is! { !push_segment(out, &mut seg_len, span, kind), return seg_len }
135                }
136                let break_start = char_i;
137                char_i += 1;
138                if ch == '\r' && matches!(chars.peek_char(), Some('\n')) {
139                    let _ = chars.next_char();
140                    char_i += 1;
141                }
142                let span = TextLayoutSpan::from_prim(break_start, char_i, 0);
143                if !push_segment(out, &mut seg_len, span, Kind::HardBreak) {
144                    return seg_len;
145                }
146                continue;
147            }
148            let kind = is! { ch.is_whitespace(), Kind::Glue, Kind::Box };
149            match run_kind {
150                None => {
151                    run_start = char_i;
152                    run_units = 1;
153                    run_kind = Some(kind);
154                }
155                Some(prev) if prev.eq(kind) => {
156                    is! { kind.is_box(), run_units += 1 }
157                    // Glue stays width 1, regardless of how many whitespace chars it covers.
158                }
159                Some(prev) => {
160                    let span = TextLayoutSpan::from_prim(run_start, char_i, run_units);
161                    is! { !push_segment(out, &mut seg_len, span, prev), return seg_len }
162                    run_start = char_i;
163                    run_units = 1;
164                    run_kind = Some(kind);
165                }
166            }
167            char_i += 1;
168        }
169        if let Some(kind) = run_kind {
170            let span = TextLayoutSpan::from_prim(run_start, char_i, run_units);
171            let _ = push_segment(out, &mut seg_len, span, kind);
172        }
173        seg_len
174    }
175}
176
177const fn push_segment(
178    out: &mut [TextSegment],
179    len: &mut usize,
180    span: TextLayoutSpan,
181    kind: TextSegmentKind,
182) -> bool {
183    is! { *len == out.len(), return false }
184    out[*len] = TextSegment::new(span, kind);
185    *len += 1;
186    true
187}
