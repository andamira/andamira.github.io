37#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
38pub struct TextLine {

69/// Iterates wrapped visual lines over text segments.
70#[doc = crate::_doc_meta!{
71    location("text/layout", struct TextWrapIter),
72    #[cfg(target_pointer_width = "32")]

77///
78/// `TextWrapIter` applies simple prose wrapping over semantic segments:
79/// boxes stay together when possible, glue collapses between boxes, and hard
80/// breaks force a line boundary.
81#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
82pub struct TextWrapIter<'a> {
83    segments: &'a [TextSegment],
84    cursor: usize,
85    width: TextUnit,
86    split: Option<TextIndex>,
87}
88
89impl<'a> TextWrapIter<'a> {
90    /// Creates a text wrapping iterator.
91    #[must_use]
92    pub const fn new(segments: &'a [TextSegment], width: TextUnit) -> Self {
93        Self { segments, cursor: 0, width, split: None }
94    }
95
96    /// Returns the current segment cursor.
97    #[must_use]
98    pub const fn cursor(&self) -> usize {
99        self.cursor
100    }
101    /// Returns the wrapping width.
102    #[must_use]
103    pub const fn width(&self) -> TextUnit {
104        self.width
105    }
106    /// Returns whether a segment is currently being split across lines.
107    #[must_use]
108    pub const fn is_splitting(&self) -> bool {
109        self.split.is_some()
110    }
111
112    /// Computes the next wrapped visual line.
113    ///
114    /// Advances the internal segment cursor and returns the next [`TextLine`],
115    /// or `None` when no further line can be produced.
116    ///
117    /// The returned line records the covered source range,
118    /// consumed inline units, continuation cursor, and break kind.
119    ///
120    /// This performs semantic wrapping over [`TextSegment`]s.
121    /// It does not shape, render, scroll, place, or interact with text.
122    pub const fn next_line(&mut self) -> Option<TextLine> {
123        is! { self.width == 0, return None }
124        let (mut i, line_start, mut line_units, mut last_end);
125        let mut break_kind = Break::Between;
126        if let Some(split_start) = self.split {
127            let seg = self.segments[self.cursor];
128            let seg_end = seg.span.end();
129            let remaining = seg_end.0 - split_start.0;
130            let take = if remaining > self.width { self.width } else { remaining };
131            let split_end = TextIndex(split_start.0 + take);
132            line_start = split_start;
133            line_units = take;
134            last_end = split_end;
135            if remaining > self.width {
136                self.split = Some(split_end);
137                let span = TextLayoutSpan::new(line_start, split_end, take);
138                return Some(TextLine::new(span, Some(TextCursor::new(split_end)), Break::Inside));
139            }
140            self.split = None;
141            self.cursor += 1;
142            i = self.cursor;
143            if line_units == self.width {
144                let carry = is! { i < self.segments.len(), Some(TextCursor::new(last_end)), None };
145                let span = TextLayoutSpan::new(line_start, last_end, line_units);
146                return Some(TextLine::new(span, carry, Break::Between));
147            }
148        } else {
149            while self.cursor < self.segments.len() && self.segments[self.cursor].kind.is_glue() {
150                self.cursor += 1;
151            }
152            is! { self.cursor >= self.segments.len(), return None }
153            i = self.cursor;
154            line_start = self.segments[i].span.start();
155            line_units = 0;
156            last_end = line_start;
157        }
158        let mut pending_glue = false;
159        while i < self.segments.len() {
160            let seg = self.segments[i];
161            match seg.kind {
162                TextSegmentKind::HardBreak => {
163                    break_kind = Break::Hard;
164                    i += 1;
165                    self.cursor = i;
166                    let carry = if i < self.segments.len() {
167                        Some(TextCursor::new(seg.span.end()))
168                    } else {
169                        None
170                    };
171                    let span = TextLayoutSpan::new(line_start, last_end, line_units);
172                    return Some(TextLine::new(span, carry, break_kind));
173                }
174                TextSegmentKind::Glue => {
175                    pending_glue = line_units > 0;
176                    i += 1;
177                }
178                TextSegmentKind::Box => {
179                    let glue_units = if pending_glue { 1 } else { 0 };
180                    let needed = glue_units + seg.span.units;
181                    if line_units > 0 && line_units + needed > self.width {
182                        break_kind = Break::Between;
183                        break;
184                    }
185                    if line_units == 0 && seg.span.units > self.width {
186                        if can_split_by_unit(seg) {
187                            let take = self.width;
188                            let split_end = TextIndex(seg.span.start().0 + take);
189                            self.split = Some(split_end);
190                            self.cursor = i;
191                            let span = TextLayoutSpan::new(seg.span.start(), split_end, take);
192                            let cursor = TextCursor::new(split_end);
193                            return Some(TextLine::new(span, Some(cursor), Break::Inside));
194                        } else {
195                            // If the segment cannot be split by simple unit/index arithmetic,
196                            // keep it whole. This may produce an over-wide line,
197                            // but preserves correct source ranges.
198                            line_units = seg.span.units;
199                            last_end = seg.span.end();
200                            i += 1;
201                            break;
202                        }
203                    }
204                    line_units += needed;
205                    last_end = seg.span.end();
206                    pending_glue = false;
207                    i += 1;
208                }
209            }
210        }
211        self.cursor = i;
212        let carry = if self.cursor < self.segments.len() || self.split.is_some() {
213            Some(TextCursor::new(last_end))
214        } else {
215            None
216        };
217        let span = TextLayoutSpan::new(line_start, last_end, line_units);
218        Some(TextLine::new(span, carry, break_kind))
219    }
220}
221
222impl Iterator for TextWrapIter<'_> {
223    type Item = TextLine;
224    fn next(&mut self) -> Option<Self::Item> {
225        self.next_line()
226    }
227}
228
229const fn can_split_by_unit(seg: TextSegment) -> bool {
230    seg.span.end().0 - seg.span.start().0 == seg.span.units
