8#[doc = crate::_tags!(text layout result)]
9/// Result of testing whether text fits within an inline extent.
10#[doc = crate::_doc_meta!{
11    location("text/layout", enum TextFit),
12    test_size_of(TextFit = 1|8; niche Option),
13}]
14#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
15pub enum TextFit {
16    /// All remaining symbols were resolved; no continuation is required.
17    Full,
18
19    /// Some inline space was consumed, but continuation is still required.
20    Partial,
21
22    #[doc = crate::_tags!(init)]
23    /// No inline space was consumed and continuation is still required.
24    #[default]
25    None,
26}
27_impl_init![Self::None => TextFit];

29#[doc = crate::_tags!(text layout result)]
30/// Result of a single text layout step.
31#[doc = crate::_doc_meta!{
32    location("text/layout", struct TextLayoutStep),
33    #[cfg(target_pointer_width = "32")]
34    test_size_of(TextLayoutStep = 20|160; niche Option),
35    #[cfg(target_pointer_width = "64")]
36    test_size_of(TextLayoutStep = 24|192; niche Option),
37}]
38///
39/// A layout step consumes a prefix of the symbol stream within a given inline extent,
40/// and reports:
41/// - how much space was used
42/// - which spans were produced
43/// - whether layout should continue
44#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
45pub struct TextLayoutStep {
46    /// Number of valid [`TextLayoutSpan`][crate::TextLayoutSpan]
47    /// records written to the output buffer.
48    pub span_count: usize,
49
50    /// Total inline space consumed by this step.
51    pub consumed: TextUnit,
52
53    /// Continuation cursor, if not all symbols were consumed.
54    ///
55    /// `None` indicates that layout reached the end of the symbol stream.
56    pub carry: Option<TextCursor>,
57
58    /// Result of testing whether the text fit the available extent.
59    pub fit: TextFit,
60}
61impl TextLayoutStep {
62    /// Creates a new text layout step outcome.
63    pub const fn new(
64        span_count: usize,
65        consumed: TextUnit,
66        carry: Option<TextCursor>,
67        fit: TextFit,
68    ) -> Self {
69        TextLayoutStep { span_count, consumed, carry, fit }
70    }
71}
72_impl_init![Self::new(0, TextUnit::INIT, None, TextFit::INIT) => TextLayoutStep];
