8/// # Methods over `u8` slice.
9#[rustfmt::skip]
10impl Char<&[u8]> {
11    /// Decodes a UTF-8 scalar at `index`.
12    ///
13    /// Returns `Some((char, len))` if the input is a valid UTF-8 sequence
14    /// and the decoded value is a valid Unicode scalar.
15    ///
16    /// Returns `None` if:
17    /// - The index is out of bounds.
18    /// - The bytes do not form a valid UTF-8 sequence.
19    /// - The decoded value is not a valid Unicode scalar.
20    ///
21    /// This is implemented via `Char::`[`to_scalar()`][Self::to_scalar].
22    ///
23    /// # Examples
24    /// ```
25    /// use devela::Char;
26    ///
27    /// // Valid UTF-8 sequence
28    /// let result = Char(b"\xE2\x82\xAC").to_char(0); // €
29    /// assert_eq!(result, Some(('€', 3)));
30    ///
31    /// // Invalid continuation bytes
32    /// let invalid_continuation = Char(b"\xE2\x41\xAC").to_char(0);
33    /// assert_eq!(invalid_continuation, None);
34    ///
35    /// // Surrogate code point
36    /// let surrogate = Char(b"\xED\xA0\x80").to_char(0); // U+D800
37    /// assert_eq!(surrogate, None);
38    ///
39    /// // Out of bounds index
40    /// let out_of_bounds = Char(b"hello").to_char(10);
41    /// assert_eq!(out_of_bounds, None);
42    ///
43    /// // Incomplete sequence
44    /// let incomplete = Char(b"\xE2\x82").to_char(0); // Missing third byte
45    /// assert_eq!(incomplete, None);
46    /// ```
47    /// # Features
48    /// Uses the `unsafe_str` feature to skip duplicated validation checks.
49    #[must_use]
50    pub const fn to_char(self, index: usize) -> Option<(char, usize)> {
51        let (cp, len) = unwrap![some? self.to_scalar(index)]; // check cp is a valid scalar
52        cfg_select! { all(feature = "unsafe_str", not(feature = "safe_text")) => {
53            Some((unsafe { char::from_u32_unchecked(cp) }, len)) // SAFETY: we just checked
54        } _ => { Some((unwrap![some? char::from_u32(cp)], len)) }}
55    }

57    /// Decodes a UTF-8 scalar leniently at `index`, validating only the final Unicode scalar.
58    ///
59    /// This method is forgiving of UTF-8 encoding errors but ensures the result
60    /// is a valid Unicode scalar value.
61    ///
62    /// - Does not validate UTF-8 continuation bytes (may decode malformed sequences).
63    /// - If the leading byte is invalid it returns the replacement character (`�`).
64    ///
65    /// This is implemented via `Char::`[to_scalar_unchecked()][Self::to_scalar_unchecked].
66    ///
67    /// # Panics
68    /// Panics if the decoded value is not a valid Unicode scalar value,
69    /// or if the `index` is out of bounds.
70    ///
71    /// # Examples
72    /// ```
73    /// use devela::Char;
74    ///
75    /// // Valid UTF-8 sequence
76    /// let result = Char(b"\xE2\x82\xAC").to_char_lenient(0); // €
77    /// assert_eq!(result, ('€', 3));
78    ///
79    /// // Invalid UTF-8 but decodes to valid scalar - behavior depends on input
80    /// // This may return unexpected characters rather than panicking
81    /// let result = Char(b"\xE2\x41\xAC").to_char_lenient(0);
82    /// assert_eq!(result, ('\u{206c}', 3));
83    ///
84    /// // Surrogate code point - will panic
85    /// // let result = Char(b"\xED\xA0\x80").to_char_lenient(0); // PANIC: U+D800 is invalid
86    ///
87    /// // Out of bounds index - will panic
88    /// // let result = Char(b"hello").to_char_lenient(10); // PANIC: index out of bounds
89    /// ```
90    #[must_use]
91    pub const fn to_char_lenient(self, index: usize) -> (char, usize) {
92        let (cp, len) = self.to_scalar_unchecked(index);
93        (unwrap![some char::from_u32(cp)], len)
94    }

96    /// Decodes a UTF-8 scalar at `index` without any validation.
97    ///
98    /// If the leading byte is invalid it returns the replacement character (`�`).
99    ///
100    /// This is implemented via `Char::`[`to_scalar_unchecked`][Self::to_scalar_unchecked].
101    ///
102    /// # Safety
103    /// The caller must ensure that:
104    /// - `index` is within bounds of `bytes`
105    /// - `bytes[index..]` contains a valid UTF-8 sequence
106    /// - The decoded value is a valid Unicode scalar.
107    ///
108    /// Violating these conditions may lead to undefined behavior.
109    #[must_use]
110    #[cfg(all(not(feature = "safe_text"), feature = "unsafe_str"))]
111    #[cfg_attr(nightly_doc, doc(cfg(all(not(feature = "safe_text"), feature = "unsafe_str"))))]
112    pub const unsafe fn to_char_unchecked(self, index: usize) -> (char, usize) {
113        let (cp, len) = self.to_scalar_unchecked(index);
114        (unsafe { char::from_u32_unchecked(cp) }, len)
115    }

117    /// Decodes a UTF-8 scalar from the given byte slice, starting at `index`.
118    ///
119    /// Returns `(scalar, len)`, where `scalar` is the decoded Unicode scalar,
120    /// and `len` is the number of bytes consumed.
121    ///
122    /// Returns `None` if:
123    /// - The index is out of bounds.
124    /// - The bytes do not form a valid UTF-8 sequence.
125    /// - The decoded value is not a valid Unicode scalar.
126    ///
127    /// # Examples
128    /// ```
129    /// use devela::Char;
130    ///
131    /// assert_eq!(Char("Ħ".as_bytes()).to_scalar(0), Some((u32::from('Ħ'), 2)));
132    ///
133    /// let invalid = b"\x80"; // Invalid leading byte
134    /// assert_eq!(Char(invalid).to_scalar(0), None);
135    /// ```
136    #[must_use]
137    pub const fn to_scalar(self, index: usize) -> Option<(u32, usize)> {
138        if index >= self.0.len() { return None; } // out of bounds
139        let (bytes, first) = (self.0, self.0[index]);
140        if first < 0x80 { return Some((first as u32, 1)); } // ASCII fast path
141        let len = unwrap![some? Char(bytes[index]).len_utf8()]; // invalid leading byte?
142        if index + len > bytes.len() { return None; } // not enough bytes?
143        if !self.has_valid_continuation(index, len) { return None; } // malformed utf-8?
144        if self.has_overlong_encoding(index, len) { return None; } // overlong encoding?
145        let scalar = self.decode_scalar(index, len);
146        is![Char(scalar).is_valid_scalar(), Some((scalar, len)), None] // invalid scalar?
147    }

149    /// Decodes a UTF-8 scalar from the given byte slice, starting at `index`, without validation.
150    ///
151    /// Returns `(scalar, len)`, where `scalar` is the decoded Unicode scalar,
152    /// and `len` is the number of bytes consumed.
153    ///
154    /// It assumes `bytes[index..]` contains a valid UTF-8 sequence,
155    /// and it doesn't validate the resulting Unicode scalar.
156    ///
157    /// If the leading byte is invalid it returns the replacement character (`�`).
158    ///
159    /// # Panics
160    /// It will panic if the index is out of bounds.
161    #[must_use]
162    pub const fn to_scalar_unchecked(self, index: usize) -> (u32, usize) {
163        let first = self.0[index];
164        if first < 0x80 { return (first as u32, 1); } // ASCII fast path
165        let len = Char(first).len_utf8_unchecked();
166        if len == 0 { return (char::REPLACEMENT_CHARACTER as u32, 1); } // invalid leading byte?
167        (self.decode_scalar(index, len), len)
168    }
169
170    #[must_use]
171    const fn decode_scalar(self, index: usize, len: usize) -> u32 {
172        let (bytes, first) = (self.0, self.0[index]);
173        match len {
174            1 => first as u32,
175            2 => ((first as u32 & 0x1F) << 6) | (bytes[index + 1] as u32 & Char::<u32>::CONT_MASK),
176            3 => ((first as u32 & 0x0F) << 12)
177                | ((bytes[index + 1] as u32 & Char::<u32>::CONT_MASK) << 6)
178                | (bytes[index + 2] as u32 & Char::<u32>::CONT_MASK),
179            4 => ((first as u32 & 0x07) << 18)
180                | ((bytes[index + 1] as u32 & Char::<u32>::CONT_MASK) << 12)
181                | ((bytes[index + 2] as u32 & Char::<u32>::CONT_MASK) << 6)
182                | (bytes[index + 3] as u32 & Char::<u32>::CONT_MASK),
183            _ => char::REPLACEMENT_CHARACTER as u32,
184        }
185    }

187    /// Returns `true` if the UTF-8 sequence starting at `index` is overlong encoded.
188    ///
189    /// This method only checks for overlong encodings, but not other UTF-8 validity rules.
190    /// It does not verify continuation byte patterns nor invalid scalar values.
191    ///
192    /// Overlong encodings use more bytes than necessary to represent a character,
193    /// which is invalid in well-formed UTF-8.
194    ///
195    /// # Examples
196    /// ```
197    /// use devela::Char;
198    ///
199    /// assert!(Char(b"\xE0\x80\x80").has_overlong_encoding(0, 3)); // overlong encoding
200    /// assert!(!Char(b"\xE0\xA0\x80").has_overlong_encoding(0, 3)); // valid 3-byte sequence
201    /// ```
202    #[must_use] #[rustfmt::skip]
203    pub const fn has_overlong_encoding(self, index: usize, len: usize) -> bool {
204        let bytes = self.0;
205        if index + len > bytes.len() { return false; }
206        let first = bytes[index];
207        match len {
208            // should've been 1: C0, C1 are always overlong
209            2 => { first == 0xC0 || first == 0xC1 }
210            // E0 80..9F are overlong (should be 1-2 bytes)
211            3 if first == 0xE0 => { let second = bytes[index + 1]; second < 0xA0 }
212            // F0 80..8F are overlong (should be 1-3 bytes)
213            4 if first == 0xF0 => { let second = bytes[index + 1]; second < 0x90 }
214            _ => false, // 1-byte sequences can't be overlong
215        }
216    }

218    /// Verifies that the continuation bytes following a UTF-8 leading byte are properly formatted.
219    ///
220    /// Each continuation byte must match the pattern `10xxxxxx` (i.e., have the high bits `0b10`).
221    /// This ensures the byte sequence follows proper UTF-8 encoding rules.
222    ///
223    /// This method only verifies correct syntax, but not correct semantics.
224    /// It does not check for overlong encodings nor invalid scalar values.
225    ///
226    /// # Examples
227    /// ```
228    /// use devela::Char;
229    ///
230    /// assert!(Char(b"\xE2\x82\xAC").has_valid_continuation(0, 3)); // euro sign €
231    /// assert!(!Char(b"\xE2\x41\xAC").has_valid_continuation(0, 3)); // second byte is ASCII 'A'
232    /// assert!(!Char(b"\xC2").has_valid_continuation(0, 2)); // incomplete sequence
233    /// ```
234    pub const fn has_valid_continuation(self, index: usize, len: usize) -> bool {
235        let bytes = self.0;
236        is![bytes.len() < index + len, return false]; // ensure sufficient len
237        match len {
238            1 => true, // no continuation bytes needed for ASCII
239            2 => bytes[index + 1] & 0xC0 == 0x80,
240            3 => bytes[index + 1] & 0xC0 == 0x80 && bytes[index + 2] & 0xC0 == 0x80,
241            4 => bytes[index + 1] & 0xC0 == 0x80 && bytes[index + 2] & 0xC0 == 0x80
242                && bytes[index + 3] & 0xC0 == 0x80,
243            _ => false, // invalid length
244        }
245    }

247    /// Returns `true` if the byte at `index` is a valid starting point for a UTF-8 sequence.
248    ///
249    /// This checks if the byte is not a UTF-8 continuation byte (i.e., it's either
250    /// an ASCII character or a valid leading byte of a multi-byte sequence).
251    ///
252    /// Useful for safely starting UTF-8 decoding from an arbitrary position in a byte slice.
253    #[must_use]
254    pub const fn is_utf8_boundary(self, index: usize) -> bool {
255        index == self.0.len()
256            || (index < self.0.len() && Char(self.0[index]).is_utf8_boundary())
257    }
258    /// Returns the smallest UTF-8 boundary `>= index`.
259    #[must_use]
260    pub const fn ceil_utf8_boundary(self, index: usize) -> usize {
261        let bytes = self.0;
262        let mut i = is![index < bytes.len(), index, bytes.len()];
263        while i < bytes.len() && !Char(bytes).is_utf8_boundary(i) { i += 1; }
264        i
265    }
266    /// Returns the greatest UTF-8 boundary `<= index`.
267    ///
268    /// If `index > self.len()`, starts from `self.len()`.
269    ///
270    /// This only checks boundary shape, not full UTF-8 validity.
271    /// It is intended for already-valid UTF-8 byte slices.
272    #[must_use]
273    pub const fn floor_utf8_boundary(self, index: usize) -> usize {
274        let bytes = self.0;
275        let mut i = is![index < bytes.len(), index, bytes.len()];
276        while i > 0 && !Char(bytes).is_utf8_boundary(i) { i -= 1; }
277        i
278    }
