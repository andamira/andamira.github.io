164            #[doc = "Creates a `" $name "` from the first UTF-8 encoded scalar value in a byte
165            /// slice."]
166            ///
167            /// Returns `None` if the slice is empty or doesn't contain a valid UTF-8 sequence
168            /// at the beginning.
169            ///
170            /// # Examples
171            /// ```
172            /// use devela::charu;
173            ///
174            /// let ascii = charu::from_utf8(b"A").unwrap();
175            /// assert_eq!(ascii.to_utf8_bytes(), [b'A', 0, 0, 0]);
176            ///
177            /// let multi_byte = charu::from_utf8(b"\xC2\xA2 rest").unwrap(); // ¢
178            /// assert_eq!(multi_byte.to_utf8_bytes(), [0xC2, 0xA2, 0, 0]);
179            ///
180            /// assert!(charu::from_utf8(b"").is_none());
181            /// assert!(charu::from_utf8(b"\xC2").is_none()); // incomplete sequence
182            /// ```
183            /// # Features
184            /// `unsafe_hint` enables unchecked handling of unreachable branches.
185            #[must_use] #[rustfmt::skip]
186            pub const fn from_utf8(bytes: &[u8]) -> Option<$name> {
187                is![bytes.is_empty(), return None];
188                let len = unwrap![some? Char(bytes[0]).len_utf8()]; // invalid leading byte?
189                is![!Char(bytes).has_valid_continuation(0, len), return None]; // malformed utf-8?
190                is![Char(bytes).has_overlong_encoding(0, len), return None]; // overlong encoding?
191                is![len == 3 && bytes[0] == 0xED && bytes[1] >= 0xA0, return None]; // surrogate?
192                Some(Self::_from_utf8_prefix_trusted(bytes, len))
193            }

319            // Private helper to create a charu* from the first scalar value in a UTF-8 byte slice.
320            // Most efficient implementation with jump table, bit shifting and oring.
321            // This doesn't perform any validation and should be used carefully.
322            #[rustfmt::skip]
323            pub(crate) const fn _from_utf8_prefix_trusted(bytes: &[u8], len: usize) -> $name {
324                let scalar = match len {
325                    1 =>  (bytes[0] as u32) << 24,
326                    2 => ((bytes[0] as u32) << 24)
327                       | ((bytes[1] as u32) << 16),
328                    3 => ((bytes[0] as u32) << 24)
329                       | ((bytes[1] as u32) << 16)
330                       | ((bytes[2] as u32) <<  8),
331                    4 => ((bytes[0] as u32) << 24)
332                       | ((bytes[1] as u32) << 16)
333                       | ((bytes[2] as u32) <<  8)
334                       |  (bytes[3] as u32),
335                    _ => cfg_select! { all(feature = "unsafe_hint", not(feature = "safe_text")) => {
336                        unsafe { ::core::hint::unreachable_unchecked() } }
337                        _ => { unreachable!() }}
338                };
339                Self::new_unchecked(scalar)
340            }

371            /// Writes the UTF-8 representation to a buffer and returns it as a string slice.
372            ///
373            /// # Examples
374            /// ```
375            #[doc = "use devela::" $name ";\n\n"]
376            #[doc = "let c = " $name "::from_char('A');"]
377            /// let mut buf = [0u8; 4];
378            /// assert_eq!(c.as_str_into(&mut buf), "A");

380            #[doc = "let c = " $name "::from_char('¢');"]

382            /// assert_eq!(c.as_str_into(&mut buf), "¢");
383            /// ```
384            /// # Features
385            /// Uses the `unsafe_str` feature to avoid duplicated validation.
386            #[must_use]
387            pub const fn as_str_into<'a>(&self, buf: &'a mut [u8; 4]) -> &'a str {
388                *buf = self.to_utf8_bytes();
389                let len = self.len_utf8();
390                cfg_select! { all(feature = "unsafe_str", not(feature = "safe_text")) => {
391                    // SAFETY: charu always contains valid UTF-8
392                    unsafe { Str::from_utf8_unchecked(slice![mut buf, ..len]) }
393                } _ => {
394                    unwrap![ok Str::from_utf8(slice![mut buf, ..len])]
395                }}
396            }

405            /// Returns the UTF-8 byte representation as a 4-byte array.
406            ///
407            /// The bytes are arranged with the **first UTF-8 byte at index 0**,
408            /// exactly matching how UTF-8 is stored in `&str` and `[u8]`.
409            /// Unused bytes in the sequence are set to 0.
410            ///
411            /// # Examples
412            /// ```
413            /// use devela::charu;
414            ///
415            /// let c = charu::from_char('A');
416            /// assert_eq!(c.to_utf8_bytes(), [0x41, 0, 0, 0]);
417            ///
418            /// let c = charu::from_char('¢');
419            /// assert_eq!(c.to_utf8_bytes(), [0xC2, 0xA2, 0, 0]);
420            ///
421            /// let c = charu::from_char('€');
422            /// assert_eq!(c.to_utf8_bytes(), [0xE2, 0x82, 0xAC, 0]);
423            ///
424            /// let c = charu::from_char('𝅘𝅥𝅮');
425            /// assert_eq!(c.to_utf8_bytes(), [0xF0, 0x9D, 0x85, 0xA0]);
