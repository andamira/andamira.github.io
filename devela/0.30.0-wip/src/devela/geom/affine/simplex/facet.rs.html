17/// A facet is a codimension-one face of a simplex. This view borrows the
18/// original simplex and exposes its vertices in their original order, except
19/// for the vertex at [`omitted_index`](Self::omitted_index).
20///
21/// No vertices are copied and no storage is allocated.
22///
23/// # Orientation
24///
25/// In the oriented boundary of a simplex, the facet omitting vertex `i` has
26/// coefficient `(-1)^i`. This coefficient is returned by
27/// [`boundary_sign`](Self::boundary_sign).
28///
29/// # Empty facets
30///
31/// The empty facet of a point simplex is not represented. Consequently,
32/// [`Simplex::facet`] returns `None` for every `Simplex<T, D, 1>`.
33#[must_use]
34#[derive(Clone, Copy, Debug)]
35pub struct SimplexFacetView<'a, T, const D: usize, const V: usize> {
36    pub(super) simplex: &'a Simplex<T, D, V>,
37    pub(super) omitted: usize,
38}
39#[rustfmt::skip]
40impl<'a, T, const D: usize, const V: usize> SimplexFacetView<'a, T, D, V> {
41    /// Returns the simplex from which this facet is projected.
42    pub const fn simplex(&self) -> &'a Simplex<T, D, V> { self.simplex }
43
44    /// Returns the index of the omitted vertex.
45    #[must_use]
46    pub const fn omitted_index(&self) -> usize { self.omitted }
47
48    /// Returns the vertex opposite this facet.
49    pub const fn opposite_vertex(&self) -> &Point<T, D> { &self.simplex.vertices[self.omitted] }
50
51    /// Returns the number of vertices in this facet.
52    #[must_use]
53    pub const fn vertex_count(&self) -> usize { V - 1 }
54
55    /// Returns the dimension implied by this facet's vertex count.
56    #[must_use]
57    pub const fn dimension(&self) -> usize { V - 2 }
58
59    /// Returns the facet vertex at `index`.
60    ///
61    /// The returned vertices retain their relative order in the parent simplex.
62    #[must_use]
63    pub const fn vertex(&self, index: usize) -> Option<&Point<T, D>> {
64        is! { index >= V - 1, return None }
65        let source = if index < self.omitted { index } else { index + 1 };
66        Some(&self.simplex.vertices[source])
67    }
68    /// Returns this facet's coefficient in the oriented simplex boundary.
69    ///
70    /// This is positive when the omitted index is even and negative when it
71    /// is odd.
72    pub const fn boundary_sign(&self) -> Sign {
73        is! { self.omitted & 1 == 0, Sign::Positive, Sign::Negative }
74    }
75}

78/// An iterator over the facets of a [`Simplex`].
79#[doc = crate::_doc_meta!{
80    location("geom/affine", struct SimplexFacetIter),
81    #[cfg(target_pointer_width = "32")]
82    test_size_of(SimplexFacetIter<f32, 2, 4> = 12|96; niche Option),
83    #[cfg(target_pointer_width = "64")]
84    test_size_of(SimplexFacetIter<f32, 2, 4> = 24|192; niche Option),
85}]
86/// Facets are yielded in omitted-vertex order.
87#[must_use]
88#[derive(Clone, Debug)]
89pub struct SimplexFacetIter<'a, T, const D: usize, const V: usize> {
90    pub(super) simplex: &'a Simplex<T, D, V>,
91    pub(super) front: usize,
92    pub(super) back: usize,
93}
94#[rustfmt::skip]
95impl<'a, T, const D: usize, const V: usize> SimplexFacetIter<'a, T, D, V> {
96    /// Returns the number of facets not yet yielded.
97    #[must_use]
98    #[allow(clippy::len_without_is_empty)]
99    pub const fn len(&self) -> usize {
100        self.back - self.front
101    }
102    /// Returns the next facet.
103    #[must_use]
104    pub const fn next(&mut self) -> Option<SimplexFacetView<'a, T, D, V>> {
105        is! { Self::len(self) == 0, return None }
106        let omitted = self.front;
107        self.front += 1;
108        Some(SimplexFacetView { simplex: self.simplex, omitted })
109    }
110    /// Returns the next facet from the back.
111    #[must_use]
112    pub const fn next_back(&mut self) -> Option<SimplexFacetView<'a, T, D, V>> {
113        is! { Self::len(self) == 0, return None }
114        self.back -= 1;
115        Some(SimplexFacetView { simplex: self.simplex, omitted: self.back })
116    }
117}

119impl<'a, T, const D: usize, const V: usize> Iterator for SimplexFacetIter<'a, T, D, V> {
120    type Item = SimplexFacetView<'a, T, D, V>;
121
122    fn next(&mut self) -> Option<Self::Item> {
123        Self::next(self)
124    }
125    fn count(self) -> usize {
126        Self::len(&self)
127    }
128    fn size_hint(&self) -> (usize, Option<usize>) {
129        let len = self.len();
130        (len, Some(len))
131    }
132}

137    }
